2//! tamias-sign — optional content signing for published packages, via
3//! `ssh-keygen -Y sign`/`-Y verify` (the same mechanism sturn uses for
4//! its WAL ops; SSH signing is stdlib tooling — Ed25519 under the
5//! hood, no new crypto deps).
6//!
7//! **What is signed.** The canonical payload is the package's
8//! provenance anchors — name, scope, version, and the `sha256:` digest
9//! of the *unpacked* payload tree ([`tamias_core::dir_digest`]):
10//!
11//! ```text
12//! tamias-package-v1
13//! name: <bare name>
14//! scope: <scope | ->
15//! version: <version | ->
16//! digest: sha256:<hex>
17//! ```
18//!
19//! The digest is the same value the registry index records and the
20//! store record carries, so a signature binds "this exact payload,
21//! under this name@version" — tampering with either side fails
22//! verification. See [`signing_bytes`].
23//!
24//! **Transport.** Signatures ride the registry index: `publish --sign`
25//! stores the armored PEM in the entry's `signature` field, which
26//! `index.json` carries to every `cf:`/registry install. Installed
27//! entries keep it in the store record so `tamias verify` can re-check.
28//!
29//! **Boundary (v1).** Only registry-published packages (`cf:` refs)
30//! can carry a signature — a signature can't live inside a `git:` or
31//! bare `https:` ref without a sidecar convention, so those verify by
32//! digest-pin alone. `signing.required` therefore effectively mandates
33//! registry-published packages. cosign/OCI signing is future work —
34//! it needs its own binary + identity model.
35//!
36//! **Secrets.** The private key is only ever a *path* — resolved from
37//! `--key`, `$TAMIAS_SIGNING_KEY` (whose value is a path),
38//! `signing.key_env`'s named env var, or `signing.key`. Key material
39//! never appears in config, args, or logs.

61/// Canonical bytes for signing — the provenance anchors, one
62/// `key: value` per line, `-` for absent optional fields. Fixed
63/// order, no trailing ambiguity; both sides build it from the same
64/// inputs so it can't drift.
65#[must_use]
66pub fn signing_bytes(
67    name: &str,
68    scope: Option<&str>,
69    version: Option<&str>,
70    digest: &str,
71) -> Vec<u8> {
72    format!(
73        "tamias-package-v1\nname: {name}\nscope: {}\nversion: {}\ndigest: {digest}\n",
74        scope.unwrap_or("-"),
75        version.unwrap_or("-")
76    )
77    .into_bytes()
78}

93/// Sign `bytes` with the SSH private key at `key` —
94/// `ssh-keygen -Y sign -n <namespace>`. Returns the armored signature
95/// (`-----BEGIN SSH SIGNATURE-----`).
96///
97/// # Errors
98/// Fails when `ssh-keygen` is absent or refuses the key/payload.
99pub fn sign(bytes: &[u8], key: &Path, namespace: &str) -> Result<String, Error> {
100    let data = scratch_file("tamias-sign")?;
101    std::fs::write(&data, bytes).map_err(Error::io(format!("write {}", data.display())))?;
102    let result = Command::new("ssh-keygen")
103        .args(["-Y", "sign", "-f"])
104        .arg(key)
105        .args(["-n", namespace])
106        .arg(&data)
107        .stdin(Stdio::null())
108        .stdout(Stdio::null())
109        .stderr(Stdio::null())
110        .status();
111    let sig_path = PathBuf::from(format!("{}.sig", data.display()));
112    let sig = std::fs::read_to_string(&sig_path).ok();
113    let _ = std::fs::remove_file(&data);
114    let _ = std::fs::remove_file(&sig_path);
115    match result {
116        Ok(status) if status.success() => {
117            sig.ok_or_else(|| Error::Failed("ssh-keygen -Y sign produced no signature".to_string()))
118        }
119        Ok(_) => Err(Error::Failed(format!(
120            "ssh-keygen -Y sign failed for key {}",
121            key.display()
122        ))),
123        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Err(Error::Failed(
124            "ssh-keygen not found — install openssh to use package signing".to_string(),
125        )),
126        Err(e) => Err(Error::io("run ssh-keygen -Y sign")(e)),
127    }
128}

130/// Verify `signature` (armored PEM) over `bytes` against the
131/// `allowed_signers` file — `ssh-keygen -Y verify -I <principal>`.
132/// `principal` selects the allowed_signers line; tamias uses the bare
133/// package name, so a key authorized for `ci-gate` can't vouch for
134/// `other-pkg`.
135///
136/// # Errors
137/// Fails when `ssh-keygen` is absent or can't run; `Ok(false)` means
138/// the signature didn't verify (wrong key, tampered payload, unknown
139/// principal).
140pub fn verify(
141    bytes: &[u8],
142    signature: &str,
143    allowed_signers: &Path,
144    principal: &str,
145    namespace: &str,
146) -> Result<bool, Error> {
147    let data = scratch_file("tamias-verify")?;
148    let sig = scratch_file("tamias-verify-sig")?;
149    std::fs::write(&data, bytes).map_err(Error::io(format!("write {}", data.display())))?;
150    std::fs::write(&sig, signature).map_err(Error::io(format!("write {}", sig.display())))?;
151    let input =
152        std::fs::File::open(&data).map_err(Error::io(format!("open {}", data.display())))?;
153    let status = Command::new("ssh-keygen")
154        .args(["-Y", "verify", "-f"])
155        .arg(allowed_signers)
156        .args(["-I", principal, "-n", namespace, "-s"])
157        .arg(&sig)
158        .stdin(Stdio::from(input))
159        .stdout(Stdio::null())
160        .stderr(Stdio::null())
161        .status();
162    let _ = std::fs::remove_file(&data);
163    let _ = std::fs::remove_file(&sig);
164    match status {
165        Ok(s) => Ok(s.success()),
166        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Err(Error::Failed(
167            "ssh-keygen not found — install openssh to verify package signatures".to_string(),
168        )),
169        Err(e) => Err(Error::io("run ssh-keygen -Y verify")(e)),
170    }
171}

173/// A unique scratch path in the temp dir — concurrent sign/verify
174/// calls in one process (tests, `tamias serve`) must not share files.
175fn scratch_file(tag: &str) -> Result<PathBuf, Error> {
176    static SEQ: AtomicU64 = AtomicU64::new(0);
177    let seq = SEQ.fetch_add(1, Ordering::Relaxed);
178    let nanos = std::time::SystemTime::now()
179        .duration_since(std::time::UNIX_EPOCH)
180        .map_or(0, |d| d.as_nanos());
181    Ok(std::env::temp_dir().join(format!("{tag}-{}-{nanos}-{seq}", std::process::id())))
182}

184/// `~` / `~/…` expansion for config-supplied paths (`$HOME`, else
185/// `/` — mirrors `tamias_source::expand_path` without the dep).
186#[must_use]
187pub fn expand_tilde(path: &str) -> PathBuf {
188    if path == "~" {
189        return home_dir();
190    }
191    if let Some(rest) = path.strip_prefix("~/") {
192        return home_dir().join(rest);
193    }
194    PathBuf::from(path)
195}

197fn home_dir() -> PathBuf {
198    std::env::var_os("HOME")
199        .or_else(|| std::env::var_os("USERPROFILE"))
200        .map_or_else(|| PathBuf::from("/"), PathBuf::from)
201}

203/// The resolved verification policy — what installs and `tamias
204/// verify` enforce. Built from config + env by [`SigningPolicy::resolve`].
205#[derive(Clone, Debug, PartialEq, Eq)]
206pub struct SigningPolicy {
207    /// `allowed_signers` path (`~` expanded) — `None` when nothing
208    /// configured and the default file doesn't exist.
209    pub allowed_signers: Option<PathBuf>,
210    /// Every install must carry a valid signature (only `cf:`/registry
211    /// refs can supply one in v1 — see the crate docs).
212    pub required: bool,
213    /// The `ssh-keygen -Y` namespace signatures verify under.
214    pub namespace: String,
215}
216
217impl SigningPolicy {
218    /// Resolve from `config` + env:
219    ///
220    /// - `allowed_signers`: `$TAMIAS_ALLOWED_SIGNERS` →
221    ///   `signing.allowed_signers` → `~/.config/tamias/allowed_signers`
222    ///   (kept only when the file exists — a configured-but-missing
223    ///   path still wins so the error names it).
224    /// - `required`/`namespace`: `signing.required`,
225    ///   `signing.namespace` or [`DEFAULT_NAMESPACE`].
226    #[must_use]
227    pub fn resolve(config: Option<&tamias_config::Config>) -> Self {
228        let signing = config.map(|c| &c.signing);
229        let allowed_signers = std::env::var(ALLOWED_SIGNERS_ENV)
230            .ok()
231            .filter(|v| !v.is_empty())
232            .map(|v| expand_tilde(&v))
233            .or_else(|| {
234                signing
235                    .and_then(|s| s.allowed_signers.as_deref())
236                    .map(expand_tilde)
237            })
238            .or_else(|| {
239                let p = home_dir().join(".config/tamias/allowed_signers");
240                p.is_file().then_some(p)
241            });
242        Self {
243            allowed_signers,
244            required: signing.is_some_and(|s| s.required),
245            namespace: signing
246                .and_then(|s| s.namespace.clone())
247                .filter(|n| !n.is_empty())
248                .unwrap_or_else(|| DEFAULT_NAMESPACE.to_string()),
249        }
250    }
251}
252
253impl Default for SigningPolicy {
254    /// The no-config policy — optional signing, default namespace.
255    fn default() -> Self {
256        Self::resolve(None)
257    }
258}

260/// The private-key path for `publish --sign`: explicit `--key` →
261/// `$TAMIAS_SIGNING_KEY` (value is a path) → the env var named by
262/// `signing.key_env` → `signing.key` → `~/.ssh/id_ed25519`.
263///
264/// # Errors
265/// Fails when no candidate exists — the message lists what was tried.
266pub fn resolve_signing_key(
267    config: Option<&tamias_config::Config>,
268    explicit: Option<&Path>,
269) -> Result<PathBuf, Error> {
270    if let Some(key) = explicit {
271        return Ok(key.to_path_buf());
272    }
273    let signing = config.map(|c| &c.signing);
274    let candidates: Vec<(&'static str, PathBuf)> = [
275        (SIGNING_KEY_ENV, std::env::var(SIGNING_KEY_ENV).ok()),
276        (
277            "signing.key_env",
278            signing
279                .and_then(|s| s.key_env.as_deref())
280                .and_then(|var| std::env::var(var).ok()),
281        ),
282        ("signing.key", signing.and_then(|s| s.key.clone())),
283    ]
284    .into_iter()
285    .filter_map(|(label, value)| {
286        value
287            .filter(|v| !v.is_empty())
288            .map(|v| (label, expand_tilde(&v)))
289    })
290    .collect();
291    for (_, path) in &candidates {
292        if path.is_file() {
293            return Ok(path.clone());
294        }
295    }
296    let fallback = home_dir().join(".ssh/id_ed25519");
297    if fallback.is_file() {
298        return Ok(fallback);
299    }
300    let tried = if candidates.is_empty() {
301        "$TAMIAS_SIGNING_KEY, signing.key_env, signing.key, ~/.ssh/id_ed25519".to_string()
302    } else {
303        candidates
304            .iter()
305            .map(|(label, path)| format!("{label} ({})", path.display()))
306            .collect::<Vec<_>>()
307            .join(", ")
308            + ", ~/.ssh/id_ed25519"
309    };
310    Err(Error::Failed(format!(
311        "no signing key found — tried {tried}; set {SIGNING_KEY_ENV} to the key path"
312    )))
313}

315/// What a signature check concluded — the status `tamias verify`
316/// reports and the install pipeline enforces.
317#[derive(Clone, Copy, Debug, PartialEq, Eq)]
318pub enum SignatureStatus {
319    /// Signature present and `ssh-keygen -Y verify` passed.
320    Verified,
321    /// Signature present but verification failed — wrong signer,
322    /// tampered payload, or unknown principal.
323    Failed,
324    /// No signature on the package.
325    Unsigned,
326    /// Signature present but no `allowed_signers` is configured —
327    /// can't be checked either way.
328    Unchecked,
329}
330
331impl SignatureStatus {
332    /// Stable wire/report spelling.
333    #[must_use]
334    pub fn as_str(self) -> &'static str {
335        match self {
336            Self::Verified => "verified",
337            Self::Failed => "bad",
338            Self::Unsigned => "unsigned",
339            Self::Unchecked => "unchecked",
340        }
341    }
342}

344/// Check a package's signature status against `allowed_signers`
345/// (`None` → [`SignatureStatus::Unchecked`] when a sig is present).
346/// The payload signed is the canonical [`signing_bytes`] of the
347/// package's provenance anchors; `principal` is the bare package name.
348///
349/// # Errors
350/// Propagates `ssh-keygen` spawn/IO failures from [`verify`].
351pub fn signature_status(
352    name: &str,
353    scope: Option<&str>,
354    version: Option<&str>,
355    digest: &str,
356    signature: Option<&str>,
357    allowed_signers: Option<&Path>,
358    namespace: &str,
359) -> Result<SignatureStatus, Error> {
360    let Some(sig) = signature else {
361        return Ok(SignatureStatus::Unsigned);
362    };
363    let Some(signers) = allowed_signers else {
364        return Ok(SignatureStatus::Unchecked);
365    };
366    if !signers.is_file() {
367        return Ok(SignatureStatus::Unchecked);
368    }
369    let bytes = signing_bytes(name, scope, version, digest);
370    let ok = verify(&bytes, sig, signers, name, namespace)?;
371    Ok(if ok {
372        SignatureStatus::Verified
373    } else {
374        SignatureStatus::Failed
375    })
376}

378/// The store-record convenience form of [`signature_status`].
379pub fn record_signature_status(
380    record: &StoreRecord,
381    allowed_signers: Option<&Path>,
382    namespace: &str,
383) -> Result<SignatureStatus, Error> {
384    signature_status(
385        &record.name,
386        record.scope.as_deref(),
387        record.version.as_deref(),
388        &record.digest,
389        record.signature.as_deref(),
390        allowed_signers,
391        namespace,
392    )
393}

395/// Sign a payload dir's canonical provenance — `dir_digest(root)` is
396/// the same digest the registry index records and the store record
397/// carries, so the signature binds exactly what install/verify check.
398///
399/// # Errors
400/// Propagates digest/sign failures.
401pub fn sign_package(
402    root: &Path,
403    manifest: &Manifest,
404    key: &Path,
405    namespace: &str,
406) -> Result<String, Error> {
407    let digest = dir_digest(root)?;
408    sign(
409        &signing_bytes(
410            &manifest.name,
411            manifest.scope.as_deref(),
412            manifest.version.as_deref(),
413            &digest,
414        ),
415        key,
416        namespace,
417    )
418}

420/// The install-pipeline gate — call after the payload manifest loads,
421/// before driver dispatch and the store commit. `signature` is what
422/// the source carried (registry index entries today); `payload_digest`
423/// is `dir_digest` of the payload root.
424///
425/// # Errors
426/// Fails on a bad signature, a signed package with no usable
427/// `allowed_signers`, or — under `policy.required` — any unsigned
428/// install.
429pub fn enforce_signature(
430    name: &str,
431    scope: Option<&str>,
432    version: Option<&str>,
433    payload_digest: &str,
434    signature: Option<&str>,
435    policy: &SigningPolicy,
436) -> Result<(), Error> {
437    match signature_status(
438        name,
439        scope,
440        version,
441        payload_digest,
442        signature,
443        policy.allowed_signers.as_deref(),
444        &policy.namespace,
445    )? {
446        SignatureStatus::Verified => Ok(()),
447        SignatureStatus::Unsigned if !policy.required => Ok(()),
448        SignatureStatus::Unsigned => Err(Error::Failed(format!(
449            "package '{name}' is unsigned and signing.required is set — \
450             only signed registry (`cf:`) packages can be installed"
451        ))),
452        SignatureStatus::Failed => Err(Error::Failed(format!(
453            "package '{name}': signature verification failed \
454             (allowed_signers: {})",
455            policy
456                .allowed_signers
457                .as_ref()
458                .map_or_else(|| "?".to_string(), |p| p.display().to_string())
459        ))),
460        SignatureStatus::Unchecked => Err(Error::Failed(format!(
461            "package '{name}' is signed but no allowed_signers is configured — \
462             set signing.allowed_signers, {ALLOWED_SIGNERS_ENV}, or \
463             ~/.config/tamias/allowed_signers"
464        ))),
465    }
466}

628        // signed + no allowed_signers → err
629        policy.allowed_signers = None;
630        assert!(enforce_signature("pkg", None, None, "sha256:d", Some(&sig), &policy).is_err());
631        // bad sig → err
632        policy.allowed_signers = Some(signers);
633        let bad = signing_bytes("pkg", None, None, "sha256:other");
634        let badsig = sign(&bad, &key, DEFAULT_NAMESPACE).unwrap();
635        assert!(enforce_signature("pkg", None, None, "sha256:d", Some(&badsig), &policy).is_err());
636    }
