1//! cuora-identity — the identity port: *where the age identity
2//! comes from*.
3//!
4//! The age identity is the master key every vault impl decrypts
5//! with (ADR-0001: all stores hold opaque age ciphertext). How that
6//! key material is held is a port — same convention as
7//! `cuora-vault`/`cuora-config`/`cuora-deploy`:
8//!
9//! - `cuora-identity-file` — the `~/.config/cuora/identity.txt`
10//!   keyfile (mode `0600`); the headless/CI path, overridable via
11//!   `CUORA_IDENTITY_FILE` or `[vaults.*].identity`.
12//! - `cuora-identity-keyring` — the platform session secret store
13//!   (secret-service / Keychain / Credential Manager) via the
14//!   keyring-core `CredentialStore` model.
15//!
16//! Resolution order (in `cuora-ops`): an explicit file override
17//! wins; else the session keyring is preferred; else the default
18//! keyfile. A later arm of this port — a passphrase-wrapped
19//! identity file for portable identities — is a new impl, not a
20//! new port.
21
22use age::x25519;

24/// A source of (and sink for) the agent's age identity.
25///
26/// Implementations hold raw `AGE-SECRET-KEY-…` material however
27/// their backend prefers; the port speaks parsed
28/// [`x25519::Identity`]s so vault impls never parse text. All
29/// methods are sync — no async runtime in the binary.
30pub trait IdentitySource: Send + Sync {
31    /// Load the stored identity. `Ok(None)` means *this source
32    /// holds nothing* — distinct from an error, so callers can
33    /// fall through to the next source. Unreadable or corrupt
34    /// material is `Err`, never silently `None` (that would risk
35    /// clobbering a good identity on `store`).
36    fn load(&self) -> Result<Option<x25519::Identity>, IdentityError>;
37
38    /// Persist `identity`, overwriting any existing entry. Callers
39    /// wanting no-clobber semantics check [`load`](Self::load)
40    /// first (see [`ensure`]).
41    fn store(&self, identity: &x25519::Identity) -> Result<(), IdentityError>;
42
43    /// Short backend name for messages — `"file"`, `"keyring"`.
44    fn kind(&self) -> &'static str;
45
46    /// Human-facing locator — a keyfile path or a store entry name.
47    /// Never contains key material.
48    fn describe(&self) -> String;
49}
50
51/// Load-or-generate against one source: `Ok((identity, created))`.
52/// The `init`/bootstrap path — generates a fresh identity iff the
53/// source holds none, then stores it. A corrupt entry is an error,
54/// not an overwrite.
55pub fn ensure(source: &dyn IdentitySource) -> Result<(x25519::Identity, bool), IdentityError> {
56    match source.load()? {
57        Some(id) => Ok((id, false)),
58        None => {
59            let id = x25519::Identity::generate();
60            source.store(&id)?;
61            Ok((id, true))
62        }
63    }
64}
65
66/// Identity port errors. Impls map backend failures onto these so
67/// callers can match without knowing the store.
68#[derive(Debug, thiserror::Error)]
69pub enum IdentityError {
70    /// Filesystem I/O failure.
71    #[error("identity i/o: {0}")]
72    Io(String),
73    /// Stored material isn't a usable age identity.
74    #[error("identity parse: {0}")]
75    Parse(String),
76    /// The backend exists but the read/write failed (daemon down,
77    /// access denied, session locked).
78    #[error("identity store: {0}")]
79    Store(String),
80    /// The backend itself doesn't exist on this host/session —
81    /// e.g. no reachable session keyring on a headless box.
82    #[error("identity backend unavailable: {0}")]
83    Unavailable(String),
84}
85
86impl From<std::io::Error> for IdentityError {
87    fn from(e: std::io::Error) -> Self {
88        IdentityError::Io(e.to_string())
89    }
90}

92#[cfg(test)]
93#[allow(clippy::unwrap_used)]
94mod tests {
95    use super::*;
96    use std::sync::Mutex;
97
98    /// An in-memory `IdentitySource` exercising the port contract.
99    struct Mem {
100        cell: Mutex<Option<x25519::Identity>>,
101        fail: Mutex<Option<IdentityError>>,
102    }
103
104    impl Mem {
105        fn new() -> Self {
106            Mem {
107                cell: Mutex::new(None),
108                fail: Mutex::new(None),
109            }
110        }
111
112        fn failing(e: IdentityError) -> Self {
113            Mem {
114                cell: Mutex::new(None),
115                fail: Mutex::new(Some(e)),
116            }
117        }
118    }
119
120    impl IdentitySource for Mem {
121        fn load(&self) -> Result<Option<x25519::Identity>, IdentityError> {
122            let cell = self.cell.lock().unwrap().clone();
123            Ok(cell)
124        }
125
126        fn store(&self, identity: &x25519::Identity) -> Result<(), IdentityError> {
127            if let Some(e) = self.fail.lock().unwrap().take() {
128                return Err(e);
129            }
130            *self.cell.lock().unwrap() = Some(identity.clone());
131            Ok(())
132        }
133
134        fn kind(&self) -> &'static str {
135            "mem"
136        }
137
138        fn describe(&self) -> String {
139            "in-memory test source".into()
140        }
141    }
142
143    #[test]
144    fn ensure_generates_once_then_loads() {
145        let mem = Mem::new();
146        let (a, created) = ensure(&mem).unwrap();
147        assert!(created);
148        let (b, created) = ensure(&mem).unwrap();
149        assert!(!created);
150        assert_eq!(a.to_public(), b.to_public());
151    }
152
153    #[test]
154    fn ensure_propagates_store_failures() {
155        let mem = Mem::failing(IdentityError::Store("backend dead".into()));
156        // `.err()` is infallible here — a failing store must not
157        // yield an identity, so no dead panic arm.
158        let e = ensure(&mem).err().unwrap();
159        assert!(e.to_string().contains("backend dead"), "{e}");
160    }
161
162    #[test]
163    fn mem_names_itself_like_any_source() {
164        let mem = Mem::new();
165        assert_eq!(mem.kind(), "mem");
166        assert_eq!(mem.describe(), "in-memory test source");
167    }
168
169    #[test]
170    fn io_error_maps_to_the_io_variant() {
171        let e = IdentityError::from(std::io::Error::new(std::io::ErrorKind::NotFound, "gone"));
172        assert!(
173            matches!(e, IdentityError::Io(ref s) if s.contains("gone")),
174            "{e}"
175        );
176    }
