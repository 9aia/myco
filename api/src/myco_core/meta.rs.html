1//! `meta.toml` model + the pure id-allocation logic.
2//!
3//! Counters are **persistent** — done tasks leave `TODO.md` and ids
4//! must never be reused, so `TODO.md` alone can't tell you the next
5//! `T-NNN`. [`Registry`] computes next ids from counters plus a set
6//! of observed ids (reconcile = counter ≥ max-seen + 1), which is
7//! what the store uses to recover from hand-edits.
8
9use crate::{ItemId, ItemKind};
10
11const fn one() -> u32 {
12    1
13}
14
15/// A `[next]` counter — ids are 1-based, so `0` mints a `*-000`
16/// id no file can be named for. A hand-edited `t = 0` is a parse
17/// error, not a counter reset.
18fn nonzero<'de, D>(d: D) -> Result<u32, D::Error>
19where
20    D: serde::Deserializer<'de>,
21{
22    let n = <u32 as serde::Deserialize>::deserialize(d)?;
23    if n == 0 {
24        return Err(serde::de::Error::custom(
25            "counter must be ≥ 1 — 0 mints an unrepresentable `*-000` id",
26        ));
27    }
28    Ok(n)
29}

31/// Per-kind next-counters — `meta.toml`'s `[next]` table.
32/// Missing keys default to 1 so a partial `[next]` survives a
33/// hand-edit; *unknown* keys are rejected — a typo (`tx = 9`)
34/// must surface, not silently reset the counter to 1 and reissue
35/// ids.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
38#[derive(serde::Serialize, serde::Deserialize)]
39#[serde(deny_unknown_fields)]
40pub struct NextCounters {
41    /// Next `T-NNN`.
42    #[serde(default = "one", deserialize_with = "nonzero")]
43    pub t: u32,
44    /// Next `I-NNN`.
45    #[serde(default = "one", deserialize_with = "nonzero")]
46    pub i: u32,
47    /// Next `P-NNN`.
48    #[serde(default = "one", deserialize_with = "nonzero")]
49    pub p: u32,
50    /// Next `ADR-NNNN`.
51    #[serde(default = "one", deserialize_with = "nonzero")]
52    pub adr: u32,
53    /// Next `PL-NNN`.
54    #[serde(default = "one", deserialize_with = "nonzero")]
55    pub pl: u32,
56}
57
58impl NextCounters {
59    /// Counter for `kind`.
60    pub fn get(&self, kind: ItemKind) -> u32 {
61        match kind {
62            ItemKind::Task => self.t,
63            ItemKind::Issue => self.i,
64            ItemKind::Proposal => self.p,
65            ItemKind::Adr => self.adr,
66            ItemKind::Plan => self.pl,
67        }
68    }
69
70    /// Set counter for `kind`.
71    pub fn set(&mut self, kind: ItemKind, n: u32) {
72        match kind {
73            ItemKind::Task => self.t = n,
74            ItemKind::Issue => self.i = n,
75            ItemKind::Proposal => self.p = n,
76            ItemKind::Adr => self.adr = n,
77            ItemKind::Plan => self.pl = n,
78        }
79    }
80}
81
82impl Default for NextCounters {
83    /// Counters start at 1 — `T-000` is not a valid id.
84    fn default() -> Self {
85        NextCounters {
86            t: 1,
87            i: 1,
88            p: 1,
89            adr: 1,
90            pl: 1,
91        }
92    }
93}

95/// `meta.toml` — the one piece of project state that isn't
96/// derivable from the item files. Unknown keys are rejected:
97/// a typo'd table (`[netx]`) would otherwise silently drop the
98/// real counters and restart id allocation at 1 — done ids get
99/// reissued, colliding with their handoffs and history.
100#[derive(Debug, Clone, PartialEq, Eq)]
101#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
102#[derive(serde::Serialize, serde::Deserialize)]
103#[serde(deny_unknown_fields)]
104pub struct ProjectMeta {
105    /// Repo/project name — used in `# TODO — <name>` and views.
106    pub name: String,
107    /// Next-id counters per kind.
108    #[serde(default)]
109    pub next: NextCounters,
110}
111
112impl ProjectMeta {
113    /// Fresh meta for `myco init`.
114    pub fn new(name: impl Into<String>) -> Self {
115        ProjectMeta {
116            name: name.into(),
117            next: NextCounters::default(),
118        }
119    }
120
121    /// A [`Registry`] seeded from these counters.
122    pub fn registry(&self) -> Registry {
123        Registry {
124            counters: self.next,
125        }
126    }
127}
128
129/// Id allocator — pure. Seed from [`ProjectMeta::registry`],
130/// [`observe_all`][Registry::observe_all] every known id, then
131/// [`alloc`][Registry::alloc]; take the bumped counters back out via
132/// [`counters`][Registry::counters] to persist.
133#[derive(Debug, Clone, Default)]
134pub struct Registry {
135    counters: NextCounters,
136}
137
138impl Registry {
139    /// The id `alloc` would return for `kind` without bumping.
140    /// A zeroed counter floors at 1 — `T-000` is not a valid id.
141    pub fn peek(&self, kind: ItemKind) -> ItemId {
142        ItemId::new(kind, self.counters.get(kind).max(1))
143    }
144
145    /// Record an existing id — bumps its kind's counter past it.
146    /// Never lowers a counter.
147    pub fn observe(&mut self, id: ItemId) {
148        let floor = id.number().saturating_add(1);
149        let cur = self.counters.get(id.kind());
150        if floor > cur {
151            self.counters.set(id.kind(), floor);
152        }
153    }
154
155    /// Record many existing ids — the reconcile pass after a scan.
156    pub fn observe_all(&mut self, ids: impl IntoIterator<Item = ItemId>) {
157        for id in ids {
158            self.observe(id);
159        }
160    }
161
162    /// Take the next id for `kind` and bump the counter.
163    pub fn alloc(&mut self, kind: ItemKind) -> ItemId {
164        let id = self.peek(kind);
165        self.counters.set(kind, id.number().saturating_add(1));
166        id
167    }
168
169    /// The counters to write back to `meta.toml`.
170    pub fn counters(&self) -> NextCounters {
171        self.counters
172    }
173}

175#[cfg(test)]
176#[allow(clippy::unwrap_used)]
177mod tests {
178    use super::*;
179
180    #[test]
181    fn counters_start_at_one() {
182        let n = NextCounters::default();
183        for k in ItemKind::ALL {
184            assert_eq!(n.get(k), 1);
185        }
186        let meta = ProjectMeta::new("myco");
187        assert_eq!(meta.name, "myco");
188        assert_eq!(meta.next, n);
189    }
190
191    #[test]
192    fn alloc_bumps_per_kind() {
193        let mut r = Registry::default();
194        assert_eq!(r.alloc(ItemKind::Task), ItemId::new(ItemKind::Task, 1));
195        assert_eq!(r.alloc(ItemKind::Task), ItemId::new(ItemKind::Task, 2));
196        assert_eq!(r.alloc(ItemKind::Issue), ItemId::new(ItemKind::Issue, 1));
197        assert_eq!(r.peek(ItemKind::Task), ItemId::new(ItemKind::Task, 3));
198        assert_eq!(r.peek(ItemKind::Adr), ItemId::new(ItemKind::Adr, 1));
199    }
200
201    #[test]
202    fn observe_raises_floor_never_lowers() {
203        let mut r = Registry::default();
204        r.observe(ItemId::new(ItemKind::Task, 40));
205        assert_eq!(r.peek(ItemKind::Task), ItemId::new(ItemKind::Task, 41));
206        // observing a lower id doesn't lower the counter
207        r.observe(ItemId::new(ItemKind::Task, 5));
208        assert_eq!(r.peek(ItemKind::Task), ItemId::new(ItemKind::Task, 41));
209        // other kinds untouched
210        assert_eq!(r.peek(ItemKind::Plan), ItemId::new(ItemKind::Plan, 1));
211        // counters written back
212        let c = r.counters();
213        assert_eq!(c.t, 41);
214        assert_eq!(c.pl, 1);
215    }
216
217    #[test]
218    fn observe_all_reconciles() {
219        let mut meta = ProjectMeta::new("x");
220        meta.next.t = 2; // stale: file actually contains T-012
221        let mut r = meta.registry();
222        r.observe_all([
223            ItemId::new(ItemKind::Task, 12),
224            ItemId::new(ItemKind::Task, 3),
225        ]);
226        assert_eq!(r.alloc(ItemKind::Task), ItemId::new(ItemKind::Task, 13));
227    }
228
229    #[test]
230    fn counter_get_set_each_kind() {
231        let mut n = NextCounters::default();
232        for (i, k) in ItemKind::ALL.into_iter().enumerate() {
233            n.set(k, (i as u32) + 10);
234        }
235        for (i, k) in ItemKind::ALL.into_iter().enumerate() {
236            assert_eq!(n.get(k), (i as u32) + 10);
237        }
238    }
239
240    #[test]
241    fn unknown_keys_are_rejected() {
242        // a typo'd key must surface — silently defaulting a
243        // misspelled counter resets ids to 1 and reissues done ids
244        assert!(serde_json::from_value::<NextCounters>(serde_json::json!({"tx": 9})).is_err());
245        assert!(serde_json::from_value::<ProjectMeta>(serde_json::json!({
246            "name": "x",
247            "netx": {"t": 9}
248        }))
249        .is_err());
250        assert!(serde_json::from_value::<ProjectMeta>(serde_json::json!({
251            "name": "x",
252            "next": {"t": 9, "tx": 1}
253        }))
254        .is_err());
255    }
256
257    #[test]
258    fn zero_counter_is_rejected_and_floored() {
259        // `0` mints `T-000` — an id whose file name can't
260        // round-trip; reject it at the boundary…
261        assert!(serde_json::from_value::<NextCounters>(serde_json::json!({"t": 0})).is_err());
262        // …and floor it if a zeroed counter arrives anyway
263        let mut meta = ProjectMeta::new("x");
264        meta.next.t = 0;
265        let mut r = meta.registry();
266        assert_eq!(r.alloc(ItemKind::Task), ItemId::new(ItemKind::Task, 1));
267        assert_eq!(r.peek(ItemKind::Task), ItemId::new(ItemKind::Task, 2));
268    }
