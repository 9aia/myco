8/// A work item's metadata.
9///
10/// `created` is `Option<Date>` — the spec grammar
11/// (`- [ ] T-012 title · pri:high · #tag`) has no date slot, so a
12/// missing `created` must stay representable; `myco check` can flag
13/// it later. Emit always writes `created:` when present.
14///
15/// `note` is reserved for handoffs (`myco handoff --note`); kept in
16/// the model now so phase 6 doesn't re-shape it.
17#[derive(Debug, Clone, PartialEq, Eq)]
18#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
19#[derive(serde::Serialize, serde::Deserialize)]
20pub struct Item {
21    /// Typed id — determines kind, sort, filename.
22    pub id: ItemId,
23    /// One-line summary. No `·`, no newlines in line-grammar
24    /// contexts (enforced at emit, not here — `Item` is a value,
25    /// not a validator).
26    pub title: String,
27    /// Lifecycle state — transitions are validated by the caller
28    /// via [`Status::transition_to`].
29    pub status: Status,
30    /// `pri:` — defaults to [`Priority::Medium`].
31    #[serde(default)]
32    pub priority: Priority,
33    /// `owner:` — `devin`, `human:luis`, …
34    #[serde(default, skip_serializing_if = "Option::is_none")]
35    #[cfg_attr(feature = "schema", schemars(with = "Option<String>"))]
36    pub owner: Option<Actor>,
37    /// `#label`s — a sorted set, duplicates impossible.
38    #[serde(default, skip_serializing_if = "BTreeSet::is_empty")]
39    pub labels: BTreeSet<String>,
40    /// `blocks:` — ids this item blocks. Sorted set.
41    #[serde(default, skip_serializing_if = "BTreeSet::is_empty")]
42    pub blocks: BTreeSet<ItemId>,
43    /// `links:` — ids this item is associated with (its plan, the
44    /// proposal that spawned it, …). Kind-agnostic and symmetric —
45    /// a declared edge, not a prose mention. Sorted set.
46    #[serde(default, skip_serializing_if = "BTreeSet::is_empty")]
47    pub links: BTreeSet<ItemId>,
48    /// `created:` — `YYYY-MM-DD` when known. Absent (not `null`)
49    /// in JSON — every other `Option` field skips the same way.
50    #[serde(
51        default,
52        with = "crate::date::opt",
53        skip_serializing_if = "Option::is_none"
54    )]
55    #[cfg_attr(feature = "schema", schemars(with = "Option<String>"))]
56    pub created: Option<time::Date>,
57    /// Free-form note — handoff/resume payload (phase 6).
58    #[serde(default, skip_serializing_if = "Option::is_none")]
59    pub note: Option<String>,
60    /// `superseded_by:` — the item that replaced this one (ADRs).
61    #[serde(default, skip_serializing_if = "Option::is_none")]
62    pub superseded_by: Option<ItemId>,
63    /// `supersedes:` — the item this one replaces (ADRs).
64    #[serde(default, skip_serializing_if = "Option::is_none")]
65    pub supersedes: Option<ItemId>,
66    /// `epic:` — the epic (named delivery grouping) this item
67    /// belongs to. Line token `epic:<name>`; a group name, not a
68    /// label — one per item.
69    #[serde(default, skip_serializing_if = "Option::is_none")]
70    pub epic: Option<String>,
71    /// `milestone:` — the milestone this item targets. Line token
72    /// `ms:<name>` (kept short — lines stay readable).
73    #[serde(default, skip_serializing_if = "Option::is_none")]
74    pub milestone: Option<String>,
75    /// Nested sub-items — the `  - [ ]` lines indented under this
76    /// one in TODO.md/BACKLOG.md. Children are full items: same
77    /// metadata, same state machine. Depth is arbitrary (one
78    /// level is the recommended shape); `done`/`defer` act on the
79    /// whole subtree. Empty for file-backed kinds — children only
80    /// exist in the line grammar.
81    #[serde(default, skip_serializing_if = "Vec::is_empty")]
82    pub children: Vec<Item>,
83}

90/// Pre-order DFS over an item subtree — a stack of pending nodes,
91/// children pushed reversed so they pop in file order.
92struct Subtree<'a> {
93    stack: Vec<&'a Item>,
94}
95
96impl<'a> Iterator for Subtree<'a> {
97    type Item = &'a Item;
98
99    fn next(&mut self) -> Option<Self::Item> {
100        let item = self.stack.pop()?;
101        for c in item.children.iter().rev() {
102            self.stack.push(c);
103        }
104        Some(item)
105    }
106}

108impl Item {
109    /// Minimal item: id + title, open, medium, nothing else.
110    pub fn new(id: ItemId, title: impl Into<String>) -> Self {
111        Item {
112            id,
113            title: title.into(),
114            status: Status::Open,

129
130    /// `self` plus every descendant, pre-order — the subtree
131    /// flattened (each parent immediately before its children).
132    /// Id lookups, dup checks, counts, and index scans all see
133    /// nested items through this.
134    pub fn subtree(&self) -> impl Iterator<Item = &Item> {
135        Subtree { stack: vec![self] }
136    }
137
138    /// Every descendant (children, grandchildren, …), pre-order.
139    pub fn descendants(&self) -> impl Iterator<Item = &Item> {
140        self.subtree().skip(1)
141    }

223    /// `T-001 open pri:high owner:devin #core title` — the compact
224    /// human list line.
225    pub fn compact(&self) -> String {
226        let mut s = format!("{} {} pri:{}", self.id, self.status, self.priority.token());
227        if let Some(o) = &self.owner {
228            s += &format!(" owner:{o}");
229        }
230        for l in &self.labels {
231            s += &format!(" #{l}");
232        }
233        if let Some(e) = &self.epic {
234            s += &format!(" epic:{e}");
235        }
236        if let Some(m) = &self.milestone {
237            s += &format!(" ms:{m}");
238        }
239        if !self.title.is_empty() {
240            s += &format!(" {}", self.title);
241        }
242        s
243    }

246/// Optional metadata applied to a freshly-created item — the parsed
247/// `--pri/--owner/--label/--blocks/--epic/--milestone` bundle every
248/// `add`/`new` front-end shares. Validated on `apply`.
249#[derive(Debug, Clone, Default, PartialEq, Eq)]
250#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
251#[derive(serde::Serialize, serde::Deserialize)]
252pub struct NewItemMeta {
253    /// `pri:` override — absent keeps the item default.
254    pub priority: Option<Priority>,
255    /// `owner:` actor.
256    pub owner: Option<Actor>,
257    /// `#label`s to add.
258    pub labels: Vec<String>,
259    /// `blocks:` ids.
260    pub blocks: Vec<ItemId>,
261    /// `links:` ids.
262    pub links: Vec<ItemId>,
263    /// `epic:` group name.
264    pub epic: Option<String>,
265    /// `ms:` milestone name.
266    pub milestone: Option<String>,
267}
268
269impl NewItemMeta {
270    /// Apply the metadata to `item` — labels and group names are
271    /// validated (invalid tokens can't be emitted by the file
272    /// grammar). All-or-nothing: everything validates before any
273    /// field moves, so a bad `--epic` can't leave labels applied.
274    pub fn apply(&self, item: &mut Item) -> Result<(), String> {
275        for l in &self.labels {
276            if !is_valid_label(l) {
277                return Err(format!("invalid label {l:?}"));
278            }
279        }
280        for (what, name) in [("epic", &self.epic), ("milestone", &self.milestone)] {
281            if let Some(name) = name {
282                if !is_valid_group_name(name) {
283                    return Err(format!("invalid {what} name {name:?}"));
284                }
285            }
286        }
287        if let Some(p) = self.priority {
288            item.priority = p;
289        }
290        if let Some(o) = &self.owner {
291            item.owner = Some(o.clone());
292        }
293        for l in &self.labels {
294            item.labels.insert(l.clone());
295        }
296        for b in &self.blocks {
297            item.blocks.insert(*b);
298        }
299        for l in &self.links {
300            item.links.insert(*l);
301        }
302        item.epic = self.epic.clone();
303        item.milestone = self.milestone.clone();
304        Ok(())
305    }
306}

308/// Depth-tagged pre-order view — list UIs indent children under
309/// their parent by it. Iterative (explicit stack, children pushed
310/// reversed to pop in file order): trees can nest past any sane
311/// recursion depth.
312pub fn tree_view(items: &[Item]) -> Vec<(usize, &Item)> {
313    let mut out = Vec::new();
314    let mut stack: Vec<(usize, &Item)> = items.iter().rev().map(|i| (0, i)).collect();
315    while let Some((depth, item)) = stack.pop() {
316        out.push((depth, item));
317        for c in item.children.iter().rev() {
318            stack.push((depth + 1, c));
319        }
320    }
321    out
322}

324/// A `#label` is `[A-Za-z0-9_-]+` — kept permissive on the model,
325/// enforced at the file-grammar boundary.
326pub fn is_valid_label(s: &str) -> bool {
327    !s.is_empty()
328        && s.bytes()
329            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'_' | b'-'))
330}
331
332/// An `epic:`/`ms:` group name — `[A-Za-z0-9_.-]+`: the label
333/// charset plus `.` for `1.0`-style milestones. Same boundary rule
334/// as labels (the name must fit a `·`-separated line token).
335pub fn is_valid_group_name(s: &str) -> bool {
336    !s.is_empty()
337        && s.bytes()
338            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'_' | b'-' | b'.'))
339}
