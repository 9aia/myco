8/// A work item's metadata.
9///
10/// `created` is `Option<Date>` — the spec grammar
11/// (`- [ ] T-012 title · pri:high · #tag`) has no date slot, so a
12/// missing `created` must stay representable; `myco check` can flag
13/// it later. Emit always writes `created:` when present.
14///
15/// `note` is reserved for handoffs (`myco handoff --note`); kept in
16/// the model now so phase 6 doesn't re-shape it.
17#[derive(Debug, Clone, PartialEq, Eq)]
18#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
19#[derive(serde::Serialize, serde::Deserialize)]
20pub struct Item {
21    /// Typed id — determines kind, sort, filename.
22    pub id: ItemId,
23    /// One-line summary. No `·`, no newlines in line-grammar
24    /// contexts (enforced at emit, not here — `Item` is a value,
25    /// not a validator).
26    pub title: String,
27    /// Lifecycle state — transitions are validated by the caller
28    /// via [`Status::transition_to`].
29    pub status: Status,
30    /// `pri:` — defaults to [`Priority::Medium`].
31    #[serde(default)]
32    pub priority: Priority,
33    /// `owner:` — `devin`, `human:luis`, …
34    #[serde(default, skip_serializing_if = "Option::is_none")]
35    #[cfg_attr(feature = "schema", schemars(with = "Option<String>"))]
36    pub owner: Option<Actor>,
37    /// `#label`s — a sorted set, duplicates impossible.
38    #[serde(default, skip_serializing_if = "BTreeSet::is_empty")]
39    pub labels: BTreeSet<String>,
40    /// `blocks:` — ids this item blocks. Sorted set.
41    #[serde(default, skip_serializing_if = "BTreeSet::is_empty")]
42    pub blocks: BTreeSet<ItemId>,
43    /// `created:` — `YYYY-MM-DD` when known.
44    #[serde(default, with = "crate::date::opt")]
45    #[cfg_attr(feature = "schema", schemars(with = "Option<String>"))]
46    pub created: Option<time::Date>,
47    /// Free-form note — handoff/resume payload (phase 6).
48    #[serde(default, skip_serializing_if = "Option::is_none")]
49    pub note: Option<String>,
50    /// `superseded_by:` — the item that replaced this one (ADRs).
51    #[serde(default, skip_serializing_if = "Option::is_none")]
52    pub superseded_by: Option<ItemId>,
53    /// `supersedes:` — the item this one replaces (ADRs).
54    #[serde(default, skip_serializing_if = "Option::is_none")]
55    pub supersedes: Option<ItemId>,
56    /// `epic:` — the epic (named delivery grouping) this item
57    /// belongs to. Line token `epic:<name>`; a group name, not a
58    /// label — one per item.
59    #[serde(default, skip_serializing_if = "Option::is_none")]
60    pub epic: Option<String>,
61    /// `milestone:` — the milestone this item targets. Line token
62    /// `ms:<name>` (kept short — lines stay readable).
63    #[serde(default, skip_serializing_if = "Option::is_none")]
64    pub milestone: Option<String>,
65    /// Nested sub-items — the `  - [ ]` lines indented under this
66    /// one in TODO.md/BACKLOG.md. Children are full items: same
67    /// metadata, same state machine. Depth is arbitrary (one
68    /// level is the recommended shape); `done`/`defer` act on the
69    /// whole subtree. Empty for file-backed kinds — children only
70    /// exist in the line grammar.
71    #[serde(default, skip_serializing_if = "Vec::is_empty")]
72    pub children: Vec<Item>,
73}

80/// Pre-order DFS over an item subtree — a stack of pending nodes,
81/// children pushed reversed so they pop in file order.
82struct Subtree<'a> {
83    stack: Vec<&'a Item>,
84}
85
86impl<'a> Iterator for Subtree<'a> {
87    type Item = &'a Item;
88
89    fn next(&mut self) -> Option<Self::Item> {
90        let item = self.stack.pop()?;
91        for c in item.children.iter().rev() {
92            self.stack.push(c);
93        }
94        Some(item)
95    }
96}

98impl Item {
99    /// Minimal item: id + title, open, medium, nothing else.
100    pub fn new(id: ItemId, title: impl Into<String>) -> Self {
101        Item {
102            id,
103            title: title.into(),
104            status: Status::Open,
105            priority: Priority::Medium,
106            owner: None,
107            labels: BTreeSet::new(),
108            blocks: BTreeSet::new(),
109            created: None,
110            note: None,
111            superseded_by: None,
112            supersedes: None,
113            epic: None,
114            milestone: None,
115            children: Vec::new(),
116        }
117    }
118
119    /// `self` plus every descendant, pre-order — the subtree
120    /// flattened (each parent immediately before its children).
121    /// Id lookups, dup checks, counts, and index scans all see
122    /// nested items through this.
123    pub fn subtree(&self) -> impl Iterator<Item = &Item> {
124        Subtree { stack: vec![self] }
125    }
126
127    /// Every descendant (children, grandchildren, …), pre-order.
128    pub fn descendants(&self) -> impl Iterator<Item = &Item> {
129        self.subtree().skip(1)
130    }
131
132    /// Builder-style `status` setter (no transition validation —
133    /// use [`Status::transition_to`] when advancing a real item).
134    pub fn status(mut self, status: Status) -> Self {
135        self.status = status;
136        self
137    }

139    /// Builder-style `priority` setter.

207/// A `#label` is `[A-Za-z0-9_-]+` — kept permissive on the model,
208/// enforced at the file-grammar boundary.
209pub fn is_valid_label(s: &str) -> bool {
210    !s.is_empty()
211        && s.bytes()
212            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'_' | b'-'))
213}
214
215/// An `epic:`/`ms:` group name — `[A-Za-z0-9_.-]+`: the label
216/// charset plus `.` for `1.0`-style milestones. Same boundary rule
217/// as labels (the name must fit a `·`-separated line token).
218pub fn is_valid_group_name(s: &str) -> bool {
219    !s.is_empty()
220        && s.bytes()
221            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'_' | b'-' | b'.'))
222}

230    #[test]
231    fn builder() {
232        let id = ItemId::new(ItemKind::Task, 12);
233        let d = crate::date::parse("2026-02-09").unwrap();
234        let item = Item::new(id, "write phase 2")
235            .status(Status::Claimed)
236            .priority(Priority::High)
237            .owner("devin".parse().unwrap())
238            .created(d)
239            .note("session 7")
240            .label("core")
241            .label("store")
242            .blocks(ItemId::new(ItemKind::Task, 10));
243        assert_eq!(item.id, id);
244        assert_eq!(item.title, "write phase 2");
245        assert_eq!(item.status, Status::Claimed);
246        assert_eq!(item.priority, Priority::High);
247        assert_eq!(item.owner.as_ref().map(Actor::as_str), Some("devin"));
248        assert_eq!(item.created, Some(d));
249        assert_eq!(item.note.as_deref(), Some("session 7"));
250        assert!(item.labels.contains("core") && item.labels.contains("store"));
251        assert!(item.blocks.contains(&ItemId::new(ItemKind::Task, 10)));
252    }

320    #[test]
321    fn subtree_walks_preorder() {
322        let t = |n: u32| Item::new(ItemId::new(ItemKind::Task, n), format!("t{n}"));
323        let root = t(1).child(t(2).child(t(4))).child(t(3));
324        let ids: Vec<u32> = root.subtree().map(|i| i.id.number()).collect();
325        assert_eq!(ids, [1, 2, 4, 3]);
326        let deep: Vec<u32> = root.descendants().map(|i| i.id.number()).collect();
327        assert_eq!(deep, [2, 4, 3]);
328        // a leaf subtree is just itself
329        assert_eq!(t(9).subtree().count(), 1);
330        assert_eq!(t(9).descendants().count(), 0);
331    }
