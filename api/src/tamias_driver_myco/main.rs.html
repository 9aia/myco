1//! tamias-driver-myco — the pavo view driver for a repo's
2//! `project/` ledger.
3//!
4//! Verbs (the tamias driver contract):
5//!
6//! - `manifest` — print the [`DriverManifest`] JSON (discovery
7//!   probe; never touches project state).
8//! - `capabilities` — `{"capabilities": ["pavo"], "verbs": […]}`.
9//! - `serve` — ndjson JSON-RPC on stdio: `{"op":"manifest"}`,
10//!   `{"op":"views"}`, `{"op":"view","view":…,"params":{…}}`,
11//!   `{"op":"query","query":…,"params":{…}}` →
12//!   `{"ok":true,"result":…}` / `{"ok":false,"error":…}`.
13//!
14//! Which repo: each `view`/`query` resolves its own target —
15//! `params.dir`/`params.root` → `$MYCO_DIR` → cwd discovery
16//! (see `data.rs` for the exact order). `manifest`/`views` are
17//! project-free so pavo can enumerate drivers anywhere.
18
19mod data;
20mod views;
21
22use pavo_driver_kit as kit;
23use pavo_protocol::wire::{self, Request, Response, ViewSummary};
24
25fn capabilities_json() -> serde_json::Value {
26    let m = views::manifest();
27    serde_json::json!({
28        "capabilities": m.capabilities,
29        "verbs": m.verbs,
30        "views": m.views.len(),
31    })
32}
33
34/// Dispatch one parsed request to its response.
35fn dispatch(req: &Request) -> Response {
36    match req {
37        Request::Manifest => Response::ok_typed(&views::manifest()),
38        Request::Views => {
39            let summaries: Vec<ViewSummary> =
40                views::specs().iter().map(ViewSummary::from).collect();
41            Response::ok_typed(&summaries)
42        }
43        Request::View { view, params } => match views::view_result(view, params) {
44            Ok(result) => Response::ok_typed(&result),
45            Err(e) => Response::err(e),
46        },
47        Request::Query { query, params } => {
48            // Unknown names fail the op; everything else answers
49            // `ok:true` — degraded snapshots travel as
50            // `{"error": …}` data (the marker renderers understand),
51            // matching the reference driver's contract.
52            if !views::known_query(query) {
53                return Response::err(format!("unknown query: {query}"));
54            }
55            let snap = data::load(params);
56            let payload = views::query_data(query, params, &snap);
57            Response::ok_typed(&wire::QueryResult {
58                query: query.clone(),
59                data: payload,
60            })
61        }
62        Request::Render { kind, .. } => Response::err(format!(
63            "tamias-driver-myco claims no renderers (kind '{kind}')"
64        )),
65        // The protocol grew `submit`/`jobs`/`notifications` ops after
66        // this driver was written; myco is read-only with no feeds,
67        // so they answer `ok:false` like any unsupported op.
68        Request::Submit { .. } | Request::Jobs | Request::Notifications => {
69            Response::err("tamias-driver-myco implements no submit/jobs/notifications ops")
70        }
71    }
72}
73
74fn serve() -> i32 {
75    // The kit owns the ndjson loop — stdin EOF and a dead client
76    // both end it quietly, and either way the driver exits 0.
77    let _ = kit::serve_stdio(dispatch);
78    0
79}
80
81fn main() {
82    let mut args = std::env::args().skip(1);
83    match args.next().as_deref() {
84        Some("manifest") => println!("{}", kit::manifest_json(&views::manifest())),
85        Some("capabilities") => {
86            println!(
87                "{}",
88                serde_json::to_string_pretty(&capabilities_json()).unwrap_or_default()
89            );
90        }
91        Some("serve") => std::process::exit(serve()),
92        _ => {
93            eprintln!("usage: tamias-driver-myco manifest|capabilities|serve");
94            std::process::exit(1);
95        }
96    }
97}
