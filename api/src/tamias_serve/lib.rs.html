2//! tamias-serve — the ndjson JSON-RPC surface for embedding: read one JSON
3//! request per line on stdin, write one JSON response per line on stdout.
4//! Same wire as `sepia serve` / `cuora serve` / `myco serve`.
5//!
6//! Ops: `install`, `sync`, `list`, `info`, `verify`, `remove`, `search`,
7//! `publish`, `registry.list`, `registry.get`, `registry.add`,
8//! `registry.remove`. Requests are `{"op":"…", …}`; responses are
9//! `{"ok":true,"result":…}` or `{"ok":false,"error":"…"}`.
10//!
11//! `sync` reconciles the *server's* cwd project dep set against the
12//! store + `tamias.lock` — the same `tamias sync` semantics
13//! (`{"op":"sync","locked":…,"from_lock":…,"allow_drivers":…,
14//! "dir":…}`; `dir` defaults to `.`). `from_lock: true` treats
15//! `tamias.lock` itself as the spec (no manifest consult, no lock
16//! rewrite). Per-dep outcomes come back in `result.deps`; a
17//! bundled-driver consent refusal is an `ok:false` error like
18//! `install`'s.
19//!
20//! `search`/`publish`/`registry.*` resolve named registry sources and
21//! config writes through `tamias-registry-open` — the same code the
22//! `tamias` CLI verbs use, so the wire surface can't diverge.

32/// Store + cache + registry roots, plus the config file the
33/// registry-aware ops read/write.
34#[derive(Clone, Debug)]
35pub struct Env {
36    /// The local store root.
37    pub store_root: PathBuf,
38    /// The fetch-cache root.
39    pub cache_root: PathBuf,
40    /// The default local-registry root (`TAMIAS_REGISTRY`/XDG) — the
41    /// target when a `search`/`publish` `registry` value is absent.
42    pub registry_root: PathBuf,
43    /// Explicit config file for the registry-aware ops — `None` honors
44    /// `TAMIAS_CONFIG` then `./tamias.{yml,yaml,toml}` discovery, and
45    /// the file is re-read per op so a `registry.add` is visible to a
46    /// later `search`/`publish` in the same session.
47    pub config_file: Option<PathBuf>,
48}

50/// The effective config for read-side resolution (`search`/`publish`
51/// routing, `install`'s download cap) — mirrors the CLI's
52/// warn-and-ignore posture: a broken config warns and yields defaults
53/// rather than failing every op. Management ops (`registry.*`) read
54/// through `config_target` directly and propagate parse errors.
55fn env_config(env: &Env) -> tamias_config::Config {
56    match tamias_registry_open::load_config(env.config_file.as_deref()) {
57        Ok(config) => config,
58        Err(e) => {
59            eprintln!("tamias-serve: warning — ignoring config file: {e}");
60            tamias_config::Config::default()
61        }
62    }
63}

65/// Run the loop until stdin closes.
66///
67/// # Errors
68/// Propagates stdin/stdout failures.
69pub fn serve(env: Env) -> Result<(), Error> {
70    serve_loop(env)
71}
72
73/// The real blocking stdin loop — compiled out under `cargo-llvm-cov`
74/// (`cfg(coverage)`) so the coverage build doesn't hang on stdin.
75#[cfg(not(coverage))]
76fn serve_loop(env: Env) -> Result<(), Error> {
77    let stdin = std::io::stdin();
78    let mut stdout = std::io::stdout();
79    for line in stdin.lock().lines() {
80        let line = line.map_err(Error::io("read stdin"))?;
81        if line.trim().is_empty() {
82            continue;
83        }
84        let response = handle(&env, &line);
85        writeln!(stdout, "{response}").map_err(Error::io("write stdout"))?;
86        stdout.flush().map_err(Error::io("flush stdout"))?;
87    }
88    Ok(())
89}

91/// The coverage probe — a fast exit so the loop's lines aren't measured.
92#[cfg(coverage)]
93fn serve_loop(_env: Env) -> Result<(), Error> {
94    Err(Error::Failed(
95        "serve loop is excluded under coverage".to_string(),
96    ))
97}

99fn handle(env: &Env, line: &str) -> String {
100    match dispatch(env, line) {
101        Ok(result) => json!({"ok": true, "result": result}).to_string(),
102        Err(e) => json!({"ok": false, "error": e.to_string()}).to_string(),
103    }
104}

106fn str_field<'a>(req: &'a Value, key: &str) -> Result<&'a str, Error> {
107    req.get(key)
108        .and_then(Value::as_str)
109        .ok_or_else(|| Error::Failed(format!("missing '{key}'")))
110}

112fn dispatch(env: &Env, line: &str) -> Result<Value, Error> {
113    let req: Value =
114        serde_json::from_str(line).map_err(|e| Error::Failed(format!("bad request: {e}")))?;
115    let op = str_field(&req, "op")?;
116    match op {
117        "install" => {
118            let reference = str_field(&req, "ref")?;
119            // Optional consent for payload-shipped `drivers/` — same rule as
120            // `tamias install --allow-drivers` / `TAMIAS_ALLOW_DRIVERS=1`.
121            let allow_drivers = req
122                .get("allow_drivers")
123                .and_then(Value::as_bool)
124                .unwrap_or(false);
125            // The shared install pipeline — silent progress, `"allow_drivers"`
126            // consent hints (see `tamias_install::Surface::Rpc`).
127            let config = env_config(env);
128            let entry = tamias_install::install(
129                &tamias_install::Env {
130                    store_root: env.store_root.clone(),
131                    cache_root: env.cache_root.clone(),
132                    // Global `limits.max_download` caps http(s)/`cf:`
133                    // fetches — same resolution as the CLI's install.
134                    max_download: Some(config.effective_limits(None, None).max_download),
135                    // `signing.*` — package-signature verification policy.
136                    signing: tamias_sign::SigningPolicy::resolve(Some(&config)),
137                },
138                reference,
139                allow_drivers,
140            )?;
141            Ok(json!({
142                "label": entry.label(),
143                "resolved": entry.record.resolved,
144                "digest": entry.record.digest,
145                "signed": entry.record.signature.is_some(),
146            }))
147        }
148        "sync" => {
149            // `tamias sync` — reconcile the project dep set against
150            // the store + `tamias.lock`. `dir` (default `.`) is the
151            // project dir; fs refs inside `deps:` still resolve
152            // against the server's cwd, same as `install`.
153            // `from_lock: true` swaps the spec to `tamias.lock`
154            // itself (the manifest's `deps:` is not consulted).
155            let allow_drivers = req
156                .get("allow_drivers")
157                .and_then(Value::as_bool)
158                .unwrap_or(false);
159            let locked = req.get("locked").and_then(Value::as_bool).unwrap_or(false);
160            let from_lock = req
161                .get("from_lock")
162                .and_then(Value::as_bool)
163                .unwrap_or(false);
164            let dir = req
165                .get("dir")
166                .and_then(Value::as_str)
167                .map_or_else(|| Path::new("."), Path::new);
168            let config = env_config(env);
169            let report = tamias_install::sync_project(
170                &tamias_install::Env {
171                    store_root: env.store_root.clone(),
172                    cache_root: env.cache_root.clone(),
173                    max_download: Some(config.effective_limits(None, None).max_download),
174                    signing: tamias_sign::SigningPolicy::resolve(Some(&config)),
175                },
176                dir,
177                tamias_install::Surface::Rpc,
178                tamias_install::SyncOptions {
179                    allow_drivers,
180                    locked,
181                    source: if from_lock {
182                        tamias_install::SyncSource::Lock
183                    } else {
184                        tamias_install::SyncSource::Deps
185                    },
186                },
187                &mut |_| {},
188            )?;
189            let deps: Vec<Value> = report
190                .outcomes
191                .iter()
192                .map(|o| {
193                    json!({
194                        "dep": o.dep,
195                        "status": o.status.as_str(),
196                        "detail": o.detail,
197                    })
198                })
199                .collect();
200            Ok(json!({"deps": deps, "failed": report.failed()}))
201        }
202        "list" => {
203            let entries = Store::new(&env.store_root).entries()?;
204            let items: Vec<Value> = entries
205                .iter()
206                .map(|e| {
207                    json!({
208                        "label": e.label(),
209                        "source_ref": e.record.source_ref,
210                        "digest": e.record.digest,
211                    })
212                })
213                .collect();
214            Ok(json!(items))
215        }
216        "info" => {
217            let name = str_field(&req, "name")?;
218            let entry = Store::new(&env.store_root).find(name)?;
219            Ok(json!({
220                "label": entry.label(),
221                "resolved": entry.record.resolved,
222                "digest": entry.record.digest,
223                "size": entry.size()?,
224            }))
225        }
226        "verify" => {
227            let name = str_field(&req, "name")?;
228            let store = Store::new(&env.store_root);
229            let report = store.verify(name)?;
230            // Signature status alongside the digest check — the record's
231            // `signature` re-verified against `signing.allowed_signers`.
232            let entry = store.find(name)?;
233            let config = env_config(env);
234            let policy = tamias_sign::SigningPolicy::resolve(Some(&config));
235            let signature = tamias_sign::record_signature_status(
236                &entry.record,
237                policy.allowed_signers.as_deref(),
238                &policy.namespace,
239            )?
240            .as_str();
241            Ok(json!({
242                "name": report.name,
243                "ok": report.ok,
244                "signature": signature,
245            }))
246        }
247        "remove" => {
248            let name = str_field(&req, "name")?;
249            // The shared lightweight uninstall (`tamias_install::remove`)
250            // — same teardown rule as `tamias ui`'s Remove job.
251            let record = tamias_install::remove(&env.store_root, name, &mut |m| {
252                eprintln!("tamias-serve: warning — {m}");
253            })?;
254            Ok(json!({"name": record.name}))
255        }
256        // --- registry sources (shared with the CLI verbs) ---
257        "registry.list" => {
258            let (_path, config) = tamias_registry_open::config_target(env.config_file.as_deref())?;
259            let items: Vec<Value> = config
260                .registries
261                .iter()
262                .map(|(name, reg)| tamias_registry_open::registry_json(name, reg))
263                .collect();
264            Ok(json!({"registries": items}))
265        }
266        "registry.get" => {
267            let name = str_field(&req, "name")?;
268            let reg = tamias_registry_open::get_registry(env.config_file.as_deref(), name)?;
269            Ok(tamias_registry_open::registry_json(name, &reg))
270        }
271        "registry.add" => {
272            let name = str_field(&req, "name")?;
273            let kind = str_field(&req, "kind")?;
274            let url = str_field(&req, "url")?;
275            let token_env = req
276                .get("token_env")
277                .and_then(Value::as_str)
278                .map(str::to_string);
279            let path = tamias_registry_open::add_registry(
280                env.config_file.as_deref(),
281                name,
282                kind,
283                url,
284                token_env,
285            )?;
286            Ok(json!({"name": name, "config": path.display().to_string()}))
287        }
288        "registry.remove" => {
289            let name = str_field(&req, "name")?;
290            let path = tamias_registry_open::remove_registry(env.config_file.as_deref(), name)?;
291            Ok(json!({"name": name, "config": path.display().to_string()}))
292        }
293        "search" => {
294            let query = str_field(&req, "query")?;
295            let flag = req.get("registry").and_then(Value::as_str);
296            let config = env_config(env);
297            // Same routing as `tamias search`: flag → TAMIAS_REGISTRY →
298            // search.default → publish.default → local default root —
299            // the shared `search` (`tamias-registry-open`).
300            let (found, _target) =
301                tamias_registry_open::search(&env.registry_root, Some(&config), flag, query)?;
302            Ok(json!({"results": found.iter().map(registry_entry_json).collect::<Vec<_>>()}))
303        }
304        "publish" => {
305            let dir = str_field(&req, "dir")?;
306            let flag = req.get("registry").and_then(Value::as_str);
307            let config = env_config(env);
308            // Same routing as `tamias publish`: flag → TAMIAS_REGISTRY →
309            // publish.default → local default root — the shared
310            // `publish_plan` resolution (`tamias-registry-open`).
311            // `"keep_includes":true` ships the raw payload — the
312            // default flattens `include:` bases read-only.
313            let keep_includes = req
314                .get("keep_includes")
315                .and_then(Value::as_bool)
316                .unwrap_or(false);
317            let plan = tamias_registry_open::publish_plan(
318                Path::new(dir),
319                &env.registry_root,
320                Some(&config),
321                flag,
322                &[config.publish.default.as_deref()],
323                &env.store_root,
324                keep_includes,
325                &mut |_, _| {},
326            )?;
327            // `"sign":true` — ssh-keygen -Y sign the provenance anchors;
328            // the key resolves like the CLI's `publish --sign`.
329            let signature = if req.get("sign").and_then(Value::as_bool).unwrap_or(false) {
330                let key = tamias_sign::resolve_signing_key(Some(&config), None)?;
331                let policy = tamias_sign::SigningPolicy::resolve(Some(&config));
332                Some(tamias_sign::sign_package(
333                    &plan.root,
334                    &plan.manifest,
335                    &key,
336                    &policy.namespace,
337                )?)
338            } else {
339                None
340            };
341            let entry = plan.publish(signature.as_deref())?;
342            Ok(registry_entry_json(&entry))
343        }
344        other => Err(Error::Failed(format!("unknown op '{other}'"))),
345    }
346}

348/// One registry index entry as a JSON object — the `search`/`publish`
349/// result shape (same fields the CLI's `--json` prints).
350fn registry_entry_json(entry: &tamias_registry::RegistryEntry) -> Value {
351    json!({
352        "label": entry.label(),
353        "name": entry.name,
354        "scope": entry.scope,
355        "version": entry.version,
356        "kind": entry.kind,
357        "digest": entry.digest,
358        "signed": entry.signature.is_some(),
359        "file": entry.file,
360    })
361}

378    /// Serializes tests that mutate process env (`TAMIAS_HOME`,
379    /// `TAMIAS_ALLOW_DRIVERS`, `TAMIAS_REGISTRY`, config env). `cargo
380    /// test` runs tests on threads in one process, so mutations must
381    /// be serialized *and* restored even on panic.
382    static DRVENV: std::sync::Mutex<()> = std::sync::Mutex::new(());
383
384    /// RAII guard for [`DRVENV`]: restores the process environment and
385    /// working directory on drop — panic-safe. Poison is ignored: an
386    /// unwinding sibling already restored everything via its `Drop`.
387    struct EnvGuard {
388        _lock: std::sync::MutexGuard<'static, ()>,
389        env: Vec<(std::ffi::OsString, std::ffi::OsString)>,
390        cwd: std::path::PathBuf,
391    }
392
393    impl EnvGuard {
394        fn new() -> Self {
395            let lock = DRVENV.lock().unwrap_or_else(|e| e.into_inner());
396            let env = std::env::vars_os().collect();
397            let cwd = std::env::current_dir().unwrap();
398            Self {
399                _lock: lock,
400                env,
401                cwd,
402            }
403        }
404    }
405
406    impl Drop for EnvGuard {
407        fn drop(&mut self) {
408            for (k, _) in std::env::vars_os() {
409                if !self.env.iter().any(|(sk, _)| *sk == k) {
410                    std::env::remove_var(&k);
411                }
412            }
413            for (k, v) in &self.env {
414                std::env::set_var(k, v);
415            }
416            let _ = std::env::set_current_dir(&self.cwd);
417        }
418    }

844    #[test]
845    fn registry_ops_round_trip() {
846        let tmp = tempfile::tempdir().unwrap();
847        let env = registry_env(tmp.path());
848        // empty to start
849        let list = dispatch(&env, r#"{"op":"registry.list"}"#).unwrap();
850        assert_eq!(list["registries"].as_array().unwrap().len(), 0);
851        // add → list → get → remove
852        let reg_dir = tmp.path().join("reg").display().to_string();
853        let add = format!(
854            "{{\"op\":\"registry.add\",\"name\":\"mine\",\"kind\":\"local\",\"url\":{reg_dir:?},\"token_env\":\"MY_REG_TOKEN\"}}"
855        );
856        let res = dispatch(&env, &add).unwrap();
857        assert_eq!(res["name"], "mine");
858        let list = dispatch(&env, r#"{"op":"registry.list"}"#).unwrap();
859        let regs = list["registries"].as_array().unwrap();
860        assert_eq!(regs.len(), 1);
861        assert_eq!(regs[0]["name"], "mine");
862        assert_eq!(regs[0]["effective_kind"], "local");
863        // the env var *name* is exposed; no token value exists to leak
864        assert_eq!(regs[0]["token_env"], "MY_REG_TOKEN");
865        let get = dispatch(&env, r#"{"op":"registry.get","name":"mine"}"#).unwrap();
866        assert_eq!(get["name"], "mine");
867        assert_eq!(get["url"], reg_dir);
868        // the file on disk round-tripped through the YAML renderer
869        let text = std::fs::read_to_string(tmp.path().join("tamias.yml")).unwrap();
870        assert!(
871            text.contains("registries:") && text.contains("mine"),
872            "{text}"
873        );
874        // duplicate add + unknown get/remove error
875        assert!(dispatch(&env, &add).is_err());
876        assert!(dispatch(&env, r#"{"op":"registry.get","name":"ghost"}"#).is_err());
877        assert!(dispatch(&env, r#"{"op":"registry.remove","name":"ghost"}"#).is_err());
878        let rem = dispatch(&env, r#"{"op":"registry.remove","name":"mine"}"#).unwrap();
879        assert_eq!(rem["name"], "mine");
880        assert!(dispatch(&env, r#"{"op":"registry.get","name":"mine"}"#).is_err());
881    }
