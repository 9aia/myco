1//! Git plumbing for myco — shells out to the `git` CLI (the
2//! bupha-git/sturn-git convention; no git2/gix). Phase 3 builds
3//! `init` / `check` / `migrate` / auto-commit on top of these
4//! primitives: the `project` orphan branch + worktree, and one
5//! atomic commit per verb.
6
7use std::path::{Path, PathBuf};
8use std::process::{Command, Stdio};

10/// Run `git -C <repo> <args>`; true on exit status 0. stdout is
11/// silenced — progress chatter (`HEAD is now at …`, commit summaries)
12/// must never pollute a verb's `--json` stream; stderr stays
13/// inherited so failures are still diagnosable.
14pub fn git(repo: &Path, args: &[&str]) -> bool {
15    Command::new("git")
16        .arg("-C")
17        .arg(repo)
18        .args(args)
19        .stdout(Stdio::null())
20        .status()
21        .map(|s| s.success())
22        .unwrap_or(false)
23}
24
25/// A git invocation that failed, couldn't spawn, or printed non-UTF8.
26#[derive(Debug, thiserror::Error)]
27pub enum Error {
28    /// Non-zero exit — stderr captured for the message.
29    #[error("git {}: {}", args, stderr)]
30    Git {
31        /// The argv after `git -C <repo>` (for context).
32        args: String,
33        /// Trimmed stderr.
34        stderr: String,
35    },
36    /// The `git` binary couldn't be spawned.
37    #[error("git spawn: {0}")]
38    Spawn(#[from] std::io::Error),
39    /// stdout wasn't valid UTF-8.
40    #[error("git output was not utf-8")]
41    Utf8,
42    /// A path can't be carried on git's argv — refuse to smuggle it
43    /// through as an empty string.
44    #[error("path is not utf-8: {}", .0.display())]
45    NonUtf8Path(PathBuf),
46    /// The checkout's HEAD is detached — a commit there belongs to no
47    /// branch, so the ledger's audit trail would be unreachable the
48    /// moment the worktree moves. `myco check --fix` reattaches it.
49    #[error(
50        "{} is on a detached HEAD — a commit would be unreachable; run `myco check --fix`",
51        .0.display()
52    )]
53    Detached(PathBuf),
54}

56/// `git -C <repo> <args>` → trimmed stdout. stdin is `/dev/null` —
57/// plumbing commands that read stdin (`mktree`) get an immediate
58/// EOF instead of blocking on the caller's terminal.
59pub fn git_out(repo: &Path, args: &[&str]) -> Result<String, Error> {
60    let out = Command::new("git")
61        .arg("-C")
62        .arg(repo)
63        .args(args)
64        .stdin(Stdio::null())
65        .output()?;
66    if !out.status.success() {
67        return Err(Error::Git {
68            args: args.join(" "),
69            stderr: String::from_utf8_lossy(&out.stderr).trim().to_string(),
70        });
71    }
72    String::from_utf8(out.stdout)
73        .map(|s| s.trim().to_string())
74        .map_err(|_| Error::Utf8)
75}

77/// `git -C <repo> <args>` for commands that need a committer
78/// identity (`commit`, `commit-tree`). When the repo can't resolve
79/// one — `git var GIT_COMMITTER_IDENT` makes the same check the
80/// commit machinery does — a `-c user.name=myco
81/// -c user.email=myco@localhost` fallback is injected for that one
82/// invocation. A configured identity (or `GIT_*_NAME`/`GIT_*_EMAIL`
83/// env) always wins, since the probe sees it first.
84pub fn git_ident(repo: &Path, args: &[&str]) -> Result<String, Error> {
85    const FALLBACK: [&str; 4] = ["-c", "user.name=myco", "-c", "user.email=myco@localhost"];
86    let mut scoped = Vec::with_capacity(args.len() + FALLBACK.len());
87    if git_out(repo, &["var", "GIT_COMMITTER_IDENT"]).is_err() {
88        scoped.extend_from_slice(&FALLBACK);
89    }
90    scoped.extend_from_slice(args);
91    git_out(repo, &scoped)
92}

94/// `git rev-parse --show-toplevel` — the worktree root containing
95/// `start` (`Err` outside a repo).
96pub fn repo_root(start: &Path) -> Result<PathBuf, Error> {
97    git_out(start, &["rev-parse", "--show-toplevel"]).map(PathBuf::from)
98}

100/// `Ok(true)` when `start` is inside a git work tree. "Not a
101/// repo" (non-zero `rev-parse`) is `Ok(false)`; only spawn/output
102/// failures are `Err`, so a missing `git` binary doesn't
103/// masquerade as "not a repo".
104pub fn is_git_repo(start: &Path) -> Result<bool, Error> {
105    match git_out(start, &["rev-parse", "--is-inside-work-tree"]) {
106        Ok(out) => Ok(out == "true"),
107        Err(Error::Git { .. }) => Ok(false),
108        Err(e) => Err(e),
109    }
110}

125/// Create `name` as an orphan branch via plumbing — `mktree` an
126/// empty tree, `commit-tree` a root commit, `update-ref` the branch.
127/// Unlike `git checkout --orphan` this never disturbs the index or
128/// working tree of the current checkout.
129pub fn create_orphan_branch(repo: &Path, name: &str, msg: &str) -> Result<(), Error> {
130    let tree = git_out(repo, &["mktree"])?;
131    let commit = git_ident(repo, &["commit-tree", &tree, "-m", msg])?;
132    // empty old-value = create-only — the ref must not exist, so a
133    // race past `branch_exists` can't reset `name` to an empty
134    // root and orphan the ledger
135    git_out(
136        repo,
137        &["update-ref", &format!("refs/heads/{name}"), &commit, ""],
138    )?;
139    Ok(())
140}

142/// The branch a worktree dir has checked out (`None` when detached
143/// or not a worktree).
144pub fn worktree_branch(dir: &Path) -> Option<String> {
145    git_out(dir, &["rev-parse", "--abbrev-ref", "HEAD"])
146        .ok()
147        .filter(|b| b != "HEAD" && !b.is_empty())
148}

150/// `dir` is a git checkout whose HEAD is detached — commits made
151/// there belong to no branch and become unreachable as soon as the
152/// worktree moves. `None` when `dir` isn't a checkout at all
153/// (the plain-directory ledger shape), which is not a defect.
154pub fn is_detached(dir: &Path) -> Option<bool> {
155    head_commit(dir)?;
156    Some(worktree_branch(dir).is_none())
157}

159/// Attach the checkout at `dir` to `name`, keeping its current HEAD
160/// as the branch tip — the repair for a detached ledger worktree
161/// (`myco check --fix`). `checkout -B` is create-or-reset, so a
162/// branch left behind by an earlier run loses to the commits that
163/// are actually here; git refuses when another worktree holds the
164/// branch, which is precisely the case a human must arbitrate.
165pub fn attach_branch(dir: &Path, name: &str) -> Result<(), Error> {
166    if worktree_branch(dir).as_deref() == Some(name) {
167        return Ok(());
168    }
169    git_out(dir, &["checkout", "-B", name]).map(|_| ())
170}

172/// Does `origin`'s fetch refspec publish `branch` into
173/// `refs/remotes/origin/`? A repo cloned with `--single-branch`
174/// only tracks some branches, so `origin/<branch>` can sit stale
175/// forever while `git push` still succeeds — comparing against it
176/// would report "unpushed" no matter what the user does.
177pub fn tracks_branch(dir: &Path, remote: &str, branch: &str) -> bool {
178    let key = format!("remote.{remote}.fetch");
179    let Ok(specs) = git_out(dir, &["config", "--get-all", &key]) else {
180        return false;
181    };
182    let want = format!("refs/remotes/{remote}/{branch}");
183    let all = format!("refs/remotes/{remote}/*");
184    specs.lines().any(|line| {
185        let spec = line.trim().trim_start_matches('+');
186        matches!(spec.rsplit_once(':'), Some((_, dst)) if dst == want || dst == all)
187    })
188}

190/// How many commits `HEAD` has that `rev` doesn't — `None` when

192pub fn commits_ahead(dir: &Path, rev: &str) -> Option<usize> {
193    git_out(dir, &["rev-list", "--count", &format!("{rev}..HEAD")])
194        .ok()?
195        .trim()
196        .parse()
197        .ok()

206        &["rev-parse", "--path-format=absolute", "--git-common-dir"],
207    )
208    .map(PathBuf::from)

211/// The main (first) worktree of the repo `dir` belongs to — i.e.
212/// the repo root when `dir` is a linked worktree like `project/`.
213/// Derived from `--git-common-dir` (`<main>/.git` → `<main>`).
214pub fn main_worktree(dir: &Path) -> Option<PathBuf> {
215    let common = common_dir(dir).ok()?;
216    if common.file_name().is_some_and(|n| n == ".git") {
217        return common.parent().map(Path::to_path_buf);
218    }
219    // bare repo or worktree-of-worktree — no checkout root
220    None
221}
222
223/// Like [`main_worktree`] but tolerant of old git: `worktree list`'s
224/// first entry is always the main worktree (no `--path-format`,
225/// which needs git ≥ 2.31).
226pub fn main_worktree_compat(dir: &Path) -> Option<PathBuf> {
227    main_worktree(dir).or_else(|| {
228        let out = git_out(dir, &["worktree", "list"]).ok()?;
229        let path = out.lines().next()?.split_whitespace().next()?;
230        Some(PathBuf::from(path))
231    })
232}

294/// `git add -A -- . && git commit --allow-empty -m <msg>` —
295/// the auto-commit-per-verb path writes files first, then this.
296/// The stage is scoped to `repo` (`-- .`): when `repo` is a plain
297/// subdirectory of a bigger repo rather than a real worktree,
298/// changes elsewhere aren't swept into the ledger commit. A bare
299/// `commit` could still pick up foreign *staged* changes, so those
300/// are refused rather than committed under our message (a
301/// `commit -- <pathspec>` alternative can't express the first
302/// commit — the pathspec must match known files). `--allow-empty`
303/// keeps no-op verbs honest in the audit trail.
304pub fn commit_all(repo: &Path, msg: &str) -> Result<(), Error> {
305    // A detached checkout has no branch to commit onto: the write
306    // would succeed and the audit trail would quietly vanish. Refuse
307    // before touching the index (`check --fix` reattaches, and the
308    // repair paths — `commit`, `migrate`, `check` — take the lock
309    // directly, so they still reach a detached ledger).
310    if is_detached(repo) == Some(true) {
311        return Err(Error::Detached(repo.to_path_buf()));
312    }
313    git_out(repo, &["add", "-A", "--", "."])?;
314    // Foreign *staged* changes must not be swept into the ledger
315    // commit — refuse them. `:(exclude).` is cwd-relative, so the
316    // probe can only fire for a non-worktree dir; a real worktree
317    // can't hold staged entries outside itself. The quiet diff is
318    // byte-safe where a --name-only parse trips on non-UTF-8
319    // filenames.
320    if !git(repo, &["diff", "--cached", "--quiet", "--", ":(exclude)."]) {
321        return Err(Error::Git {
322            args: "commit".into(),
323            stderr: "refusing to sweep staged changes outside the worktree — unstage them first"
324                .into(),
325        });
326    }
327    git_ident(repo, &["commit", "--allow-empty", "-m", msg])?;
328    Ok(())
329}

356/// One `git log` line, parsed.
357#[derive(Debug, Clone, PartialEq, Eq)]
358pub struct LogEntry {
359    /// Full commit hash.
360    pub hash: String,
361    /// Author date, `YYYY-MM-DD` (`%as`).
362    pub date: String,
363    /// Subject line.
364    pub subject: String,
365}
366
367/// `git log` newest-first — `%H %as %s` over a NUL separator so
368/// subjects are safe. `limit` caps the walk (`git log -n`). An
369/// unborn branch yields an empty history, not an error.
370pub fn log(dir: &Path, limit: Option<usize>) -> Result<Vec<LogEntry>, Error> {
371    if !has_commit(dir, "HEAD") {
372        return Ok(Vec::new());
373    }
374    let mut args = vec!["log", "--format=%H%x00%as%x00%s"];
375    let n;
376    if let Some(k) = limit {
377        n = format!("-{k}");
378        args.push(&n);
379    }
380    let out = git_out(dir, &args)?;
381    Ok(out
382        .lines()
383        .filter_map(|line| {
384            let mut parts = line.split('\0');
385            Some(LogEntry {
386                hash: parts.next()?.to_string(),
387                date: parts.next()?.to_string(),
388                subject: parts.next().unwrap_or_default().to_string(),
389            })
390        })
391        .collect())
392}
