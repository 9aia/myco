1//! Git plumbing for myco — shells out to the `git` CLI (the
2//! bupha-git/sturn-git convention; no git2/gix). Phase 3 builds
3//! `init` / `check` / `migrate` / auto-commit on top of these
4//! primitives: the `project` orphan branch + worktree, and one
5//! atomic commit per verb.

7use std::path::{Path, PathBuf};
8use std::process::{Command, Stdio};

10/// Run `git -C <repo> <args>`; true on exit status 0. stdout is
11/// silenced — progress chatter (`HEAD is now at …`, commit summaries)
12/// must never pollute a verb's `--json` stream; stderr stays
13/// inherited so failures are still diagnosable.
14pub fn git(repo: &Path, args: &[&str]) -> bool {
15    Command::new("git")
16        .arg("-C")
17        .arg(repo)
18        .args(args)
19        .stdout(Stdio::null())
20        .status()
21        .map(|s| s.success())
22        .unwrap_or(false)
23}
24
25/// A git invocation that failed, couldn't spawn, or printed non-UTF8.
26#[derive(Debug, thiserror::Error)]
27pub enum Error {
28    /// Non-zero exit — stderr captured for the message.
29    #[error("git {}: {}", args, stderr)]
30    Git {
31        /// The argv after `git -C <repo>` (for context).
32        args: String,
33        /// Trimmed stderr.
34        stderr: String,
35    },
36    /// The `git` binary couldn't be spawned.
37    #[error("git spawn: {0}")]
38    Spawn(#[from] std::io::Error),
39    /// stdout wasn't valid UTF-8.
40    #[error("git output was not utf-8")]
41    Utf8,
42    /// A path can't be carried on git's argv — refuse to smuggle it
43    /// through as an empty string.
44    #[error("path is not utf-8: {}", .0.display())]
45    NonUtf8Path(PathBuf),
46}

48/// `git -C <repo> <args>` → trimmed stdout. stdin is `/dev/null` —
49/// plumbing commands that read stdin (`mktree`) get an immediate
50/// EOF instead of blocking on the caller's terminal.
51pub fn git_out(repo: &Path, args: &[&str]) -> Result<String, Error> {
52    let out = Command::new("git")
53        .arg("-C")
54        .arg(repo)
55        .args(args)
56        .stdin(Stdio::null())
57        .output()?;
58    if !out.status.success() {
59        return Err(Error::Git {
60            args: args.join(" "),
61            stderr: String::from_utf8_lossy(&out.stderr).trim().to_string(),
62        });
63    }
64    String::from_utf8(out.stdout)
65        .map(|s| s.trim().to_string())
66        .map_err(|_| Error::Utf8)
67}

69/// `git -C <repo> <args>` for commands that need a committer
70/// identity (`commit`, `commit-tree`). When the repo can't resolve
71/// one — `git var GIT_COMMITTER_IDENT` makes the same check the
72/// commit machinery does — a `-c user.name=myco
73/// -c user.email=myco@localhost` fallback is injected for that one
74/// invocation. A configured identity (or `GIT_*_NAME`/`GIT_*_EMAIL`
75/// env) always wins, since the probe sees it first.
76pub fn git_ident(repo: &Path, args: &[&str]) -> Result<String, Error> {
77    const FALLBACK: [&str; 4] = ["-c", "user.name=myco", "-c", "user.email=myco@localhost"];
78    let mut scoped = Vec::with_capacity(args.len() + FALLBACK.len());
79    if git_out(repo, &["var", "GIT_COMMITTER_IDENT"]).is_err() {
80        scoped.extend_from_slice(&FALLBACK);
81    }
82    scoped.extend_from_slice(args);
83    git_out(repo, &scoped)
84}

86/// `git rev-parse --show-toplevel` — the worktree root containing
87/// `start` (`Err` outside a repo).
88pub fn repo_root(start: &Path) -> Result<PathBuf, Error> {
89    git_out(start, &["rev-parse", "--show-toplevel"]).map(PathBuf::from)
90}

92/// `Ok(true)` when `start` is inside a git work tree. "Not a
93/// repo" (non-zero `rev-parse`) is `Ok(false)`; only spawn/output
94/// failures are `Err`, so a missing `git` binary doesn't
95/// masquerade as "not a repo".
96pub fn is_git_repo(start: &Path) -> Result<bool, Error> {
97    match git_out(start, &["rev-parse", "--is-inside-work-tree"]) {
98        Ok(out) => Ok(out == "true"),
99        Err(Error::Git { .. }) => Ok(false),
100        Err(e) => Err(e),
101    }
102}

111            "--quiet",

113        ],

117/// Create `name` as an orphan branch via plumbing — `mktree` an
118/// empty tree, `commit-tree` a root commit, `update-ref` the branch.
119/// Unlike `git checkout --orphan` this never disturbs the index or
120/// working tree of the current checkout.
121pub fn create_orphan_branch(repo: &Path, name: &str, msg: &str) -> Result<(), Error> {
122    let tree = git_out(repo, &["mktree"])?;
123    let commit = git_ident(repo, &["commit-tree", &tree, "-m", msg])?;
124    // empty old-value = create-only — the ref must not exist, so a
125    // race past `branch_exists` can't reset `name` to an empty
126    // root and orphan the ledger
127    git_out(
128        repo,
129        &["update-ref", &format!("refs/heads/{name}"), &commit, ""],
130    )?;
131    Ok(())
132}

134/// The branch a worktree dir has checked out (`None` when detached
135/// or not a worktree).
136pub fn worktree_branch(dir: &Path) -> Option<String> {
137    git_out(dir, &["rev-parse", "--abbrev-ref", "HEAD"])
138        .ok()
139        .filter(|b| b != "HEAD" && !b.is_empty())

142/// `git rev-parse --path-format=absolute --git-common-dir` — the
143/// shared git dir for all of `dir`'s worktrees (`<main>/.git`),
144/// resolved from inside any of them.
145pub fn common_dir(dir: &Path) -> Result<PathBuf, Error> {
146    git_out(
147        dir,
148        &["rev-parse", "--path-format=absolute", "--git-common-dir"],
149    )
150    .map(PathBuf::from)
151}
152
153/// The main (first) worktree of the repo `dir` belongs to — i.e.
154/// the repo root when `dir` is a linked worktree like `project/`.
155/// Derived from `--git-common-dir` (`<main>/.git` → `<main>`).
156pub fn main_worktree(dir: &Path) -> Option<PathBuf> {
157    let common = common_dir(dir).ok()?;
158    if common.file_name().is_some_and(|n| n == ".git") {
159        return common.parent().map(Path::to_path_buf);
160    }
161    // bare repo or worktree-of-worktree — no checkout root
162    None
163}
164
165/// Like [`main_worktree`] but tolerant of old git: `worktree list`'s
166/// first entry is always the main worktree (no `--path-format`,
167/// which needs git ≥ 2.31).
168pub fn main_worktree_compat(dir: &Path) -> Option<PathBuf> {
169    main_worktree(dir).or_else(|| {
170        let out = git_out(dir, &["worktree", "list"]).ok()?;
171        let path = out.lines().next()?.split_whitespace().next()?;
172        Some(PathBuf::from(path))
173    })
174}

176/// Uncommitted/untracked changes under `dir` (`status --porcelain
177/// -- .` non-empty). Scoped to `dir` itself — when `dir` is a plain
178/// subdirectory of a bigger repo rather than a real worktree,
179/// changes elsewhere in the containing repo don't count. Git
180/// failures propagate: callers must not mistake "status failed"

182pub fn has_changes(dir: &Path) -> Result<bool, Error> {
183    git_out(dir, &["status", "--porcelain", "--", "."]).map(|s| !s.is_empty())
184}

236/// `git add -A -- . && git commit --allow-empty -m <msg>` —
237/// the auto-commit-per-verb path writes files first, then this.
238/// The stage is scoped to `repo` (`-- .`): when `repo` is a plain
239/// subdirectory of a bigger repo rather than a real worktree,
240/// changes elsewhere aren't swept into the ledger commit. A bare
241/// `commit` could still pick up foreign *staged* changes, so those
242/// are refused rather than committed under our message (a
243/// `commit -- <pathspec>` alternative can't express the first
244/// commit — the pathspec must match known files). `--allow-empty`
245/// keeps no-op verbs honest in the audit trail.
246pub fn commit_all(repo: &Path, msg: &str) -> Result<(), Error> {
247    git_out(repo, &["add", "-A", "--", "."])?;
248    // Foreign *staged* changes must not be swept into the ledger
249    // commit — refuse them. `:(exclude).` is cwd-relative, so the
250    // probe can only fire for a non-worktree dir; a real worktree
251    // can't hold staged entries outside itself. The quiet diff is
252    // byte-safe where a --name-only parse trips on non-UTF-8
253    // filenames.
254    if !git(repo, &["diff", "--cached", "--quiet", "--", ":(exclude)."]) {
255        return Err(Error::Git {
256            args: "commit".into(),
257            stderr: "refusing to sweep staged changes outside the worktree — unstage them first"
258                .into(),
259        });
260    }
261    git_ident(repo, &["commit", "--allow-empty", "-m", msg])?;
262    Ok(())
263}

290/// One `git log` line, parsed.
291#[derive(Debug, Clone, PartialEq, Eq)]
292pub struct LogEntry {
293    /// Full commit hash.
294    pub hash: String,
295    /// Author date, `YYYY-MM-DD` (`%as`).
296    pub date: String,
297    /// Subject line.
298    pub subject: String,
299}
300
301/// `git log` newest-first — `%H %as %s` over a NUL separator so
302/// subjects are safe. `limit` caps the walk (`git log -n`). An
303/// unborn branch yields an empty history, not an error.
304pub fn log(dir: &Path, limit: Option<usize>) -> Result<Vec<LogEntry>, Error> {
305    if !has_commit(dir, "HEAD") {
306        return Ok(Vec::new());
307    }
308    let mut args = vec!["log", "--format=%H%x00%as%x00%s"];
309    let n;
310    if let Some(k) = limit {
311        n = format!("-{k}");
312        args.push(&n);
313    }
314    let out = git_out(dir, &args)?;
315    Ok(out
316        .lines()
317        .filter_map(|line| {
318            let mut parts = line.split('\0');
319            Some(LogEntry {
320                hash: parts.next()?.to_string(),
321                date: parts.next()?.to_string(),
322                subject: parts.next().unwrap_or_default().to_string(),
323            })
324        })
325        .collect())
326}

556        // create-only update-ref: a second attempt must fail, and
557        // the branch must still point at the original root
558        let e = create_orphan_branch(d.path(), "project", "clobber").unwrap_err();
559        assert!(matches!(e, Error::Git { .. }), "{e:?}");
560        assert_eq!(git_out(d.path(), &["rev-parse", "project"]).unwrap(), head);
561    }
