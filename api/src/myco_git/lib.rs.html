1//! Git plumbing for myco — shells out to the `git` CLI (the
2//! bupha-git/sturn-git convention; no git2/gix). Phase 3 builds
3//! `init` / `check` / `migrate` / auto-commit on top of these
4//! primitives: the `project` orphan branch + worktree, and one
5//! atomic commit per verb.
6
7use std::path::{Path, PathBuf};
8use std::process::{Command, Stdio};
9
10/// Run `git -C <repo> <args>`; true on exit status 0. stdout is
11/// silenced — progress chatter (`HEAD is now at …`, commit summaries)
12/// must never pollute a verb's `--json` stream; stderr stays
13/// inherited so failures are still diagnosable.
14pub fn git(repo: &Path, args: &[&str]) -> bool {
15    Command::new("git")
16        .arg("-C")
17        .arg(repo)
18        .args(args)
19        .stdout(Stdio::null())
20        .status()
21        .map(|s| s.success())
22        .unwrap_or(false)
23}
24
25/// A git invocation that failed, couldn't spawn, or printed non-UTF8.
26#[derive(Debug, thiserror::Error)]
27pub enum Error {
28    /// Non-zero exit — stderr captured for the message.
29    #[error("git {}: {}", args, stderr)]
30    Git {
31        /// The argv after `git -C <repo>` (for context).
32        args: String,
33        /// Trimmed stderr.
34        stderr: String,
35    },
36    /// The `git` binary couldn't be spawned.
37    #[error("git spawn: {0}")]
38    Spawn(#[from] std::io::Error),
39    /// stdout wasn't valid UTF-8.
40    #[error("git output was not utf-8")]
41    Utf8,
42}
43
44/// `git -C <repo> <args>` → trimmed stdout. stdin is `/dev/null` —
45/// plumbing commands that read stdin (`mktree`) get an immediate
46/// EOF instead of blocking on the caller's terminal.
47pub fn git_out(repo: &Path, args: &[&str]) -> Result<String, Error> {
48    let out = Command::new("git")
49        .arg("-C")
50        .arg(repo)
51        .args(args)
52        .stdin(Stdio::null())
53        .output()?;
54    if !out.status.success() {
55        return Err(Error::Git {
56            args: args.join(" "),
57            stderr: String::from_utf8_lossy(&out.stderr).trim().to_string(),
58        });
59    }
60    String::from_utf8(out.stdout)
61        .map(|s| s.trim().to_string())
62        .map_err(|_| Error::Utf8)
63}

65/// `git rev-parse --show-toplevel` — the worktree root containing
66/// `start` (`Err` outside a repo).
67pub fn repo_root(start: &Path) -> Result<PathBuf, Error> {
68    git_out(start, &["rev-parse", "--show-toplevel"]).map(PathBuf::from)
69}

76/// `refs/heads/<name>` exists.
77pub fn branch_exists(repo: &Path, name: &str) -> bool {
78    git(
79        repo,
80        &[
81            "show-ref",
82            "--verify",
83            "--quiet",
84            &format!("refs/heads/{name}"),
85        ],
86    )
87}

89/// Create `name` as an orphan branch via plumbing — `mktree` an
90/// empty tree, `commit-tree` a root commit, `update-ref` the branch.
91/// Unlike `git checkout --orphan` this never disturbs the index or
92/// working tree of the current checkout.
93pub fn create_orphan_branch(repo: &Path, name: &str, msg: &str) -> Result<(), Error> {
94    let tree = git_out(repo, &["mktree"])?;
95    let commit = git_out(repo, &["commit-tree", &tree, "-m", msg])?;
96    git_out(
97        repo,
98        &["update-ref", &format!("refs/heads/{name}"), &commit],
99    )?;
100    Ok(())
101}

103/// The branch a worktree dir has checked out (`None` when detached
104/// or not a worktree).
105pub fn worktree_branch(dir: &Path) -> Option<String> {
106    git_out(dir, &["rev-parse", "--abbrev-ref", "HEAD"])
107        .ok()
108        .filter(|b| b != "HEAD" && !b.is_empty())
109}

111/// The main (first) worktree of the repo `dir` belongs to — i.e.
112/// the repo root when `dir` is a linked worktree like `project/`.
113/// Derived from `--git-common-dir` (`<main>/.git` → `<main>`).
114pub fn main_worktree(dir: &Path) -> Option<PathBuf> {
115    let common = git_out(
116        dir,
117        &["rev-parse", "--path-format=absolute", "--git-common-dir"],
118    )
119    .ok()?;
120    let common = PathBuf::from(common);
121    if common.file_name().is_some_and(|n| n == ".git") {
122        common.parent().map(Path::to_path_buf)
123    } else {
124        // bare repo or worktree-of-worktree — no checkout root
125        None
126    }
127}

129/// Uncommitted/untracked changes present (`status --porcelain`

131pub fn has_changes(dir: &Path) -> bool {

133        .map(|s| !s.is_empty())
134        .unwrap_or(false)

190/// `git branch <name> <start>` — create a branch without switching.

197/// `git add -A && git commit --allow-empty -m <msg>` — the
198/// auto-commit-per-verb path writes files first, then this.
199/// `--allow-empty` keeps no-op verbs honest in the audit trail.
200pub fn commit_all(repo: &Path, msg: &str) -> bool {
201    git(repo, &["add", "-A"]) && git(repo, &["commit", "--allow-empty", "-m", msg])
202}

222/// `git rev-parse --verify HEAD` — the commit a worktree/repo is
223/// on, `None` when the branch is unborn. Index freshness is
224/// `stored head == head_commit(worktree)`.
225pub fn head_commit(dir: &Path) -> Option<String> {
226    git_out(dir, &["rev-parse", "--verify", "HEAD"]).ok()
227}

229/// One `git log` line, parsed.
230#[derive(Debug, Clone, PartialEq, Eq)]
231pub struct LogEntry {
232    /// Full commit hash.
233    pub hash: String,
234    /// Author date, `YYYY-MM-DD` (`%as`).
235    pub date: String,
236    /// Subject line.
237    pub subject: String,
238}
239
240/// `git log` newest-first — `%H %as %s` over a NUL separator so
241/// subjects are safe. `limit` caps the walk (`git log -n`). An
242/// unborn branch yields an empty history, not an error.
243pub fn log(dir: &Path, limit: Option<usize>) -> Result<Vec<LogEntry>, Error> {
244    if !has_commit(dir, "HEAD") {
245        return Ok(Vec::new());
246    }
247    let mut args = vec!["log", "--format=%H%x00%as%x00%s"];
248    let n;
249    if let Some(k) = limit {
250        n = format!("-{k}");
251        args.push(&n);
252    }
253    let out = git_out(dir, &args)?;
254    Ok(out
255        .lines()
256        .filter_map(|line| {
257            let mut parts = line.split('\0');
258            Some(LogEntry {
259                hash: parts.next()?.to_string(),
260                date: parts.next()?.to_string(),
261                subject: parts.next().unwrap_or_default().to_string(),
262            })
263        })
264        .collect())
265}
