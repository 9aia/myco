1//! The item-line grammar shared by `TODO.md` (`- [ ] T-012 …`) and
2//! `BACKLOG.md` (`- T-034 …`):
3//!
4//! ```text
5//! <id> <title> [· status:<s>] [· pri:<p>] [· owner:<a>]
6//!     [· created:<date>] [· #<label>]… [· blocks:<id>[,<id>]…]
7//!     [· links:<id>[,<id>]…] [· epic:<name>] [· ms:<name>]
8//!     [· note:<text>]
9//! ```
10//!
11//! Metadata tokens come after the title, `·`-separated, in the
12//! canonical order above. Titles may not contain `·`. Defaults:
13//! `status` = the doc's default (`open` for TODO, `deferred` for
14//! BACKLOG), `pri` = `med`. Unknown/malformed tokens are errors —
15//! `myco check` should catch typos, not swallow them.
16//!
17//! Items nest: a `-`/`[ ]` list indented under an item is its
18//! `children` (canonical emit = 2 spaces per level; arbitrary
19//! depth up to [`MAX_ITEM_DEPTH`], one level recommended). Child
20//! lines carry the same grammar — full items, not prose.
21

29/// Deepest item nesting the docs accept — the spec says arbitrary
30/// depth, but parse and emit walk the tree recursively, so past
31/// this bound it's an error, not a stack overflow. 64 is far past
32/// any real `TODO.md`/`BACKLOG.md` nesting.
33pub(crate) const MAX_ITEM_DEPTH: usize = 64;
34
35/// Per-document defaults for line parse/emit.
36pub(crate) struct LineSpec {
37    /// Kind ids must have (tasks only — `T-NNN`).
38    pub kind: ItemKind,
39    /// Status when no `status:` token is present.
40    pub default_status: Status,
41    /// Which file we're in — for emit errors.
42    pub file: &'static str,
43}
44
45fn bad(msg: impl Into<String>) -> String {
46    msg.into()
47}

49/// A list item's shape: its `Paragraph` head line plus at most one
50/// nested unordered `List` (the item's children). Anything else —
51/// a second paragraph, an ordered list, a second nested list —
52/// breaks the grammar.
53pub(crate) fn item_parts(
54    li: &markdown::mdast::ListItem,
55) -> Result<(&Node, Option<&markdown::mdast::List>), String> {
56    let mut children = li.children.iter();
57    let Some(head @ Node::Paragraph(_)) = children.next() else {
58        return Err(bad("item must be a single line"));
59    };
60    let mut nested = None;
61    for c in children {
62        match c {
63            Node::List(l) if !l.ordered && nested.is_none() => nested = Some(l),
64            _ => {
65                return Err(bad(
66                    "unexpected block under item — only a nested `-` list is allowed",
67                ))
68            }
69        }
70    }
71    Ok((head, nested))
72}

74/// Parse the text inside a list item (after `- [ ] ` / `- `).
75pub(crate) fn parse_text(text: &str, spec: &LineSpec) -> Result<Item, String> {
76    if text.contains('\n') {
77        return Err(bad("item must be a single line"));
78    }
79    let mut parts = text.split('·');
80    let head = parts.next().unwrap_or_default().trim();
81    let mut head_parts = head.splitn(2, char::is_whitespace);
82    let id_str = head_parts
83        .next()
84        .filter(|s| !s.is_empty())
85        .ok_or_else(|| bad("expected item id (T-NNN)"))?;
86    let id: ItemId = id_str.parse().map_err(|e| format!("{e}"))?;
87    if id.kind() != spec.kind {
88        return Err(bad(format!(
89            "wrong item kind {id} — {file} holds {} items only",
90            spec.kind.prefix(),
91            file = spec.file
92        )));
93    }
94    let title = head_parts.next().unwrap_or("").trim().to_string();
95
96    let mut status = None;
97    let mut priority = None;
98    let mut owner = None;
99    let mut created = None;
100    let mut blocks_seen = false;
101    let mut links_seen = false;
102    let mut epic = None;
103    let mut milestone = None;
104    let mut note = None;
105    let mut item = Item::new(id, title);
106
107    for raw in parts {
108        let tok = raw.trim();
109        if tok.is_empty() {
110            return Err(bad("empty metadata token — double `·` or trailing `·`"));
111        }
112        if let Some(label) = tok.strip_prefix('#') {
113            if !is_valid_label(label) {
114                return Err(bad(format!("invalid label #{label:?}")));
115            }
116            item.labels.insert(label.to_string());
117            continue;
118        }
119        let (key, value) = tok.split_once(':').ok_or_else(|| {
120            bad(format!(
121                "malformed metadata token {tok:?} — expected key:value or #label"
122            ))
123        })?;
124        let value = value.trim();
125        if value.is_empty() {
126            return Err(bad(format!("empty value in token {tok:?}")));
127        }
128        match key {
129            "status" => {
130                if status.is_some() {
131                    return Err(bad("duplicate status: token"));
132                }
133                status = Some(value.parse::<Status>().map_err(|e| format!("{e}"))?);
134            }
135            "pri" => {
136                if priority.is_some() {
137                    return Err(bad("duplicate pri: token"));
138                }
139                priority = Some(value.parse::<Priority>().map_err(|e| format!("{e}"))?);
140            }
141            "owner" => {
142                if owner.is_some() {
143                    return Err(bad("duplicate owner: token"));
144                }
145                owner = Some(Actor::new(value).map_err(|e| format!("{e}"))?);
146            }
147            "created" => {
148                if created.is_some() {
149                    return Err(bad("duplicate created: token"));
150                }
151                created = Some(date::parse(value).map_err(|e| format!("bad created: date — {e}"))?);
152            }
153            "blocks" => {
154                if blocks_seen {
155                    return Err(bad("duplicate blocks: token"));
156                }
157                blocks_seen = true;
158                for bid in value.split(',') {
159                    let bid = bid.trim();
160                    if bid.is_empty() {
161                        return Err(bad("empty id in blocks: list"));
162                    }
163                    item.blocks
164                        .insert(bid.parse::<ItemId>().map_err(|e| format!("{e}"))?);
165                }
166            }
167            "links" => {
168                if links_seen {
169                    return Err(bad("duplicate links: token"));
170                }
171                links_seen = true;
172                for lid in value.split(',') {
173                    let lid = lid.trim();
174                    if lid.is_empty() {
175                        return Err(bad("empty id in links: list"));
176                    }
177                    item.links
178                        .insert(lid.parse::<ItemId>().map_err(|e| format!("{e}"))?);
179                }
180            }
181            "epic" => {
182                if epic.is_some() {
183                    return Err(bad("duplicate epic: token"));
184                }
185                if !is_valid_group_name(value) {
186                    return Err(bad(format!("invalid epic name {value:?}")));
187                }
188                epic = Some(value.to_string());
189            }
190            "ms" => {
191                if milestone.is_some() {
192                    return Err(bad("duplicate ms: token"));
193                }
194                if !is_valid_group_name(value) {
195                    return Err(bad(format!("invalid milestone name {value:?}")));
196                }
197                milestone = Some(value.to_string());
198            }
199            "note" => {
200                if note.is_some() {
201                    return Err(bad("duplicate note: token"));
202                }
203                note = Some(value.to_string());
204            }
205            other => {
206                return Err(bad(format!(
207                    "unknown metadata key {other:?} — expected status|pri|owner|created|blocks|links|epic|ms|note or #label"
208                )));
209            }
210        }
211    }
212
213    if item.blocks.contains(&id) {
214        return Err(bad(format!("{id} can't block itself")));
215    }
216    if item.links.contains(&id) {
217        return Err(bad(format!("{id} can't link itself")));
218    }
219    item.status = status.unwrap_or(spec.default_status);
220    item.priority = priority.unwrap_or_default();
221    item.owner = owner;
222    item.created = created;
223    item.epic = epic;
224    item.milestone = milestone;
225    item.note = note;
226    Ok(item)
227}

229/// Validate what the line grammar can't express. Shared by TODO and
230/// BACKLOG emit — the caller adds status rules on top.
231pub(crate) fn check_item(item: &Item, file: &'static str) -> Result<(), EmitError> {
232    let err = |m: &str| EmitError::new(file, format!("{m} ({})", item.id));
233    if item.title != item.title.trim() {
234        return Err(err("untrimmed title"));
235    }
236    if item.title.contains('·') {
237        return Err(err("title contains `·`"));
238    }
239    if item.title.contains('\n') || item.title.contains('\r') {
240        return Err(err("title contains a newline"));
241    }
242    for label in &item.labels {
243        if !is_valid_label(label) {
244            return Err(err(&format!("invalid label #{label:?}")));
245        }
246    }
247    for (what, name) in [
248        ("epic", item.epic.as_deref()),
249        ("milestone", item.milestone.as_deref()),
250    ] {
251        if let Some(name) = name {
252            if !is_valid_group_name(name) {
253                return Err(err(&format!("invalid {what} name {name:?}")));
254            }
255        }
256    }
257    if let Some(note) = &item.note {
258        if note.is_empty() || note != note.trim() {
259            return Err(err("untrimmed/empty note"));
260        }
261        if note.contains('·') || note.contains('\n') || note.contains('\r') {
262            return Err(err("note contains `·` or a newline"));
263        }
264    }
265    if item.blocks.contains(&item.id)
266        || item.links.contains(&item.id)
267        || item.supersedes == Some(item.id)
268        || item.superseded_by == Some(item.id)
269    {
270        return Err(err(
271            "self-reference — can't block, link, or supersede itself",
272        ));
273    }
274    Ok(())
275}

277/// The line text (`<id> <title> · <tokens>`) — checkbox/marker is
278/// the caller's job.
279pub(crate) fn text(item: &Item, spec: &LineSpec) -> Result<String, EmitError> {
280    check_item(item, spec.file)?;
281    let mut s = item.id.to_string();
282    if !item.title.is_empty() {
283        s.push(' ');
284        s.push_str(&item.title);
285    }
286    let mut tokens: Vec<String> = Vec::new();
287    if item.status != spec.default_status {
288        tokens.push(format!("status:{}", item.status.token()));
289    }
290    if item.priority != Priority::Medium {
291        tokens.push(format!("pri:{}", item.priority.token()));
292    }
293    if let Some(owner) = &item.owner {
294        tokens.push(format!("owner:{owner}"));
295    }
296    if let Some(d) = item.created {
297        tokens.push(format!("created:{}", date::render(d)));
298    }
299    for label in &item.labels {
300        tokens.push(format!("#{label}"));
301    }
302    if !item.blocks.is_empty() {
303        let ids: Vec<String> = item.blocks.iter().map(ItemId::to_string).collect();
304        tokens.push(format!("blocks:{}", ids.join(",")));
305    }
306    if !item.links.is_empty() {
307        let ids: Vec<String> = item.links.iter().map(ItemId::to_string).collect();
308        tokens.push(format!("links:{}", ids.join(",")));
309    }
310    if let Some(epic) = &item.epic {
311        tokens.push(format!("epic:{epic}"));
312    }
313    if let Some(ms) = &item.milestone {
314        tokens.push(format!("ms:{ms}"));
315    }
316    if let Some(note) = &item.note {
317        tokens.push(format!("note:{note}"));
318    }
319    for t in tokens {
320        s.push_str(" · ");
321        s.push_str(&t);
322    }
323    Ok(s)
324}

341    #[test]
342    fn parse_full_line() {
343        let i = item("T-012 Write the thing · status:in-progress · pri:high · owner:devin · created:2026-02-09 · #core · #store · blocks:T-010, I-002 · epic:v2 · ms:1.0 · note:from src/x.rs:9");

428    #[test]
429    fn emit_canonical_order() {
430        let mut i = Item::new(ItemId::new(ItemKind::Task, 12), "Write the thing")
431            .status(Status::InProgress)
432            .priority(Priority::High)
433            .owner(Actor::new("devin").unwrap())
434            .created(date::parse("2026-02-09").unwrap())
435            .label("store")
436            .label("core")
437            .blocks(ItemId::new(ItemKind::Issue, 2))
438            .blocks(ItemId::new(ItemKind::Task, 10))
439            .epic("v2")
440            .milestone("1.0")
441            .note("swept from src/x.rs:9");
442        let s = text(&i, &TODO).unwrap();
443        assert_eq!(
444            s,
445            "T-012 Write the thing · status:in-progress · pri:high · owner:devin · created:2026-02-09 · #core · #store · blocks:T-010,I-002 · epic:v2 · ms:1.0 · note:swept from src/x.rs:9"
446        );
447        // round trip
448        assert_eq!(parse_text(&s, &TODO).unwrap(), i);
449        // defaults emit nothing
450        i.status = Status::Open;
451        i.priority = Priority::Medium;
452        let s2 = text(&i, &TODO).unwrap();
453        assert!(!s2.contains("status:") && !s2.contains("pri:"));
454        assert_eq!(parse_text(&s2, &TODO).unwrap(), i);
455    }
