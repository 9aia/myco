1//! The item-line grammar shared by `TODO.md` (`- [ ] T-012 …`) and
2//! `BACKLOG.md` (`- T-034 …`):
3//!
4//! ```text
5//! <id> <title> [· status:<s>] [· pri:<p>] [· owner:<a>]
6//!     [· created:<date>] [· #<label>]… [· blocks:<id>[,<id>]…]
7//!     [· epic:<name>] [· ms:<name>] [· note:<text>]
8//! ```
9//!
10//! Metadata tokens come after the title, `·`-separated, in the
11//! canonical order above. Titles may not contain `·`. Defaults:
12//! `status` = the doc's default (`open` for TODO, `deferred` for
13//! BACKLOG), `pri` = `med`. Unknown/malformed tokens are errors —
14//! `myco check` should catch typos, not swallow them.
15//!
16//! Items nest: a `-`/`[ ]` list indented under an item is its
17//! `children` (canonical emit = 2 spaces per level; arbitrary
18//! depth, one level recommended). Child lines carry the same
19//! grammar — full items, not prose.

28/// Per-document defaults for line parse/emit.
29pub(crate) struct LineSpec {
30    /// Kind ids must have (tasks only — `T-NNN`).
31    pub kind: ItemKind,
32    /// Status when no `status:` token is present.
33    pub default_status: Status,
34    /// Which file we're in — for emit errors.
35    pub file: &'static str,
36}
37
38fn bad(msg: impl Into<String>) -> String {
39    msg.into()
40}
41
42/// A list item's shape: its `Paragraph` head line plus at most one
43/// nested unordered `List` (the item's children). Anything else —
44/// a second paragraph, an ordered list, a second nested list —
45/// breaks the grammar.
46pub(crate) fn item_parts(
47    li: &markdown::mdast::ListItem,
48) -> Result<(&Node, Option<&markdown::mdast::List>), String> {
49    let mut children = li.children.iter();
50    let Some(head @ Node::Paragraph(_)) = children.next() else {
51        return Err(bad("item must be a single line"));
52    };
53    let mut nested = None;
54    for c in children {
55        match c {
56            Node::List(l) if !l.ordered && nested.is_none() => nested = Some(l),
57            _ => {
58                return Err(bad(
59                    "unexpected block under item — only a nested `-` list is allowed",
60                ))
61            }
62        }
63    }
64    Ok((head, nested))
65}

67/// Parse the text inside a list item (after `- [ ] ` / `- `).
68pub(crate) fn parse_text(text: &str, spec: &LineSpec) -> Result<Item, String> {
69    if text.contains('\n') {
70        return Err(bad("item must be a single line"));
71    }
72    let mut parts = text.split('·');
73    let head = parts.next().unwrap_or_default().trim();
74    let mut head_parts = head.splitn(2, char::is_whitespace);
75    let id_str = head_parts
76        .next()
77        .filter(|s| !s.is_empty())
78        .ok_or_else(|| bad("expected item id (T-NNN)"))?;
79    let id: ItemId = id_str.parse().map_err(|e| format!("{e}"))?;
80    if id.kind() != spec.kind {
81        return Err(bad(format!(
82            "wrong item kind {id} — {file} holds {} items only",
83            spec.kind.prefix(),
84            file = spec.file
85        )));
86    }
87    let title = head_parts.next().unwrap_or("").trim().to_string();
88
89    let mut status = None;
90    let mut priority = None;
91    let mut owner = None;
92    let mut created = None;
93    let mut blocks_seen = false;
94    let mut epic = None;
95    let mut milestone = None;
96    let mut note = None;
97    let mut item = Item::new(id, title);
98
99    for raw in parts {
100        let tok = raw.trim();
101        if tok.is_empty() {
102            return Err(bad("empty metadata token — double `·` or trailing `·`"));
103        }
104        if let Some(label) = tok.strip_prefix('#') {
105            if !is_valid_label(label) {
106                return Err(bad(format!("invalid label #{label:?}")));
107            }
108            item.labels.insert(label.to_string());
109            continue;
110        }
111        let (key, value) = tok.split_once(':').ok_or_else(|| {
112            bad(format!(
113                "malformed metadata token {tok:?} — expected key:value or #label"
114            ))
115        })?;
116        let value = value.trim();
117        if value.is_empty() {
118            return Err(bad(format!("empty value in token {tok:?}")));
119        }
120        match key {
121            "status" => {
122                if status.is_some() {
123                    return Err(bad("duplicate status: token"));
124                }
125                status = Some(value.parse::<Status>().map_err(|e| format!("{e}"))?);
126            }
127            "pri" => {
128                if priority.is_some() {
129                    return Err(bad("duplicate pri: token"));
130                }
131                priority = Some(value.parse::<Priority>().map_err(|e| format!("{e}"))?);
132            }
133            "owner" => {
134                if owner.is_some() {
135                    return Err(bad("duplicate owner: token"));
136                }
137                owner = Some(Actor::new(value).map_err(|e| format!("{e}"))?);
138            }
139            "created" => {
140                if created.is_some() {
141                    return Err(bad("duplicate created: token"));
142                }
143                created = Some(date::parse(value).map_err(|e| format!("bad created: date — {e}"))?);
144            }
145            "blocks" => {
146                if blocks_seen {
147                    return Err(bad("duplicate blocks: token"));
148                }
149                blocks_seen = true;
150                for bid in value.split(',') {
151                    let bid = bid.trim();
152                    if bid.is_empty() {
153                        return Err(bad("empty id in blocks: list"));
154                    }
155                    item.blocks
156                        .insert(bid.parse::<ItemId>().map_err(|e| format!("{e}"))?);
157                }
158            }
159            "epic" => {
160                if epic.is_some() {
161                    return Err(bad("duplicate epic: token"));
162                }
163                if !is_valid_group_name(value) {
164                    return Err(bad(format!("invalid epic name {value:?}")));
165                }
166                epic = Some(value.to_string());
167            }
168            "ms" => {
169                if milestone.is_some() {
170                    return Err(bad("duplicate ms: token"));
171                }
172                if !is_valid_group_name(value) {
173                    return Err(bad(format!("invalid milestone name {value:?}")));
174                }
175                milestone = Some(value.to_string());
176            }
177            "note" => {
178                if note.is_some() {
179                    return Err(bad("duplicate note: token"));
180                }
181                note = Some(value.to_string());
182            }
183            other => {
184                return Err(bad(format!(
185                    "unknown metadata key {other:?} — expected status|pri|owner|created|blocks|epic|ms|note or #label"
186                )));
187            }
188        }
189    }
190
191    item.status = status.unwrap_or(spec.default_status);
192    item.priority = priority.unwrap_or_default();
193    item.owner = owner;
194    item.created = created;
195    item.epic = epic;
196    item.milestone = milestone;
197    item.note = note;
198    Ok(item)
199}

201/// Validate what the line grammar can't express. Shared by TODO and
202/// BACKLOG emit — the caller adds status rules on top.
203pub(crate) fn check_item(item: &Item, file: &'static str) -> Result<(), EmitError> {
204    let err = |m: &str| EmitError::new(file, format!("{m} ({})", item.id));
205    if item.title != item.title.trim() {
206        return Err(err("untrimmed title"));
207    }
208    if item.title.contains('·') {
209        return Err(err("title contains `·`"));
210    }
211    if item.title.contains('\n') || item.title.contains('\r') {
212        return Err(err("title contains a newline"));
213    }
214    for label in &item.labels {
215        if !is_valid_label(label) {
216            return Err(err(&format!("invalid label #{label:?}")));
217        }
218    }
219    for (what, name) in [
220        ("epic", item.epic.as_deref()),
221        ("milestone", item.milestone.as_deref()),
222    ] {
223        if let Some(name) = name {
224            if !is_valid_group_name(name) {
225                return Err(err(&format!("invalid {what} name {name:?}")));
226            }
227        }
228    }
229    if let Some(note) = &item.note {
230        if note.is_empty() || note != note.trim() {
231            return Err(err("untrimmed/empty note"));
232        }
233        if note.contains('·') || note.contains('\n') || note.contains('\r') {
234            return Err(err("note contains `·` or a newline"));
235        }
236    }
237    Ok(())
238}

240/// The line text (`<id> <title> · <tokens>`) — checkbox/marker is
241/// the caller's job.
242pub(crate) fn text(item: &Item, spec: &LineSpec) -> Result<String, EmitError> {
243    check_item(item, spec.file)?;
244    let mut s = item.id.to_string();
245    if !item.title.is_empty() {
246        s.push(' ');
247        s.push_str(&item.title);
248    }
249    let mut tokens: Vec<String> = Vec::new();
250    if item.status != spec.default_status {
251        tokens.push(format!("status:{}", item.status.token()));
252    }
253    if item.priority != Priority::Medium {
254        tokens.push(format!("pri:{}", item.priority.token()));
255    }
256    if let Some(owner) = &item.owner {
257        tokens.push(format!("owner:{owner}"));
258    }
259    if let Some(d) = item.created {
260        tokens.push(format!("created:{}", date::render(d)));
261    }
262    for label in &item.labels {
263        tokens.push(format!("#{label}"));
264    }
265    if !item.blocks.is_empty() {
266        let ids: Vec<String> = item.blocks.iter().map(ItemId::to_string).collect();
267        tokens.push(format!("blocks:{}", ids.join(",")));
268    }
269    if let Some(epic) = &item.epic {
270        tokens.push(format!("epic:{epic}"));
271    }
272    if let Some(ms) = &item.milestone {
273        tokens.push(format!("ms:{ms}"));
274    }
275    if let Some(note) = &item.note {
276        tokens.push(format!("note:{note}"));
277    }
278    for t in tokens {
279        s.push_str(" · ");
280        s.push_str(&t);
281    }
282    Ok(s)
283}

300    #[test]
301    fn parse_full_line() {
302        let i = item("T-012 Write the thing · status:in-progress · pri:high · owner:devin · created:2026-02-09 · #core · #store · blocks:T-010, I-002 · epic:v2 · ms:1.0 · note:from src/x.rs:9");
303        assert_eq!(i.id, ItemId::new(ItemKind::Task, 12));
304        assert_eq!(i.title, "Write the thing");
305        assert_eq!(i.status, Status::InProgress);
306        assert_eq!(i.priority, Priority::High);
307        assert_eq!(i.owner.as_ref().unwrap().as_str(), "devin");
308        assert_eq!(i.created, Some(date::parse("2026-02-09").unwrap()));
309        assert!(i.labels.contains("core") && i.labels.contains("store"));
310        assert!(i.blocks.contains(&ItemId::new(ItemKind::Task, 10)));
311        assert!(i.blocks.contains(&ItemId::new(ItemKind::Issue, 2)));
312        assert_eq!(i.epic.as_deref(), Some("v2"));
313        assert_eq!(i.milestone.as_deref(), Some("1.0"));
314        assert_eq!(i.note.as_deref(), Some("from src/x.rs:9"));
315    }

369    #[test]
370    fn emit_canonical_order() {
371        let mut i = Item::new(ItemId::new(ItemKind::Task, 12), "Write the thing")
372            .status(Status::InProgress)
373            .priority(Priority::High)
374            .owner(Actor::new("devin").unwrap())
375            .created(date::parse("2026-02-09").unwrap())
376            .label("store")
377            .label("core")
378            .blocks(ItemId::new(ItemKind::Issue, 2))
379            .blocks(ItemId::new(ItemKind::Task, 10))
380            .epic("v2")
381            .milestone("1.0")
382            .note("swept from src/x.rs:9");
383        let s = text(&i, &TODO).unwrap();
384        assert_eq!(
385            s,
386            "T-012 Write the thing · status:in-progress · pri:high · owner:devin · created:2026-02-09 · #core · #store · blocks:T-010,I-002 · epic:v2 · ms:1.0 · note:swept from src/x.rs:9"
387        );
388        // round trip
389        assert_eq!(parse_text(&s, &TODO).unwrap(), i);
390        // defaults emit nothing
391        i.status = Status::Open;
392        i.priority = Priority::Medium;
393        let s2 = text(&i, &TODO).unwrap();
394        assert!(!s2.contains("status:") && !s2.contains("pri:"));
395        assert_eq!(parse_text(&s2, &TODO).unwrap(), i);
396    }
