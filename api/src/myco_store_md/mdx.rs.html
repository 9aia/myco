1//! mdast helpers — the `markdown`/`mdast_util_to_markdown` plumbing
2//! shared by the document parsers and emitters.
3
4use markdown::mdast::{Node, Paragraph, Text};
5use markdown::{to_mdast, Constructs, ParseOptions};
6
7/// `ParseOptions`: CommonMark + GFM task-list items only — autolinks,
8/// tables etc. stay off so item lines stay plain.
9pub(crate) fn parse_options() -> ParseOptions {
10    ParseOptions {
11        constructs: Constructs {
12            gfm_task_list_item: true,
13            ..Constructs::default()
14        },
15        ..ParseOptions::default()
16    }
17}
18
19/// `to_mdast` with myco's options; the error is flattened to text.
20pub(crate) fn parse(src: &str) -> Result<Node, String> {
21    to_mdast(src, &parse_options()).map_err(|m| m.to_string())
22}
23
24/// Emit options — canonical style: `-` bullets, `*` emphasis, ATX
25/// headings, fenced code with backticks.
26pub(crate) fn emit_options() -> mdast_util_to_markdown::Options {
27    mdast_util_to_markdown::Options {
28        bullet: '-',
29        ..Default::default()
30    }
31}

33/// Serialize one inline string — escapes markdown specials (`*`,
34/// `[`, leading `#`, …) so hand-emitted lines re-parse to the same
35/// text. Uses `mdast_util_to_markdown` but verifies the result by
36/// re-parsing: 0.0.2's escaper corrupts sequences like `· #_x`
37/// (byte/char index drift — `_` is eaten and the next char is
38/// entity-encoded). On divergence it falls back to [`escape`].
39pub(crate) fn inline(text: &str) -> String {
40    let node = Node::Paragraph(Paragraph {
41        children: vec![Node::Text(Text {
42            value: text.to_string(),
43            position: None,
44        })],
45        position: None,
46    });
47    let via_lib = mdast_util_to_markdown::to_markdown_with_options(&node, &emit_options())
48        .unwrap_or_default();
49    let via_lib = via_lib.trim_end_matches('\n');
50    let verified = parse(via_lib)
51        .ok()
52        .as_ref()
53        .and_then(Node::children)
54        .and_then(|c| c.first())
55        .and_then(single_text)
56        == Some(text);
57    if verified {
58        via_lib.to_string()
59    } else {
60        escape(text)
61    }
62}

64/// Minimal CommonMark escaper — `\` before every ASCII punctuation
65/// that can open an inline construct anywhere (always-safe: `\x`
66/// for ASCII punctuation re-parses to literal `x`), plus the
67/// block-marker chars that only matter at text start (`#` heading,
68/// `+`/`-`/`.` list, `>` quote, `=` setext). `·` is *not* escaped —
69/// it's banned from titles/values by the grammar.
70pub(crate) fn escape(text: &str) -> String {
71    const ANYWHERE: &[char] = &['\\', '`', '*', '_', '[', ']', '<', '&', '!'];
72    const LINE_START: &[char] = &['#', '+', '-', '.', '>', '='];
73    let mut out = String::with_capacity(text.len() + 8);
74    for (i, c) in text.chars().enumerate() {
75        if ANYWHERE.contains(&c) || (i == 0 && LINE_START.contains(&c)) {
76            out.push('\\');
77        }
78        out.push(c);
79    }
80    out
81}

83/// The text of a node that must be exactly one `Text` child —
84/// heading names and item lines are plain by spec.
85pub(crate) fn single_text(node: &Node) -> Option<&str> {
86    let children = node.children()?;
87    if children.len() != 1 {
88        return None;
89    }
90    match &children[0] {
91        Node::Text(t) => Some(t.value.as_str()),
92        _ => None,
93    }
94}

96/// Normalize CRLF → LF up front so spans match the stored source.
97pub(crate) fn normalize(src: &str) -> String {
98    if src.contains('\r') {
99        src.replace("\r\n", "\n").replace('\r', "\n")
100    } else {
101        src.to_string()
102    }
103}

105/// Byte ranges in `src` covered by code — fenced/indented blocks
106/// and `inline` spans — so prose scans (`myco check`'s
107/// `dangling-ref`) can whitelist them. Sorted, non-overlapping.
108/// Empty when `src` doesn't parse (callers scan unwhitelisted —
109/// a stale ref inside a broken fence is still worth flagging).
110pub fn code_spans(src: &str) -> Vec<(usize, usize)> {
111    fn walk(node: &Node, out: &mut Vec<(usize, usize)>) {
112        if matches!(node, Node::Code(_) | Node::InlineCode(_)) {
113            if let Some(p) = node.position() {
114                out.push((p.start.offset, p.end.offset));
115            }
116            return; // code nodes can't nest scannable prose
117        }
118        if let Some(children) = node.children() {
119            for c in children {
120                walk(c, out);
121            }
122        }
123    }
124    let mut out = Vec::new();
125    if let Ok(ast) = parse(src) {
126        walk(&ast, &mut out);
127    }
128    out
129}

131#[cfg(test)]
132#[allow(clippy::unwrap_used)]
133mod tests {
134    use super::*;
135    use markdown::mdast::Heading;

137    #[test]
138    fn inline_escapes_specials() {
139        assert_eq!(inline("plain"), "plain");
140        assert_eq!(inline("a *b* c"), "a \\*b\\* c");
141        assert_eq!(inline("#hash"), "\\#hash");
142        assert_eq!(
143            inline("T-1 t · #label · status:open"),
144            "T-1 t · #label · status:open"
145        );
146        assert_eq!(inline(""), "");
147    }
148
149    #[test]
150    fn inline_falls_back_on_lib_corruption() {
151        // mdast_util_to_markdown 0.0.2 mangles `· #_x` — verify our
152        // verifier catches it and the fallback round-trips.
153        let out = inline("T-001 · #_a-a");
154        assert_eq!(out, "T-001 · #\\_a-a");
155        let ast = parse(&format!("- {out}\n")).unwrap();
156        let list = &ast.children().unwrap()[0];
157        assert!(matches!(list, Node::List(_)));
158        let li = &list.children().unwrap()[0];
159        assert!(matches!(li, Node::ListItem(_)));
160        let para = &li.children().unwrap()[0];
161        assert_eq!(single_text(para), Some("T-001 · #_a-a"));
162    }
163
164    #[test]
165    fn escape_covers_punct() {
166        assert_eq!(
167            escape("a*b_c`d[e]f<g&h!i"),
168            "a\\*b\\_c\\`d\\[e\\]f\\<g\\&h\\!i"
169        );
170        assert_eq!(escape("#lead"), "\\#lead");
171        assert_eq!(escape("mid #dash"), "mid #dash");
172        assert_eq!(escape("plain, text: ok"), "plain, text: ok");
173    }
174
175    #[test]
176    fn single_text_strict() {
177        let h = Node::Heading(Heading {
178            depth: 2,
179            children: vec![Node::Text(Text {
180                value: "sec".into(),
181                position: None,
182            })],
183            position: None,
184        });
185        assert_eq!(single_text(&h), Some("sec"));
186        let empty = Node::Heading(Heading {
187            depth: 2,
188            children: vec![],
189            position: None,
190        });
191        assert_eq!(single_text(&empty), None);
192        let two = Node::Paragraph(Paragraph {
193            children: vec![
194                Node::Text(Text {
195                    value: "a".into(),
196                    position: None,
197                }),
198                Node::Text(Text {
199                    value: "b".into(),
200                    position: None,
201                }),
202            ],
203            position: None,
204        });
205        assert_eq!(single_text(&two), None);
206        // a single non-Text child is also rejected
207        let em = Node::Paragraph(Paragraph {
208            children: vec![Node::Emphasis(markdown::mdast::Emphasis {
209                children: vec![],
210                position: None,
211            })],
212            position: None,
213        });
214        assert_eq!(single_text(&em), None);
215        // root has children but no single text
216        let t = Node::Text(Text {
217            value: "x".into(),
218            position: None,
219        });
220        assert_eq!(single_text(&t), None);
221    }

236    #[test]
237    fn code_spans_cover_fences_and_inline() {
238        let src = "see `T-001` here\n\n```\nT-002 in a fence\n```\n\nplain T-003\n";
239        let spans = code_spans(src);
240        // inline code + the fenced block, nothing else
241        assert_eq!(spans.len(), 2, "{spans:?}");
242        let inline = &src[spans[0].0..spans[0].1];
243        assert_eq!(inline, "`T-001`", "{inline}");
244        let fence = &src[spans[1].0..spans[1].1];
245        assert!(fence.contains("T-002 in a fence"), "{fence}");
246        // a hand-built node without position is skipped silently —
247        // and unparseable input yields no spans rather than an error
248        assert!(code_spans("plain text only\n").is_empty());
249    }
