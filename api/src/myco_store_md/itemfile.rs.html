1//! Item files — one md file per issue/proposal/adr/plan:
2//!
3//! ```text
4//! issues/issue-NNNN-slug.md    proposals/proposal-NNNN-slug.md
5//! adrs/NNNN-slug.md            plans/pl-NNNN-slug.md
6//! ```
7//!
8//! `NNNN` = id number padded to 4 digits. TOML frontmatter in
9//! `+++` fences (id, title, status, created, owner, priority,
10//! labels, blocks, note, superseded_by, supersedes, epic,
11//! milestone — extendable: unknown keys are ignored on read for
12//! forward-compat); body = free-form markdown (verbatim; kind
13//! templates are `myco-template`'s job).
14

22/// The `project/` subdirectory a file-backed kind lives in —
23/// the spec's `dir` (`issues`, `spikes`); `doc`-storage kinds
24/// have no files (`None`).
25pub fn item_dir(spec: &KindSpec) -> Option<String> {
26    (spec.storage == Storage::Dir).then(|| spec.dir_name())
27}
28
29/// Filename prefix before the number — `issue`, `proposal`, `pl`,
30/// the spec's `file_prefix` for user kinds, or empty for ADRs
31/// (`adrs/0004-slug.md`).
32fn file_prefix(spec: &KindSpec) -> Option<String> {
33    (spec.storage == Storage::Dir).then(|| spec.file_prefix())
34}

36/// Canonical filename for an item — `issue-0003-memory-leak.md`,
37/// `0004-hybrid-store.md`. Slug = `slug::slugify(title)`,
38/// `"untitled"` when the title slugifies to nothing.
39pub fn item_filename(spec: &KindSpec, id: &ItemId, title: &str) -> Option<String> {
40    let prefix = file_prefix(spec)?;
41    let slug = {
42        let s = slug::slugify(title);
43        if s.is_empty() {
44            "untitled".to_string()
45        } else {
46            s
47        }
48    };
49    Some(if prefix.is_empty() {
50        format!("{:04}-{slug}.md", id.number())
51    } else {
52        format!("{prefix}-{:04}-{slug}.md", id.number())
53    })
54}

56/// `issues/issue-0003-x.md` → `I-003`; `None` when the filename
57/// doesn't fit the kind's pattern. The number must fit `u32` —
58/// the same bound [`ItemId`]'s own parse applies, so anything
59/// [`item_filename`] can emit parses back (id numbers are `u32`,
60/// i.e. up to 10 digits).
61pub fn parse_item_filename(spec: &KindSpec, filename: &str) -> Option<ItemId> {
62    let prefix = file_prefix(spec)?;
63    let stem = filename.strip_suffix(".md")?;
64    let rest = if prefix.is_empty() {
65        stem
66    } else {
67        stem.strip_prefix(prefix.as_str())?.strip_prefix('-')?
68    };
69    let (digits, slug) = rest.split_once('-')?;
70    if digits.is_empty() || !digits.bytes().all(|b| b.is_ascii_digit()) || slug.is_empty() {
71        return None;
72    }
73    let number: u32 = digits.parse().ok()?;
74    if number == 0 {
75        return None;
76    }
77    Some(ItemId::new(spec.id_kind(), number))
78}

80/// Deepest line the `frontmatter_id` probe will read — real
81/// frontmatter is ~10 keys; past this the `+++` block is
82/// certainly unclosed (or the file is pathological).
83const FRONTMATTER_PROBE_MAX: usize = 128;
84
85/// `id = "…"` pulled out of a *closed* `+++` frontmatter block —
86/// the cheap probe for when the strict parse isn't needed (or
87/// already failed). `None` for files with no `+++` fence, no
88/// closing fence within [`FRONTMATTER_PROBE_MAX`] lines, or no
89/// parseable `id` line — including frontmatter-less files that
90/// merely *borrow* an id number via their filename. The close
91/// requirement is what keeps a body's `id = "…"` (a code block,
92/// a quoted example) from posing as frontmatter.
93pub fn frontmatter_id(src: &str) -> Option<ItemId> {
94    let mut lines = src.lines();
95    if lines.next()?.trim() != "+++" {
96        return None;
97    }
98    let mut found = None;
99    for l in lines.take(FRONTMATTER_PROBE_MAX) {
100        let t = l.trim();
101        // the close ends the block — an id only counts when the
102        // block actually closed (else a body's `id = "…"` would
103        // pose as frontmatter)
104        if t == "+++" {
105            return found;
106        }
107        if found.is_none() {
108            if let Some(rest) = t.strip_prefix("id") {
109                if let Some(v) = rest.trim_start().strip_prefix('=') {
110                    if let Ok(id) = v.trim().trim_matches('"').parse::<ItemId>() {
111                        found = Some(id);
112                    }
113                }
114            }
115        }
116    }
117    None // window exhausted — never closed
118}

120/// Frontmatter DTO — stringly-typed so bad values point at the
121/// frontmatter block, not toml's internals.
122#[derive(Debug, Deserialize)]
123struct FrontmatterIn {
124    id: String,
125    title: String,
126    status: Status,
127    #[serde(default, deserialize_with = "de_opt_created")]
128    created: Option<String>,
129    owner: Option<String>,
130    #[serde(default, deserialize_with = "de_opt_priority")]
131    priority: Option<Priority>,
132    #[serde(default)]
133    labels: Vec<String>,
134    #[serde(default)]
135    blocks: Vec<String>,
136    note: Option<String>,
137    superseded_by: Option<String>,
138    supersedes: Option<String>,
139    epic: Option<String>,
140    milestone: Option<String>,
141}
142
143/// `priority` reads like the line grammar — `low|med|high|crit`
144/// and the full names emit writes — via [`Priority`]'s `FromStr`
145/// (lenient read, strict canonical write). Deriving `Priority`
146/// would accept full names only.
147fn de_opt_priority<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Option<Priority>, D::Error> {
148    Option::<String>::deserialize(d)?
149        .map(|s| s.parse().map_err(serde::de::Error::custom))
150        .transpose()
151}
152
153/// `created` accepts both the quoted string emit writes
154/// (`created = "2026-02-01"`) and a bare TOML date
155/// (`created = 2026-02-01`) — hand-edited files use the native
156/// form. Either way the value lands as text for [`date::parse`]
157/// to validate.
158fn de_opt_created<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Option<String>, D::Error> {
159    match Option::<toml::Value>::deserialize(d)? {
160        None => Ok(None),
161        Some(toml::Value::String(s)) => Ok(Some(s)),
162        Some(toml::Value::Datetime(dt)) => Ok(Some(dt.to_string())),
163        Some(v) => Err(serde::de::Error::custom(format!(
164            "expected a `YYYY-MM-DD` date, got {v}"
165        ))),
166    }
167}

174    status: Status,

182    #[serde(skip_serializing_if = "Vec::is_empty")]

196/// Parse an item file's source. `spec` declares the kind the
197/// directory the file was found in belongs to — an id whose
198/// prefix isn't the spec's (`P-002` inside `issues/`) is an
199/// error.
200pub fn parse_item_file(src: &str, spec: &KindSpec, path: &str) -> Result<ItemFile, ParseError> {
201    let src = mdx::normalize(src);
202    let at = |off: usize, len: usize, m: String| ParseError::at(path, src.clone(), off, len, m);
203    let whole = |m: &str| ParseError::whole_file(path, src.clone(), m);
204    let mut lines = src.split_inclusive('\n');
205    if lines.next().map(str::trim_end) != Some("+++") {
206        return Err(whole("expected `+++` TOML frontmatter fence on line 1"));
207    }
208    // find the closing `+++` line (byte offsets into src)
209    let fm_start = "+++\n".len();
210    let mut fm_end = None;
211    let mut off = fm_start;
212    for line in lines {
213        if line.trim_end() == "+++" {
214            fm_end = Some((off, off + line.len()));
215            break;
216        }
217        off += line.len();
218    }
219    let Some((fm_end, fence_end)) = fm_end else {
220        return Err(whole("unclosed `+++` frontmatter fence"));
221    };
222    let fm_src = &src[fm_start..fm_end];
223    let fm: FrontmatterIn = toml::from_str(fm_src).map_err(|e| {
224        let (o, l) = e
225            .span()
226            .map(|s| (fm_start + s.start, s.end.saturating_sub(s.start)))
227            .unwrap_or((fm_start, fm_end - fm_start));
228        at(o, l, e.to_string())
229    })?;
230    // DTO → Item; value errors point at the whole frontmatter block.
231    let fm_err = |m: String| at(fm_start, fm_end - fm_start, m);
232    let id: ItemId = fm
233        .id
234        .parse()
235        .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?;
236    if id.kind() != spec.id_kind() {
237        return Err(fm_err(format!(
238            "id {id} doesn't belong in this file (expected a {} id)",
239            spec.name
240        )));
241    }
242    // filename ↔ frontmatter: a canonical filename's number must
243    // agree with the id the file claims — `issue-0003-x.md`
244    // holding `I-009` reserves 3 in the id scan while claiming 9,
245    // which would let the allocator reissue a live id. A name
246    // that doesn't fit the kind's pattern carries no number to
247    // check against (that's a filename finding, not this error).
248    let basename = std::path::Path::new(path)
249        .file_name()
250        .and_then(|n| n.to_str())
251        .unwrap_or(path);
252    if let Some(fname_id) = parse_item_filename(spec, basename) {
253        if fname_id != id {
254            return Err(fm_err(format!(
255                "filename says {fname_id} but frontmatter id is {id}"
256            )));
257        }
258    }
259    let owner = fm
260        .owner
261        .map(|o| Actor::new(&o).map_err(|e| fm_err(e.to_string())))
262        .transpose()?;
263    let created = fm
264        .created
265        .map(|c| date::parse(&c).map_err(|e| fm_err(format!("bad created date — {e}"))))
266        .transpose()?;
267    let mut labels = BTreeSet::new();
268    for l in fm.labels {
269        if !myco_core::is_valid_label(&l) {
270            return Err(fm_err(format!("invalid label {l:?}")));
271        }
272        labels.insert(l);
273    }
274    let mut blocks = BTreeSet::new();
275    for b in fm.blocks {
276        blocks.insert(
277            b.parse::<ItemId>()
278                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?,
279        );
280    }
281    let mut item = Item::new(id, fm.title);
282    item.status = fm.status;
283    item.priority = fm.priority.unwrap_or_default();
284    item.owner = owner;
285    item.labels = labels;
286    item.blocks = blocks;
287    item.created = created;
288    item.note = fm.note;
289    item.superseded_by = fm
290        .superseded_by
291        .map(|s| {
292            s.parse::<ItemId>()
293                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
294        })
295        .transpose()?;
296    item.supersedes = fm
297        .supersedes
298        .map(|s| {
299            s.parse::<ItemId>()
300                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
301        })
302        .transpose()?;
303    for (key, name) in [("epic", &fm.epic), ("milestone", &fm.milestone)] {
304        if let Some(name) = name {
305            if !myco_core::is_valid_group_name(name) {
306                return Err(fm_err(format!("invalid {key} name {name:?}")));
307            }
308        }
309    }
310    item.epic = fm.epic;
311    item.milestone = fm.milestone;
312    if item.blocks.contains(&id) || item.supersedes == Some(id) || item.superseded_by == Some(id) {
313        return Err(fm_err(format!("{id} can't block or supersede itself")));
314    }
315    // body: everything after the closing fence, canonicalized —
316    // leading blank lines go, the first content line's own
317    // indentation survives (an indented code block stays a code
318    // block)
319    let file = ItemFile {
320        meta: item,
321        body: myco_core::canonical_body(&src[fence_end..]).to_string(),
322    };
323    // parse-safe: the body must be markdown the parser accepts
324    if !file.body.is_empty() {
325        mdx::parse(&file.body).map_err(|m| at(fence_end, src.len() - fence_end, m))?;
326    }
327    Ok(file)
328}

330/// Canonical item file — `+++\n<toml>+++\n\n<body>\n`. The model
331/// is validated first ([`line::check_item`] — the same validity
332/// source as the line grammar) so emit can't write a file parse
333/// would reject; children are an error, never silently dropped.
334pub fn emit_item_file(file: &ItemFile) -> Result<String, EmitError> {
335    line::check_item(&file.meta, "item file")?;
336    if !file.meta.children.is_empty() {
337        return Err(EmitError::new(
338            "item file",
339            format!(
340                "{} carries children — sub-items only exist in TODO.md/BACKLOG.md",
341                file.meta.id
342            ),
343        ));
344    }
345    let fm = FrontmatterOut {
346        id: file.meta.id.to_string(),
347        title: &file.meta.title,
348        status: file.meta.status,
349        created: file.meta.created.map(date::render),
350        owner: file.meta.owner.as_ref().map(ToString::to_string),
351        priority: file.meta.priority,
352        labels: file.meta.labels.iter().map(String::as_str).collect(),
353        blocks: file.meta.blocks.iter().map(ItemId::to_string).collect(),
354        note: file.meta.note.as_deref(),
355        superseded_by: file.meta.superseded_by.map(|i| i.to_string()),
356        supersedes: file.meta.supersedes.map(|i| i.to_string()),
357        epic: file.meta.epic.as_deref(),
358        milestone: file.meta.milestone.as_deref(),
359    };
360    let toml = toml::to_string(&fm).map_err(|e| EmitError::new("item file", e.to_string()))?;
361    if file.body != myco_core::canonical_body(&file.body) {
362        // a body parse can't reproduce byte-for-byte would emit
363        // then read back as a different model — non-canonical
364        // bodies are the caller's bug, not content to reshape
365        return Err(EmitError::new(
366            "item file",
367            "body isn't canonical — leading blank lines or trailing whitespace",
368        ));
369    }
370    let mut out = String::from("+++\n");
371    out.push_str(&toml);
372    out.push_str("+++\n");
373    if !file.body.is_empty() {
374        out.push('\n');
375        out.push_str(&file.body);
376        out.push('\n');
377    }
378    Ok(out)
379}

526    #[test]
527    fn frontmatter_id_needs_a_closed_block() {
528        let i1 = Some(ItemId::new(ItemKind::Issue, 1));
529        // the closed block yields its id
530        assert_eq!(frontmatter_id("+++\nid = \"I-1\"\n+++\n"), i1);
531        // id lines after the close are body, not frontmatter
532        assert_eq!(
533            frontmatter_id("+++\ntitle = \"x\"\n+++\nid = \"I-1\"\n"),
534            None
535        );
536        // an unclosed block never counts — a body/code `id = "…"`
537        // can't pose as frontmatter
538        assert_eq!(
539            frontmatter_id("+++\ntitle = \"x\"\n\nbody\nid = \"I-1\"\n"),
540            None
541        );
542        assert_eq!(frontmatter_id("+++\nid = \"I-1\"\n"), None);
543        // …and the scan is bounded even when the "block" is huge
544        let mut src = "+++\n".to_string();
545        for _ in 0..(FRONTMATTER_PROBE_MAX + 10) {
546            src.push_str("title = \"x\"\n");
547        }

678        // `issue-0003-x.md` claiming `I-009` — the scan would
679        // reserve 3 while the file holds 9; that's corruption, not
680        // a rename waiting to happen
681        let src = "+++\nid = \"I-009\"\ntitle = \"x\"\nstatus = \"open\"\n+++\n";
682        assert!(parse_item_file(src, &spec(ItemKind::Issue), "issues/issue-0003-x.md").is_err());
683        // agreement parses; a name carrying no number isn't checked
684        let src = "+++\nid = \"I-003\"\ntitle = \"x\"\nstatus = \"open\"\n+++\n";

687        // kind mismatch stays a kind error (checked first)

692        assert!(parse_item_file(src, &spec(ItemKind::Proposal), "issues/issue-0002-x.md").is_ok());

695    #[test]
696    fn emit_validates_the_model() {
697        let good = ItemFile::new(Item::new(ItemId::new(ItemKind::Issue, 1), "x"));
698        assert!(emit_item_file(&good).is_ok());
699        // an invalid label — parse would reject the emitted file
700        let mut bad = good.clone();
701        bad.meta.labels.insert("bad label".into());
702        assert!(emit_item_file(&bad).is_err());
703        // invalid group names too
704        let mut bad = good.clone();
705        bad.meta.epic = Some("bad epic".into());
706        assert!(emit_item_file(&bad).is_err());
707        bad.meta.epic = None;
708        bad.meta.milestone = Some("bad/ms".into());
709        assert!(emit_item_file(&bad).is_err());
710        // an empty/unemittable note
711        let mut bad = good.clone();
712        bad.meta.note = Some(String::new());
713        assert!(emit_item_file(&bad).is_err());
714        // children are an error, never silently dropped
715        let mut bad = good;
716        bad.meta
717            .children
718            .push(Item::new(ItemId::new(ItemKind::Task, 1), "sub"));
719        let e = emit_item_file(&bad).unwrap_err();
720        assert!(e.to_string().contains("children"), "{e}");
721    }
