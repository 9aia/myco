1//! Item files — one md file per issue/proposal/adr/plan:
2//!
3//! ```text
4//! issues/issue-NNNN-slug.md    proposals/proposal-NNNN-slug.md
5//! adrs/NNNN-slug.md            plans/pl-NNNN-slug.md
6//! ```
7//!
8//! `NNNN` = id number padded to 4 digits. TOML frontmatter in
9//! `+++` fences (id, title, status, created, owner, priority,
10//! labels, blocks, note, superseded_by, supersedes, epic,
11//! milestone — extendable: unknown keys are ignored on read for
12//! forward-compat); body = free-form markdown (verbatim; kind
13//! templates are `myco-template`'s job).
14

22/// The `project/` subdirectory a file-backed kind lives in.
23/// [`ItemKind::Task`] has no files (`None`).
24pub fn item_dir(kind: ItemKind) -> Option<&'static str> {
25    match kind {
26        ItemKind::Task => None,
27        ItemKind::Issue => Some("issues"),
28        ItemKind::Proposal => Some("proposals"),
29        ItemKind::Adr => Some("adrs"),
30        ItemKind::Plan => Some("plans"),
31    }
32}

34/// Filename prefix before the number — `issue`, `proposal`, `pl`,
35/// or empty for ADRs (`adrs/0004-slug.md`).
36fn file_prefix(kind: ItemKind) -> Option<&'static str> {
37    match kind {
38        ItemKind::Task => None,
39        ItemKind::Issue => Some("issue"),
40        ItemKind::Proposal => Some("proposal"),
41        ItemKind::Adr => Some(""),
42        ItemKind::Plan => Some("pl"),
43    }
44}

46/// Canonical filename for an item — `issue-0003-memory-leak.md`,
47/// `0004-hybrid-store.md`. Slug = `slug::slugify(title)`,
48/// `"untitled"` when the title slugifies to nothing.
49pub fn item_filename(id: &ItemId, title: &str) -> Option<String> {
50    let prefix = file_prefix(id.kind())?;
51    let slug = {
52        let s = slug::slugify(title);
53        if s.is_empty() {
54            "untitled".to_string()
55        } else {
56            s
57        }
58    };
59    Some(if prefix.is_empty() {
60        format!("{:04}-{slug}.md", id.number())
61    } else {
62        format!("{prefix}-{:04}-{slug}.md", id.number())
63    })
64}

66/// `issues/issue-0003-x.md` → `I-003`; `None` when the filename
67/// doesn't fit the kind's pattern.
68pub fn parse_item_filename(kind: ItemKind, filename: &str) -> Option<ItemId> {
69    let prefix = file_prefix(kind)?;
70    let stem = filename.strip_suffix(".md")?;
71    let rest = if prefix.is_empty() {
72        stem
73    } else {
74        stem.strip_prefix(prefix)?.strip_prefix('-')?
75    };
76    let (digits, slug) = rest.split_once('-')?;
77    if digits.is_empty()
78        || digits.len() > 9
79        || !digits.bytes().all(|b| b.is_ascii_digit())
80        || slug.is_empty()
81    {
82        return None;
83    }
84    let number: u32 = digits.parse().ok()?;
85    if number == 0 {
86        return None;
87    }
88    Some(ItemId::new(kind, number))
89}

91/// `id = "…"` pulled out of a `+++` frontmatter block — the cheap
92/// probe for when the strict parse isn't needed (or already
93/// failed). `None` for files with no `+++` fence or no parseable
94/// `id` line — including frontmatter-less files that merely
95/// *borrow* an id number via their filename.
96pub fn frontmatter_id(src: &str) -> Option<ItemId> {
97    let mut lines = src.lines();
98    if lines.next()?.trim() != "+++" {
99        return None;
100    }
101    for l in lines {
102        let t = l.trim();
103        if t == "+++" {
104            break;
105        }
106        if let Some(rest) = t.strip_prefix("id") {
107            if let Some(v) = rest.trim_start().strip_prefix('=') {
108                if let Ok(id) = v.trim().trim_matches('"').parse::<ItemId>() {
109                    return Some(id);
110                }
111            }
112        }
113    }
114    None
115}

117/// Frontmatter DTO — stringly-typed so bad values point at the
118/// frontmatter block, not toml's internals.
119#[derive(Debug, Deserialize)]
120struct FrontmatterIn {
121    id: String,
122    title: String,
123    status: Status,
124    created: Option<String>,
125    owner: Option<String>,
126    #[serde(default)]
127    priority: Option<Priority>,
128    #[serde(default)]
129    labels: Vec<String>,
130    #[serde(default)]
131    blocks: Vec<String>,
132    note: Option<String>,
133    superseded_by: Option<String>,
134    supersedes: Option<String>,
135    epic: Option<String>,
136    milestone: Option<String>,
137}

139/// Emit shape — field order IS the frontmatter key order.
140#[derive(Serialize)]
141struct FrontmatterOut<'a> {
142    id: String,
143    title: &'a str,
144    status: Status,
145    #[serde(skip_serializing_if = "Option::is_none")]
146    created: Option<String>,
147    #[serde(skip_serializing_if = "Option::is_none")]
148    owner: Option<String>,
149    priority: Priority,
150    #[serde(skip_serializing_if = "Vec::is_empty")]
151    labels: Vec<&'a str>,
152    #[serde(skip_serializing_if = "Vec::is_empty")]
153    blocks: Vec<String>,
154    #[serde(skip_serializing_if = "Option::is_none")]
155    note: Option<&'a str>,
156    #[serde(skip_serializing_if = "Option::is_none")]
157    superseded_by: Option<String>,
158    #[serde(skip_serializing_if = "Option::is_none")]
159    supersedes: Option<String>,
160    #[serde(skip_serializing_if = "Option::is_none")]
161    epic: Option<&'a str>,
162    #[serde(skip_serializing_if = "Option::is_none")]
163    milestone: Option<&'a str>,
164}

166/// Parse an item file's source. `expected_kind` comes from the
167/// directory the file was found in — a mismatch (`P-002` inside
168/// `issues/`) is an error.
169pub fn parse_item_file(
170    src: &str,
171    expected_kind: ItemKind,
172    path: &str,
173) -> Result<ItemFile, ParseError> {
174    let src = mdx::normalize(src);
175    let at = |off: usize, len: usize, m: String| ParseError::at(path, src.clone(), off, len, m);
176    let whole = |m: &str| ParseError::whole_file(path, src.clone(), m);
177    let mut lines = src.split_inclusive('\n');
178    if lines.next().map(str::trim_end) != Some("+++") {
179        return Err(whole("expected `+++` TOML frontmatter fence on line 1"));
180    }
181    // find the closing `+++` line (byte offsets into src)
182    let fm_start = "+++\n".len();
183    let mut fm_end = None;
184    let mut off = fm_start;
185    for line in lines {
186        if line.trim_end() == "+++" {
187            fm_end = Some((off, off + line.len()));
188            break;
189        }
190        off += line.len();
191    }
192    let Some((fm_end, fence_end)) = fm_end else {
193        return Err(whole("unclosed `+++` frontmatter fence"));
194    };
195    let fm_src = &src[fm_start..fm_end];
196    let fm: FrontmatterIn = toml::from_str(fm_src).map_err(|e| {
197        let (o, l) = e
198            .span()
199            .map(|s| (fm_start + s.start, s.end.saturating_sub(s.start)))
200            .unwrap_or((fm_start, fm_end - fm_start));
201        at(o, l, e.to_string())
202    })?;
203    // DTO → Item; value errors point at the whole frontmatter block.
204    let fm_err = |m: String| at(fm_start, fm_end - fm_start, m);
205    let id: ItemId = fm
206        .id
207        .parse()
208        .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?;
209    if id.kind() != expected_kind {
210        return Err(fm_err(format!(
211            "id {id} doesn't belong in this file (expected a {} id)",
212            expected_kind.prefix()
213        )));
214    }
215    let owner = fm
216        .owner
217        .map(|o| Actor::new(&o).map_err(|e| fm_err(e.to_string())))
218        .transpose()?;
219    let created = fm
220        .created
221        .map(|c| date::parse(&c).map_err(|e| fm_err(format!("bad created date — {e}"))))
222        .transpose()?;
223    let mut labels = BTreeSet::new();
224    for l in fm.labels {
225        if !myco_core::is_valid_label(&l) {
226            return Err(fm_err(format!("invalid label {l:?}")));
227        }
228        labels.insert(l);
229    }
230    let mut blocks = BTreeSet::new();
231    for b in fm.blocks {
232        blocks.insert(
233            b.parse::<ItemId>()
234                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?,
235        );
236    }
237    let mut item = Item::new(id, fm.title);
238    item.status = fm.status;
239    item.priority = fm.priority.unwrap_or_default();
240    item.owner = owner;
241    item.labels = labels;
242    item.blocks = blocks;
243    item.created = created;
244    item.note = fm.note;
245    item.superseded_by = fm
246        .superseded_by
247        .map(|s| {
248            s.parse::<ItemId>()
249                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
250        })
251        .transpose()?;
252    item.supersedes = fm
253        .supersedes
254        .map(|s| {
255            s.parse::<ItemId>()
256                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
257        })
258        .transpose()?;
259    for (key, name) in [("epic", &fm.epic), ("milestone", &fm.milestone)] {
260        if let Some(name) = name {
261            if !myco_core::is_valid_group_name(name) {
262                return Err(fm_err(format!("invalid {key} name {name:?}")));
263            }
264        }
265    }
266    item.epic = fm.epic;
267    item.milestone = fm.milestone;
268    // body: everything after the closing fence + one blank line
269    let mut body = &src[fence_end..];
270    body = body.strip_prefix('\n').unwrap_or(body);
271    let file = ItemFile {
272        meta: item,
273        body: body.trim().to_string(),
274    };
275    // parse-safe: the body must be markdown the parser accepts
276    if !file.body.is_empty() {
277        mdx::parse(&file.body).map_err(|m| at(fence_end, src.len() - fence_end, m))?;
278    }
279    Ok(file)
280}

282/// Canonical item file — `+++\n<toml>+++\n\n<body>\n`.
283pub fn emit_item_file(file: &ItemFile) -> Result<String, EmitError> {
284    let fm = FrontmatterOut {
285        id: file.meta.id.to_string(),
286        title: &file.meta.title,
287        status: file.meta.status,
288        created: file.meta.created.map(date::render),
289        owner: file.meta.owner.as_ref().map(ToString::to_string),
290        priority: file.meta.priority,
291        labels: file.meta.labels.iter().map(String::as_str).collect(),
292        blocks: file.meta.blocks.iter().map(ItemId::to_string).collect(),
293        note: file.meta.note.as_deref(),
294        superseded_by: file.meta.superseded_by.map(|i| i.to_string()),
295        supersedes: file.meta.supersedes.map(|i| i.to_string()),
296        epic: file.meta.epic.as_deref(),
297        milestone: file.meta.milestone.as_deref(),
298    };
299    let toml = toml::to_string(&fm).map_err(|e| EmitError::new("item file", e.to_string()))?;
300    let mut out = String::from("+++\n");
301    out.push_str(&toml);
302    out.push_str("+++\n");
303    if !file.body.is_empty() {
304        out.push('\n');
305        out.push_str(file.body.trim_end());
306        out.push('\n');
307    }
308    Ok(out)
309}
