1//! Item files — one md file per issue/proposal/adr/plan:
2//!
3//! ```text
4//! issues/issue-NNNN-slug.md    proposals/proposal-NNNN-slug.md
5//! adrs/NNNN-slug.md            plans/pl-NNNN-slug.md
6//! ```
7//!
8//! `NNNN` = id number padded to 4 digits. TOML frontmatter in
9//! `+++` fences (id, title, status, created, owner, priority,
10//! labels, blocks, links, note, superseded_by, supersedes, epic,
11//! milestone, due, estimate, cycle — extendable: unknown keys are
12//! ignored on read for forward-compat); body = free-form markdown
13//! (verbatim; kind templates are `myco-template`'s job).
14

22/// The `project/` subdirectory a file-backed kind lives in —
23/// the spec's `dir` (`issues`, `spikes`); `doc`-storage kinds
24/// have no files (`None`).
25pub fn item_dir(spec: &KindSpec) -> Option<String> {
26    (spec.storage == Storage::Dir).then(|| spec.dir_name())
27}
28
29/// Filename prefix before the number — `issue`, `proposal`, `pl`,
30/// the spec's `file_prefix` for user kinds, or empty for ADRs
31/// (`adrs/0004-slug.md`).
32fn file_prefix(spec: &KindSpec) -> Option<String> {
33    (spec.storage == Storage::Dir).then(|| spec.file_prefix())
34}

36/// Canonical filename for an item — `issue-0003-memory-leak.md`,
37/// `0004-hybrid-store.md`. Slug = `slug::slugify(title)`,
38/// `"untitled"` when the title slugifies to nothing.
39pub fn item_filename(spec: &KindSpec, id: &ItemId, title: &str) -> Option<String> {
40    let prefix = file_prefix(spec)?;
41    let slug = {
42        let s = slug::slugify(title);
43        if s.is_empty() {
44            "untitled".to_string()
45        } else {
46            s
47        }
48    };
49    Some(if prefix.is_empty() {
50        format!("{:04}-{slug}.md", id.number())
51    } else {
52        format!("{prefix}-{:04}-{slug}.md", id.number())
53    })
54}

56/// `issues/issue-0003-x.md` → `I-003`; `None` when the filename
57/// doesn't fit the kind's pattern. The number must fit `u32` —
58/// the same bound [`ItemId`]'s own parse applies, so anything
59/// [`item_filename`] can emit parses back (id numbers are `u32`,
60/// i.e. up to 10 digits).
61pub fn parse_item_filename(spec: &KindSpec, filename: &str) -> Option<ItemId> {
62    let prefix = file_prefix(spec)?;
63    let stem = filename.strip_suffix(".md")?;
64    let rest = if prefix.is_empty() {
65        stem
66    } else {
67        stem.strip_prefix(prefix.as_str())?.strip_prefix('-')?
68    };
69    let (digits, slug) = rest.split_once('-')?;
70    if digits.is_empty() || !digits.bytes().all(|b| b.is_ascii_digit()) || slug.is_empty() {
71        return None;
72    }
73    let number: u32 = digits.parse().ok()?;
74    if number == 0 {
75        return None;
76    }
77    Some(ItemId::new(spec.id_kind(), number))
78}

80/// Deepest line the `frontmatter_id` probe will read — real
81/// frontmatter is ~10 keys; past this the `+++` block is
82/// certainly unclosed (or the file is pathological).
83const FRONTMATTER_PROBE_MAX: usize = 128;
84
85/// `id = "…"` pulled out of a *closed* `+++` frontmatter block —
86/// the cheap probe for when the strict parse isn't needed (or
87/// already failed). `None` for files with no `+++` fence, no
88/// closing fence within [`FRONTMATTER_PROBE_MAX`] lines, or no
89/// parseable `id` line — including frontmatter-less files that
90/// merely *borrow* an id number via their filename. The close
91/// requirement is what keeps a body's `id = "…"` (a code block,
92/// a quoted example) from posing as frontmatter.
93pub fn frontmatter_id(src: &str) -> Option<ItemId> {
94    let mut lines = src.lines();
95    if lines.next()?.trim() != "+++" {
96        return None;
97    }
98    let mut found = None;
99    for l in lines.take(FRONTMATTER_PROBE_MAX) {
100        let t = l.trim();
101        // the close ends the block — an id only counts when the
102        // block actually closed (else a body's `id = "…"` would
103        // pose as frontmatter)
104        if t == "+++" {
105            return found;
106        }
107        if found.is_none() {
108            if let Some(rest) = t.strip_prefix("id") {
109                if let Some(v) = rest.trim_start().strip_prefix('=') {
110                    if let Ok(id) = v.trim().trim_matches('"').parse::<ItemId>() {
111                        found = Some(id);
112                    }
113                }
114            }
115        }
116    }
117    None // window exhausted — never closed
118}

120/// Frontmatter DTO — stringly-typed so bad values point at the
121/// frontmatter block, not toml's internals.
122#[derive(Debug, Deserialize)]
123struct FrontmatterIn {
124    id: String,
125    title: String,
126    status: Status,
127    #[serde(default, deserialize_with = "de_opt_created")]
128    created: Option<String>,
129    owner: Option<String>,
130    #[serde(default, deserialize_with = "de_opt_priority")]
131    priority: Option<Priority>,
132    #[serde(default)]
133    labels: Vec<String>,
134    #[serde(default)]
135    blocks: Vec<String>,
136    #[serde(default)]
137    links: Vec<String>,
138    note: Option<String>,
139    superseded_by: Option<String>,
140    supersedes: Option<String>,
141    epic: Option<String>,
142    milestone: Option<String>,
143    #[serde(default, deserialize_with = "de_opt_created")]
144    due: Option<String>,
145    estimate: Option<u32>,
146    cycle: Option<String>,
147}
148
149/// `priority` reads like the line grammar — `low|med|high|crit`,
150/// the spelling emit writes — plus the serde full names
151/// (`medium`, `critical`) older files and hand-edits carry, via
152/// [`Priority`]'s `FromStr` (lenient read, strict canonical
153/// write). Deriving `Priority` would accept full names only.
154fn de_opt_priority<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Option<Priority>, D::Error> {
155    Option::<String>::deserialize(d)?
156        .map(|s| s.parse().map_err(serde::de::Error::custom))
157        .transpose()
158}
159
160/// `created` accepts both the quoted string emit writes
161/// (`created = "2026-02-01"`) and a bare TOML date
162/// (`created = 2026-02-01`) — hand-edited files use the native
163/// form. Either way the value lands as text for [`date::parse`]
164/// to validate.
165fn de_opt_created<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Option<String>, D::Error> {
166    match Option::<toml::Value>::deserialize(d)? {
167        None => Ok(None),
168        Some(toml::Value::String(s)) => Ok(Some(s)),
169        Some(toml::Value::Datetime(dt)) => Ok(Some(dt.to_string())),
170        Some(v) => Err(serde::de::Error::custom(format!(
171            "expected a `YYYY-MM-DD` date, got {v}"
172        ))),
173    }
174}

215/// Parse an item file's source. `spec` declares the kind the
216/// directory the file was found in belongs to — an id whose
217/// prefix isn't the spec's (`P-002` inside `issues/`) is an
218/// error.
219pub fn parse_item_file(src: &str, spec: &KindSpec, path: &str) -> Result<ItemFile, ParseError> {
220    let src = mdx::normalize(src);
221    let at = |off: usize, len: usize, m: String| ParseError::at(path, src.clone(), off, len, m);
222    let whole = |m: &str| ParseError::whole_file(path, src.clone(), m);
223    let mut lines = src.split_inclusive('\n');
224    if lines.next().map(str::trim_end) != Some("+++") {
225        return Err(whole("expected `+++` TOML frontmatter fence on line 1"));
226    }
227    // find the closing `+++` line (byte offsets into src)
228    let fm_start = "+++\n".len();
229    let mut fm_end = None;
230    let mut off = fm_start;
231    for line in lines {
232        if line.trim_end() == "+++" {
233            fm_end = Some((off, off + line.len()));
234            break;
235        }
236        off += line.len();
237    }
238    let Some((fm_end, fence_end)) = fm_end else {
239        return Err(whole("unclosed `+++` frontmatter fence"));
240    };
241    let fm_src = &src[fm_start..fm_end];
242    let fm: FrontmatterIn = toml::from_str(fm_src).map_err(|e| {
243        let (o, l) = e
244            .span()
245            .map(|s| (fm_start + s.start, s.end.saturating_sub(s.start)))
246            .unwrap_or((fm_start, fm_end - fm_start));
247        at(o, l, e.to_string())
248    })?;
249    // DTO → Item; value errors point at the whole frontmatter block.
250    let fm_err = |m: String| at(fm_start, fm_end - fm_start, m);
251    let id: ItemId = fm
252        .id
253        .parse()
254        .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?;
255    if id.kind() != spec.id_kind() {
256        return Err(fm_err(format!(
257            "id {id} doesn't belong in this file (expected a {} id)",
258            spec.name
259        )));
260    }
261    // filename ↔ frontmatter: a canonical filename's number must
262    // agree with the id the file claims — `issue-0003-x.md`
263    // holding `I-009` reserves 3 in the id scan while claiming 9,
264    // which would let the allocator reissue a live id. A name
265    // that doesn't fit the kind's pattern carries no number to
266    // check against (that's a filename finding, not this error).
267    let basename = std::path::Path::new(path)
268        .file_name()
269        .and_then(|n| n.to_str())
270        .unwrap_or(path);
271    if let Some(fname_id) = parse_item_filename(spec, basename) {
272        if fname_id != id {
273            return Err(fm_err(format!(
274                "filename says {fname_id} but frontmatter id is {id}"
275            )));
276        }
277    }
278    let owner = fm
279        .owner
280        .map(|o| Actor::new(&o).map_err(|e| fm_err(e.to_string())))
281        .transpose()?;
282    let created = fm
283        .created
284        .map(|c| date::parse(&c).map_err(|e| fm_err(format!("bad created date — {e}"))))
285        .transpose()?;
286    let mut labels = BTreeSet::new();
287    for l in fm.labels {
288        if !myco_core::is_valid_label(&l) {
289            return Err(fm_err(format!("invalid label {l:?}")));
290        }
291        labels.insert(l);
292    }
293    let mut blocks = BTreeSet::new();
294    for b in fm.blocks {
295        blocks.insert(
296            b.parse::<ItemId>()
297                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?,
298        );
299    }
300    let mut links = BTreeSet::new();
301    for l in fm.links {
302        links.insert(
303            l.parse::<ItemId>()
304                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?,
305        );
306    }
307    let mut item = Item::new(id, fm.title);
308    item.status = fm.status;
309    item.priority = fm.priority.unwrap_or_default();
310    item.owner = owner;
311    item.labels = labels;
312    item.blocks = blocks;
313    item.links = links;
314    item.created = created;
315    item.note = fm.note;
316    item.superseded_by = fm
317        .superseded_by
318        .map(|s| {
319            s.parse::<ItemId>()
320                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
321        })
322        .transpose()?;
323    item.supersedes = fm
324        .supersedes
325        .map(|s| {
326            s.parse::<ItemId>()
327                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
328        })
329        .transpose()?;
330    for (key, name) in [
331        ("epic", &fm.epic),
332        ("milestone", &fm.milestone),
333        ("cycle", &fm.cycle),
334    ] {
335        if let Some(name) = name {
336            if !myco_core::is_valid_group_name(name) {
337                return Err(fm_err(format!("invalid {key} name {name:?}")));
338            }
339        }
340    }
341    item.epic = fm.epic;
342    item.milestone = fm.milestone;
343    item.due = fm
344        .due
345        .map(|d| date::parse(&d).map_err(|e| fm_err(format!("bad due date — {e}"))))
346        .transpose()?;
347    if let Some(e) = fm.estimate {
348        if e > myco_core::MAX_ESTIMATE {
349            return Err(fm_err(format!(
350                "estimate {e} exceeds max {}",
351                myco_core::MAX_ESTIMATE
352            )));
353        }
354        item.estimate = Some(e);
355    }
356    item.cycle = fm.cycle;
357    if item.blocks.contains(&id)
358        || item.links.contains(&id)
359        || item.supersedes == Some(id)
360        || item.superseded_by == Some(id)
361    {
362        return Err(fm_err(format!(
363            "{id} can't block, link, or supersede itself"
364        )));
365    }
366    // body: everything after the closing fence, canonicalized —
367    // leading blank lines go, the first content line's own
368    // indentation survives (an indented code block stays a code
369    // block)
370    let file = ItemFile {
371        meta: item,
372        body: myco_core::canonical_body(&src[fence_end..]).to_string(),
373    };
374    // parse-safe: the body must be markdown the parser accepts
375    if !file.body.is_empty() {
376        mdx::parse(&file.body).map_err(|m| at(fence_end, src.len() - fence_end, m))?;
377    }
378    Ok(file)
379}

381/// Canonical item file — `+++\n<toml>+++\n\n<body>\n`. The model
382/// is validated first ([`line::check_item`] — the same validity
383/// source as the line grammar) so emit can't write a file parse
384/// would reject; children are an error, never silently dropped.
385pub fn emit_item_file(file: &ItemFile) -> Result<String, EmitError> {
386    line::check_item(&file.meta, "item file")?;
387    if !file.meta.children.is_empty() {
388        return Err(EmitError::new(
389            "item file",
390            format!(
391                "{} carries children — sub-items only exist in TODO.md/BACKLOG.md",
392                file.meta.id
393            ),
394        ));
395    }
396    let fm = FrontmatterOut {
397        id: file.meta.id.to_string(),
398        title: &file.meta.title,
399        status: file.meta.status,
400        created: file.meta.created.map(date::render),
401        owner: file.meta.owner.as_ref().map(ToString::to_string),
402        priority: file.meta.priority.token(),
403        labels: file.meta.labels.iter().map(String::as_str).collect(),
404        blocks: file.meta.blocks.iter().map(ItemId::to_string).collect(),
405        links: file.meta.links.iter().map(ItemId::to_string).collect(),
406        note: file.meta.note.as_deref(),
407        superseded_by: file.meta.superseded_by.map(|i| i.to_string()),
408        supersedes: file.meta.supersedes.map(|i| i.to_string()),
409        epic: file.meta.epic.as_deref(),
410        milestone: file.meta.milestone.as_deref(),
411        due: file.meta.due.map(date::render),
412        estimate: file.meta.estimate,
413        cycle: file.meta.cycle.as_deref(),
414    };
415    let toml = toml::to_string(&fm).map_err(|e| EmitError::new("item file", e.to_string()))?;
416    if file.body != myco_core::canonical_body(&file.body) {
417        // a body parse can't reproduce byte-for-byte would emit
418        // then read back as a different model — non-canonical
419        // bodies are the caller's bug, not content to reshape
420        return Err(EmitError::new(
421            "item file",
422            "body isn't canonical — leading blank lines or trailing whitespace",
423        ));
424    }
425    let mut out = String::from("+++\n");
426    out.push_str(&toml);
427    out.push_str("+++\n");
428    if !file.body.is_empty() {
429        out.push('\n');
430        out.push_str(&file.body);
431        out.push('\n');
432    }
433    Ok(out)
434}

634    fn frontmatter_id_needs_a_closed_block() {
635        let i1 = Some(ItemId::new(ItemKind::Issue, 1));
636        // the closed block yields its id
637        assert_eq!(frontmatter_id("+++\nid = \"I-1\"\n+++\n"), i1);
638        // id lines after the close are body, not frontmatter
639        assert_eq!(
640            frontmatter_id("+++\ntitle = \"x\"\n+++\nid = \"I-1\"\n"),

643        // an unclosed block never counts — a body/code `id = "…"`
644        // can't pose as frontmatter

654        }
655        src.push_str("id = \"I-1\"\n+++\n");

659    #[test]
660    fn frontmatter_normalizes_full_names() {
661        // serde's full names and bare TOML dates — the shapes a
662        // hand-edit (or an older myco) produces — read as their
663        // canonical values (I-021, I-032); emit rewrites the
664        // strict `pri:`-token / quoted-date forms
665        let src = "+++\nid = \"I-001\"\ntitle = \"x\"\nstatus = \"open\"\npriority = \"medium\"\ncreated = 2026-02-01\n+++\n";
666        let file = parse_item_file(src, &spec(ItemKind::Issue), "issues/x.md").unwrap();
667        assert_eq!(file.meta.priority, Priority::Medium);
668        assert_eq!(file.meta.created, Some(date::parse("2026-02-01").unwrap()));
669        let out = emit_item_file(&file).unwrap();
670        assert!(out.contains("priority = \"med\""));
671        assert!(out.contains("created = \"2026-02-01\""));
672        // `critical` normalizes to `crit` the same way
673        let src =
674            "+++\nid = \"I-001\"\ntitle = \"x\"\nstatus = \"open\"\npriority = \"critical\"\n+++\n";
675        let file = parse_item_file(src, &spec(ItemKind::Issue), "issues/x.md").unwrap();
676        assert_eq!(file.meta.priority, Priority::Critical);
677        assert!(emit_item_file(&file)
678            .unwrap()
679            .contains("priority = \"crit\""));
680        // token spellings round-trip byte-identical
681        let src =
682            "+++\nid = \"I-001\"\ntitle = \"x\"\nstatus = \"open\"\npriority = \"med\"\n+++\n";
683        let file = parse_item_file(src, &spec(ItemKind::Issue), "issues/x.md").unwrap();
684        assert_eq!(emit_item_file(&file).unwrap(), src);
685        // a datetime-with-time is still a bad `YYYY-MM-DD` date
686        let bad = "+++\nid = \"I-001\"\ntitle = \"x\"\nstatus = \"open\"\ncreated = 2026-02-01T10:00:00Z\n+++\n";
687        assert!(parse_item_file(bad, &spec(ItemKind::Issue), "x.md").is_err());
688        // …and a wrong-typed value errors, not silently coerces
689        let bad = "+++\nid = \"I-001\"\ntitle = \"x\"\nstatus = \"open\"\ncreated = 42\n+++\n";
690        assert!(parse_item_file(bad, &spec(ItemKind::Issue), "x.md").is_err());
691    }
