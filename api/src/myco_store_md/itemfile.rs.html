1//! Item files — one md file per issue/proposal/adr/plan:
2//!
3//! ```text
4//! issues/issue-NNNN-slug.md    proposals/proposal-NNNN-slug.md
5//! adrs/NNNN-slug.md            plans/pl-NNNN-slug.md
6//! ```
7//!
8//! `NNNN` = id number padded to 4 digits. TOML frontmatter in
9//! `+++` fences (id, title, status, created, owner, priority,
10//! labels, blocks, note, superseded_by, supersedes, epic,
11//! milestone — extendable: unknown keys are ignored on read for
12//! forward-compat); body = free-form markdown (verbatim; kind
13//! templates are `myco-template`'s job).
14

22/// The `project/` subdirectory a file-backed kind lives in.
23/// [`ItemKind::Task`] has no files (`None`).
24pub fn item_dir(kind: ItemKind) -> Option<&'static str> {
25    match kind {
26        ItemKind::Task => None,
27        ItemKind::Issue => Some("issues"),
28        ItemKind::Proposal => Some("proposals"),
29        ItemKind::Adr => Some("adrs"),
30        ItemKind::Plan => Some("plans"),
31    }
32}

34/// Filename prefix before the number — `issue`, `proposal`, `pl`,
35/// or empty for ADRs (`adrs/0004-slug.md`).
36fn file_prefix(kind: ItemKind) -> Option<&'static str> {
37    match kind {
38        ItemKind::Task => None,
39        ItemKind::Issue => Some("issue"),
40        ItemKind::Proposal => Some("proposal"),
41        ItemKind::Adr => Some(""),
42        ItemKind::Plan => Some("pl"),
43    }
44}

46/// Canonical filename for an item — `issue-0003-memory-leak.md`,
47/// `0004-hybrid-store.md`. Slug = `slug::slugify(title)`,
48/// `"untitled"` when the title slugifies to nothing.
49pub fn item_filename(id: &ItemId, title: &str) -> Option<String> {
50    let prefix = file_prefix(id.kind())?;
51    let slug = {
52        let s = slug::slugify(title);
53        if s.is_empty() {
54            "untitled".to_string()
55        } else {
56            s
57        }
58    };
59    Some(if prefix.is_empty() {
60        format!("{:04}-{slug}.md", id.number())
61    } else {
62        format!("{prefix}-{:04}-{slug}.md", id.number())
63    })
64}

66/// `issues/issue-0003-x.md` → `I-003`; `None` when the filename
67/// doesn't fit the kind's pattern.
68pub fn parse_item_filename(kind: ItemKind, filename: &str) -> Option<ItemId> {
69    let prefix = file_prefix(kind)?;
70    let stem = filename.strip_suffix(".md")?;
71    let rest = if prefix.is_empty() {
72        stem
73    } else {
74        stem.strip_prefix(prefix)?.strip_prefix('-')?
75    };
76    let (digits, slug) = rest.split_once('-')?;
77    if digits.is_empty()
78        || digits.len() > 9
79        || !digits.bytes().all(|b| b.is_ascii_digit())
80        || slug.is_empty()
81    {
82        return None;
83    }
84    let number: u32 = digits.parse().ok()?;
85    if number == 0 {
86        return None;
87    }
88    Some(ItemId::new(kind, number))
89}

91/// Deepest line the `frontmatter_id` probe will read — real
92/// frontmatter is ~10 keys; past this the `+++` block is
93/// certainly unclosed (or the file is pathological).
94const FRONTMATTER_PROBE_MAX: usize = 128;
95
96/// `id = "…"` pulled out of a *closed* `+++` frontmatter block —
97/// the cheap probe for when the strict parse isn't needed (or
98/// already failed). `None` for files with no `+++` fence, no
99/// closing fence within [`FRONTMATTER_PROBE_MAX`] lines, or no
100/// parseable `id` line — including frontmatter-less files that
101/// merely *borrow* an id number via their filename. The close
102/// requirement is what keeps a body's `id = "…"` (a code block,
103/// a quoted example) from posing as frontmatter.
104pub fn frontmatter_id(src: &str) -> Option<ItemId> {
105    let mut lines = src.lines();
106    if lines.next()?.trim() != "+++" {
107        return None;
108    }
109    let mut found = None;
110    for l in lines.take(FRONTMATTER_PROBE_MAX) {
111        let t = l.trim();
112        // the close ends the block — an id only counts when the
113        // block actually closed (else a body's `id = "…"` would
114        // pose as frontmatter)
115        if t == "+++" {
116            return found;
117        }
118        if found.is_none() {
119            if let Some(rest) = t.strip_prefix("id") {
120                if let Some(v) = rest.trim_start().strip_prefix('=') {
121                    if let Ok(id) = v.trim().trim_matches('"').parse::<ItemId>() {
122                        found = Some(id);
123                    }
124                }
125            }
126        }
127    }
128    None // window exhausted — never closed
129}

131/// Frontmatter DTO — stringly-typed so bad values point at the

153/// Emit shape — field order IS the frontmatter key order.
154#[derive(Serialize)]
155struct FrontmatterOut<'a> {
156    id: String,
157    title: &'a str,
158    status: Status,
159    #[serde(skip_serializing_if = "Option::is_none")]
160    created: Option<String>,
161    #[serde(skip_serializing_if = "Option::is_none")]
162    owner: Option<String>,
163    priority: Priority,
164    #[serde(skip_serializing_if = "Vec::is_empty")]
165    labels: Vec<&'a str>,
166    #[serde(skip_serializing_if = "Vec::is_empty")]
167    blocks: Vec<String>,
168    #[serde(skip_serializing_if = "Option::is_none")]
169    note: Option<&'a str>,
170    #[serde(skip_serializing_if = "Option::is_none")]

178}

180/// Parse an item file's source. `expected_kind` comes from the
181/// directory the file was found in — a mismatch (`P-002` inside
182/// `issues/`) is an error.
183pub fn parse_item_file(
184    src: &str,
185    expected_kind: ItemKind,
186    path: &str,
187) -> Result<ItemFile, ParseError> {
188    let src = mdx::normalize(src);
189    let at = |off: usize, len: usize, m: String| ParseError::at(path, src.clone(), off, len, m);
190    let whole = |m: &str| ParseError::whole_file(path, src.clone(), m);
191    let mut lines = src.split_inclusive('\n');
192    if lines.next().map(str::trim_end) != Some("+++") {
193        return Err(whole("expected `+++` TOML frontmatter fence on line 1"));
194    }
195    // find the closing `+++` line (byte offsets into src)
196    let fm_start = "+++\n".len();
197    let mut fm_end = None;
198    let mut off = fm_start;
199    for line in lines {
200        if line.trim_end() == "+++" {
201            fm_end = Some((off, off + line.len()));
202            break;
203        }
204        off += line.len();
205    }
206    let Some((fm_end, fence_end)) = fm_end else {
207        return Err(whole("unclosed `+++` frontmatter fence"));
208    };
209    let fm_src = &src[fm_start..fm_end];
210    let fm: FrontmatterIn = toml::from_str(fm_src).map_err(|e| {
211        let (o, l) = e
212            .span()
213            .map(|s| (fm_start + s.start, s.end.saturating_sub(s.start)))
214            .unwrap_or((fm_start, fm_end - fm_start));
215        at(o, l, e.to_string())
216    })?;
217    // DTO → Item; value errors point at the whole frontmatter block.
218    let fm_err = |m: String| at(fm_start, fm_end - fm_start, m);
219    let id: ItemId = fm
220        .id
221        .parse()
222        .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?;
223    if id.kind() != expected_kind {
224        return Err(fm_err(format!(
225            "id {id} doesn't belong in this file (expected a {} id)",
226            expected_kind.prefix()
227        )));
228    }
229    // filename ↔ frontmatter: a canonical filename's number must
230    // agree with the id the file claims — `issue-0003-x.md`
231    // holding `I-009` reserves 3 in the id scan while claiming 9,
232    // which would let the allocator reissue a live id. A name
233    // that doesn't fit the kind's pattern carries no number to
234    // check against (that's a filename finding, not this error).
235    let basename = std::path::Path::new(path)
236        .file_name()
237        .and_then(|n| n.to_str())
238        .unwrap_or(path);
239    if let Some(fname_id) = parse_item_filename(expected_kind, basename) {
240        if fname_id != id {
241            return Err(fm_err(format!(
242                "filename says {fname_id} but frontmatter id is {id}"
243            )));
244        }
245    }
246    let owner = fm
247        .owner
248        .map(|o| Actor::new(&o).map_err(|e| fm_err(e.to_string())))
249        .transpose()?;
250    let created = fm
251        .created
252        .map(|c| date::parse(&c).map_err(|e| fm_err(format!("bad created date — {e}"))))
253        .transpose()?;
254    let mut labels = BTreeSet::new();
255    for l in fm.labels {
256        if !myco_core::is_valid_label(&l) {
257            return Err(fm_err(format!("invalid label {l:?}")));
258        }
259        labels.insert(l);
260    }
261    let mut blocks = BTreeSet::new();
262    for b in fm.blocks {
263        blocks.insert(
264            b.parse::<ItemId>()
265                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?,
266        );
267    }
268    let mut item = Item::new(id, fm.title);
269    item.status = fm.status;
270    item.priority = fm.priority.unwrap_or_default();
271    item.owner = owner;
272    item.labels = labels;
273    item.blocks = blocks;
274    item.created = created;
275    item.note = fm.note;
276    item.superseded_by = fm
277        .superseded_by
278        .map(|s| {
279            s.parse::<ItemId>()
280                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
281        })
282        .transpose()?;
283    item.supersedes = fm
284        .supersedes
285        .map(|s| {
286            s.parse::<ItemId>()
287                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
288        })
289        .transpose()?;
290    for (key, name) in [("epic", &fm.epic), ("milestone", &fm.milestone)] {
291        if let Some(name) = name {
292            if !myco_core::is_valid_group_name(name) {
293                return Err(fm_err(format!("invalid {key} name {name:?}")));
294            }
295        }
296    }
297    item.epic = fm.epic;
298    item.milestone = fm.milestone;
299    // body: everything after the closing fence + one blank line
300    let mut body = &src[fence_end..];
301    body = body.strip_prefix('\n').unwrap_or(body);
302    let file = ItemFile {
303        meta: item,
304        body: body.trim().to_string(),
305    };
306    // parse-safe: the body must be markdown the parser accepts
307    if !file.body.is_empty() {
308        mdx::parse(&file.body).map_err(|m| at(fence_end, src.len() - fence_end, m))?;
309    }
310    Ok(file)
311}

313/// Canonical item file — `+++\n<toml>+++\n\n<body>\n`. The model
314/// is validated first ([`line::check_item`] — the same validity
315/// source as the line grammar) so emit can't write a file parse
316/// would reject; children are an error, never silently dropped.
317pub fn emit_item_file(file: &ItemFile) -> Result<String, EmitError> {
318    line::check_item(&file.meta, "item file")?;
319    if !file.meta.children.is_empty() {
320        return Err(EmitError::new(
321            "item file",
322            format!(
323                "{} carries children — sub-items only exist in TODO.md/BACKLOG.md",
324                file.meta.id
325            ),
326        ));
327    }
328    let fm = FrontmatterOut {
329        id: file.meta.id.to_string(),
330        title: &file.meta.title,
331        status: file.meta.status,
332        created: file.meta.created.map(date::render),
333        owner: file.meta.owner.as_ref().map(ToString::to_string),
334        priority: file.meta.priority,
335        labels: file.meta.labels.iter().map(String::as_str).collect(),
336        blocks: file.meta.blocks.iter().map(ItemId::to_string).collect(),
337        note: file.meta.note.as_deref(),
338        superseded_by: file.meta.superseded_by.map(|i| i.to_string()),
339        supersedes: file.meta.supersedes.map(|i| i.to_string()),
340        epic: file.meta.epic.as_deref(),
341        milestone: file.meta.milestone.as_deref(),
342    };
343    let toml = toml::to_string(&fm).map_err(|e| EmitError::new("item file", e.to_string()))?;
344    let mut out = String::from("+++\n");
345    out.push_str(&toml);
346    out.push_str("+++\n");
347    if !file.body.is_empty() {
348        out.push('\n');
349        out.push_str(file.body.trim_end());
350        out.push('\n');
351    }
352    Ok(out)
353}

552        assert!(parse_item_file(src, ItemKind::Issue, "issues/x.md").is_ok());
553        // kind mismatch stays a kind error (checked first)
554        let src = "+++\nid = \"P-002\"\ntitle = \"x\"\nstatus = \"open\"\n+++\n";
555        assert!(parse_item_file(src, ItemKind::Issue, "issues/issue-0002-x.md").is_err());
556        // but parses under its own kind even inside issues/ — the
557        // filename's `issue` prefix claims no proposal number
558        assert!(parse_item_file(src, ItemKind::Proposal, "issues/issue-0002-x.md").is_ok());
559    }

561    #[test]
562    fn emit_validates_the_model() {
563        let good = ItemFile::new(Item::new(ItemId::new(ItemKind::Issue, 1), "x"));
564        assert!(emit_item_file(&good).is_ok());
565        // an invalid label — parse would reject the emitted file
566        let mut bad = good.clone();
567        bad.meta.labels.insert("bad label".into());
568        assert!(emit_item_file(&bad).is_err());
569        // invalid group names too
570        let mut bad = good.clone();
571        bad.meta.epic = Some("bad epic".into());
572        assert!(emit_item_file(&bad).is_err());
573        bad.meta.epic = None;
574        bad.meta.milestone = Some("bad/ms".into());
575        assert!(emit_item_file(&bad).is_err());
576        // an empty/unemittable note
577        let mut bad = good.clone();
578        bad.meta.note = Some(String::new());
579        assert!(emit_item_file(&bad).is_err());
580        // children are an error, never silently dropped
581        let mut bad = good;
582        bad.meta
583            .children
584            .push(Item::new(ItemId::new(ItemKind::Task, 1), "sub"));
585        let e = emit_item_file(&bad).unwrap_err();
586        assert!(e.to_string().contains("children"), "{e}");
587    }
