1//! myco-index-redb — the redb impl of [`myco_index::Index`].
2//!
3//! Layout under the index dir (`$MYCO_HOME`, else
4//! `$XDG_DATA_HOME/myco`, else `~/.local/share/myco`):
5//!
6//! - `index.redb` — one `snapshots` table: canonical repo path →
7//!   JSON [`RepoSnapshot`].
8//! - `repos.toml` — the registry (`[[repo]] path name`), kept as a
9//!   plain file so it stays human-inspectable.
10//!
11//! Everything here is a derived cache — `rm -rf` the dir and
12//! `myco index rebuild` derives it back from the worktrees.
13
14use std::path::PathBuf;
15use std::time::{Duration, Instant};
16
17use camino::{Utf8Path, Utf8PathBuf};
18use myco_index::{Error, Index, RepoEntry, RepoSnapshot};
19use redb::{Database, ReadableDatabase, TableDefinition};
20
21/// `snapshots` table — repo path → JSON [`RepoSnapshot`].
22const SNAPSHOTS: TableDefinition<&str, &str> = TableDefinition::new("snapshots");
23
24/// The registry file name inside the index dir.
25const REGISTRY: &str = "repos.toml";
26
27/// The redb file name inside the index dir.
28const DB: &str = "index.redb";
29
30/// Total time a contended index lock is retried before giving up —
31/// lets concurrent `myco` processes queue instead of crash.
32const LOCK_RETRY_BUDGET: Duration = Duration::from_secs(2);
33
34/// First backoff after a busy lock; doubles each retry.
35const LOCK_RETRY_MIN: Duration = Duration::from_millis(10);
36
37/// Longest backoff between retries.
38const LOCK_RETRY_MAX: Duration = Duration::from_millis(320);
39
40/// The index dir: `$MYCO_HOME` → `$XDG_DATA_HOME/myco` →
41/// `~/.local/share/myco`. The `$MYCO_HOME` override is how tests
42/// (and unusual setups) relocate the whole cache.
43pub fn data_dir() -> PathBuf {
44    if let Some(d) = std::env::var_os("MYCO_HOME") {
45        return PathBuf::from(d);
46    }
47    if let Some(d) = std::env::var_os("XDG_DATA_HOME") {
48        return PathBuf::from(d).join("myco");
49    }
50    std::env::var("HOME")
51        .map(PathBuf::from)
52        .unwrap_or_else(|_| PathBuf::from("."))
53        .join(".local/share/myco")
54}

56/// `repos.toml` on disk.
57#[derive(Debug, Default, serde::Serialize, serde::Deserialize)]
58struct Registry {
59    #[serde(default)]
60    repo: Vec<RepoEntry>,
61}

63/// Open `index.redb` at `path`, retrying on lock contention
64/// ([`redb::DatabaseError::DatabaseAlreadyOpen`]) with exponential
65/// backoff until `budget` is spent. Anything else — corrupt file,
66/// IO, upgrade required — fails immediately.
67fn open_db(path: &Utf8Path, budget: Duration) -> Result<Database, Error> {
68    let start = Instant::now();
69    let mut backoff = LOCK_RETRY_MIN;
70    let mut attempt = 0u64;
71    loop {
72        match Database::create(path.as_std_path()) {
73            Ok(db) => return Ok(db),
74            Err(redb::DatabaseError::DatabaseAlreadyOpen) => {
75                let left = budget.saturating_sub(start.elapsed());
76                if left.is_zero() {
77                    return Err(Error::Backend(format!(
78                        "index busy — another myco process holds {DB} (waited ~{:.1}s)",
79                        budget.as_secs_f64()
80                    )));
81                }
82                // counter-based stagger — spreads contenders, no rand dep
83                let jitter = Duration::from_millis((attempt % 4) * 5);
84                std::thread::sleep((backoff + jitter).min(left));
85                backoff = (backoff * 2).min(LOCK_RETRY_MAX);
86                attempt += 1;
87            }
88            Err(e) => return Err(Error::Backend(format!("open {path}: {e}"))),
89        }
90    }
91}

93/// redb-backed index rooted at an index dir.
94pub struct RedbIndex {
95    dir: Utf8PathBuf,
96}
97
98impl RedbIndex {
99    /// Open the default index ([`data_dir`]), creating it.
100    pub fn open() -> Result<Self, Error> {
101        let dir = Utf8PathBuf::from_path_buf(data_dir())
102            .map_err(|p| Error::Backend(format!("{}: non-utf8 index dir", p.display())))?;
103        Self::open_at(dir)
104    }
105
106    /// Open an index rooted at `dir`, creating it when missing.
107    pub fn open_at(dir: impl Into<Utf8PathBuf>) -> Result<Self, Error> {
108        let dir = dir.into();
109        std::fs::create_dir_all(&dir)?;
110        Ok(RedbIndex { dir })
111    }
112
113    /// The index dir.
114    pub fn dir(&self) -> &Utf8Path {
115        &self.dir
116    }
117
118    fn registry_path(&self) -> Utf8PathBuf {
119        self.dir.join(REGISTRY)
120    }
121
122    fn db_path(&self) -> Utf8PathBuf {
123        self.dir.join(DB)
124    }
125
126    fn load_registry(&self) -> Result<Registry, Error> {
127        match std::fs::read_to_string(self.registry_path()) {
128            Ok(src) => toml::from_str(&src).map_err(|e| Error::Codec(format!("{REGISTRY}: {e}"))),
129            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(Registry::default()),
130            Err(e) => Err(Error::Io(e)),
131        }
132    }
133
134    fn save_registry(&self, reg: &Registry) -> Result<(), Error> {
135        let src = toml::to_string(reg).map_err(|e| Error::Codec(format!("{REGISTRY}: {e}")))?;
136        let path = self.registry_path();
137        let tmp = path.with_extension("toml.tmp");
138        std::fs::write(&tmp, src)?;
139        std::fs::rename(&tmp, &path)?;
140        Ok(())
141    }
142
143    fn db(&self) -> Result<Database, Error> {
144        open_db(&self.db_path(), LOCK_RETRY_BUDGET)
145    }
146
147    fn write_snapshot(&self, path: &Utf8Path, value: Option<&str>) -> Result<(), Error> {
148        let db = self.db()?;
149        let tx = db
150            .begin_write()
151            .map_err(|e| Error::Backend(format!("begin_write: {e}")))?;
152        {
153            let mut t = tx
154                .open_table(SNAPSHOTS)
155                .map_err(|e| Error::Backend(format!("open snapshots: {e}")))?;
156            match value {
157                Some(v) => {
158                    t.insert(path.as_str(), v)
159                        .map_err(|e| Error::Backend(format!("insert: {e}")))?;
160                }
161                None => {
162                    t.remove(path.as_str())
163                        .map_err(|e| Error::Backend(format!("remove: {e}")))?;
164                }
165            }
166        }
167        tx.commit()
168            .map_err(|e| Error::Backend(format!("commit: {e}")))
169    }
170}

172impl Index for RedbIndex {
173    fn register(&self, repo: &RepoEntry) -> Result<(), Error> {
174        let mut reg = self.load_registry()?;
175        match reg.repo.iter_mut().find(|r| r.path == repo.path) {
176            Some(r) => r.name = repo.name.clone(),
177            None => reg.repo.push(repo.clone()),
178        }
179        self.save_registry(&reg)
180    }
181
182    fn remove(&self, path: &Utf8Path) -> Result<bool, Error> {
183        let mut reg = self.load_registry()?;
184        let before = reg.repo.len();
185        reg.repo.retain(|r| r.path != path);
186        if reg.repo.len() == before {
187            return Ok(false);
188        }
189        self.save_registry(&reg)?;
190        self.write_snapshot(path, None)?;
191        Ok(true)
192    }
193
194    fn repos(&self) -> Result<Vec<RepoEntry>, Error> {
195        let mut repos = self.load_registry()?.repo;
196        repos.sort_by(|a, b| a.path.cmp(&b.path));
197        Ok(repos)
198    }
199
200    fn snapshot(&self, path: &Utf8Path) -> Result<Option<RepoSnapshot>, Error> {
201        let db = self.db()?;
202        let tx = db
203            .begin_read()
204            .map_err(|e| Error::Backend(format!("begin_read: {e}")))?;
205        let t = match tx.open_table(SNAPSHOTS) {
206            Ok(t) => t,
207            Err(_) => return Ok(None), // no table → no snapshots yet
208        };
209        let Some(v) = t
210            .get(path.as_str())
211            .map_err(|e| Error::Backend(format!("get: {e}")))?
212        else {
213            return Ok(None);
214        };
215        serde_json::from_str(v.value())
216            .map(Some)
217            .map_err(|e| Error::Codec(format!("snapshot {path}: {e}")))
218    }
219
220    fn save_snapshot(&self, snapshot: &RepoSnapshot) -> Result<(), Error> {
221        let json = serde_json::to_string(snapshot)
222            .map_err(|e| Error::Codec(format!("snapshot {}: {e}", snapshot.path)))?;
223        self.write_snapshot(&snapshot.path, Some(&json))
224    }
225}

244    fn snap(e: &RepoEntry, head: &str) -> RepoSnapshot {
245        RepoSnapshot {
246            path: e.path.clone(),
247            name: e.name.clone(),
248            head_commit: head.into(),
249            scanned_at: "2026-02-09T00:00:00Z".into(),

255    #[test]
256    fn register_list_remove_cycle() {
257        let (_d, i) = idx();
258        assert!(i.repos().unwrap().is_empty());
259        i.register(&entry("b")).unwrap();
260        i.register(&entry("a")).unwrap();
261        // re-register updates the name, doesn't duplicate
262        i.register(&RepoEntry::new(Utf8PathBuf::from("/repos/b"), "b-renamed"))
263            .unwrap();
264        let repos = i.repos().unwrap();
265        assert_eq!(repos.len(), 2);
266        // sorted by path
267        assert_eq!(repos[0].path.as_str(), "/repos/a");
268        assert_eq!(repos[1].name, "b-renamed");
269        // registry persists on disk as plain toml
270        let src = std::fs::read_to_string(i.registry_path()).unwrap();
271        assert!(src.contains("/repos/a"));
272        // remove drops the entry + its snapshot
273        i.save_snapshot(&snap(&repos[1], "h1")).unwrap();
274        assert!(i.remove(&repos[1].path).unwrap());
275        assert!(!i.remove(&repos[1].path).unwrap());
276        assert_eq!(i.repos().unwrap().len(), 1);
277        assert!(i.snapshot(&repos[1].path).unwrap().is_none());
278    }

280    #[test]
281    fn snapshot_roundtrip_and_overwrite() {
282        let (_d, i) = idx();
283        let e = entry("x");
284        assert!(i.snapshot(&e.path).unwrap().is_none());
285        i.save_snapshot(&snap(&e, "h1")).unwrap();
286        let got = i.snapshot(&e.path).unwrap().unwrap();
287        assert_eq!(got.head_commit, "h1");
288        assert_eq!(got.items.len(), 1);
289        // overwrite — freshness check sees the new head
290        i.save_snapshot(&snap(&e, "h2")).unwrap();
291        assert_eq!(i.snapshot(&e.path).unwrap().unwrap().head_commit, "h2");
292        // survives a reopen
293        let i2 = RedbIndex::open_at(i.dir().to_path_buf()).unwrap();
294        assert_eq!(i2.snapshot(&e.path).unwrap().unwrap().head_commit, "h2");
295    }

349    #[test]
350    fn contended_open_retries_until_free() {
351        let (_d, i) = idx();
352        // a second "myco" holds the lock for a beat, then drops
353        let holder = Database::create(i.db_path().as_std_path()).unwrap();
354        let release = std::thread::spawn(move || {
355            std::thread::sleep(Duration::from_millis(80));
356            drop(holder);
357        });
358        // write queues behind the holder instead of dying
359        let e = entry("busy");
360        i.save_snapshot(&snap(&e, "h1")).unwrap();
361        release.join().unwrap();
362        assert_eq!(i.snapshot(&e.path).unwrap().unwrap().head_commit, "h1");
363    }
