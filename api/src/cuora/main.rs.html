1//! cuora — secrets broker for agents. Sealed, not spilled.
2//!
3//! Code and config carry `cuora://` references, never values; `cuora
4//! run` materializes declared secrets as env vars in the child only.
5//! Every materialization — and every denial/failure — emits an
6//! audit event: ndjson on stderr, plus OTLP spans when
7//! `OTEL_EXPORTER_OTLP_ENDPOINT` is set.
8//!
9//! Values never enter via CLI args — `put`/`rotate` read stdin or a
10//! hidden `/dev/tty` prompt (args land in shell history and
11//! `/proc`). `get` is the escape hatch that prints a value to
12//! stdout; `[guardrails]` can advisory-deny it for agents.
13

22/// cuora — secrets broker for agents: `cuora://` refs, redaction,
23/// env-injection instead of plaintext.
24///
25/// Refs name logical secrets (`cuora://billing/stripe`); `--env`
26/// picks the resolution context. `[secrets]` in cuora.toml bounds
27/// what `run`/`env` may inject. `manifest`/`schema` describe the
28/// surface for machines; every verb honors `--json`.
29#[derive(Parser)]
30#[command(name = "cuora", version, about, long_about)]
31struct Cli {
32    /// Emit JSON — stable machine shapes on every verb; errors are
33    /// `{"error": "…"}` on stderr with a non-zero exit.
34    #[arg(long, global = true)]
35    json: bool,
36
37    /// Env context — `dev`/`staging`/`prod`/custom. Overrides
38    /// `CUORA_ENV`; default `dev`.
39    #[arg(long, global = true, value_name = "ENV")]
40    env: Option<String>,
41
42    /// Config file, overriding the search
43    /// ($CUORA_CONFIG → ./cuora.toml → ~/.config/cuora/cuora.toml).
44    #[arg(long, global = true, value_name = "PATH")]
45    config: Option<String>,
46
47    /// Actor attribution for audit events (default $CUORA_ACTOR,
48    /// else $USER).
49    #[arg(long, global = true, value_name = "ACTOR")]
50    actor: Option<String>,
51
52    #[command(subcommand)]
53    command: Cmd,
54}

56#[derive(Subcommand)]
57enum Cmd {
58    /// Bootstrap: store the age identity in the system keyring
59    /// when a session secret store is reachable — importing an
60    /// existing identity.txt rather than orphaning it — else the
61    /// `~/.config/cuora/identity.txt` keyfile (0600). Verifies the
62    /// vault opens.
63    Init {
64        /// Force the identity keyfile even when a keyring is
65        /// available — the headless/CI mode.
66        #[arg(long)]
67        with_file: bool,
68    },
69    /// Store a secret. The value comes from stdin (`printf … | cuora
70    /// put cuora://ns/name`) or a hidden terminal prompt — never
71    /// args, which land in shell history and `/proc`.
72    Put {
73        /// `cuora://` ref, e.g. cuora://billing/stripe.
74        rref: String,
75    },
76    /// Print a secret's value to stdout — the escape hatch. Prefer
77    /// `cuora run` so values never enter agent context; guardrails
78    /// can deny this op.
79    Get {
80        /// `cuora://` ref.
81        rref: String,
82    },
83    /// List the refs stored in the selected env — names only, never
84    /// values.
85    List,
86    /// Remove a secret from the selected env.
87    Rm {
88        /// `cuora://` ref.
89        rref: String,
90    },
91    /// Rotate a secret to a new value (stdin or hidden prompt, like
92    /// `put`); bumps version + rotated_at.
93    Rotate {
94        /// `cuora://` ref.
95        rref: String,
96    },
97    /// Run a command with declared secrets injected as env vars.
98    ///
99    /// Injection surface = `[secrets]` in cuora.toml plus `-s
100    /// VAR=cuora://ref` ad-hoc pairs; nothing else is injectable.
101    /// Two exec modes after `--`:
102    ///
103    /// `cuora run -- 'pnpm test | tee log'` — one string → `bash -c`
104    ///
105    /// `cuora run -- pnpm test --filter x` — argv → direct exec
106    ///
107    /// argv mode needs no shell quoting — prefer it when the
108    /// command has arguments. Exit code is the child's.
109    Run {
110        /// Extra `VAR=cuora://ref` pairs to inject (refused when
111        /// `[guardrails] deny_adhoc` is set).
112        #[arg(short = 's', long = "secret", value_name = "VAR=CUORA_REF")]
113        secrets: Vec<String>,
114        /// Command to run (after `--`). One arg → `bash -c`;
115        /// several → `argv[0]` exec'd directly with the rest.
116        #[arg(last = true, required = true)]
117        cmd: Vec<String>,
118    },
119    /// Print `export VAR='…'` lines for `eval "$(cuora env)"` — the
120    /// human path (`run` is the agent path). Values print to
121    /// stdout: `--json` emits a `VAR → value` map.
122    Env {
123        /// Extra `VAR=cuora://ref` pairs to resolve.
124        #[arg(short = 's', long = "secret", value_name = "VAR=CUORA_REF")]
125        secrets: Vec<String>,
126    },
127    /// Validate config + the declared surface: `[envs]` routes
128    /// point at declared vaults, vaults decrypt, every `[secrets]`
129    /// ref resolves in its routed env. Exit 1 on findings.
130    Check,
131    /// Rewrite stored records for a wire/config upgrade. Currently
132    /// one transform: `--private-refs` re-keys every plaintext-named
133    /// row in each env routed to a cf vault with
134    /// `[vaults.*] private_refs = true` — ciphertext moves verbatim,
135    /// old keys are deleted after the new rows land.
136    Migrate {
137        /// Re-key cf-vault rows under HMAC-derived storage keys,
138        /// attaching the encrypted `name` blob (P-004).
139        #[arg(long)]
140        private_refs: bool,
141    },
142    /// Push a stored secret to a write-only deploy target
143    /// (`[deploy.NAME]` in cuora.toml, e.g. a CF Secrets Store).
144    /// The vault is the source of truth; the sink never gives the
145    /// value back. With one configured target, `--target` may be
146    /// omitted.
147    Deploy {
148        /// `cuora://` ref to push.
149        rref: String,
150        /// Deploy target name (`[deploy.NAME]`); required when
151        /// several are configured.
152        #[arg(long, value_name = "NAME")]
153        target: Option<String>,
154    },
155    /// Serve the Vault port over ndjson — one JSON request per
156    /// stdin line, one response per stdout line, audit events on
157    /// stderr unchanged. For orchestrators embedding cuora
158    /// (peregrin); with `[serve.pdp]` configured, a policy
159    /// decision point is consulted before every vault-touching
160    /// op. `{"op":"manifest"}` on the wire describes the protocol.
161    Serve,
162    /// Capabilities, JSON shapes, env vars, audit contract — agent
163    /// self-discovery (always JSON).
164    Manifest,
165    /// JSON Schema for `cuora.toml`.
166    Schema,
167    /// Print the CLI reference as markdown (docs pipeline —
168    /// `bun run docs:cli` pipes it to `docs/reference/cli.md`).
169    #[command(hide = true, name = "gen-cli-docs")]
170    GenCliDocs,
171}

173fn main() -> ExitCode {
174    sigpipe::reset();
175    let cli = Cli::parse();
176    let code = dispatch(&cli);
177    // Buffered audit events export as OTLP spans here — every
178    // exit path funnels through this one flush.
179    cuora_ops::audit::flush();
180    code
181}
182
183fn dispatch(cli: &Cli) -> ExitCode {
184    match &cli.command {
185        // Pure introspection — before config load, no audit event.
186        Cmd::Manifest => json_out(&manifest()),
187        Cmd::GenCliDocs => out(&clap_markdown::help_markdown::<Cli>()),
188        Cmd::Schema => out(&schema_json()),
189        cmd => {
190            let op = cmd_op(cmd);
191            let rref = cmd_rref(cmd);
192            let ops = match load(cli) {
193                Ok(ops) => ops,
194                Err(e) => {
195                    audit_fallback(cli, op, rref, Outcome::Error);
196                    return fail(cli, &e.to_string());
197                }
198            };
199            // Advisory guardrails — accident prevention, not
200            // policy; enforced before the op touches the vault.
201            if let Some(g) = ops.config().guardrails.as_ref() {
202                if !g.allows(op) {
203                    let rref = norm_rref(&ops, rref);
204                    ops.emit(op, rref.as_deref(), Outcome::Denied);
205                    // guardrails only exist when a config file
206                    // loaded — config_path is always Some here
207                    #[allow(clippy::unwrap_used)]
208                    let at = ops
209                        .config_path()
210                        .expect("guardrails imply a loaded config file")
211                        .display()
212                        .to_string();
213                    return fail(
214                        cli,
215                        &format!(
216                            "op '{op}' denied by guardrails in {at} \
217                             (advisory — remove/relax the rule to allow)"
218                        ),
219                    );
220                }
221            }
222            match run_cmd(cmd, cli, &ops) {
223                Ok(code) => code,
224                Err(e) => {
225                    let rref = norm_rref(&ops, rref);
226                    ops.emit(op, rref.as_deref(), Outcome::Error);
227                    fail(cli, &e.to_string())
228                }
229            }
230        }
231    }
232}

234/// The [`Op`] a subcommand maps to — what guardrails and audit
235/// record.
236fn cmd_op(cmd: &Cmd) -> Op {
237    match cmd {
238        Cmd::Init { .. } => Op::Init,
239        Cmd::Put { .. } => Op::Put,
240        Cmd::Get { .. } => Op::Get,
241        Cmd::List => Op::List,
242        Cmd::Rm { .. } => Op::Rm,
243        Cmd::Rotate { .. } => Op::Rotate,
244        Cmd::Run { .. } => Op::Run,
245        Cmd::Env { .. } => Op::Env,
246        Cmd::Check => Op::Check,
247        Cmd::Migrate { .. } => Op::Migrate,
248        Cmd::Deploy { .. } => Op::Deploy,
249        Cmd::Serve => Op::Serve,
250        Cmd::Manifest => Op::Manifest,
251        Cmd::Schema => Op::Schema,
252        // Handled before ops load — this arm is only for
253        // exhaustiveness, never reached.
254        Cmd::GenCliDocs => Op::Schema,
255    }
256}

258/// The raw ref arg for verbs that take one — for denied/error
259/// audit events, before (or regardless of) parse.
260fn cmd_rref(cmd: &Cmd) -> Option<&str> {
261    match cmd {
262        Cmd::Put { rref }
263        | Cmd::Get { rref }
264        | Cmd::Rm { rref }
265        | Cmd::Rotate { rref }
266        | Cmd::Deploy { rref, .. } => Some(rref),
267        _ => None,
268    }
269}

271/// Normalize a requested ref to the resolved-path form success
272/// events carry (`env pin` stripped, no scheme) so the audit
273/// stream correlates regardless of outcome. Falls back to the raw
274/// arg when it doesn't parse — the honest record of a bad request.
275fn norm_rref(ops: &Ops, raw: Option<&str>) -> Option<String> {
276    raw.map(|s| {
277        Ref::new(s)
278            .map(|r| ops.resolve(&r).1.path().to_string())
279            .unwrap_or_else(|_| s.to_string())
280    })
281}

283/// Audit an op that never got an `Ops` (config/env/actor
284/// resolution failed) — best-effort actor + env from flags and
285/// process env.
286fn audit_fallback(cli: &Cli, op: Op, rref: Option<&str>, outcome: Outcome) {
287    let actor = cli
288        .actor
289        .clone()
290        .or_else(|| env_var("CUORA_ACTOR"))
291        .or_else(|| std::env::var("USER").ok())
292        .unwrap_or_else(|| "anon".into());
293    let env = cli
294        .env
295        .clone()
296        .or_else(|| env_var("CUORA_ENV"))
297        .unwrap_or_else(|| "dev".into());
298    let rref = rref.map(|s| {
299        Ref::new(s)
300            .map(|r| r.path().to_string())
301            .unwrap_or_else(|_| s.to_string())
302    });
303    cuora_ops::audit::emit(&AuditEvent::op(&actor, op, rref.as_deref(), &env, outcome));
304}

306fn load(cli: &Cli) -> Result<Ops, OpsError> {
307    Ops::load(LoadOpts {
308        config_path: cli.config.clone(),
309        env: cli.env.clone(),
310        actor: cli.actor.clone(),
311    })

314fn run_cmd(cmd: &Cmd, cli: &Cli, ops: &Ops) -> Result<ExitCode, OpsError> {
315    match cmd {
316        Cmd::Init { with_file } => {
317            let info = ops.init(&cuora_ops::InitOpts {
318                with_file: *with_file,
319            })?;
320            ops.emit(Op::Init, None, Outcome::Ok);
321            if cli.json {
322                return Ok(json_out(&serde_json::json!({
323                    "identity": info.identity,
324                    "backend": info.identity_backend,
325                    "created": info.identity_created,
326                    "migrated": info.migrated,
327                    "file_remains": info.file_remains,
328                    "recipient": info.recipient,
329                    "vault": info.vault_path,
330                })));
331            }
332            eprintln!(
333                "identity: {}{}",
334                info.identity,
335                if info.identity_created {
336                    " (generated)"
337                } else {
338                    ""
339                }
340            );
341            // Requires a session secret store holding no identity —
342            // unreachable on headless CI where platform() always
343            // fails; exercised at ops level where sources inject.
344            #[cfg(not(coverage))]
345            if info.migrated {
346                if let Some(p) = &info.file_remains {
347                    eprintln!(
348                        "migrated:  imported into the keyring; {} kept as a \
349                         backup — remove it when ready",
350                        p.display()
351                    );
352                }
353            }
354            eprintln!("recipient: {}", info.recipient);
355            eprintln!("vault:     {}", info.vault_path.display());
356            Ok(ExitCode::SUCCESS)
357        }
358        Cmd::Put { rref } => {
359            let parsed = parse_ref(rref)?;
360            let secret = read_secret(&format!("value for {rref}: "))?;
361            let rec = ops.put(&parsed, secret)?;
362            if cli.json {
363                let mut v = record_json(ops, &parsed, &rec);
364                v["ok"] = true.into();
365                return Ok(json_out(&v));
366            }
367            eprintln!("stored {rref} (v{})", rec.version);
368            Ok(ExitCode::SUCCESS)
369        }
370        Cmd::Get { rref } => {
371            let parsed = parse_ref(rref)?;
372            match ops.get(&parsed)? {
373                Some(rec) => {
374                    if cli.json {
375                        let mut v = record_json(ops, &parsed, &rec);
376                        v["value"] = rec.value.expose().into();
377                        return Ok(json_out(&v));
378                    }
379                    if std::io::stdout().is_terminal() {
380                        eprintln!("note: secret value printed to terminal");
381                    }
382                    println!("{}", rec.value.expose());
383                    Ok(ExitCode::SUCCESS)
384                }
385                None => Err(OpsError::Msg(format!(
386                    "{rref} is not stored in env {}",
387                    ops.resolve(&parsed).0
388                ))),
389            }
390        }
391        Cmd::List => {
392            let (env, refs) = ops.list(None)?;
393            ops.emit(Op::List, None, Outcome::Ok);
394            if cli.json {
395                let (_, records) = ops.list_records(None)?;
396                let arr: Vec<serde_json::Value> = records
397                    .iter()
398                    .map(|(r, rec)| {
399                        serde_json::json!({
400                            "ref": r.to_string(),
401                            "env": env.as_str(),
402                            "version": rec.version,
403                            "created_at": rec.created_at.to_string(),
404                            "rotated_at": rec.rotated_at.map(|t| t.to_string()),
405                        })
406                    })
407                    .collect();
408                return Ok(json_out(&serde_json::Value::Array(arr)));
409            }
410            for r in refs {
411                println!("{r}");
412            }
413            Ok(ExitCode::SUCCESS)
414        }
415        Cmd::Rm { rref } => {
416            let parsed = parse_ref(rref)?;
417            let (env, _) = ops.resolve(&parsed);
418            if !ops.rm(&parsed)? {
419                return Err(OpsError::Msg(format!("{rref} is not stored in env {env}")));
420            }
421            if cli.json {
422                return Ok(json_out(&serde_json::json!({
423                    "ok": true,
424                    "removed": true,
425                    "ref": parsed.to_string(),
426                    "env": env.as_str(),
427                })));
428            }
429            eprintln!("removed {rref}");
430            Ok(ExitCode::SUCCESS)
431        }
432        Cmd::Rotate { rref } => {
433            let parsed = parse_ref(rref)?;
434            let secret = read_secret(&format!("new value for {rref}: "))?;
435            let rec = ops.rotate(&parsed, secret)?;
436            if cli.json {
437                let mut v = record_json(ops, &parsed, &rec);
438                v["ok"] = true.into();
439                return Ok(json_out(&v));
440            }
441            eprintln!("rotated {rref} (v{})", rec.version);
442            Ok(ExitCode::SUCCESS)
443        }
444        Cmd::Run { secrets, cmd } => {
445            let vars = ops.resolve_for_injection(Op::Run, &parse_pairs(secrets)?)?;
446            let exposed: Vec<(String, String)> = vars
447                .iter()
448                .map(|(k, v)| (k.clone(), v.expose().to_string()))
449                .collect();
450            let cwd = std::env::current_dir()?;
451            // One arg → `bash -c` (shell syntax); several → exec
452            // argv[0] directly, no shell quoting.
453            let status = if cmd.len() == 1 {
454                cuora_shell::run_with_env(&cwd, &cmd[0], &exposed)?
455            } else {
456                cuora_shell::exec_with_env(&cwd, cmd, &exposed)?
457            };
458            Ok(ExitCode::from(status.code().unwrap_or(1) as u8))
459        }
460        Cmd::Env { secrets } => {
461            let vars = ops.resolve_for_injection(Op::Env, &parse_pairs(secrets)?)?;
462            if cli.json {
463                let map: BTreeMap<&String, &str> =
464                    vars.iter().map(|(k, v)| (k, v.expose())).collect();
465                return Ok(json_out(&serde_json::json!(map)));
466            }
467            let stdout = std::io::stdout();
468            let mut w = stdout.lock();
469            for (k, v) in &vars {
470                writeln!(w, "export {k}={}", sh_escape(v.expose()))?;
471            }
472            Ok(ExitCode::SUCCESS)
473        }
474        Cmd::Check => {
475            let report = ops.check();
476            // A check that found problems exits 1 — the audit
477            // outcome mirrors it, not a blanket `ok`.
478            ops.emit(
479                Op::Check,
480                None,
481                if report.problems.is_empty() {
482                    Outcome::Ok
483                } else {
484                    Outcome::Error
485                },
486            );
487            let ok = report.problems.is_empty();
488            if cli.json {
489                let secrets: serde_json::Map<String, serde_json::Value> = report
490                    .secrets
491                    .iter()
492                    .map(|s| {
493                        (
494                            s.var.clone(),
495                            serde_json::json!({
496                                "ref": s.rref,
497                                "env": s.env,
498                                "resolves": s.resolves,
499                            }),
500                        )
501                    })
502                    .collect();
503                let vaults: Vec<serde_json::Value> = report
504                    .vaults
505                    .iter()
506                    .map(|v| {
507                        serde_json::json!({
508                            "name": v.name,
509                            "kind": v.kind,
510                            "path": v.path,
511                            "ok": v.ok,
512                            "error": v.error,
513                        })
514                    })
515                    .collect();
516                let guardrails = report.guardrails.as_ref().map(|g| {
517                    serde_json::json!({
518                        "default": match g.default {
519                            cuora_core::GuardDefault::Allow => "allow",
520                            cuora_core::GuardDefault::Deny => "deny",
521                        },
522                        "allow": g.allow.iter().map(|o| o.as_str()).collect::<Vec<_>>(),
523                        "deny": g.deny.iter().map(|o| o.as_str()).collect::<Vec<_>>(),
524                        "deny_adhoc": g.deny_adhoc,
525                    })
526                });
527                let code = json_out(&serde_json::json!({
528                    "config": report.config_path,
529                    "env": report.env.as_str(),
530                    "vault": report.vault,
531                    "vaults": vaults,
532                    "secrets": secrets,
533                    "guardrails": guardrails,
534                    "errors": report.problems,
535                }));
536                debug_assert_eq!(code, ExitCode::SUCCESS);
537            } else {
538                match &report.config_path {
539                    Some(p) => eprintln!("config: {}", p.display()),
540                    None => eprintln!("config: (none found — defaults)"),
541                }
542                eprintln!("env: {} → vault {:?}", report.env, report.vault,);
543                for v in &report.vaults {
544                    match &v.error {
545                        None => eprintln!("vault {:?} ({}): readable", v.name, v.kind),
546                        Some(e) => eprintln!("vault {:?} ({}): {e}", v.name, v.kind),
547                    }
548                }
549                for s in &report.secrets {
550                    eprintln!(
551                        "secret {} → {} (env {}): {}",
552                        s.var,
553                        s.rref,
554                        s.env,
555                        if s.resolves { "resolves" } else { "NOT STORED" }
556                    );
557                }
558                if let Some(g) = &report.guardrails {
559                    let deny: Vec<_> = g.deny.iter().map(|o| o.as_str()).collect();
560                    eprintln!("guardrails: configured (deny: {})", deny.join(", "));
561                }
562                if report.problems.is_empty() {
563                    eprintln!("ok");
564                } else {
565                    for p in &report.problems {
566                        eprintln!("problem: {p}");
567                    }
568                }
569            }
570            Ok(if ok {
571                ExitCode::SUCCESS
572            } else {
573                ExitCode::FAILURE
574            })
575        }
576        Cmd::Migrate { private_refs } => {
577            if !private_refs {
578                return Err(OpsError::Msg(
579                    "migrate needs a target — currently only `--private-refs`".into(),
580                ));
581            }
582            let reports = ops.migrate_private_refs()?;
583            ops.emit(Op::Migrate, None, Outcome::Ok);
584            if cli.json {
585                let migrated: Vec<serde_json::Value> = reports
586                    .iter()
587                    .map(|r| {
588                        serde_json::json!({
589                            "vault": r.vault,
590                            "env": r.env.as_str(),
591                            "rewritten": r.rewritten,
592                            "already_private": r.already_private,
593                        })
594                    })
595                    .collect();
596                return Ok(json_out(&serde_json::json!({
597                    "ok": true,
598                    "migrated": migrated,
599                })));
600            }
601            for r in &reports {
602                eprintln!(
603                    "vault {:?} env {}: rewrote {} ref(s) ({} already private)",
604                    r.vault,
605                    r.env,
606                    r.rewritten.len(),
607                    r.already_private
608                );
609            }
610            Ok(ExitCode::SUCCESS)
611        }
612        Cmd::Deploy { rref, target } => {
613            let parsed = parse_ref(rref)?;
614            let report = ops.deploy(&parsed, target.as_deref())?;
615            if cli.json {
616                return Ok(json_out(&serde_json::json!({
617                    "ok": true,
618                    "ref": report.rref.to_string(),
619                    "env": report.env.as_str(),
620                    "target": report.target,
621                    "secret_name": report.secret_name,
622                    "created": report.created,
623                })));
624            }
625            eprintln!(
626                "deployed {} → {:?} as {} ({})",
627                report.rref,
628                report.target,
629                report.secret_name,
630                if report.created { "created" } else { "updated" }
631            );
632            Ok(ExitCode::SUCCESS)
633        }
634        Cmd::Serve => {
635            // Trust-root warning: a `[serve.pdp] exec` line is
636            // config that authorizes shell exec. When it arrives
637            // via the *implicit* ./cuora.toml search — not a pinned
638            // --config/$CUORA_CONFIG and not the user's home
639            // config — the working directory itself is the trust
640            // root, so say so loudly before serving.
641            let implicit_local = cli.config.is_none()
642                && std::env::var_os("CUORA_CONFIG").is_none()
643                && std::env::var_os("ARCA_CONFIG").is_none()
644                && ops.config_path().is_some_and(|p| {
645                    p == std::path::Path::new("cuora.toml")
646                        || p == std::path::Path::new("arca.toml")
647                });
648            if implicit_local
649                && ops
650                    .config()
651                    .serve
652                    .as_ref()
653                    .and_then(|s| s.pdp.as_ref())
654                    .is_some()
655            {
656                let config_name = ops
657                    .config_path()
658                    .map(|p| p.display().to_string())
659                    .unwrap_or_else(|| "cuora.toml".into());
660                eprintln!(
661                    "warning: {config_name} declares [serve.pdp] exec — \
662                     this config is a trust root authorizing shell execution \
663                     and was found on the implicit search path. \
664                     Pin it with --config or $CUORA_CONFIG if this is intended."
665                );
666            }
667            let stdin = std::io::stdin();
668            let stdout = std::io::stdout();
669            cuora_serve::run(ops, &mut stdin.lock(), &mut stdout.lock())?;
670            ops.emit(Op::Serve, None, Outcome::Ok);
671            Ok(ExitCode::SUCCESS)
672        }
673        // Introspection verbs are handled before ops load — this
674        // arm exists only for exhaustiveness and must never run.
675        Cmd::Manifest | Cmd::Schema | Cmd::GenCliDocs => {
676            eprintln!("internal error: introspection verb reached run_cmd");
677            Ok(ExitCode::FAILURE)
678        }
679    }
680}

682fn parse_ref(s: &str) -> Result<Ref, OpsError> {
683    Ok(s.parse::<Ref>()?)

686/// `-s VAR=cuora://ref` pairs.
687fn parse_pairs(raw: &[String]) -> Result<Vec<(String, String)>, OpsError> {
688    raw.iter()
689        .map(|s| {
690            s.split_once('=')
691                .map(|(k, v)| (k.to_string(), v.to_string()))
692                .ok_or_else(|| OpsError::Msg(format!("-s wants VAR=cuora://ref, got {s:?}")))
693        })
694        .collect()
695}

697/// Secrets arrive on stdin (the scripted path) or a hidden
698/// `/dev/tty` prompt (the human path) — never args. One trailing
699/// line terminator is stripped; the rest is the value verbatim
700/// (newlines allowed on the pipe path).
701fn read_secret(prompt: &str) -> Result<Secret, OpsError> {
702    let stdin = std::io::stdin();
703    let raw = if stdin.is_terminal() {
704        rpassword::prompt_password(prompt)
705            .map_err(|e| OpsError::Msg(format!("read secret: {e}")))?
706    } else {
707        let mut buf = String::new();
708        stdin
709            .lock()
710            .read_to_string(&mut buf)
711            .map_err(|e| OpsError::Msg(format!("read stdin: {e}")))?;
712        buf
713    };
714    let v = raw
715        .strip_suffix("\r\n")
716        .or_else(|| raw.strip_suffix('\n'))
717        .unwrap_or(&raw);
718    if v.is_empty() {
719        return Err(OpsError::Msg("empty secret".into()));
720    }
721    Ok(Secret::new(v))
722}

724/// Shell single-quote escape for `cuora env` output — `'` becomes
725/// `'\''`, so any value is safe inside the quotes.
726fn sh_escape(v: &str) -> String {
727    format!("'{}'", v.replace('\'', "'\\''"))
728}

878/// Error path honoring `--json`: `{"error": …}` on stderr for
879/// machines, `cuora: …` for humans.
880fn fail(cli: &Cli, msg: &str) -> ExitCode {
881    if cli.json {
882        eprintln!("{}", serde_json::json!({ "error": msg }));
883    } else {
884        eprintln!("cuora: {msg}");
885    }
886    ExitCode::FAILURE
887}
