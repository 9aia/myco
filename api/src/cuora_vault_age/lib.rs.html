1//! cuora-vault-age — the local vault, and the default backend
2//! forever (per PL-001 hard constraints).
3//!
4//! One age-x25519-encrypted JSON doc holds every env's records;
5//! `put`/`rm`/`rotate` rewrite the whole file (fine at v1 scale).
6//! Decrypted payload shape (`v` is the format version):
7//!
8//! ```json
9//! {
10//!   "v": 1,
11//!   "envs": {
12//!     "dev": {
13//!       "billing/stripe": {
14//!         "value": "…", "version": 1,
15//!         "created_at": "…", "rotated_at": "…"
16//!       }
17//!     }
18//!   }
19//! }
20//! ```
21//!
22//! The vault takes an already-resolved [`x25519::Identity`] —
23//! where the identity *lives* is the `cuora-identity` port's
24//! business (keyfile via `cuora-identity-file`, session keyring via
25//! `cuora-identity-keyring`; `cuora-ops` resolves it). Recipients
26//! default to that identity's public key plus any extra
27//! `recipients` in the vault's config.
28//! [`AgeVault::open_or_init`] is the bootstrap path — it runs
29//! [`cuora_identity::ensure`] on the caller's chosen source.
30

48/// The decrypted vault document. **No `Debug`** — the whole point
49/// of the payload is that it holds plaintext; a derived dump would
50/// print every value in the vault.
51#[derive(Default, Serialize, Deserialize)]
52struct Payload {
53    v: u32,
54    #[serde(default)]
55    envs: BTreeMap<String, BTreeMap<String, Entry>>,
56}

58/// One stored secret, as persisted (the plaintext lives only inside
59/// the ciphertext — this struct is the *decrypted* in-memory form).
60/// `Zeroizing` scrubs the value buffer on drop; no `Debug` for the
61/// same reason as [`Payload`].
62#[derive(Clone, Serialize, Deserialize)]
63struct Entry {
64    value: Zeroizing<String>,
65    #[serde(default = "one")]
66    version: u64,
67    #[serde(with = "time::serde::rfc3339")]
68    created_at: OffsetDateTime,
69    #[serde(default, with = "time::serde::rfc3339::option")]
70    rotated_at: Option<OffsetDateTime>,
71}

77/// How long a mutating verb waits for a held vault lock before
78/// failing — vault writes are milliseconds, so this is generous.
79const LOCK_WAIT: std::time::Duration = std::time::Duration::from_millis(2_000);
80/// A lockfile older than this is reclaimed — its holder is
81/// presumed dead (crash between create and remove).
82const LOCK_STALE: std::time::Duration = std::time::Duration::from_secs(120);
83
84/// Cross-process exclusion for the vault's read-modify-write
85/// cycle (MED-3). `flock`/`fcntl` aren't in std and a new
86/// dependency for one syscall is heavier than this: `<vault>.lock`
87/// is created with `create_new` (atomic `O_EXCL`), so exactly one
88/// process holds it. A stale lock from a dead holder is reclaimed
89/// after [`LOCK_STALE`]; `Drop` removes it.
90struct VaultLock {
91    path: PathBuf,
92}
93
94impl VaultLock {
95    fn acquire(vault: &Path) -> Result<Self, VaultError> {
96        let path = vault.with_extension("age.lock");
97        let deadline = std::time::Instant::now() + LOCK_WAIT;
98        loop {
99            match std::fs::OpenOptions::new()
100                .write(true)
101                .create_new(true)
102                .open(&path)
103            {
104                Ok(mut f) => {
105                    // pid + creation timestamp — provenance for a
106                    // human debugging a stuck lock, and the
107                    // staleness signal for the next acquirer
108                    let _ = writeln!(f, "{} {}", std::process::id(), unix_ms());
109                    return Ok(VaultLock { path });
110                }
111                Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
112                    if Self::held_is_stale(&path) {
113                        // best-effort: if another process wins the
114                        // race to re-create, our next loop loses
115                        // create_new anyway
116                        let _ = std::fs::remove_file(&path);
117                        continue;
118                    }
119                    if std::time::Instant::now() >= deadline {
120                        return Err(VaultError::Backend(format!(
121                            "vault is locked by another process ({})",
122                            path.display()
123                        )));
124                    }
125                    std::thread::sleep(std::time::Duration::from_millis(20));
126                }
127                Err(e) => return Err(e.into()),
128            }
129        }
130    }
131
132    /// A held lock is stale when the creation timestamp inside it
133    /// is older than [`LOCK_STALE`] — an unreadable or
134    /// foreign-shaped file is treated as held, never clobbered.
135    fn held_is_stale(path: &Path) -> bool {
136        std::fs::read_to_string(path)
137            .ok()
138            .and_then(|s| s.split_whitespace().nth(1)?.parse::<u64>().ok())
139            .map(|created| unix_ms().saturating_sub(created) > LOCK_STALE.as_millis() as u64)
140            .unwrap_or(false)
141    }
142}
143
144/// Milliseconds since the epoch — the lockfile's creation stamp.
145fn unix_ms() -> u64 {
146    std::time::SystemTime::now()
147        .duration_since(std::time::UNIX_EPOCH)
148        .map(|d| d.as_millis() as u64)
149        .unwrap_or(0)
150}

152impl Drop for VaultLock {
153    fn drop(&mut self) {
154        let _ = std::fs::remove_file(&self.path);
155    }
156}

169/// The age-encrypted local vault.
170pub struct AgeVault {
171    path: PathBuf,
172    identity: x25519::Identity,
173    recipients: Vec<x25519::Recipient>,
174    state: Mutex<Payload>,
175}

177impl AgeVault {
178    /// The default vault file — `$XDG_DATA_HOME/cuora/vault.age`,
179    /// else `~/.local/share/cuora/vault.age`. A pre-rename
180    /// `…/arca/vault.age` that still exists wins over the fresh
181    /// cuora path so existing vaults keep resolving.
182    pub fn default_vault_path() -> PathBuf {
183        let cuora = data_home().join("cuora/vault.age");
184        let legacy = data_home().join("arca/vault.age");
185        if cuora.exists() || !legacy.exists() {
186            cuora
187        } else {
188            legacy
189        }
190    }

192    /// Open the vault with an already-resolved identity — where the
193    /// identity came from (keyfile, keyring) is the caller's
194    /// business. A missing vault file is an empty vault.
195    /// `extra_recipients` are `age1…` strings encrypted to alongside
196    /// the identity's public key.
197    pub fn open(
198        vault_path: &Path,
199        identity: x25519::Identity,
200        extra_recipients: &[String],
201    ) -> Result<Self, VaultError> {
202        let mut recipients = vec![identity.to_public()];
203        for r in extra_recipients {
204            recipients.push(
205                x25519::Recipient::from_str(r)
206                    .map_err(|e| VaultError::Config(format!("invalid recipient {r:?}: {e}")))?,
207            );
208        }
209        let payload = match std::fs::read(vault_path) {
210            Ok(bytes) => Self::decrypt(&identity, &bytes)?,
211            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Payload::default(),
212            Err(e) => return Err(e.into()),
213        };
214        // F3: loose vault material is a note, never a failure —
215        // the file may not exist yet (a first-run open is an
216        // empty vault) and missing paths simply don't warn.
217        warn_perms(vault_path, false);
218        if let Some(d) = vault_path.parent() {
219            warn_perms(d, true);
220        }
221        Ok(AgeVault {
222            path: vault_path.to_path_buf(),
223            identity,
224            recipients,
225            state: Mutex::new(payload),
226        })
227    }
228
229    /// [`cuora_identity::ensure`] + [`Self::open`] — bootstrap the
230    /// identity on `source` when it holds none, then open. Returns
231    /// the vault and whether a fresh identity was generated.
232    pub fn open_or_init(
233        vault_path: &Path,
234        source: &dyn IdentitySource,
235        extra_recipients: &[String],
236    ) -> Result<(Self, bool), VaultError> {
237        let (identity, created) = cuora_identity::ensure(source)
238            .map_err(|e| VaultError::Config(format!("identity from {}: {e}", source.describe())))?;
239        Ok((Self::open(vault_path, identity, extra_recipients)?, created))
240    }
241
242    /// The identity's public recipient (`age1…`).
243    pub fn public_recipient(&self) -> String {
244        self.identity.to_public().to_string()
245    }
246
247    fn decrypt(identity: &x25519::Identity, bytes: &[u8]) -> Result<Payload, VaultError> {
248        let decryptor = age::Decryptor::new_buffered(bytes)
249            .map_err(|e| VaultError::Crypto(format!("read vault: {e}")))?;
250        if decryptor.is_scrypt() {
251            return Err(VaultError::Crypto(
252                "vault is passphrase-encrypted; recipient key required".into(),
253            ));
254        }
255        let mut reader = decryptor
256            .decrypt(std::iter::once(identity as &dyn age::Identity))
257            .map_err(|e| VaultError::Crypto(format!("decrypt vault: {e}")))?;
258        // Zeroizing — the decrypted plaintext is scrubbed on drop
259        // rather than left in a freed heap buffer (MED-4).
260        let mut buf = Zeroizing::new(Vec::new());
261        reader.read_to_end(&mut buf)?;
262        let payload: Payload = serde_json::from_slice(&buf)
263            .map_err(|e| VaultError::Crypto(format!("vault payload: {e}")))?;
264        if payload.v > FORMAT_VERSION {
265            return Err(VaultError::Crypto(format!(
266                "vault format v{} is newer than supported v{FORMAT_VERSION}",
267                payload.v
268            )));
269        }
270        Ok(payload)
271    }
272
273    /// Serialize + encrypt a payload to all recipients. In-memory
274    /// encryption cannot realistically fail — `recipients` is never
275    /// empty (the identity's public key is always first) and a
276    /// `Vec` sink never errors — so the `expect`s document
277    /// invariants rather than guard reachable paths.
278    #[allow(clippy::unwrap_used)]
279    fn encrypt_payload(&self, payload: &Payload) -> Vec<u8> {
280        let encryptor = age::Encryptor::with_recipients(
281            self.recipients.iter().map(|r| r as &dyn age::Recipient),
282        )
283        .expect("at least one recipient");
284        let plain = Zeroizing::new(serde_json::to_vec(payload).expect("payload serializes"));
285        let mut cipher = Vec::new();
286        {
287            let mut w = encryptor.wrap_output(&mut cipher).expect("in-memory wrap");
288            w.write_all(&plain).expect("in-memory write");
289            w.finish().expect("in-memory finish");
290        }
291        cipher
292    }
293
294    /// Encrypt + atomically rewrite the vault file — tmp + fsync +
295    /// rename + dir fsync, so a crash can't leave a truncated or
296    /// unflushed vault (MED-3).
297    #[allow(clippy::unwrap_used)]
298    fn save(&self, payload: &Payload) -> Result<(), VaultError> {
299        let cipher = self.encrypt_payload(payload);
300        if let Some(parent) = self.path.parent() {
301            std::fs::create_dir_all(parent)?;
302        }
303        let tmp = self.path.with_extension("age.tmp");
304        {
305            let mut f = open_tmp_exclusive(&tmp)?;
306            f.write_all(&cipher)?;
307            f.sync_all()?;
308        }
309        set_private(&tmp).expect("chmod on a freshly written file");
310        std::fs::rename(&tmp, &self.path)?;
311        sync_dir(self.path.parent());
312        Ok(())
313    }
314
315    /// Lock the payload; poison = unrecoverable backend error.
316    fn state(&self) -> Result<std::sync::MutexGuard<'_, Payload>, VaultError> {
317        self.state
318            .lock()
319            .map_err(|_| VaultError::Backend("vault state lock poisoned".into()))
320    }
321
322    /// The on-disk payload — a missing file is an empty vault.
323    /// Only [`Self::mutate`] re-reads (under the lock, before
324    /// applying its change): reads keep the open-time cache —
325    /// re-reading on every `get`/`list` would both stall on
326    /// special files and cost a decrypt per verb (MED-3).
327    fn read_disk(&self) -> Result<Payload, VaultError> {
328        match std::fs::read(&self.path) {
329            Ok(bytes) => Self::decrypt(&self.identity, &bytes),
330            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(Payload::default()),
331            Err(e) => Err(e.into()),
332        }
333    }
334
335    /// The mutating-verb spine (MED-3): hold the process mutex,
336    /// take the cross-process [`VaultLock`], **re-read** the vault
337    /// file so another process's completed write isn't clobbered,
338    /// apply `f`, persist when `f` reports a change, and leave
339    /// `state` mirroring the new on-disk payload.
340    fn mutate<T>(
341        &self,
342        f: impl FnOnce(&mut Payload) -> Result<(T, bool), VaultError>,
343    ) -> Result<T, VaultError> {
344        let mut p = self.state()?;
345        // the lockfile lives next to the vault — the dir may not
346        // exist yet on a first write (save creates it too)
347        if let Some(parent) = self.path.parent() {
348            std::fs::create_dir_all(parent)?;
349            // F3: a vault dir others can read/write is worth a
350            // note — it weakens every file-permission we set
351            warn_perms(parent, true);
352        }
353        let _lock = VaultLock::acquire(&self.path)?;
354        *p = self.read_disk()?;
355        let (out, dirty) = f(&mut p)?;
356        if dirty {
357            self.save(&p)?;
358        }
359        Ok(out)
360    }
361}

363/// The tmp file's open options — `create_new` is `O_EXCL`:
364/// `File::create` follows symlinks, so a planted `vault.age.tmp`
365/// symlink would redirect the write onto a victim file (F2). On
366/// unix it's `0600` at create — no group/world-readable window
367/// before the post-write chmod.
368fn tmp_open_options() -> std::fs::OpenOptions {
369    let mut opts = std::fs::OpenOptions::new();
370    opts.write(true).create_new(true);
371    #[cfg(unix)]
372    {
373        use std::os::unix::fs::OpenOptionsExt;
374        opts.mode(0o600);
375    }
376    opts
377}
378
379/// Open `path` for the tmp write, refusing to follow or clobber
380/// anything already there. The [`VaultLock`] serializes writers,
381/// so an existing tmp is crash debris: a plain file is reclaimed
382/// and retried once (a second `AlreadyExists` means a racer —
383/// the io error surfaces honestly); a symlink or anything exotic
384/// was planted — the save fails rather than touching it (F2).
385fn open_tmp_exclusive(path: &Path) -> Result<std::fs::File, VaultError> {
386    match tmp_open_options().open(path) {
387        Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
388            match std::fs::symlink_metadata(path) {
389                Ok(m) if m.file_type().is_file() => std::fs::remove_file(path)?,
390                _ => {
391                    return Err(VaultError::Backend(format!(
392                        "vault tmp {} exists and isn't a plain file — \
393                         refusing to touch it",
394                        path.display()
395                    )))
396                }
397            }
398            Ok(tmp_open_options().open(path)?)
399        }
400        other => other.map_err(Into::into),
401    }
402}

404/// Best-effort fsync of the vault's directory — the durability
405/// hint for the rename. Failure isn't a vault failure.
406#[cfg(unix)]
407fn sync_dir(parent: Option<&Path>) {
408    if let Some(d) = parent {
409        if let Ok(f) = std::fs::File::open(d) {
410            let _ = f.sync_all();
411        }
412    }
413}

418impl Vault for AgeVault {
419    fn get(&self, env: &Env, rref: &Ref) -> Result<Option<SecretRecord>, VaultError> {
420        let p = self.state()?;
421        Ok(p.envs
422            .get(env.as_str())
423            .and_then(|m| m.get(&rref.path()))
424            .map(|e| e.record()))
425    }
426
427    fn put(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
428        self.mutate(|p| {
429            let map = p.envs.entry(env.as_str().to_string()).or_default();
430            let entry = match map.get_mut(&rref.path()) {
431                Some(e) => {
432                    e.value = Zeroizing::new(value.expose().to_string());
433                    e.version += 1;
434                    e.clone()
435                }
436                None => {
437                    let e = Entry {
438                        value: Zeroizing::new(value.expose().to_string()),
439                        version: 1,
440                        created_at: OffsetDateTime::now_utc(),
441                        rotated_at: None,
442                    };
443                    map.insert(rref.path(), e.clone());
444                    e
445                }
446            };
447            Ok((entry.record(), true))
448        })
449    }
450
451    fn rm(&self, env: &Env, rref: &Ref) -> Result<bool, VaultError> {
452        self.mutate(|p| {
453            let map = match p.envs.get_mut(env.as_str()) {
454                Some(m) => m,
455                None => return Ok((false, false)),
456            };
457            let removed = map.remove(&rref.path()).is_some();
458            Ok((removed, removed))
459        })
460    }
461
462    fn list(&self, env: &Env) -> Result<Vec<Ref>, VaultError> {
463        let p = self.state()?;
464        let mut out = Vec::new();
465        if let Some(map) = p.envs.get(env.as_str()) {
466            for path in map.keys() {
467                out.push(
468                    Ref::from_path(path)
469                        .map_err(|e| VaultError::Backend(format!("stored key {e}")))?,
470                );
471            }
472        }
473        out.sort();
474        Ok(out)
475    }
476
477    fn rotate(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
478        self.mutate(|p| {
479            let map = p.envs.entry(env.as_str().to_string()).or_default();
480            let e = map
481                .get_mut(&rref.path())
482                .ok_or_else(|| VaultError::NotFound {
483                    rref: rref.path(),
484                    env: env.as_str().to_string(),
485                })?;
486            e.value = Zeroizing::new(value.expose().to_string());
487            e.version += 1;
488            e.rotated_at = Some(OffsetDateTime::now_utc());
489            let entry = e.clone();
490            Ok((entry.record(), true))
491        })
492    }
493}

508/// The warning text when `path` is more permissive than vault
509/// material should be — `None` when it's strict enough or doesn't
510/// exist (F3). Files must not exceed `0600`, directories `0700`;
511/// the check is "any bit outside the allowed set", like ssh's
512/// strict-modes rule. Returned (not printed) so tests can assert.
513#[cfg(unix)]
514fn perm_note(path: &Path, is_dir: bool) -> Option<String> {
515    use std::os::unix::fs::PermissionsExt;
516    let mode = std::fs::symlink_metadata(path).ok()?.permissions().mode() & 0o777;
517    let allowed: u32 = if is_dir { 0o700 } else { 0o600 };
518    (mode & !allowed != 0).then(|| {
519        let kind = if is_dir { "directory" } else { "file" };
520        format!(
521            "{kind} {} is group/world-accessible (mode {mode:o}) — \
522             vault material should be {allowed:o}; run \
523             `chmod {allowed:o} {}`",
524            path.display(),
525            path.display()
526        )
527    })
528}
529
530#[cfg(not(unix))]
531fn perm_note(_path: &Path, _is_dir: bool) -> Option<String> {
532    None
533}
534
535/// F3 stderr note — a loose permission is a warning, never a
536/// vault failure.
537fn warn_perms(path: &Path, is_dir: bool) {
538    if let Some(note) = perm_note(path, is_dir) {
539        cuora_notice::note(&note);
540    }

683    fn decrypt_reports_every_failure_mode() {
684        let id = x25519::Identity::generate();
685        // not an age file at all
686        assert!(AgeVault::decrypt(&id, b"garbage").is_err());
687        // passphrase-encrypted → we hold no passphrase
688        let scrypt = age::Encryptor::with_user_passphrase(age::secrecy::SecretString::from(
689            "fake-pw".to_string(),
690        ));
691        let mut blob = Vec::new();
692        {
693            let mut w = scrypt.wrap_output(&mut blob).unwrap();
694            w.write_all(b"{}").unwrap();
695            w.finish().unwrap();
696        }
697        let e = AgeVault::decrypt(&id, &blob).err().unwrap();
698        assert!(e.to_string().contains("passphrase"), "{e}");
699        // encrypted to someone else → can't decrypt
700        let other = x25519::Identity::generate();
701        assert!(AgeVault::decrypt(&id, &encrypt_to(&other.to_public(), b"{}")).is_err());
702        // valid envelope, garbage plaintext → payload error
703        let blob = encrypt_to(&id.to_public(), b"not json");
704        assert!(AgeVault::decrypt(&id, &blob).is_err());
705        // a newer format version is refused, honestly
706        let blob = encrypt_to(&id.to_public(), br#"{"v":99,"envs":{}}"#);
707        let e = AgeVault::decrypt(&id, &blob).err().unwrap();
708        assert!(e.to_string().contains("v99"), "{e}");
709        // truncated mid-ciphertext → the stream read fails
710        let good = encrypt_to(&id.to_public(), br#"{"v":1,"envs":{}}"#);
711        assert!(AgeVault::decrypt(&id, &good[..good.len() / 2]).is_err());
712    }

807        // Two handles on one file — the multi-process shape. The
808        // second handle's put must merge the first's write (the
809        // locked re-read), not clobber it (MED-3).
810        let dir = tempfile::tempdir().unwrap();
811        let v_path = dir.path().join("vault.age");
812        let id = x25519::Identity::generate();

819        // both writes landed — the writer's own view is merged
820        let refs = v2.list(&Env::dev()).unwrap();
821        assert!(refs.contains(&r1) && refs.contains(&r2), "{refs:?}");

836        assert!(!v_path.with_extension("age.lock").exists());
