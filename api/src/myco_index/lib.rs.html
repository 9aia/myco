1//! Index port — the derived, disposable cache that powers
2//! index-sourced `myco view` (the `org-*` presets).
3//!
4//! Markdown in `project/` stays canonical; a [`RepoSnapshot`] is a
5//! frozen scan of one repo's ledger keyed by the project-branch
6//! HEAD commit it was taken at. Freshness is a one-line check —
7//! `snapshot.head_commit == git rev-parse HEAD` — so callers can
8//! rebuild exactly the stale repos. Deleting the whole index is
9//! always safe: `myco index rebuild` derives it back.
10//!
11//! Impls: `myco-index-redb` (`~/.local/share/myco/`).
12//!
13//! The second face is the link index ([`links`], P-005):
14//! `project/` files → `(src, dst, kind)` edges, invalidated by
15//! content fingerprint — same "derived, disposable" contract.
16

26/// A registered repo — an entry in the index's registry.
27#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
28pub struct RepoEntry {
29    /// Canonical repo-root path — the registry key.
30    pub path: Utf8PathBuf,
31    /// Display name (project `meta.toml` name at registration).
32    pub name: String,
33    /// The `project/` worktree the index scans — set at
34    /// registration so `--dir`/`project_dir` worktrees (not at
35    /// `<path>/project`) stay reachable. Empty means the
36    /// conventional location (registries written before this
37    /// field existed).
38    #[serde(default)]
39    pub worktree: Utf8PathBuf,
40}
41
42impl RepoEntry {
43    /// New registry entry.
44    pub fn new(path: Utf8PathBuf, name: impl Into<String>) -> Self {
45        RepoEntry {
46            path,
47            name: name.into(),
48            worktree: Utf8PathBuf::new(),
49        }
50    }
51
52    /// Record the scanned worktree path.
53    pub fn with_worktree(mut self, worktree: Utf8PathBuf) -> Self {
54        self.worktree = worktree;
55        self
56    }
57
58    /// The worktree to scan — the registered `worktree`, else the
59    /// conventional `<path>/project`.
60    pub fn worktree_path(&self) -> Utf8PathBuf {
61        if self.worktree.as_str().is_empty() {
62            self.path.join("project")
63        } else {
64            self.worktree.clone()
65        }
66    }
67}

69/// A frozen scan of one repo's `project/` ledger.
70///
71/// `items` holds every item found — TODO/BACKLOG lines and
72/// file-backed metas, live and archived. `archived` marks which ids
73/// live under `<dir>/archive/` so views can split "resolved" from
74/// "live".
75#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
76pub struct RepoSnapshot {
77    /// Canonical repo-root path — the snapshot key.
78    pub path: Utf8PathBuf,
79    /// Project name (from `meta.toml` / `TODO.md`).
80    pub name: String,
81    /// The project-worktree `HEAD` this scan was taken at —
82    /// freshness is `head_commit == current HEAD`.
83    pub head_commit: String,
84    /// RFC 3339 timestamp of the scan.
85    pub scanned_at: String,
86    /// Every item in the ledger (any status, live + archived).
87    pub items: Vec<Item>,
88    /// Ids whose files live under `archive/`.
89    #[serde(default)]
90    pub archived: BTreeSet<ItemId>,
91}
92
93impl RepoSnapshot {
94    /// Items not archived — the live ledger. Nested children ride
95    /// inside their parents in `items`; they flatten out here so
96    /// org views count them like any item.
97    pub fn live_items(&self) -> impl Iterator<Item = &Item> {
98        self.items
99            .iter()
100            .flat_map(Item::subtree)
101            .filter(|i| !self.archived.contains(&i.id))
102    }
103}

105/// The index port — registry + snapshot store.
106pub trait Index {
107    /// Register (or re-register) a repo root.
108    fn register(&self, repo: &RepoEntry) -> Result<(), Error>;
109
110    /// Unregister a repo and drop its snapshot. `Ok(true)` when it
111    /// was registered.
112    fn remove(&self, path: &Utf8Path) -> Result<bool, Error>;
113
114    /// Every registered repo, sorted by path.
115    fn repos(&self) -> Result<Vec<RepoEntry>, Error>;
116
117    /// The cached snapshot for `path`, `None` when absent.
118    fn snapshot(&self, path: &Utf8Path) -> Result<Option<RepoSnapshot>, Error>;
119
120    /// Write (or overwrite) a repo's snapshot.
121    fn save_snapshot(&self, snapshot: &RepoSnapshot) -> Result<(), Error>;
122}
123
124/// Port error — impls map backend failures into these variants.
125#[derive(Debug, thiserror::Error)]
126pub enum Error {
127    /// Filesystem failure (registry file, db path).
128    #[error("io: {0}")]
129    Io(#[from] std::io::Error),
130
131    /// The snapshot couldn't be (de)serialized.
132    #[error("snapshot codec: {0}")]
133    Codec(String),
134
135    /// Backend failure (redb, …).
136    #[error("index: {0}")]
137    Backend(String),
138}

140#[cfg(test)]
141#[allow(clippy::unwrap_used)]
142mod tests {
143    use super::*;
144    use myco_core::{ItemId, ItemKind, Status};
145
146    #[test]
147    fn entry_and_snapshot_serde() {
148        let e = RepoEntry::new(Utf8PathBuf::from("/repo"), "myco");
149        let snap = RepoSnapshot {
150            path: e.path.clone(),
151            name: "myco".into(),
152            head_commit: "deadbeef".into(),
153            scanned_at: "2026-02-09T00:00:00Z".into(),
154            items: vec![
155                Item::new(ItemId::new(ItemKind::Task, 1), "open"),
156                Item::new(ItemId::new(ItemKind::Issue, 2), "done").status(Status::Done),
157                Item::new(ItemId::new(ItemKind::Issue, 3), "arch"),
158            ],
159            archived: BTreeSet::from([ItemId::new(ItemKind::Issue, 3)]),
160        };
161        let json = serde_json::to_string(&snap).unwrap();
162        let back: RepoSnapshot = serde_json::from_str(&json).unwrap();
163        assert_eq!(back, snap);
164        // live = non-archived (done still counts — views filter it)
165        let live: Vec<ItemId> = back.live_items().map(|i| i.id).collect();
166        assert_eq!(
167            live,
168            [
169                ItemId::new(ItemKind::Task, 1),
170                ItemId::new(ItemKind::Issue, 2)
171            ]
172        );
173    }
174
175    #[test]
176    fn error_display() {
177        assert_eq!(Error::Codec("x".into()).to_string(), "snapshot codec: x");
178        assert_eq!(Error::Backend("y".into()).to_string(), "index: y");
179    }
