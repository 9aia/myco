1//! tamias-driver-cuora — the pavo view driver for cuora.
2//!
3//! Verbs (the tamias driver contract):
4//!
5//! - `manifest` — print the [`DriverManifest`] JSON (discovery
6//!   probe; never touches vault state).
7//! - `capabilities` — `{"capabilities": ["pavo"], "verbs": […]}`.
8//! - `serve` — ndjson JSON-RPC on stdio: `{"op":"manifest"}`,
9//!   `{"op":"views"}`, `{"op":"view","view":…,"params":{…}}`,
10//!   `{"op":"query","query":…,"params":{…}}` →
11//!   `{"ok":true,"result":…}` / `{"ok":false,"error":…}`.
12//!
13//! Target resolution: each `view`/`query` builds `cuora_ops::Ops`
14//! from the request params — `config`/`env`/`actor` map to the
15//! CLI's `--config`/`--env`/`--actor`; absent params fall through
16//! to `$CUORA_CONFIG` → `./cuora.toml` → `~/.config/cuora/cuora.toml`,
17//! `$CUORA_ENV`, `$CUORA_ACTOR`/`$USER` — identical to `cuora`
18//! verbs. See `data.rs` for the secrets-safety contract: this
19//! driver renders vault *metadata* only, never values.
20
21mod data;
22mod views;
23
24use pavo_driver_kit as kit;
25use pavo_protocol::wire::{self, Request, Response, ViewSummary};
26
27fn capabilities_json() -> serde_json::Value {
28    let m = views::manifest();
29    serde_json::json!({
30        "capabilities": m.capabilities,
31        "verbs": m.verbs,
32        "views": m.views.len(),
33    })
34}
35
36/// Dispatch one parsed request to its response.
37fn dispatch(req: &Request) -> Response {
38    match req {
39        Request::Manifest => Response::ok_typed(&views::manifest()),
40        Request::Views => {
41            let summaries: Vec<ViewSummary> =
42                views::specs().iter().map(ViewSummary::from).collect();
43            Response::ok_typed(&summaries)
44        }
45        Request::View { view, params } => match views::view_result(view, params) {
46            Ok(result) => Response::ok_typed(&result),
47            Err(e) => Response::err(e),
48        },
49        Request::Render { kind, .. } => Response::err(format!(
50            "tamias-driver-cuora claims no renderers (kind '{kind}')"
51        )),
52        Request::Query { query, params } => {
53            // Unknown names fail the op; everything else answers
54            // `ok:true` — degraded reads travel as `{"error": …}`
55            // data (the marker renderers understand), matching the
56            // reference driver's contract.
57            if !views::known_query(query) {
58                return Response::err(format!("unknown query: {query}"));
59            }
60            let mut ctx = data::Ctx::open(params);
61            let payload = views::query_data(&mut ctx, query, params);
62            Response::ok_typed(&wire::QueryResult {
63                query: query.clone(),
64                data: payload,
65            })
66        }
67        // The protocol grew `submit`/`jobs`/`notifications` ops after
68        // this driver was written; cuora is read-only with no feeds,
69        // so they answer `ok:false` like any unsupported op.
70        Request::Submit { .. } | Request::Jobs | Request::Notifications => {
71            Response::err("tamias-driver-cuora implements no submit/jobs/notifications ops")
72        }
73    }
74}
75
76fn serve() -> i32 {
77    // The kit owns the ndjson loop — stdin EOF and a dead client
78    // both end it quietly, and either way the driver exits 0.
79    let _ = kit::serve_stdio(dispatch);
80    0
81}
82
83fn main() {
84    let mut args = std::env::args().skip(1);
85    match args.next().as_deref() {
86        Some("manifest") => {
87            println!("{}", kit::manifest_json(&views::manifest()));
88        }
89        Some("capabilities") => {
90            println!(
91                "{}",
92                serde_json::to_string_pretty(&capabilities_json()).unwrap_or_default()
93            );
94        }
95        Some("serve") => std::process::exit(serve()),
96        _ => {
97            eprintln!("usage: tamias-driver-cuora manifest|capabilities|serve");
98            std::process::exit(1);
99        }
100    }
101}
