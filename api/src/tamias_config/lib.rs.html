22/// A store/cache location.

40/// A named registry.
41#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
42#[serde(default)]
43pub struct Registry {
44    /// Explicit kind — `local` | `http` | `cf`. Unset → inferred from
45    /// `url` (http(s) → `http`, anything else → `local`), so configs
46    /// written before kinds were explicit keep working.
47    #[serde(skip_serializing_if = "Option::is_none")]
48    pub kind: Option<String>,
49    /// Base URL (http) or root dir.
50    #[serde(skip_serializing_if = "Option::is_none")]
51    pub url: Option<String>,
52    /// Env var **name** holding the bearer token — never the token.
53    #[serde(skip_serializing_if = "Option::is_none")]
54    pub token_env: Option<String>,
55    /// Per-registry limit overrides (values + package rules).
56    #[serde(default, skip_serializing_if = "Limits::is_empty")]
57    pub limits: Limits,
58}

60impl Registry {
61    /// The effective kind — explicit `kind` (lowercased) when set,
62    /// else inferred: an http(s) `url` → `http`, anything else →
63    /// `local`. `cf` is never inferred — it's the explicit opt-in for
64    /// worker-managed registries (same client transport as `http`).
65    #[must_use]
66    pub fn effective_kind(&self) -> String {
67        if let Some(kind) = &self.kind {
68            return kind.to_lowercase();
69        }
70        match self.url.as_deref() {
71            Some(url) if url.starts_with("http://") || url.starts_with("https://") => {
72                "http".to_string()
73            }
74            _ => "local".to_string(),
75        }
76    }
77}

133/// Deserialize an optional byte count — a plain integer or a human
134/// string like `"256MiB"` / `"1GiB"` / `"512KiB"` (binary suffixes;
135/// bare `K`/`M`/`G`/`T`/`B` and `KB`-style spellings accepted).
136fn de_bytes_opt<'de, D>(deserializer: D) -> Result<Option<u64>, D::Error>
137where
138    D: serde::Deserializer<'de>,
139{
140    #[derive(Deserialize)]
141    #[serde(untagged)]
142    enum Repr {
143        Num(u64),
144        Str(String),
145    }
146    match Option::<Repr>::deserialize(deserializer)? {
147        None => Ok(None),
148        Some(Repr::Num(n)) => Ok(Some(n)),
149        Some(Repr::Str(s)) => parse_bytes(&s).map(Some).map_err(serde::de::Error::custom),
150    }
151}

153/// Serialize an optional byte count as a plain integer.
154fn ser_bytes_opt<S>(value: &Option<u64>, serializer: S) -> Result<S::Ok, S::Error>
155where
156    S: serde::Serializer,
157{
158    match value {
159        Some(n) => serializer.serialize_some(n),
160        None => serializer.serialize_none(),
161    }
162}

164/// Parse `<number>[ <unit>]` into bytes. Units are binary:
165/// `B`/`K`/`KB`/`KiB` = 1024ⁿ — `1K`, `1KB`, `1KiB` are all 1024.
166/// Fractions allowed (`1.5GiB`). Also the parser behind CLI size flags
167/// (`--max-upload 256MiB`).
168///
169/// # Errors
170/// Fails on a non-numeric size, an unknown unit, or u64 overflow.
171pub fn parse_bytes(text: &str) -> Result<u64, String> {
172    let text = text.trim();
173    let split = text
174        .find(|c: char| !(c.is_ascii_digit() || c == '.'))
175        .unwrap_or(text.len());
176    let (num, unit) = text.split_at(split);
177    let value: f64 = num
178        .trim()
179        .parse()
180        .map_err(|_| format!("invalid byte size '{text}'"))?;
181    if !value.is_finite() || value < 0.0 {
182        return Err(format!("invalid byte size '{text}'"));
183    }
184    let multiplier: u128 = match unit.trim().to_lowercase().as_str() {
185        "" | "b" => 1,
186        "k" | "kb" | "kib" => 1 << 10,
187        "m" | "mb" | "mib" => 1 << 20,
188        "g" | "gb" | "gib" => 1 << 30,
189        "t" | "tb" | "tib" => 1 << 40,
190        "p" | "pb" | "pib" => 1 << 50,
191        other => return Err(format!("unknown byte unit '{other}' in '{text}'")),
192    };
193    let bytes = (value * multiplier as f64) as u128;
194    u64::try_from(bytes).map_err(|_| format!("byte size '{text}' overflows u64"))
195}

197/// Byte-size limit knobs. Every field is optional — `None` inherits
198/// the next level of the resolution chain (per-package override →
199/// per-registry `limits` → global `limits` → built-in default; see
200/// [`Config::effective_limits`]).
201///
202/// `packages` maps a package name — or a `*` glob (`foo*`, `*foo`,
203/// `*foo*`, `*`) — to overrides for that package. It lives on the
204/// global `limits` (and may also appear under a registry's `limits`
205/// for registry-scoped overrides); `packages` nested *inside* a
206/// package entry is ignored — overrides are one level deep.
207#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
208#[serde(default)]
209pub struct Limits {
210    /// Largest payload tarball a publish may upload, in bytes.
211    #[serde(
212        skip_serializing_if = "Option::is_none",
213        deserialize_with = "de_bytes_opt",
214        serialize_with = "ser_bytes_opt"
215    )]
216    pub max_upload: Option<u64>,
217    /// Largest tarball a fetch may download, in bytes.
218    #[serde(
219        skip_serializing_if = "Option::is_none",
220        deserialize_with = "de_bytes_opt",
221        serialize_with = "ser_bytes_opt"
222    )]
223    pub max_download: Option<u64>,
224    /// Per-package overrides — exact name or `*` glob → limits.
225    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
226    pub packages: BTreeMap<String, Limits>,
227}
228
229impl Limits {
230    /// True when no knob or package rule is set — used to skip empty
231    /// `limits:` blocks on render.
232    #[must_use]
233    pub fn is_empty(&self) -> bool {
234        self.max_upload.is_none() && self.max_download.is_none() && self.packages.is_empty()
235    }
236}

253/// `*` wildcard match — `*` matches any (possibly empty) run; every
254/// literal part must appear in order. `foo*`, `*foo`, `*foo*` and `*`
255/// all work; patterns without `*` are exact matches.
256fn glob_match(pattern: &str, name: &str) -> bool {
257    if !pattern.contains('*') {
258        return pattern == name;
259    }
260    // Literal segments between `*`s: the first anchors at the start
261    // (unless the pattern opens with `*`), the last at the end (unless
262    // it closes with `*`), the rest must appear in order.
263    let segs: Vec<&str> = pattern.split('*').filter(|s| !s.is_empty()).collect();
264    let mut rest = name;
265    let mut lo = 0usize;
266    let mut hi = segs.len();
267    if !pattern.starts_with('*') && lo < hi {
268        let seg = segs[lo];
269        if !rest.starts_with(seg) {
270            return false;
271        }
272        rest = &rest[seg.len()..];
273        lo += 1;
274    }
275    let tail = if !pattern.ends_with('*') && hi > lo {
276        hi -= 1;
277        Some(segs[hi])
278    } else {
279        None
280    };
281    for seg in &segs[lo..hi] {
282        match rest.find(seg) {
283            Some(i) => rest = &rest[i + seg.len()..],
284            None => return false,
285        }
286    }
287    tail.is_none_or(|seg| rest.ends_with(seg))
288}

290/// The override for `package` in `limits.packages` — exact name match
291/// first, then `*` glob patterns in key order.
292fn package_limits<'a>(limits: &'a Limits, package: &str) -> Option<&'a Limits> {
293    if let Some(found) = limits.packages.get(package) {
294        return Some(found);
295    }
296    limits
297        .packages
298        .iter()
299        .find(|(pattern, _)| glob_match(pattern, package))
300        .map(|(_, found)| found)
301}

303/// The tamias config — discovered as `./tamias.{yml,yaml,toml}` (or
304/// `TAMIAS_CONFIG`/`--file`). `store`/`cache` roots feed the CLI's
305/// `Env` (env vars win); `registries`/`publish`/`search` route the
306/// registry verbs; `limits` caps uploads/downloads.
307#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
308#[serde(default)]
309pub struct Config {
310    /// Install store location — `root` applies when `kind` is `local`/unset
311    /// and `TAMIAS_HOME`/XDG env is unset (env wins).
312    pub store: Location,
313    /// Fetch-cache location — same resolution rule as `store`.
314    pub cache: Location,
315    /// Named registries — `publish`/`search --registry <name>` and the
316    /// `publish.default`/`search.default` routing resolve through these.
317    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
318    pub registries: BTreeMap<String, Registry>,
319    /// Publish routing.
320    pub publish: Publish,
321    /// Search routing.
322    pub search: Search,
323    /// Global byte limits + per-package overrides.
324    #[serde(default, skip_serializing_if = "Limits::is_empty")]
325    pub limits: Limits,
326    /// Package signing policy — `publish --sign` key resolution and
327    /// install/verify `allowed_signers` enforcement (`tamias-sign`).
328    pub signing: Signing,
329}
330
331impl Config {
332    /// Resolve the effective byte limits for `package` on `registry`
333    /// (a `registries` key, or `None` for the unnamed/default registry).
334    ///
335    /// Per knob, first set wins:
336    /// package override — the registry's own `limits.packages` first,
337    /// then the global `limits.packages` (exact name, else `*` glob) →
338    /// per-registry `limits` → global `limits` → built-in default
339    /// ([`DEFAULT_MAX_UPLOAD`] / [`DEFAULT_MAX_DOWNLOAD`], 512 MiB).
340    #[must_use]
341    pub fn effective_limits(
342        &self,
343        registry: Option<&str>,
344        package: Option<&str>,
345    ) -> EffectiveLimits {
346        let reg = registry.and_then(|name| self.registries.get(name));
347        let pkg = package.and_then(|p| {
348            reg.and_then(|r| package_limits(&r.limits, p))
349                .or_else(|| package_limits(&self.limits, p))
350        });
351        let pick = |f: fn(&Limits) -> Option<u64>| {
352            pkg.and_then(f)
353                .or_else(|| reg.and_then(|r| f(&r.limits)))
354                .or_else(|| f(&self.limits))
355        };
356        EffectiveLimits {
357            max_upload: pick(|l| l.max_upload).unwrap_or(DEFAULT_MAX_UPLOAD),
358            max_download: pick(|l| l.max_download).unwrap_or(DEFAULT_MAX_DOWNLOAD),
359        }
360    }
361}

363/// A config file format — parse text → [`Config`], and render back.
364pub trait ConfigFormat: Send + Sync {
365    /// The file extensions this format claims (no leading dot).
366    fn extensions(&self) -> &'static [&'static str];
367
368    /// Parse `text`.
369    ///
370    /// # Errors
371    /// Fails on a syntax or type error.
372    fn parse(&self, text: &str) -> Result<Config, Error>;
373
374    /// Render `config` back to this format.
375    ///
376    /// # Errors
377    /// Fails on a serialize error.
378    fn render(&self, config: &Config) -> Result<String, Error>;
379}
380
381/// The format that claims `path`'s extension, if any.
382#[must_use]
383pub fn format_for<'a>(
384    path: &Path,
385    formats: &'a [&'a dyn ConfigFormat],
386) -> Option<&'a dyn ConfigFormat> {
387    let ext = path.extension().and_then(|e| e.to_str())?;
388    formats
389        .iter()
390        .copied()
391        .find(|f| f.extensions().contains(&ext))
392}
393
394/// Load a specific config file, parsing with the format that claims its
395/// extension.
396///
397/// # Errors
398/// Fails on a read/parse error or an unclaimed extension.
399pub fn load_file(path: &Path, formats: &[&dyn ConfigFormat]) -> Result<Config, Error> {
400    let format = format_for(path, formats)
401        .ok_or_else(|| Error::Failed(format!("no config format for {}", path.display())))?;
402    let text =
403        std::fs::read_to_string(path).map_err(Error::io(format!("read {}", path.display())))?;
404    format.parse(&text)
405}
406
407/// Load the first config file found in `dir` (YAML first — [`CONFIG_NAMES`]).
408///
409/// # Errors
410/// Fails on a read/parse error; `Ok(None)` when no config file exists.
411pub fn load_in(
412    dir: &Path,
413    formats: &[&dyn ConfigFormat],
414) -> Result<Option<(PathBuf, Config)>, Error> {
415    for name in CONFIG_NAMES {
416        let path = dir.join(name);
417        if !path.is_file() {
418            continue;
419        }
420        let config = load_file(&path, formats)?;
421        return Ok(Some((path, config)));
422    }
423    Ok(None)
424}

579            },
