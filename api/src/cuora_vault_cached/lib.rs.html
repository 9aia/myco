1//! cuora-vault-cached — a read-through, age-encrypted local cache in
2//! front of any [`Vault`] backend (P-007).
3//!
4//! The cache is itself a `Vault` impl — a composite, not a bolt-on:
5//! `get`/`list` serve the local cache while it's fresh (TTL),
6//! miss → remote fetch → write-through; `put`/`rotate`/`rm` always
7//! hit the remote first, then update or invalidate the cached row —
8//! writes never serve stale. Rows carry the remote `version`/
9//! `rotated_at` **verbatim** — a cached `get` answers with the
10//! metadata the remote returned, not the cache's own bookkeeping.
11//!
12//! Freshness is TTL-bounded ([`CachePolicy::ttl`]): a cached row
13//! or listing younger than the TTL answers without a remote
14//! round-trip; past it the remote is re-fetched and the cache
15//! updated. A remote *failure* falls back to the stale copy — that
16//! degraded read is the offline mode the cache exists for.
17//!
18//! Env scoping is the safety rail: [`CachePolicy::envs`] decides
19//! which envs may enter the cache at all — `dev` may cache
20//! aggressively while `prod` live-fetches (a stale prod secret is a
21//! footgun). `cuora-ops` builds the policy from `[vaults.*] cache`.
22//!
23//! Layout: rows live in a `Vault` of the caller's choice (ops wires
24//! an `AgeVault` at `$XDG_CACHE_HOME/cuora/<vault>.age`, the honest
25//! disposable-data placement); the cached value slot holds a JSON
26//! row `{v, value, version, created_at, rotated_at, fetched_at}` —
27//! ref paths and values stay inside the ciphertext like any vault
28//! row, so the cache needs no crypto of its own. Per-env list
29//! markers live in a small plaintext sidecar (`<vault>.listed.json`)
30//! — env names and timestamps only, never refs or values. Every
31//! cache interaction is best-effort: a corrupt row is a miss, a
32//! cache that won't open degrades to remote-only pass-through, and
33//! the remote is always the source of truth.
34

46/// Default freshness window — 300s. Long enough that a burst of
47/// `cuora run`s costs one remote round-trip, short enough that a
48/// rotation shows up inside a work session.
49pub const DEFAULT_TTL: Duration = Duration::from_secs(300);
50
51/// On-disk cache-row format version.
52const ROW_VERSION: u32 = 1;
53
54/// Which envs the cache may hold + how fresh a row must be to
55/// answer without a remote round-trip.
56#[derive(Debug, Clone)]
57pub struct CachePolicy {
58    /// Envs allowed into the cache — `None` caches every env. An
59    /// env outside the set passes straight through to the remote:
60    /// its rows never land on disk.
61    pub envs: Option<BTreeSet<String>>,
62    /// Freshness window for rows and list markers.
63    pub ttl: Duration,
64}
65
66impl CachePolicy {
67    /// Cache every env.
68    pub fn all(ttl: Duration) -> Self {
69        CachePolicy { envs: None, ttl }
70    }
71
72    /// Cache only `envs` — every other env live-fetches.
73    pub fn scoped(envs: impl IntoIterator<Item = Env>, ttl: Duration) -> Self {
74        CachePolicy {
75            envs: Some(envs.into_iter().map(|e| e.as_str().to_string()).collect()),
76            ttl,
77        }
78    }
79
80    /// Is `env` inside the cache scope?
81    pub fn caches(&self, env: &Env) -> bool {
82        match &self.envs {
83            None => true,
84            Some(set) => set.contains(env.as_str()),
85        }
86    }
87}

89/// One cached row — what the cache vault's value slot holds (JSON).
90/// `version`/`created_at`/`rotated_at` are the *remote's* values,
91/// preserved verbatim so a cached `get` reports the same metadata a
92/// live one would; `fetched_at` is the cache's freshness stamp.
93#[derive(Serialize, Deserialize)]
94struct CacheRow {
95    v: u32,
96    value: Zeroizing<String>,
97    version: u64,
98    #[serde(with = "time::serde::rfc3339")]
99    created_at: OffsetDateTime,
100    #[serde(default, with = "time::serde::rfc3339::option")]
101    rotated_at: Option<OffsetDateTime>,
102    #[serde(with = "time::serde::rfc3339")]
103    fetched_at: OffsetDateTime,
104}
105
106impl CacheRow {
107    fn wrap(rec: &SecretRecord) -> Self {
108        CacheRow {
109            v: ROW_VERSION,
110            value: Zeroizing::new(rec.value.expose().to_string()),
111            version: rec.version,
112            created_at: rec.created_at,
113            rotated_at: rec.rotated_at,
114            fetched_at: OffsetDateTime::now_utc(),
115        }
116    }
117
118    /// Within the freshness window?
119    fn fresh(&self, ttl: Duration) -> bool {
120        fresh(self.fetched_at, ttl)
121    }
122
123    /// The record the remote returned — verbatim metadata.
124    fn record(&self) -> SecretRecord {
125        SecretRecord {
126            value: Secret::new(self.value.as_str()),
127            version: self.version,
128            created_at: self.created_at,
129            rotated_at: self.rotated_at,
130        }
131    }
132}

134/// Is `t` younger than `ttl`? A `fetched_at` in the future (clock
135/// skew) counts as *stale* — a timestamp that can't be trusted to
136/// be young is distrusted, erring toward one extra remote fetch.
137fn fresh(t: OffsetDateTime, ttl: Duration) -> bool {
138    let age = OffsetDateTime::now_utc()
139        .unix_timestamp()
140        .saturating_sub(t.unix_timestamp());
141    age >= 0 && (age as u64) < ttl.as_secs()
142}

144/// The list-marker sidecar — `{env → rfc3339}` per env whose
145/// `list` answer is a complete remote snapshot. Plaintext on
146/// purpose: it holds env names and timestamps, never refs or
147/// values (those stay inside the cache vault's ciphertext).
148#[derive(Default, Serialize, Deserialize)]
149struct MetaFile {
150    #[serde(default)]
151    listed_at: BTreeMap<String, String>,
152}
153
154/// A [`Vault`] that reads through a local encrypted cache to a
155/// remote source of truth. All cache access is best-effort — a
156/// cache that's absent (`None`), corrupt, or fails mid-op degrades
157/// to remote-only behavior, never to a wrong answer.
158pub struct CachedVault {
159    remote: Box<dyn Vault>,
160    cache: Option<Box<dyn Vault>>,
161    meta_path: Option<PathBuf>,
162    policy: CachePolicy,
163}
164
165impl CachedVault {
166    /// Wrap `remote` with `cache` as the local mirror. `cache` may
167    /// be `None` — the vault then behaves as a pass-through (the
168    /// degraded mode ops picks when the cache file won't open).
169    /// `meta_path` is the list-marker sidecar; `None` means `list`
170    /// never serves from the cache (no persisted freshness signal).
171    pub fn new(
172        remote: Box<dyn Vault>,
173        cache: Option<Box<dyn Vault>>,
174        meta_path: Option<PathBuf>,
175        policy: CachePolicy,
176    ) -> Self {
177        CachedVault {
178            remote,
179            cache,
180            meta_path,
181            policy,
182        }
183    }
184
185    /// The default cache vault file for a `[vaults]` name —
186    /// `$XDG_CACHE_HOME/cuora/<name>.age`, else `~/.cache/cuora/`.
187    /// XDG cache dir on purpose: disposable-by-convention data.
188    pub fn default_cache_file(vault_name: &str) -> PathBuf {
189        cache_dir().join(format!("{}.age", sanitize(vault_name)))
190    }
191
192    /// The default list-marker sidecar —
193    /// `$XDG_CACHE_HOME/cuora/<name>.listed.json`.
194    pub fn default_meta_file(vault_name: &str) -> PathBuf {
195        cache_dir().join(format!("{}.listed.json", sanitize(vault_name)))
196    }
197
198    /// The cache vault + policy gate, or `None` when `env` isn't
199    /// cacheable or no cache is wired.
200    fn cache_for(&self, env: &Env) -> Option<&dyn Vault> {
201        if !self.policy.caches(env) {
202            return None;
203        }
204        self.cache.as_deref()
205    }
206
207    /// The cached row, or `None` — a missing row, a cache error,
208    /// and an unparseable (corrupt) row are all just "not cached".
209    /// A corrupt row is removed so the next write-through starts
210    /// clean.
211    fn cache_row(&self, cache: &dyn Vault, env: &Env, rref: &Ref) -> Option<CacheRow> {
212        let rec = cache.get(env, rref).ok()??;
213        match serde_json::from_str::<CacheRow>(rec.value.expose()) {
214            Ok(row) => Some(row),
215            Err(_) => {
216                let _ = cache.rm(env, rref);
217                None
218            }
219        }
220    }
221
222    /// Write-through: `rec` into the cache, best-effort — the
223    /// remote already answered; a cache write failure must not
224    /// turn a good remote op into an error.
225    fn store_row(&self, cache: &dyn Vault, env: &Env, rref: &Ref, rec: &SecretRecord) {
226        if let Ok(json) = serde_json::to_string(&CacheRow::wrap(rec)) {
227            let _ = cache.put(env, rref, Secret::new(json));
228        }
229    }
230
231    /// Drop refs the remote no longer has — called after a
232    /// successful remote `list` so deletions propagate into the
233    /// mirror. Lazy and best-effort: unknown remote refs stay
234    /// uncached until their first `get`.
235    fn reconcile(&self, cache: &dyn Vault, env: &Env, remote: &[Ref]) {
236        if let Ok(cached) = cache.list(env) {
237            for r in cached.iter().filter(|r| !remote.contains(r)) {
238                let _ = cache.rm(env, r);
239            }
240        }
241    }
242
243    /// The env's list-freshness marker, or `None` — a missing,
244    /// unreadable, or unparsable sidecar is "never listed".
245    fn listed_at(&self, env: &Env) -> Option<OffsetDateTime> {
246        let text = std::fs::read_to_string(self.meta_path.as_ref()?).ok()?;
247        let meta: MetaFile = serde_json::from_str(&text).ok()?;
248        OffsetDateTime::parse(meta.listed_at.get(env.as_str())?, &Rfc3339).ok()
249    }
250
251    /// Stamp `env`'s list marker — best-effort; a failed write just
252    /// means the next `list` re-fetches.
253    fn mark_listed(&self, env: &Env) {
254        let Some(path) = &self.meta_path else {
255            return;
256        };
257        let mut meta: MetaFile = std::fs::read_to_string(path)
258            .ok()
259            .and_then(|t| serde_json::from_str(&t).ok())
260            .unwrap_or_default();
261        meta.listed_at.insert(
262            env.as_str().to_string(),
263            OffsetDateTime::now_utc()
264                .format(&Rfc3339)
265                .unwrap_or_default(),
266        );
267        if let Ok(json) = serde_json::to_string(&meta) {
268            if let Some(parent) = path.parent() {
269                let _ = std::fs::create_dir_all(parent);
270            }
271            if std::fs::write(path, json).is_ok() {
272                let _ = set_private(path);
273            }
274        }
275    }
276}

278impl Vault for CachedVault {
279    fn get(&self, env: &Env, rref: &Ref) -> Result<Option<SecretRecord>, VaultError> {
280        let Some(cache) = self.cache_for(env) else {
281            return self.remote.get(env, rref);
282        };
283        let row = self.cache_row(cache, env, rref);
284        if let Some(fresh) = row.as_ref().filter(|r| r.fresh(self.policy.ttl)) {
285            return Ok(Some(fresh.record()));
286        }
287        match self.remote.get(env, rref) {
288            Ok(Some(rec)) => {
289                self.store_row(cache, env, rref, &rec);
290                Ok(Some(rec))
291            }
292            Ok(None) => {
293                // remote is authoritative — a stale row for a
294                // deleted ref is worse than a miss
295                let _ = cache.rm(env, rref);
296                Ok(None)
297            }
298            Err(e) => match row {
299                // remote unreachable but a stale copy exists — the
300                // degraded read that makes the cache worth having
301                Some(row) => Ok(Some(row.record())),
302                None => Err(e),
303            },
304        }
305    }
306
307    fn put(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
308        // remote-first: writes never serve stale. A remote failure
309        // leaves the cache untouched — the old row still mirrors
310        // the old remote state.
311        let rec = self.remote.put(env, rref, value)?;
312        if let Some(cache) = self.cache_for(env) {
313            self.store_row(cache, env, rref, &rec);
314        }
315        Ok(rec)
316    }
317
318    fn rm(&self, env: &Env, rref: &Ref) -> Result<bool, VaultError> {
319        let removed = self.remote.rm(env, rref)?;
320        if let Some(cache) = self.cache_for(env) {
321            // invalidate even when `removed` is false — the remote
322            // not having it means any cached row is stale
323            let _ = cache.rm(env, rref);
324        }
325        Ok(removed)
326    }
327
328    fn list(&self, env: &Env) -> Result<Vec<Ref>, VaultError> {
329        let Some(cache) = self.cache_for(env) else {
330            return self.remote.list(env);
331        };
332        let marker = self.listed_at(env);
333        if marker.is_some_and(|t| fresh(t, self.policy.ttl)) {
334            if let Ok(refs) = cache.list(env) {
335                return Ok(refs);
336            }
337        }
338        match self.remote.list(env) {
339            Ok(refs) => {
340                self.reconcile(cache, env, &refs);
341                self.mark_listed(env);
342                Ok(refs)
343            }
344            Err(e) => match marker {
345                // ever listed + remote down → the stale snapshot is
346                // still the best answer
347                Some(_) => cache.list(env),
348                None => Err(e),
349            },
350        }
351    }
352
353    fn rotate(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
354        let rec = self.remote.rotate(env, rref, value)?;
355        if let Some(cache) = self.cache_for(env) {
356            self.store_row(cache, env, rref, &rec);
357        }
358        Ok(rec)
359    }
360}

362/// `$XDG_CACHE_HOME/cuora` or `~/.cache/cuora`. A pre-rename
363/// `…/arca` cache dir that still exists is reused so warm caches
364/// survive the rename.
365fn cache_dir() -> PathBuf {
366    let base = std::env::var("XDG_CACHE_HOME")
367        .map(PathBuf::from)
368        .unwrap_or_else(|_| home().join(".cache"));
369    let cuora = base.join("cuora");
370    let legacy = base.join("arca");
371    if cuora.exists() || !legacy.exists() {
372        cuora
373    } else {
374        legacy
375    }
376}
377
378/// `~` for path defaults.
379fn home() -> PathBuf {
380    std::env::var("HOME")
381        .map(PathBuf::from)
382        .unwrap_or_else(|_| PathBuf::from("."))
383}

385/// A `[vaults]` name is a TOML key — anything goes, including
386/// `..` and `/`. Map it onto a single safe filename component.
387fn sanitize(name: &str) -> String {
388    let out: String = name
389        .chars()
390        .map(|c| {
391            if c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-') {
392                c
393            } else {
394                '_'
395            }
396        })
397        .collect();
398    if out.is_empty() || out == "." || out == ".." {
399        "_".into()
400    } else {
401        out
402    }
403}

405/// `chmod 0600` — the sidecar is non-sensitive but
406/// private-by-default is still right.
407#[cfg(unix)]
408fn set_private(path: &Path) -> std::io::Result<()> {
409    use std::os::unix::fs::PermissionsExt;
410    std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o600))
411}
412
413#[cfg(not(unix))]
414fn set_private(_path: &Path) -> std::io::Result<()> {
415    Ok(())
416}

423    #[test]
424    fn policy_scopes_envs() {
425        let all = CachePolicy::all(Duration::from_secs(60));
426        assert!(all.caches(&Env::dev()));
427        assert!(all.caches(&Env::new("prod").unwrap()));
428        let scoped = CachePolicy::scoped([Env::dev()], Duration::from_secs(60));
429        assert!(scoped.caches(&Env::dev()));
430        assert!(!scoped.caches(&Env::new("prod").unwrap()));
431    }

474    #[test]
475    fn freshness_window() {
476        let ttl = Duration::from_secs(60);
477        assert!(fresh(OffsetDateTime::now_utc(), ttl));
478        // a row older than the TTL is stale; zero TTL is never fresh
479        assert!(!fresh(
480            OffsetDateTime::now_utc() - time::Duration::hours(1),
481            ttl
482        ));
483        assert!(!fresh(OffsetDateTime::now_utc(), Duration::ZERO));
484        // fetched_at in the future (clock skew) is distrusted → stale
485        assert!(!fresh(
486            OffsetDateTime::now_utc() + time::Duration::hours(1),
487            ttl
488        ));
