1//! TOML impl of `ConfigSource` — explicit path → $MYCO_CONFIG →
2//! ./myco.toml → ~/.config/myco/myco.toml.
3
4use myco_config::ConfigSource;
5use std::path::PathBuf;
6
7/// File-backed config source.
8pub struct TomlSource {
9    /// Explicit `--config` path, or None for the default search.
10    pub path: Option<String>,
11}
12
13impl ConfigSource for TomlSource {
14    fn load(&self) -> Result<myco_core::Config, String> {
15        let candidates: Vec<PathBuf> = match self
16            .path
17            .clone()
18            .or_else(|| std::env::var("MYCO_CONFIG").ok())
19        {
20            Some(p) => vec![PathBuf::from(p)],
21            None => vec![
22                PathBuf::from("myco.toml"),
23                std::env::var("HOME")
24                    .map(PathBuf::from)
25                    .unwrap_or(".".into())
26                    .join(".config/myco/myco.toml"),
27            ],
28        };
29        for p in candidates {
30            if p.exists() {
31                let text = std::fs::read_to_string(&p)
32                    .map_err(|e| format!("read {}: {e}", p.display()))?;
33                return toml::from_str(&text).map_err(|e| format!("parse {}: {e}", p.display()));
34            }
35        }
36        Err("no myco.toml found (search: $MYCO_CONFIG, ., ~/.config/myco/)".into())
37    }
38}

40#[cfg(test)]
41#[allow(clippy::unwrap_used)]
42mod tests {
43    use super::*;
44
45    #[test]
46    fn explicit_missing_file_errors() {
47        let src = TomlSource {
48            path: Some("/nonexistent/myco.toml".into()),
49        };
50        let err = src.load().err().unwrap_or_default();
51        assert!(err.contains("myco.toml"));
52    }
53
54    #[test]
55    fn parses_project_dir() {
56        let dir = std::env::temp_dir().join(format!("myco-cfg-{}", std::process::id()));
57        std::fs::create_dir_all(&dir).unwrap_or_default();
58        let f = dir.join("myco.toml");
59        std::fs::write(&f, "project_dir = \"project\"\n").unwrap_or_default();
60        let src = TomlSource {
61            path: Some(f.to_string_lossy().into_owned()),
62        };
63        let cfg = src.load().unwrap_or_default();
64        assert_eq!(cfg.project_dir.as_deref(), Some("project"));
65        let _ = std::fs::remove_dir_all(&dir);
66    }
67
68    #[test]
69    fn default_search_builds_candidates() {
70        // no explicit path, no MYCO_CONFIG → ./myco.toml +
71        // ~/.config/myco/myco.toml are probed (and miss in a
72        // bare cwd)
73        let src = TomlSource { path: None };
74        temp_env::with_vars(
75            [
76                ("MYCO_CONFIG", None::<&str>),
77                ("HOME", Some("/nonexistent-home")),
78            ],
79            || {
80                let dir =
81                    std::env::temp_dir().join(format!("myco-cfg-none-{}", std::process::id()));
82                std::fs::create_dir_all(&dir).unwrap();
83                let cwd = std::env::current_dir().unwrap();
84                std::env::set_current_dir(&dir).unwrap();
85                // no myco.toml anywhere reachable → search error
86                let first = src.load();
87                assert!(first.is_err(), "expected search error: {first:?}");
88                // ./myco.toml is found on the default search
89                std::fs::write(dir.join("myco.toml"), "project_dir = \"proj\"\n").unwrap();
90                let out = src.load().unwrap();
91                std::env::set_current_dir(cwd).unwrap();
92                assert_eq!(out.project_dir.as_deref(), Some("proj"));
93                let _ = std::fs::remove_dir_all(&dir);
94            },
95        );
96    }
97}
