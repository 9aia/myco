1//! TOML impl of `ConfigSource` — explicit path → $MYCO_CONFIG →
2//! ./myco.toml → ~/.config/myco/myco.toml.
3//!
4//! Scalar fields keep first-match semantics (the repo layer wins
5//! over the user layer). `[views]` is the one *merged* section
6//! (P-006): both layers contribute named views and the user layer
7//! wins name collisions — repo presets are shared, personal
8//! presets always reach through.
9
10use myco_config::ConfigSource;
11use std::path::PathBuf;
12
13/// File-backed config source.
14pub struct TomlSource {
15    /// Explicit `--config` path, or None for the default search.
16    pub path: Option<String>,
17}
18
19/// Parse exactly one config file — no layer search, no merge.
20/// For callers that need a specific layer (e.g. `myco check`
21/// linting only the repo's own myco.toml).
22pub fn load_file(path: &PathBuf) -> Result<myco_core::Config, String> {
23    let text =
24        std::fs::read_to_string(path).map_err(|e| format!("read {}: {e}", path.display()))?;
25    toml::from_str(&text).map_err(|e| format!("parse {}: {e}", path.display()))
26}
27
28fn parse(path: &PathBuf) -> Result<myco_core::Config, String> {
29    load_file(path)
30}
31
32/// First existing path wins within the user layer — XDG dir
33/// preferred over ~/.config.
34fn user_layer() -> Option<PathBuf> {
35    let mut candidates = Vec::new();
36    if let Ok(xdg) = std::env::var("XDG_CONFIG_HOME") {
37        candidates.push(PathBuf::from(xdg).join("myco/myco.toml"));
38    }
39    if let Ok(home) = std::env::var("HOME") {
40        candidates.push(PathBuf::from(home).join(".config/myco/myco.toml"));
41    }
42    candidates.into_iter().find(|p| p.exists())
43}
44
45impl ConfigSource for TomlSource {
46    fn load(&self) -> Result<myco_core::Config, String> {
47        // an explicit path (--config or $MYCO_CONFIG) that doesn't
48        // exist is an error naming it — not a silent skip
49        let explicit = self
50            .path
51            .clone()
52            .or_else(|| std::env::var("MYCO_CONFIG").ok());
53        let repo_layer: Option<PathBuf> = match explicit {
54            Some(p) => {
55                let p = PathBuf::from(p);
56                if !p.exists() {
57                    return Err(format!("config {} not found", p.display()));
58                }
59                Some(p)
60            }
61            None => {
62                let p = PathBuf::from("myco.toml");
63                p.exists().then_some(p)
64            }
65        };
66        let user_path = user_layer();
67        if repo_layer.is_none() && user_path.is_none() {
68            return Err(
69                "no myco.toml found (search: $MYCO_CONFIG, ., $XDG_CONFIG_HOME/myco/, ~/.config/myco/)"
70                    .into(),
71            );
72        }
73
74        let repo_cfg = repo_layer.as_ref().map(parse).transpose()?;
75        let user_cfg = user_path.as_ref().map(parse).transpose()?;
76
77        // scalars: repo layer wins (today's first-match behavior);
78        // views: merged, user layer wins name collisions
79        let mut merged = myco_core::Config {
80            project_dir: repo_cfg
81                .as_ref()
82                .and_then(|c| c.project_dir.clone())
83                .or_else(|| user_cfg.as_ref().and_then(|c| c.project_dir.clone())),
84            ..Default::default()
85        };
86        for cfg in [repo_cfg, user_cfg].into_iter().flatten() {
87            merged.views.extend(cfg.views);
88        }
89        Ok(merged)
90    }
91}

93#[cfg(test)]
94#[allow(clippy::unwrap_used)]
95mod tests {
96    use super::*;
97
98    #[test]
99    fn explicit_missing_file_errors() {
100        let src = TomlSource {
101            path: Some("/nonexistent/myco.toml".into()),
102        };
103        let err = src.load().err().unwrap_or_default();
104        assert!(err.contains("myco.toml"));
105    }
106
107    #[test]
108    fn parses_project_dir() {
109        let dir = std::env::temp_dir().join(format!("myco-cfg-{}", std::process::id()));
110        std::fs::create_dir_all(&dir).unwrap_or_default();
111        let f = dir.join("myco.toml");
112        std::fs::write(&f, "project_dir = \"project\"\n").unwrap_or_default();
113        let src = TomlSource {
114            path: Some(f.to_string_lossy().into_owned()),
115        };
116        let cfg = src.load().unwrap_or_default();
117        assert_eq!(cfg.project_dir.as_deref(), Some("project"));
118        let _ = std::fs::remove_dir_all(&dir);
119    }
120
121    #[test]
122    fn default_search_builds_candidates() {
123        // no explicit path, no MYCO_CONFIG → ./myco.toml +
124        // ~/.config/myco/myco.toml are probed (and miss in a
125        // bare cwd)
126        let src = TomlSource { path: None };
127        temp_env::with_vars(
128            [
129                ("MYCO_CONFIG", None::<&str>),
130                ("XDG_CONFIG_HOME", None::<&str>),
131                ("HOME", Some("/nonexistent-home")),
132            ],
133            || {
134                let dir =
135                    std::env::temp_dir().join(format!("myco-cfg-none-{}", std::process::id()));
136                std::fs::create_dir_all(&dir).unwrap();
137                let cwd = std::env::current_dir().unwrap();
138                std::env::set_current_dir(&dir).unwrap();
139                // no myco.toml anywhere reachable → search error
140                let first = src.load();
141                assert!(first.is_err(), "expected search error: {first:?}");
142                // ./myco.toml is found on the default search
143                std::fs::write(dir.join("myco.toml"), "project_dir = \"proj\"\n").unwrap();
144                let out = src.load().unwrap();
145                std::env::set_current_dir(cwd).unwrap();
146                assert_eq!(out.project_dir.as_deref(), Some("proj"));
147                let _ = std::fs::remove_dir_all(&dir);
148            },
149        );
150    }
151
152    #[test]
153    fn views_merge_with_user_winning() {
154        let base = std::env::temp_dir().join(format!("myco-cfg-merge-{}", std::process::id()));
155        let repo_dir = base.join("repo");
156        let home = base.join("home");
157        std::fs::create_dir_all(&repo_dir).unwrap();
158        std::fs::create_dir_all(home.join(".config/myco")).unwrap();
159        std::fs::write(
160            repo_dir.join("myco.toml"),
161            "[views.shared]\nfilter = { kind = [\"issue\"] }\n[views.repo_only]\nfilter = { kind = [\"task\"] }\n",
162        )
163        .unwrap();
164        std::fs::write(
165            home.join(".config/myco/myco.toml"),
166            "[views.shared]\nfilter = { kind = [\"task\"] }\n[views.user_only]\nfilter = { kind = [\"plan\"] }\n",
167        )
168        .unwrap();
169        temp_env::with_vars(
170            [
171                ("MYCO_CONFIG", None::<&str>),
172                ("XDG_CONFIG_HOME", None::<&str>),
173                ("HOME", Some(home.to_string_lossy().as_ref())),
174            ],
175            || {
176                let cwd = std::env::current_dir().unwrap();
177                std::env::set_current_dir(&repo_dir).unwrap();
178                let cfg = TomlSource { path: None }.load().unwrap();
179                std::env::set_current_dir(cwd).unwrap();
180                assert_eq!(cfg.views.len(), 3);
181                // user layer wins the collision
182                assert_eq!(
183                    cfg.views["shared"].filter.kind,
184                    vec![myco_core::ItemKind::Task]
185                );
186                assert!(cfg.views.contains_key("repo_only"));
187                assert!(cfg.views.contains_key("user_only"));
188            },
189        );
190        let _ = std::fs::remove_dir_all(&base);
191    }
