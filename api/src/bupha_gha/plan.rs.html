1//! plan — the pure workflow core: `.github/workflows/*.yml` text in,
2//! validated [`Job`] plans out. No I/O — the runner hands in YAML
3//! strings and an [`Event`] (ref + changed files); every construct
4//! outside the v1 subset is an explicit error naming the YAML key.

9/// The push event a workflow is selected against.
10pub struct Event<'a> {
11    /// Full git ref ("refs/heads/main"); "" when the source can't
12    /// say — `branches` filters then fail loudly rather than guess.
13    pub git_ref: &'a str,
14    /// Repo-relative paths changed by the push (v1: tip-commit
15    /// diff — see the crate README for the honest boundary).
16    pub changed_files: &'a [String],
17}

19/// GitHub-side facts `${{ }}` expressions can reference.
20#[derive(Clone, Copy)]
21pub struct GhaContext<'a> {
22    pub sha: &'a str,
23    pub git_ref: &'a str,
24    /// "owner/name" (no ":job" suffix).
25    pub repository: &'a str,
26    /// Always "push" today.
27    pub event_name: &'a str,
28}

79/// One runnable job after matrix fan-out, `${{ }}` interpolation and
80/// `needs` resolution. Indices in `needs` point into the slice this

82pub struct Job {
83    /// Workflow display name (for logs).
84    pub workflow: String,

95    pub shell: Option<String>,

125fn kind(v: &Value) -> &'static str {
126    match v {
127        Value::Null => "null",
128        Value::Bool(_) => "bool",
129        Value::Number(_) => "number",
130        Value::String(_) => "string",
131        Value::Sequence(_) => "sequence",
132        Value::Mapping(_) => "mapping",
133        Value::Tagged(_) => "tagged value",
134    }
135}

137fn scalar(v: &Value, what: &str) -> Result<String, String> {
138    match v {
139        Value::String(s) => Ok(s.clone()),
140        Value::Number(n) => Ok(n.to_string()),
141        Value::Bool(b) => Ok(b.to_string()),
142        _ => Err(format!("{what}: expected a scalar, got {}", kind(v))),
143    }
144}

146/// Reject keys outside the v1 allowlist — unsupported syntax is a
147/// hard error, never a silent skip. `unsupported` names get a
148/// clearer message than truly unknown keys.
149fn check_keys(
150    m: &serde_norway::Mapping,
151    allowed: &[&str],
152    unsupported: &[&str],
153    what: &str,
154) -> Result<(), String> {
155    for (k, _) in m {
156        let key = k.as_str().unwrap_or("");
157        if unsupported.contains(&key) {
158            return Err(format!(
159                "{what}: unsupported key `{key}` (outside the gha v1 subset)"
160            ));
161        }
162        if !allowed.contains(&key) {
163            return Err(format!("{what}: unknown key `{key}`"));
164        }
165    }
166    Ok(())
167}

169fn string_list(v: &Value, what: &str) -> Result<Vec<String>, String> {
170    match v {
171        Value::String(s) => Ok(vec![s.clone()]),
172        Value::Sequence(seq) => {
173            if seq.is_empty() {
174                return Err(format!("{what}: empty list"));
175            }
176            seq.iter()
177                .map(|i| match i {
178                    Value::String(s) => Ok(s.clone()),
179                    _ => Err(format!("{what}: entries must be strings")),
180                })
181                .collect()
182        }
183        _ => Err(format!("{what}: expected a string or list of strings")),
184    }
185}
186
187fn env_map(m: &serde_norway::Mapping, what: &str) -> Result<Vec<(String, String)>, String> {
188    let mut out = Vec::with_capacity(m.len());
189    for (k, v) in m {
190        let key = k
191            .as_str()
192            .ok_or_else(|| format!("{what}: env names must be strings"))?;
193        let valid = key
194            .chars()
195            .next()
196            .map(|c| c.is_ascii_alphabetic() || c == '_')
197            .unwrap_or(false)
198            && key.chars().all(|c| c.is_ascii_alphanumeric() || c == '_');
199        if !valid {
200            return Err(format!("{what}: invalid env name `{key}`"));
201        }
202        out.push((key.to_string(), scalar(v, &format!("{what}.{key}"))?));
203    }
204    Ok(out)
205}

211fn parse_push_filters(v: &Value, what: &str) -> Result<PushFilters, String> {
212    let m = match v {
213        Value::Null => return Ok(PushFilters::default()),
214        Value::Mapping(m) => m,
215        _ => return Err(format!("{what}: `on.push` must be a mapping or null")),
216    };
217    check_keys(
218        m,
219        &["branches", "branches-ignore", "paths", "paths-ignore"],
220        &["tags", "types", "workflows"],
221        what,
222    )?;
223    let mut f = PushFilters::default();
224    for (k, v) in m {
225        let key = k.as_str().unwrap_or("");
226        let list = string_list(v, &format!("{what}.{key}"))?;
227        match key {
228            "branches" => f.branches = Some(list),
229            "branches-ignore" => f.branches_ignore = Some(list),
230            "paths" => f.paths = Some(list),
231            "paths-ignore" => f.paths_ignore = Some(list),
232            _ => unreachable!(),
233        }
234    }
235    Ok(f)
236}

238/// `on:` may be a string, a list, or a map of trigger → config.
239/// Triggers other than `push` never fire here — ignored, not errors.
240fn parse_on(v: Option<&Value>, what: &str) -> Result<PushFilters, String> {
241    let Some(v) = v else {
242        return Err(format!("{what}: missing `on`"));
243    };
244    match v {
245        Value::String(s) => Ok(if s == "push" {
246            PushFilters::default()
247        } else {
248            PushFilters {
249                never: true,
250                ..PushFilters::default()
251            }
252        }),
253        Value::Sequence(seq) => {
254            let mut saw_push = false;
255            for item in seq {
256                match item.as_str() {
257                    Some("push") => saw_push = true,
258                    Some(_) => {}
259                    None => return Err(format!("{what}: `on` entries must be strings")),
260                }
261            }
262            Ok(PushFilters {
263                never: !saw_push,
264                ..PushFilters::default()
265            })
266        }
267        Value::Mapping(m) => {
268            let mut filters = PushFilters {
269                never: true,
270                ..PushFilters::default()
271            };
272            for (k, v) in m {
273                if k.as_str() == Some("push") {
274                    filters = parse_push_filters(v, &format!("{what}.on.push"))?;
275                }
276            }
277            Ok(filters)
278        }
279        _ => Err(format!("{what}: unsupported `on` shape ({})", kind(v))),
280    }
281}

283// ------------------------------------------------------------------
284// glob matching — `*`/`?` stay inside a path segment, `**` crosses `/`
285// ------------------------------------------------------------------
286
287fn glob_match(pat: &[u8], text: &[u8]) -> bool {
288    if pat.is_empty() {
289        return text.is_empty();
290    }
291    if pat.starts_with(b"**") {
292        return (0..=text.len()).any(|skip| glob_match(&pat[2..], &text[skip..]));
293    }
294    if text.is_empty() {
295        return false;
296    }
297    match pat[0] {
298        b'*' => (0..=text.len())
299            .take_while(|&skip| !text[..skip].contains(&b'/'))
300            .any(|skip| glob_match(&pat[1..], &text[skip..])),
301        b'?' => text[0] != b'/' && glob_match(&pat[1..], &text[1..]),
302        c => text[0] == c && glob_match(&pat[1..], &text[1..]),
303    }
304}
305
306fn any_glob(patterns: &[String], text: &str) -> bool {
307    patterns
308        .iter()
309        .any(|p| glob_match(p.as_bytes(), text.as_bytes()))
310}

312fn short_ref(git_ref: &str) -> &str {
313    git_ref
314        .strip_prefix("refs/heads/")
315        .or_else(|| git_ref.strip_prefix("refs/tags/"))
316        .unwrap_or(git_ref)
317}

319impl Workflow {
320    /// Does this workflow's `on.push` select the event?
321    pub fn matches_push(&self, ev: &Event) -> Result<bool, String> {
322        if self.filters.never {
323            return Ok(false);
324        }
325        let f = &self.filters;
326        let short = short_ref(ev.git_ref);
327        if let Some(pats) = &f.branches {
328            if ev.git_ref.is_empty() {
329                return Err("on.push.branches: cannot evaluate — event ref is unknown".into());
330            }
331            if !any_glob(pats, short) {
332                return Ok(false);
333            }
334        }
335        if let Some(pats) = &f.branches_ignore {
336            if ev.git_ref.is_empty() {
337                return Err(
338                    "on.push.branches-ignore: cannot evaluate — event ref is unknown".into(),
339                );
340            }
341            if any_glob(pats, short) {
342                return Ok(false);
343            }
344        }
345        if let Some(pats) = &f.paths {
346            if !ev.changed_files.iter().any(|f| any_glob(pats, f)) {
347                return Ok(false);
348            }
349        }
350        if let Some(pats) = &f.paths_ignore {
351            // Runs unless EVERY changed file matches an ignore
352            // pattern; an empty diff has nothing to ignore.
353            let all_ignored =
354                !ev.changed_files.is_empty() && ev.changed_files.iter().all(|f| any_glob(pats, f));
355            if all_ignored {
356                return Ok(false);
357            }
358        }
359        Ok(true)
360    }

362    /// Fan out matrix combos, interpolate `${{ }}`, resolve `needs`
363    /// and topo-order. `ctx` supplies the github.* facts.
364    pub fn expand(&self, ctx: GhaContext) -> Result<Vec<Job>, String> {
365        // Matrix fan-out, declaration order preserved.
366        let mut flat: Vec<(usize, Vec<(String, String)>)> = Vec::new();
367        for (ji, j) in self.jobs.iter().enumerate() {
368            for combo in matrix_combos(&j.matrix, &format!("jobs.{}", j.id))? {
369                flat.push((ji, combo));
370            }
371        }
372        // needs id → every expanded index of that job.
373        let mut by_id: HashMap<&str, Vec<usize>> = HashMap::new();
374        for (idx, (ji, _)) in flat.iter().enumerate() {
375            by_id
376                .entry(self.jobs[*ji].id.as_str())
377                .or_default()
378                .push(idx);
379        }
380        let mut jobs = Vec::with_capacity(flat.len());
381        for (ji, combo) in &flat {
382            jobs.push(self.build_job(&self.jobs[*ji], combo, &by_id, ctx)?);
383        }
384        topo_order(jobs)
385    }
386
387    fn build_job(
388        &self,
389        j: &RawJob,
390        combo: &[(String, String)],
391        by_id: &HashMap<&str, Vec<usize>>,
392        ctx: GhaContext,
393    ) -> Result<Job, String> {
394        let what = format!("jobs.{}", j.id);
395        let mut needs: Vec<usize> = j
396            .needs
397            .iter()
398            .flat_map(|n| by_id.get(n.as_str()).cloned().unwrap_or_default())
399            .collect();
400        needs.sort_unstable();
401        needs.dedup();
402
403        // env.* is unavailable while defining env — github.*/matrix.*
404        // only. Workflow env merges under job env.
405        let no_env = Eval {
406            ctx,
407            matrix: combo,
408            env: None,
409        };
410        let wf_env = interpolate_pairs(&self.env, &no_env, &format!("{what}.env"))?;
411        let job_env = merge_env(
412            &wf_env,
413            &interpolate_pairs(&j.env, &no_env, &format!("{what}.env"))?,
414        );
415        let eval = Eval {
416            ctx,
417            matrix: combo,
418            env: Some(&job_env),
419        };
420
421        let mut steps = Vec::with_capacity(j.steps.len());
422        for (si, s) in j.steps.iter().enumerate() {
423            steps.push(build_step(
424                s,
425                &eval,
426                &job_env,
427                ctx,
428                &format!("{what}.steps[{si}]"),
429            )?);
430        }
431        let defaults = Defaults {
432            working_directory: pick(
433                &j.defaults.working_directory,
434                &self.defaults.working_directory,
435            )
436            .map(|d| interpolate(d, &eval, &format!("{what}.defaults.run.working-directory")))
437            .transpose()?,
438            shell: pick(&j.defaults.shell, &self.defaults.shell)
439                .map(|s| interpolate(s, &eval, &format!("{what}.defaults.run.shell")))
440                .transpose()?,
441        };
442        Ok(Job {
443            workflow: self.name.clone(),
444            label: j.name.clone().unwrap_or_else(|| j.id.clone()),
445            runs_on: j
446                .runs_on
447                .iter()
448                .map(|r| interpolate(r, &eval, &format!("{what}.runs-on")))
449                .collect::<Result<_, _>>()?,
450            needs,
451            env: job_env,
452            working_directory: defaults.working_directory,
453            shell: defaults.shell,
454            steps,
455        })
456    }
457}

463/// Last-wins merge of two env lists.
464fn merge_env(base: &[(String, String)], over: &[(String, String)]) -> Vec<(String, String)> {
465    let mut out = base.to_vec();
466    for (k, v) in over {
467        match out.iter_mut().find(|(ek, _)| ek == k) {
468            Some(slot) => slot.1 = v.clone(),
469            None => out.push((k.clone(), v.clone())),
470        }
471    }
472    out
473}

475/// Interpolate every value in an env list (env.* unavailable).
476fn interpolate_pairs(
477    raw: &[(String, String)],
478    eval: &Eval,
479    what: &str,
480) -> Result<Vec<(String, String)>, String> {
481    raw.iter()
482        .map(|(k, v)| Ok((k.clone(), interpolate(v, eval, &format!("{what}.{k}"))?)))
483        .collect()
484}

486/// Interpolation context for one expansion — github facts plus the
487/// matrix bindings and merged env visible at this point.
488pub struct Eval<'a> {
489    pub ctx: GhaContext<'a>,
490    pub matrix: &'a [(String, String)],
491    pub env: Option<&'a [(String, String)]>,
492}
493
494fn build_step(
495    s: &RawStep,
496    eval: &Eval,
497    job_env: &[(String, String)],
498    ctx: GhaContext,
499    what: &str,
500) -> Result<Step, String> {
501    match s {
502        RawStep::Checkout => Ok(Step::Checkout),
503        RawStep::Run {
504            shell,
505            working_directory,
506            env,
507            script,
508        } => {
509            // Step env merges over job env; its own pairs are what
510            // the step block exports.
511            let step_own = interpolate_pairs(env, eval, &format!("{what}.env"))?;
512            let step_env = merge_env(job_env, &step_own);
513            let step_eval = Eval {
514                ctx,
515                matrix: eval.matrix,
516                env: Some(&step_env),
517            };
518            Ok(Step::Run(RunStep {
519                shell: shell
520                    .as_ref()
521                    .map(|s| interpolate(s, &step_eval, &format!("{what}.shell")))
522                    .transpose()?,
523                working_directory: working_directory
524                    .as_ref()
525                    .map(|w| interpolate(w, &step_eval, &format!("{what}.working-directory")))
526                    .transpose()?,
527                env: step_own,
528                script: interpolate(script, &step_eval, &format!("{what}.run"))?,
529            }))
530        }
531    }
532}

534/// Kahn topo sort, declaration order breaking ties. Indices in
535/// `needs` refer to the input vec; the output re-maps them.
536fn topo_order(jobs: Vec<Job>) -> Result<Vec<Job>, String> {
537    let n = jobs.len();
538    let mut slots: Vec<Option<Job>> = jobs.into_iter().map(Some).collect();
539    let mut out_pos = vec![usize::MAX; n];
540    let mut emitted = vec![false; n];
541    let mut order: Vec<usize> = Vec::with_capacity(n);
542    for _ in 0..n {
543        let mut progressed = false;
544        for i in 0..n {
545            if emitted[i] {
546                continue;
547            }
548            let ready = slots[i]
549                .as_ref()
550                .map(|j| j.needs.iter().all(|&d| emitted[d]))
551                .unwrap_or(false);
552            if !ready {
553                continue;
554            }
555            emitted[i] = true;
556            out_pos[i] = order.len();
557            order.push(i);
558            progressed = true;
559        }
560        if order.len() == n {
561            break;
562        }
563        if !progressed {
564            return Err("jobs: dependency cycle in `needs`".into());
565        }
566    }
567    let mut out = Vec::with_capacity(n);
568    for i in order {
569        if let Some(mut j) = slots[i].take() {
570            j.needs = j.needs.iter().map(|&d| out_pos[d]).collect();
571            j.needs.sort_unstable();
572            j.needs.dedup();
573            out.push(j);
574        }
575    }
576    Ok(out)
577}

583/// Replace every `${{ expr }}` in `tpl`. Anything that isn't a
584/// plain dotted path is an explicit error.
585pub fn interpolate(tpl: &str, eval: &Eval, what: &str) -> Result<String, String> {
586    let mut out = String::with_capacity(tpl.len());
587    let mut rest = tpl;
588    while let Some(start) = rest.find("${{") {
589        out.push_str(&rest[..start]);
590        let after = &rest[start + 3..];
591        let Some(end) = after.find("}}") else {
592            return Err(format!("{what}: unclosed `${{{{`"));
593        };
594        out.push_str(&resolve(after[..end].trim(), eval, what)?);
595        rest = &after[end + 2..];
596    }
597    out.push_str(rest);
598    Ok(out)
599}
600
601fn resolve(inner: &str, eval: &Eval, what: &str) -> Result<String, String> {
602    let bad = || {
603        format!(
604            "{what}: unsupported expression `${{{{ {inner} }}}}` — supported: \
605             env.*, github.{{sha,ref,ref_name,repository,event_name}}, matrix.*"
606        )
607    };
608    let dotted_ok = !inner.is_empty()
609        && inner
610            .chars()
611            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '.' | '-'));
612    if !dotted_ok {
613        return Err(bad());
614    }
615    let (root, rest) = inner.split_once('.').unwrap_or((inner, ""));
616    match root {
617        "env" => {
618            if rest.is_empty() {
619                return Err(bad());
620            }
621            let Some(env) = eval.env else {
622                return Err(format!("{what}: `env.*` is not available in this position"));
623            };
624            // Undefined env expands empty, like GitHub.
625            Ok(env
626                .iter()
627                .rev()
628                .find(|(k, _)| k == rest)
629                .map(|(_, v)| v.clone())
630                .unwrap_or_default())
631        }
632        "github" => match rest {
633            "sha" => Ok(eval.ctx.sha.to_string()),
634            "ref" => Ok(eval.ctx.git_ref.to_string()),
635            "ref_name" => Ok(short_ref(eval.ctx.git_ref).to_string()),
636            "repository" => Ok(eval.ctx.repository.to_string()),
637            "event_name" => Ok(eval.ctx.event_name.to_string()),
638            _ => Err(format!(
639                "{what}: unsupported github context `{inner}` — supported: \
640                 sha, ref, ref_name, repository, event_name"
641            )),
642        },
643        "matrix" => {
644            if rest.is_empty() {
645                return Err(bad());
646            }
647            eval.matrix
648                .iter()
649                .find(|(k, _)| k == rest)
650                .map(|(_, v)| v.clone())
651                .ok_or_else(|| format!("{what}: matrix has no `{rest}` binding"))
652        }
653        "needs" => Err(format!(
654            "{what}: `needs.*` contexts (outputs/result) are not supported in v1"
655        )),
656        _ => Err(bad()),
657    }
658}

664/// Parse + validate one workflow file. `source` names the file in
665/// errors.
666pub fn parse_workflow(source: &str, text: &str) -> Result<Workflow, String> {
667    let v: Value =
668        serde_norway::from_str(text).map_err(|e| format!("{source}: YAML parse error: {e}"))?;
669    let top = as_map(&v, source)?;
670    check_keys(
671        top,
672        &[
673            "name",
674            "run-name",
675            "on",
676            "env",
677            "defaults",
678            "jobs",
679            "permissions",
680        ],
681        &[],
682        source,
683    )?;
684    let name = get(top, "name")
685        .map(|n| scalar(n, &format!("{source}.name")))
686        .transpose()?
687        .unwrap_or_else(|| {
688            source
689                .rsplit('/')
690                .next()
691                .unwrap_or(source)
692                .trim_end_matches(".yml")
693                .trim_end_matches(".yaml")
694                .to_string()
695        });
696    let env = match get(top, "env") {
697        Some(Value::Mapping(m)) => env_map(m, &format!("{source}.env"))?,
698        Some(v) => return Err(format!("{source}.env: expected a mapping, got {}", kind(v))),
699        None => Vec::new(),
700    };
701    let defaults = parse_defaults(get(top, "defaults"), &format!("{source}.defaults"))?;
702    let filters = parse_on(get(top, "on"), &format!("{source}.on"))?;
703    let jobs_v = get(top, "jobs").ok_or_else(|| format!("{source}: missing `jobs`"))?;
704    let jobs_m = as_map(jobs_v, &format!("{source}.jobs"))?;
705    if jobs_m.is_empty() {
706        return Err(format!("{source}: `jobs` is empty"));
707    }
708    let mut jobs = Vec::with_capacity(jobs_m.len());
709    for (k, jv) in jobs_m {
710        let id = k
711            .as_str()
712            .ok_or_else(|| format!("{source}.jobs: job ids must be strings"))?
713            .to_string();
714        jobs.push(parse_job(
715            as_map(jv, &format!("{source}.jobs.{id}"))?,
716            &id,
717            source,
718        )?);
719    }
720    let ids: HashSet<&str> = jobs.iter().map(|j| j.id.as_str()).collect();
721    for j in &jobs {
722        for n in &j.needs {
723            if !ids.contains(n.as_str()) {
724                return Err(format!(
725                    "{source}.jobs.{}: `needs` references unknown job `{n}`",
726                    j.id
727                ));
728            }
729        }
730    }
731    Ok(Workflow {
732        name,
733        env,
734        defaults,
735        filters,
736        jobs,
737    })
738}

740fn parse_defaults(v: Option<&Value>, what: &str) -> Result<Defaults, String> {
741    let Some(v) = v else {
742        return Ok(Defaults::default());
743    };
744    let m = as_map(v, what)?;
745    check_keys(m, &["run"], &[], what)?;
746    let Some(rv) = get(m, "run") else {
747        return Ok(Defaults::default());
748    };
749    let rm = as_map(rv, &format!("{what}.run"))?;
750    check_keys(
751        rm,
752        &["working-directory", "shell"],
753        &[],
754        &format!("{what}.run"),
755    )?;
756    Ok(Defaults {
757        working_directory: get(rm, "working-directory")
758            .map(|s| scalar(s, &format!("{what}.run.working-directory")))
759            .transpose()?,
760        shell: get(rm, "shell")
761            .map(|s| scalar(s, &format!("{what}.run.shell")))
762            .transpose()?,
763    })
764}

766fn parse_job(m: &serde_norway::Mapping, id: &str, source: &str) -> Result<RawJob, String> {
767    let what = format!("{source}.jobs.{id}");
768    check_keys(
769        m,
770        &[
771            "name",
772            "needs",
773            "runs-on",
774            "env",
775            "defaults",
776            "steps",
777            "strategy",
778            "permissions",
779        ],
780        // The v1 boundary, loudly: conditions, containers,
781        // services, timeouts, outputs, concurrency.
782        &[
783            "if",
784            "timeout-minutes",
785            "services",
786            "container",
787            "environment",
788            "concurrency",
789            "outputs",
790            "continue-on-error",
791            "uses",
792            "secrets",
793        ],
794        &what,
795    )?;
796    let name = get(m, "name")
797        .map(|n| scalar(n, &format!("{what}.name")))
798        .transpose()?;
799    let needs = match get(m, "needs") {
800        Some(v) => string_list(v, &format!("{what}.needs"))?,
801        None => Vec::new(),
802    };
803    let runs_on = match get(m, "runs-on") {
804        Some(v) => string_list(v, &format!("{what}.runs-on"))?,
805        None => Vec::new(),
806    };
807    let env = match get(m, "env") {
808        Some(Value::Mapping(em)) => env_map(em, &format!("{what}.env"))?,
809        Some(v) => return Err(format!("{what}.env: expected a mapping, got {}", kind(v))),
810        None => Vec::new(),
811    };
812    let defaults = parse_defaults(get(m, "defaults"), &format!("{what}.defaults"))?;
813    let matrix = parse_matrix(get(m, "strategy"), &what)?;
814    let steps_v = get(m, "steps").ok_or_else(|| format!("{what}: missing `steps`"))?;
815    let Value::Sequence(seq) = steps_v else {
816        return Err(format!("{what}.steps: expected a sequence"));
817    };
818    if seq.is_empty() {
819        return Err(format!("{what}.steps: empty"));
820    }
821    let mut steps = Vec::with_capacity(seq.len());
822    for (si, sv) in seq.iter().enumerate() {
823        let sw = format!("{what}.steps[{si}]");
824        let sm = as_map(sv, &sw)?;
825        check_keys(
826            sm,
827            &[
828                "id",
829                "name",
830                "run",
831                "uses",
832                "working-directory",
833                "env",
834                "shell",
835            ],
836            &["if", "timeout-minutes", "continue-on-error"],
837            &sw,
838        )?;
839        let run = get(sm, "run");
840        let uses = get(sm, "uses");
841        match (run, uses) {
842            (Some(_), Some(_)) => {
843                return Err(format!("{sw}: step cannot have both `run` and `uses`"))
844            }
845            (None, None) => return Err(format!("{sw}: step needs `run` or `uses`")),
846            (Some(rv), None) => {
847                let Value::String(script) = rv else {
848                    return Err(format!("{sw}.run: expected a string"));
849                };
850                let env = match get(sm, "env") {
851                    Some(Value::Mapping(em)) => env_map(em, &format!("{sw}.env"))?,
852                    Some(v) => {
853                        return Err(format!("{sw}.env: expected a mapping, got {}", kind(v)))
854                    }
855                    None => Vec::new(),
856                };
857                steps.push(RawStep::Run {
858                    shell: get(sm, "shell")
859                        .map(|s| scalar(s, &format!("{sw}.shell")))
860                        .transpose()?,
861                    working_directory: get(sm, "working-directory")
862                        .map(|s| scalar(s, &format!("{sw}.working-directory")))
863                        .transpose()?,
864                    env,
865                    script: script.clone(),
866                });
867            }
868            (None, Some(uv)) => {
869                let uses = scalar(uv, &format!("{sw}.uses"))?;
870                parse_uses(&uses, sm, &sw)?;
871                steps.push(RawStep::Checkout);
872            }
873        }
874    }
875    Ok(RawJob {
876        id: id.to_string(),
877        name,
878        needs,
879        runs_on,
880        env,
881        defaults,
882        matrix,
883        steps,
884    })
885}

887/// `uses:` — only `actions/checkout` exists in v1 (a verification
888/// no-op: the worktree is already at the pushed sha). Everything
889/// else, `actions/setup-*` included, is an explicit error.
890fn parse_uses(uses: &str, sm: &serde_norway::Mapping, what: &str) -> Result<(), String> {
891    let unsupported = || {
892        format!(
893            "{what}: unsupported `uses: {uses}` — v1 supports only \
894             `actions/checkout` (`actions/setup-*` is a later milestone)"
895        )
896    };
897    let body = uses.split('@').next().unwrap_or(uses);
898    if body.starts_with("docker://") || body.starts_with("./") || body.starts_with('/') {
899        return Err(unsupported());
900    }
901    if !body.eq_ignore_ascii_case("actions/checkout") {
902        return Err(unsupported());
903    }
904    match get(sm, "with") {
905        Some(Value::Mapping(w)) => {
906            if get(w, "path").is_some() {
907                return Err(format!(
908                    "{what}: `with.path` on actions/checkout is unsupported — \
909                     the worktree root is fixed"
910                ));
911            }
912            // Other checkout inputs (ref, fetch-depth, token, …)
913            // don't change local semantics: the sha is checked out.
914        }
915        Some(_) => return Err(format!("{what}.with: expected a mapping")),
916        None => {}
917    }
918    Ok(())
919}

921fn parse_matrix(v: Option<&Value>, what: &str) -> Result<Vec<(String, Vec<String>)>, String> {
922    let Some(v) = v else { return Ok(Vec::new()) };
923    let m = as_map(v, &format!("{what}.strategy"))?;
924    check_keys(
925        m,
926        &["matrix"],
927        &["fail-fast", "max-parallel"],
928        &format!("{what}.strategy"),
929    )?;
930    let Some(mv) = get(m, "matrix") else {
931        return Ok(Vec::new());
932    };
933    let mm = as_map(mv, &format!("{what}.strategy.matrix"))?;
934    let mut out = Vec::with_capacity(mm.len());
935    for (k, vv) in mm {
936        let key = k
937            .as_str()
938            .ok_or_else(|| format!("{what}.strategy.matrix: keys must be strings"))?;
939        let Value::Sequence(seq) = vv else {
940            return Err(format!(
941                "{what}.strategy.matrix.{key}: expected a list — \
942                 `include`/`exclude` are not supported in v1"
943            ));
944        };
945        if seq.is_empty() {
946            return Err(format!("{what}.strategy.matrix.{key}: empty list"));
947        }
948        let vals = seq
949            .iter()
950            .map(|i| scalar(i, &format!("{what}.strategy.matrix.{key}")))
951            .collect::<Result<Vec<_>, _>>()?;
952        out.push((key.to_string(), vals));
953    }
954    Ok(out)
955}

957/// Cartesian product of matrix axes; no matrix → one empty combo.
958fn matrix_combos(
959    axes: &[(String, Vec<String>)],
960    what: &str,
961) -> Result<Vec<Vec<(String, String)>>, String> {
962    let mut combos: Vec<Vec<(String, String)>> = vec![Vec::new()];
963    for (name, vals) in axes {
964        let mut next = Vec::with_capacity(combos.len() * vals.len());
965        for c in &combos {
966            for v in vals {
967                let mut c = c.clone();
968                c.push((name.clone(), v.clone()));
969                next.push(c);
970            }
971        }
972        combos = next;
973    }
974    if combos.is_empty() {
975        return Err(format!("{what}: matrix produced zero combinations"));
976    }
977    Ok(combos)
978}
