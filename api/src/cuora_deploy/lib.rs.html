1//! cuora-deploy — the write-only push port.
2//!
3//! Deploy targets are **sinks, not sources**. Some stores accept a
4//! value and never give it back — Cloudflare's account Secrets
5//! Store is the motivating case: `POST`/`PATCH` a value in and the
6//! API literally cannot return it. A [`Deploy`] therefore has
7//! `push`/`remove` and deliberately no `get`/`list` — anything
8//! readable belongs behind the [`cuora_vault::Vault`] port instead.
9//!
10//! The value pushed is **plaintext** (decrypted at the edge by the
11//! caller): the sink is the thing Workers/CI read the secret from
12//! directly — the age-ciphertext boundary lives at cuora's own
13//! vaults, not at external secret stores that must serve values to
14//! runtimes that can't hold an cuora identity.
15//!
16//! Naming: impls map `(env, ref)` onto their sink's name scheme;
17//! [`PushReport::name`] reports what the value was written as.
18//!
19//! Sync by design — no async runtime; impls use blocking HTTP.
20
21use cuora_core::{Env, Ref, Secret};
22
23/// What a successful [`Deploy::push`] did at the sink.
24#[derive(Debug, Clone, PartialEq, Eq)]
25pub struct PushReport {
26    /// The sink-side name the value was written under (e.g.
27    /// `prod_billing_stripe`).
28    pub name: String,
29    /// `true` when the sink created a new entry; `false` when it
30    /// updated an existing one.
31    pub created: bool,
32}
33
34/// A write-only secret sink.
35pub trait Deploy: Send + Sync {
36    /// The configured target name (`[deploy.<name>]`) — for
37    /// messages and audit context.
38    fn name(&self) -> &str;
39
40    /// Write `value` at the sink under the name derived from
41    /// `(env, rref)`, creating or updating as needed.
42    fn push(&self, env: &Env, rref: &Ref, value: &Secret) -> Result<PushReport, DeployError>;
43
44    /// Delete the sink entry for `(env, rref)`; `true` when
45    /// something was removed.
46    fn remove(&self, env: &Env, rref: &Ref) -> Result<bool, DeployError>;
47}
48
49/// Deploy port errors — impls map their own failure modes onto
50/// these so callers match without knowing the sink.
51#[derive(Debug, thiserror::Error)]
52pub enum DeployError {
53    /// Misconfigured target (missing account/store id, unset token
54    /// env var).
55    #[error("deploy config: {0}")]
56    Config(String),
57    /// Network/transport failure reaching the sink API.
58    #[error("deploy i/o: {0}")]
59    Io(String),
60    /// The sink refused the credential (HTTP 401/403).
61    #[error("deploy unauthorized: {0}")]
62    Unauthorized(String),
63    /// Anything else the sink wants to say.
64    #[error("deploy backend: {0}")]
65    Backend(String),
66}
