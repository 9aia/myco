1//! `include:` base resolution — the install half of payload
2//! composition (PL-0007 / P-004).
3//!
4//! Each `Manifest::include` entry resolves **in manifest order** to a
5//! base payload dir:
6//!
7//! - an **installed store name** — full label (`@scope/name@version`),
8//!   scoped name (`@scope/name`), or bare name (`name`, ambiguous
9//!   across scopes → error) — via [`find_store_entry`] over
10//!   `Store::entries`. A missing name is a hard error naming it with a
11//!   hint to use an installable ref; `@`-prefixed entries never fall
12//!   through to path resolution.
13//! - an **installable ref** — anything the ref grammar claims
14//!   (`git:`/`cf:`/`http(s)`/paths) that isn't a store name — recurses
15//!   through [`crate::install_inner`], so the base gets the full
16//!   pipeline (fetch → *its own* includes → drivers → store commit)
17//!   and the outer package composes over the committed payload.
18//!
19//! Recursion is bounded two ways ([`IncludeState`]): a depth cap
20//! ([`MAX_INCLUDE_DEPTH`]) and a cycle set of in-progress package
21//! labels — a package including itself or an ancestor fails loudly.
22//!
23//! Composition itself is *not* transitive on the store-name path: a
24//! store entry's payload is whatever was committed at *its* install
25//! (already composed if it had `include:` then). We compose stored
26//! payloads as-is — never re-resolve their includes.
27

38/// Include-recursion cap — a chain of ref-includes deeper than this
39/// fails instead of recursing forever (cycles are caught earlier by
40/// [`IncludeState::in_progress`]; the cap bounds pathological DAGs).
41pub(crate) const MAX_INCLUDE_DEPTH: usize = 8;
42
43/// Cycle/depth state threaded through recursive installs. Created
44/// fresh per top-level [`crate::install_with_progress`] call.
45#[derive(Default)]
46pub(crate) struct IncludeState {
47    /// Labels (`@scope/name@version`, `name@<sha12>`) of packages whose
48    /// `include:` lists are currently resolving — the ancestor chain.
49    /// [`install_inner`] inserts its own label before resolving and
50    /// removes it after; [`resolve_base`] rejects a base whose store
51    /// label is already in the set.
52    pub in_progress: HashSet<String>,
53    /// Current include-recursion depth — `0` at the top-level install,
54    /// incremented around each ref-include's recursive install.
55    pub depth: usize,
56}
57
58/// The store-entry label for a not-yet-committed package — same rule
59/// as [`StoreEntry::label`] (`@scope/name@version`, sha short form
60/// when unversioned) so in-progress keys compare equal to store hits.
61pub(crate) fn pending_label(
62    name: &str,
63    scope: Option<&str>,
64    version: Option<&str>,
65    resolved: &str,
66) -> String {
67    let record = StoreRecord {
68        name: name.to_string(),
69        scope: scope.map(str::to_string),
70        version: version.map(str::to_string),
71        kind: None,
72        source_ref: String::new(),
73        resolved: resolved.to_string(),
74        digest: String::new(),
75        signature: None,
76        installed_at: 0,
77    };
78    StoreEntry {
79        dir: PathBuf::new(),
80        record,
81    }
82    .label()
83}

85/// Resolve every `include:` entry of `includes` to a base payload dir,
86/// **in manifest order** (earlier entries outrank later ones —
87/// [`tamias_core::overlay`] applies them last-to-first).
88///
89/// `report` sees a [`Progress::Including`] event per resolved base so
90/// surfaces can show composition happening; ref-includes additionally
91/// stream the recursive install's own events.
92///
93/// # Errors
94/// Fails on an ambiguous/missing store name, an unparseable ref, a
95/// cycle, depth-cap breach, or any recursive-install failure.
96#[allow(clippy::too_many_arguments)]
97pub(crate) fn resolve_bases(
98    env: &Env,
99    includes: &[String],
100    state: &mut IncludeState,
101    allow_drivers: bool,
102    surface: Surface,
103    report: &mut dyn FnMut(Progress),
104) -> Result<Vec<PathBuf>, Error> {
105    let store = Store::new(&env.store_root);
106    let mut bases = Vec::with_capacity(includes.len());
107    for text in includes {
108        // Re-read per entry: a ref-include earlier in the list installs
109        // into the store, and a later store-name entry may name it.
110        let entries = store.entries()?;
111        bases.push(resolve_base(
112            env,
113            &entries,
114            text,
115            state,
116            allow_drivers,
117            surface,
118            report,
119        )?);
120    }
121    Ok(bases)
122}

124/// One include entry → one base payload dir. See the module docs for
125/// the store-name-then-ref order.
126#[allow(clippy::too_many_arguments)]
127fn resolve_base(
128    env: &Env,
129    entries: &[StoreEntry],
130    text: &str,
131    state: &mut IncludeState,
132    allow_drivers: bool,
133    surface: Surface,
134    report: &mut dyn FnMut(Progress),
135) -> Result<PathBuf, Error> {
136    // 1. Store name — label / `@scope/name` / bare `name`.
137    if let Some(entry) = find_store_entry(entries, text)? {
138        let label = entry.label();
139        if state.in_progress.contains(&label) {
140            return Err(Error::Failed(format!(
141                "include cycle: '{text}' resolves to {label}, which is already being installed"
142            )));
143        }
144        report(Progress::Including { base: label });
145        return Ok(entry.dir);
146    }
147    // `@scope/name` is a *name*, never a path — a missing one errors
148    // here instead of falling through to fs-ref resolution.
149    if text.starts_with('@') {
150        return Err(missing_base(text));
151    }
152    // 2. Installable ref — recurse through the full pipeline.
153    let spec = crate::refs().parse(text).map_err(|_| missing_base(text))?;
154    if spec.is::<FsSpec>() {
155        // `FsSpec.path` is `expand_path(input)` — recompute it rather
156        // than downcasting the opaque spec.
157        if !expand_path(Path::new(text)).exists() {
158            return Err(missing_base(text));
159        }
160    }
161    if state.depth >= MAX_INCLUDE_DEPTH {
162        return Err(Error::Failed(format!(
163            "include '{text}' exceeds the {MAX_INCLUDE_DEPTH}-deep nesting cap"
164        )));
165    }
166    report(Progress::Including {
167        base: text.to_string(),
168    });
169    state.depth += 1;
170    let result = install_inner(env, text, allow_drivers, surface, report, state);
171    state.depth -= 1;
172    let entry = result?;
173    Ok(entry.dir)
174}

176/// The hard error for an include entry that is neither an installed
177/// store name nor a resolvable ref.
178fn missing_base(text: &str) -> Error {
179    Error::Failed(format!(
180        "include '{text}' is not installed — store names must already be in the store; \
181         use an installable ref (path, git:, cf:, http:) or `tamias install {text}` first"
182    ))
183}

185/// Pin the composed payload's manifest to the *local* package's
186/// manifest file, byte-for-byte. [`tamias_core::overlay`] doc-merges
187/// `.yaml`/`.json` files — right for payload docs, wrong for the
188/// manifest itself (a base's `deps:`/`include:` arrays would concat
189/// into the composed manifest). The top package's manifest is
190/// authoritative: `include:` stays for provenance, nothing inherits.
191/// Stray manifest files a base contributed under a different
192/// [`MANIFEST_NAMES`](tamias_core::MANIFEST_NAMES) spelling are
193/// removed so `Manifest::load` sees exactly one.
194///
195/// # Errors
196/// Propagates copy/remove failures.
197pub(crate) fn pin_manifest(local: &Path, dest: &Path) -> Result<(), Error> {
198    let Some(src) = tamias_core::Manifest::path_in(local) else {
199        return Ok(());
200    };
201    let file_name = src
202        .file_name()
203        .map(|n| n.to_string_lossy().into_owned())
204        .unwrap_or_default();
205    for candidate in tamias_core::MANIFEST_NAMES {
206        let target = dest.join(candidate);
207        if candidate == file_name {
208            std::fs::copy(&src, &target).map_err(tamias_core::Error::io(format!(
209                "pin manifest {}",
210                target.display()
211            )))?;
212        } else if target.exists() {
213            std::fs::remove_file(&target).map_err(tamias_core::Error::io(format!(
214                "remove stray manifest {}",
215                target.display()
216            )))?;
217        }
218    }
219    Ok(())
220}
