1//! xtask — repo chores. `cargo xtask <verb>` via the alias in
2//! `.cargo/config.toml`.
3//!
4//! - `check`    — `cargo fmt --check` + `cargo clippy -D warnings`
5//! - `test`     — `cargo nextest run` when installed, else `cargo test`;
6//!   plus the worker smoke test when `bun` is installed
7//! - `worker`   — the worker smoke test: `bun test` in
8//!   crates/cuora-vault-cf/worker — executes the REAL index.js over
9//!   a bun:sqlite D1 shim (the Rust mock can't catch worker bugs)
10//! - `coverage` — `cargo llvm-cov` (HTML → target/llvm-cov/html);
11//!   extra args pass through, e.g. `--fail-under-lines 100`
12//! - `install`  — release-install `cuora` into ~/.cargo/bin
13//! - `site`     — build the docs site (docs/ worktree, VitePress)
14
15use std::process::{Command, ExitStatus};
16
17fn run(args: &[&str]) -> ExitStatus {
18    let empty: &[&str] = &[];
19    let (prog, rest) = args.split_first().unwrap_or((&"true", empty));
20    Command::new(prog).args(rest).status().unwrap_or_else(|e| {
21        eprintln!("xtask: failed to spawn {prog}: {e}");
22        std::process::exit(1)
23    })
24}
25
26/// `run` with a working directory — for commands that must execute
27/// inside a child tree (the docs/ worktree).
28fn run_in(dir: &str, args: &[&str]) -> ExitStatus {
29    let empty: &[&str] = &[];
30    let (prog, rest) = args.split_first().unwrap_or((&"true", empty));
31    Command::new(prog)
32        .args(rest)
33        .current_dir(dir)
34        .status()
35        .unwrap_or_else(|e| {
36            eprintln!("xtask: failed to spawn {prog}: {e}");
37            std::process::exit(1)
38        })
39}

41fn have(tool: &str) -> bool {
42    Command::new(tool)
43        .arg("--version")
44        .stdout(std::process::Stdio::null())
45        .stderr(std::process::Stdio::null())
46        .status()
47        .map(|s| s.success())
48        .unwrap_or(false)
49}
50
51fn have_cargo_sub(sub: &str) -> bool {
52    Command::new("cargo")
53        .args([sub, "--version"])
54        .stdout(std::process::Stdio::null())
55        .stderr(std::process::Stdio::null())
56        .status()
57        .map(|s| s.success())
58        .unwrap_or(false)
59}

61fn check() -> i32 {
62    let ok = run(&["cargo", "fmt", "--all", "--", "--check"]).success()
63        && run(&[
64            "cargo",
65            "clippy",
66            "--workspace",
67            "--all-targets",
68            "--",
69            "-D",
70            "warnings",
71        ])
72        .success();
73    if ok {
74        0
75    } else {
76        1
77    }
78}
79
80fn test() -> i32 {
81    let code = if have_cargo_sub("nextest") {
82        run(&["cargo", "nextest", "run", "--workspace"])
83    } else {
84        run(&["cargo", "test", "--workspace"])
85    }
86    .code()
87    .unwrap_or(1);
88    // The worker smoke test rides the suite when bun is around —
89    // skipped (with a note) in lean envs so the Rust tests still
90    // run everywhere.
91    if have("bun") {
92        let w = run_in("crates/cuora-vault-cf/worker", &["bun", "test"])
93            .code()
94            .unwrap_or(1);
95        return if code == 0 { w } else { code };
96    }
97    eprintln!("xtask: bun not installed — skipping the worker smoke test (cargo xtask worker)");
98    code
99}

101/// `bun test` in the worker dir — the smoke test imports the real
102/// index.js and drives it over a bun:sqlite D1 shim.
103fn worker() -> i32 {
104    if !have("bun") {
105        eprintln!("bun not installed — needed for the worker smoke test");
106        return 1;
107    }
108    run_in("crates/cuora-vault-cf/worker", &["bun", "test"])
109        .code()
110        .unwrap_or(1)
111}
112
113fn coverage(extra: &[String]) -> i32 {
114    if !have_cargo_sub("llvm-cov") {
115        eprintln!("cargo-llvm-cov not installed — cargo install cargo-llvm-cov");
116        return 1;
117    }
118    let mut args = vec!["cargo", "llvm-cov"];
119    if have_cargo_sub("nextest") {
120        args.push("nextest");
121        args.extend(["--workspace", "--no-tests=pass"]);
122    } else {
123        args.push("--workspace");
124    }
125    if extra.is_empty() {
126        args.push("--html");
127    }
128    let owned: Vec<&str> = extra.iter().map(String::as_str).collect();
129    args.extend(owned);
130    run(&args).code().unwrap_or(1)
131}
132
133fn install() -> i32 {
134    run(&["cargo", "install", "--path", "crates/cuora-cli", "--force"])
135        .code()
136        .unwrap_or(1)
137}
138
139fn site() -> i32 {
140    if !std::path::Path::new("docs/package.json").exists() {
141        eprintln!("no docs/ worktree — `git worktree add docs docs` first");
142        return 1;
143    }
144    // run inside docs/ — works for bun and npm alike (`bun --cwd`
145    // isn't a `bun run` flag, so flag plumbing gets fragile)
146    let pkg = if have("bun") { "bun" } else { "npm" };
147    let _ = run_in("docs", &[pkg, "install"]).success();
148    run_in("docs", &[pkg, "run", "docs:build"])
149        .code()
150        .unwrap_or(1)
151}

153fn main() {
154    let mut args = std::env::args().skip(1);
155    let cmd = args.next().unwrap_or_default();
156    let rest: Vec<String> = args.collect();
157    let code = match cmd.as_str() {
158        "check" => check(),
159        "test" => test(),
160        "worker" => worker(),
161        "coverage" => coverage(&rest),
162        "install" => install(),
163        "site" => site(),
164        other => {
165            eprintln!(
166                "usage: cargo xtask <check|test|worker|coverage|install|site>\n  unknown verb: {other:?}"
167            );
168            2
169        }
170    };
171    std::process::exit(code);
172}
