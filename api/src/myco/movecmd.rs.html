1//! `myco item move` — the cross-repo transfer. The item is
2//! re-created in the destination ledger's own id sequence (kind,
3//! title, body, `pri`/`owner`/`#label`/`epic`/`ms` carried;
4//! `blocks:`/`links:` dropped — they name source-ledger ids), both ledgers
5//! get a provenance handoff (`moved from`/`moved to`), then the
6//! source walks to `done`: a file-backed item keeps its file with
7//! status `done`; a task's line leaves the ledger and its handoff
8//! file drops to `handoffs/archive/` — the `check`-prescribed
9//! home for items that are gone. Two commits, one per worktree,
10//! both under the usual `--no-commit`/`$MYCO_NO_COMMIT` rules.
11//!
12//! There's no distributed transaction across the two worktrees —
13//! the destination half lands first (a copy left behind beats an
14//! item deleted on both ends), and a source-side failure reports
15//! the new id so the close can be finished by hand.
16

27/// What a move produced — the CLI report and the wire payload
28/// share it.
29pub(crate) struct Moved {
30    /// The source id (closed `done` / line removed).
31    pub from: ItemId,
32    /// The destination copy — fresh id in the dest sequence.
33    pub item: Item,
34    /// Destination repo label (`meta.toml` name, else the root
35    /// basename).
36    pub repo: String,
37    /// The destination repo root.
38    pub root: PathBuf,
39    /// The source-side handoff entry — the audit pointer.
40    pub handoff: HandoffEntry,
41    /// Source-worktree autocommit result.
42    pub committed_source: bool,
43    /// Destination-worktree autocommit result.
44    pub committed_dest: bool,
45}

47/// Canonical form for comparisons — never fails, like discovery's.
48fn canon(p: &Path) -> PathBuf {
49    p.canonicalize().unwrap_or_else(|_| p.to_path_buf())
50}
51
52/// Same ledger on both ends — the same worktree path, or two
53/// worktrees sharing one git common dir (linked worktrees of a
54/// single repo). Plain-dir ledgers (no git) compare by path only.
55fn same_repo(a: &Path, b: &Path) -> bool {
56    if canon(a) == canon(b) {
57        return true;
58    }
59    match (myco_git::common_dir(a), myco_git::common_dir(b)) {
60        (Ok(x), Ok(y)) => x == y,
61        _ => false,
62    }
63}

65/// `--to` → the destination project. A path resolves like
66/// `index add`'s arg (repo root *or* its `project/` worktree); a
67/// non-path falls back to the registry's repo *names* (`myco
68/// index list`).
69fn resolve_dest(to: &str) -> Result<discover::Project, String> {
70    if let Ok(p) = indexcmd::repo_at(Path::new(to)) {
71        return Ok(p);
72    }
73    if let Ok(index) = indexcmd::open_index() {
74        if let Ok(repos) = index.repos() {
75            if let Some(r) = repos.iter().find(|r| r.name == to) {
76                return indexcmd::repo_at(r.path.as_std_path());
77            }
78        }
79    }
80    Err(format!(
81        "{to}: not a myco repo — pass a repo root, its project/ worktree, \
82         or a repo name from `myco index list`"
83    ))
84}

86/// The `repo` half of the `repo:ID` pointer — `meta.toml`'s name,
87/// else the root basename.
88fn repo_label(store: &impl Store, root: &Path) -> String {
89    store
90        .meta()
91        .map(|m| m.name)
92        .ok()
93        .filter(|n| !n.is_empty())
94        .or_else(|| root.file_name().map(|n| n.to_string_lossy().into_owned()))
95        .unwrap_or_else(|| "repo".into())
96}

98/// The metadata that survives the crossing. `blocks:`/`links:` are
99/// dropped on purpose — they name source-ledger ids and would
100/// dangle in the destination (`check` would flag them).
101fn carried_meta(src: &Item) -> NewItemMeta {
102    NewItemMeta {
103        priority: Some(src.priority),
104        owner: src.owner.clone(),
105        labels: src.labels.iter().cloned().collect(),
106        blocks: Vec::new(),
107        links: Vec::new(),
108        epic: src.epic.clone(),
109        milestone: src.milestone.clone(),
110        due: src.due,
111        estimate: src.estimate,
112        cycle: src.cycle.clone(),
113    }
114}

116/// Re-create `src` in `dest` — a task lands at TODO.md's tail via
117/// the `todo add` path; file kinds go through `save_item` with the
118/// source body verbatim. The id is the destination sequence's
119/// next; `created` carries over (a moved item keeps its filing
120/// date). Caller commits.
121fn create_in(dest: &impl Store, src: &Item, body: Option<&str>) -> Result<Item, String> {
122    let meta = carried_meta(src);
123    if src.id.kind() == ItemKind::Task {
124        let (item, _file) = todo::add_item(dest, &src.title, None, None, &meta)?;
125        return Ok(item);
126    }
127    let id = dest.alloc_id(src.id.kind()).map_err(|e| e.to_string())?;
128    let mut item = Item::new(id, &src.title).created(src.created.unwrap_or_else(items::today));
129    meta.apply(&mut item)?;
130    let file = ItemFile::new(item.clone()).body(body.unwrap_or_default().to_string());
131    dest.save_item(&file).map_err(|e| e.to_string())?;
132    Ok(item)
133}

135/// The source close after the audit handoff lands. A task's line
136/// is taken out of its ledger doc and its handoff file moves to
137/// `handoffs/archive/` (where `check` wants the files of gone
138/// items); a file-backed item walks the state machine to `done`
139/// (an already-done item just keeps its file).
140fn close_source(store: &impl Store, worktree: &Path, id: ItemId) -> Result<(), String> {
141    if id.kind() == ItemKind::Task {
142        let mut ledger = items::load_ledger(store)?;
143        let loc = ledger
144            .locate(id)
145            .ok_or_else(|| format!("{id}: no such task"))?;
146        ledger.take(&loc);
147        items::save_ledger(store, &ledger)?;
148        // the item is gone — its handoff goes cold, exactly the
149        // rename `check --fix` prescribes for orphan files. Both
150        // files existing is an already-inconsistent ledger: don't
151        // clobber the cold one, leave the live file for `check`
152        // to flag.
153        let name = myco_store_md::handoff_filename(&id);
154        let live = worktree.join("handoffs").join(&name);
155        let archive = worktree.join("handoffs").join("archive").join(&name);
156        if live.is_file() && !archive.exists() {
157            std::fs::create_dir_all(archive.parent().unwrap_or(worktree))
158                .map_err(|e| format!("handoffs/archive: {e}"))?;
159            std::fs::rename(&live, &archive).map_err(|e| format!("archive {name}: {e}"))?;
160        }
161        return Ok(());
162    }
163    let mut file = items::item_for_edit(store, id)?;
164    if file.meta.status != Status::Done {
165        items::walk_to(&mut file.meta, Status::Done)?;
166        store.save_item(&file).map_err(|e| e.to_string())?;
167    }
168    Ok(())
169}

171/// The move itself — shared by the CLI handler and `item.move` on
172/// the wire. `dir`/`root` are the global discovery flags (the
173/// source side); `to` resolves the destination.
174pub(crate) fn do_move(
175    dir: Option<&Path>,
176    root: Option<&Path>,
177    id: ItemId,
178    to: &str,
179    note: Option<&str>,
180    actor: Option<Actor>,
181    no_commit: bool,
182) -> Result<Moved, String> {
183    let (src_proj, src) = items::open(dir, root)?;
184    let (item, _location, body) =
185        items::find_item(&src, id)?.ok_or_else(|| format!("{id}: no such item"))?;
186    if src.is_archived(id).map_err(|e| e.to_string())? {
187        return Err(format!("{id} is archived — unarchive it before moving"));
188    }
189    if !item.children.is_empty() {
190        return Err(format!(
191            "{id} has {} child item(s) — move them individually first \
192             (`done`/`defer` acts on the whole subtree)",
193            item.children.len()
194        ));
195    }
196    let dest_proj = resolve_dest(to)?;
197    if same_repo(&src_proj.worktree, &dest_proj.worktree) {
198        return Err(format!(
199            "{to}: the destination is this repo — move is cross-repo only"
200        ));
201    }
202    let (dest_proj, dst) = items::open(Some(&dest_proj.worktree), Some(&dest_proj.root))?;
203    let src_name = repo_label(&src, &src_proj.root);
204    let dest_name = repo_label(&dst, &dest_proj.root);
205
206    // destination first: create + provenance + its own commit
207    let new_item =
208        create_in(&dst, &item, body.as_deref()).map_err(|e| format!("{dest_name}: {e}"))?;
209    continuity::record_handoff(
210        &dst,
211        &dest_proj.worktree,
212        new_item.id,
213        format!("moved from {src_name}:{id}"),
214        actor.clone(),
215        None,
216    )
217    .map_err(|e| format!("{dest_name}: {e}"))?;
218    let committed_dest = items::commit(
219        &dest_proj.worktree,
220        &format!("chore: move {} in from {src_name}:{id}", new_item.id),
221        actor.as_ref(),
222        no_commit,
223    )?;
224
225    // then the source: audit pointer, close, its own commit — a
226    // failure here names the created id so the close can be
227    // finished by hand
228    let mut note_text = format!("moved to {dest_name}:{}", new_item.id);
229    if let Some(extra) = note.map(str::trim).filter(|n| !n.is_empty()) {
230        note_text.push_str(&format!(" — {extra}"));
231    }
232    let left = |e: String| {
233        format!(
234            "{e} — destination {dest_name}:{} was created; \
235             close {id} here by hand",
236            new_item.id
237        )
238    };
239    let (handoff, _rel) =
240        continuity::record_handoff(&src, &src_proj.worktree, id, note_text, actor.clone(), None)
241            .map_err(left)?;
242    close_source(&src, &src_proj.worktree, id).map_err(left)?;
243    let committed_source = items::commit(
244        &src_proj.worktree,
245        &format!("chore: move {id} to {dest_name}:{}", new_item.id),
246        actor.as_ref(),
247        no_commit,
248    )?;
249
250    Ok(Moved {
251        from: id,
252        item: new_item,
253        repo: dest_name,
254        root: dest_proj.root.clone(),
255        handoff,
256        committed_source,
257        committed_dest,
258    })
259}

261/// The `--json` payload — the destination item plus both sides'
262/// pointers and commit flags (two worktrees, two autocommits).
263/// `committed` itself stays the scalar [`items::report`] merges.
264pub(crate) fn moved_json(m: &Moved) -> serde_json::Value {
265    json!({
266        "id": m.from.to_string(),
267        "item": m.item,
268        "to": {"repo": m.repo, "root": m.root, "id": m.item.id.to_string()},
269        "handoff": m.handoff,
270        "committed_source": m.committed_source,
271        "committed_destination": m.committed_dest,
272    })
273}

275/// The `item move` verb.
276pub fn h_move(ctx: &Ctx, a: &Args) -> Result<i32, String> {
277    let id = a.req::<ItemId>("id")?;
278    let to = a.text("to")?;
279    if crate::confirm::needed(a.flag("yes"))
280        && !crate::confirm::ask(&format!("move {id} → {to}? the source closes."))
281    {
282        eprintln!("myco: cancelled");
283        return Ok(1);
284    }
285    let m = do_move(
286        ctx.dir.as_deref(),
287        ctx.root.as_deref(),
288        id,
289        to,
290        a.opt("note"),
291        a.parsed("actor")?,
292        ctx.no_commit,
293    )?;
294    items::report(
295        moved_json(&m),
296        m.committed_source && m.committed_dest,
297        ctx.json,
298        &format!("moved {} → {}:{}", m.from, m.repo, m.item.id),
299    )
300}
