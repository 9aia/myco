1//! `myco next` — the single highest-priority open item:
2//! "what should I work on" (I-020, falls out of P-006). It's a
3//! view eval with `limit = 1` — the filter flags are `view`'s
4//! [`FilterFlags`] merged over an open tasks+issues default.
5//! `--claim` is the only new surface: it mutates the emitted
6//! item to `claimed` + an owner in one per-verb commit.
7

17/// The `next` flag bundle — `view`'s filter flags plus `--sort`
18/// (the engine's `pri,id` when unset) and `--claim`. No
19/// `--limit`/`--group-by`/`--format`: the verb *is* limit 1
20/// rendered as one compact line.
21#[derive(Debug, Default, Args)]
22pub struct NextFlags {
23    #[command(flatten)]
24    pub(crate) filter: view::FilterFlags,
25    /// Sort keys — `pri,id,-created`; `-` flips a key's default
26    /// direction. Default `pri,id`.
27    #[arg(long, value_delimiter = ',')]
28    sort: Vec<SortKey>,
29    /// Claim the emitted item — status `claimed`, owner set to
30    /// the `--owner` actor (else `cli`); `--owner` names the
31    /// claimer here and doesn't filter. Mutating: one commit.
32    #[arg(long)]
33    pub(crate) claim: bool,
34}
35
36/// The default `next` spec — the builtin `next` view (open
37/// tasks+issues, top item only), before config shadows/flags.
38#[cfg(test)]
39pub(crate) fn base_spec() -> ViewSpec {
40    myco_ops::view::builtins()
41        .into_iter()
42        .find(|v| v.name == "next")
43        .map(|v| v.spec)
44        .unwrap_or_default()
45}
46
47/// Resolve the `next` named view — a repo's `[views.next]`
48/// shadows the builtin like any other name — then merge flags,
49/// validated. Under `--claim` the `--owner` flag names the
50/// claimer, so it must not also restrict the pick to items that
51/// owner already holds.
52pub(crate) fn spec(root: &std::path::Path, flags: &NextFlags) -> Result<ViewSpec, String> {
53    let (view, warnings) = myco_ops::view::resolve_lenient(root, "next", ops::CONFIG)?;
54    view::warn(&warnings);
55    let mut spec = view.spec;
56    flags.filter.apply(&mut spec.filter);
57    if flags.claim {
58        spec.filter.owner = Vec::new();
59    }
60    if !flags.sort.is_empty() {
61        spec.sort = flags.sort.clone();
62    }
63    spec.validate()?;
64    Ok(spec)
65}

67/// The highest-priority live item under `spec` — the pick
68/// behind the verb and `serve`'s `next` method. The universe is
69/// top-level items only — children ride with their parent (the
70/// `myco view` rule).
71pub(crate) fn pick<'a>(scan: &'a digest::Scan, spec: &ViewSpec) -> Option<&'a Item> {
72    let universe = scan.items.iter().filter(|i| !scan.archived.contains(&i.id));
73    eval(universe, spec)
74        .iter()
75        .find_map(|g| g.items.first().map(|s| s.item))
76}

78/// Who a `--claim` assigns — `--owner`'s single actor (under
79/// `--claim` it names the claimer, not a filter), else the
80/// `handoff`/`interrupt` fallback (`cli`). Several `--owner`s
81/// can't name one claimer, so that's an error.
82fn claim_owner(flags: &NextFlags) -> Result<Actor, String> {
83    match flags.filter.owner.as_slice() {
84        [] => Actor::new(continuity::DEFAULT_ACTOR).map_err(|e| e.to_string()),
85        [owner] => Ok(owner.clone()),
86        _ => Err("next --claim: pass one --owner (the claimer), not several".into()),
87    }
88}

90/// The one-line human output — `items::compact` + the `⚡`
91/// interrupted marker, same as a `list`-format view row.
92fn line(store: &impl Store, item: &Item) -> Result<String, String> {
93    let mut out = items::compact(item);
94    if continuity::is_interrupted(store, item.id)? {
95        out.push_str(" ⚡");
96    }
97    Ok(out)
98}

100/// The `next` verb — read-only unless `--claim`, which takes
101/// the mutating open path + per-verb commit (`chore: claim
102/// <ID>`; `--no-commit`/`$MYCO_NO_COMMIT` honored).
103pub fn run(
104    flags: NextFlags,
105    dir: Option<PathBuf>,
106    root: Option<PathBuf>,
107    json_out: bool,
108    no_commit: bool,
109) -> Result<i32, String> {
110    let owner = flags.claim.then(|| claim_owner(&flags)).transpose()?;
111    let (proj, store) = if flags.claim {
112        items::open(dir.as_deref(), root.as_deref())?
113    } else {
114        items::open_read(dir.as_deref(), root.as_deref())?
115    };
116    let spec = spec(&proj.root, &flags)?;
117    let scan = digest::scan(&store)?;
118    let Some(item) = pick(&scan, &spec).cloned() else {
119        if json_out {
120            ops::print_json(&json!({"item": null}));
121        } else {
122            println!("{}: nothing next — backlog is clear", scan.name);
123        }
124        return Ok(0);
125    };
126    let Some(owner) = owner else {
127        if json_out {
128            ops::print_json(&json!({"item": item}));
129        } else {
130            println!("{}", line(&store, &item)?);
131        }
132        return Ok(0);
133    };
134    let id = item.id;
135    let set = itemcmd::SetFlags {
136        // identity isn't a transition — an already-claimed item
137        // just re-stamps the owner
138        status: (item.status != Status::Claimed).then_some(Status::Claimed),
139        owner: Some(owner),
140        ..Default::default()
141    };
142    let (item, _location) = itemcmd::mutate_item(&store, id, &set)?;
143    let committed = items::commit(&proj.worktree, &format!("chore: claim {id}"), no_commit)?;
144    let human = match &item {
145        Some(i) => line(&store, i)?,
146        None => format!("claimed {id}"),
147    };
148    items::report(json!({"item": item}), committed, json_out, &human)
149}

151#[cfg(test)]
152#[allow(clippy::unwrap_used)]
153mod tests {
154    use super::*;
155    use myco_core::{ItemId, ItemKind};
156
157    /// A hermetic root — no myco.toml, no user config leaks.
158    fn spec_at(dir: &tempfile::TempDir, flags: &NextFlags) -> ViewSpec {
159        temp_env::with_var("MYCO_CONFIG", None::<&str>, || {
160            spec(dir.path(), flags).unwrap()
161        })
162    }
163
164    #[test]
165    fn default_spec_is_open_tasks_and_issues_limit_1() {
166        let dir = tempfile::tempdir().unwrap();
167        let spec = spec_at(&dir, &NextFlags::default());
168        assert_eq!(spec.limit, Some(1));
169        assert_eq!(spec.filter.kind, [ItemKind::Task, ItemKind::Issue]);
170        assert_eq!(spec.filter.status, [Status::Open]);
171        assert!(spec.sort.is_empty()); // engine default: pri,id
172    }
173
174    #[test]
175    fn flags_merge_over_the_default() {
176        let flags = NextFlags {
177            filter: view::FilterFlags {
178                kind: vec![ItemKind::Issue],
179                label: vec!["bug".into()],
180                ..Default::default()
181            },
182            sort: vec!["-id".parse().unwrap()],
183            claim: false,
184        };
185        let dir = tempfile::tempdir().unwrap();
186        let spec = spec_at(&dir, &flags);
187        assert_eq!(spec.filter.kind, [ItemKind::Issue]);
188        // unset fields keep the base: status stays [open]
189        assert_eq!(spec.filter.status, [Status::Open]);
190        assert_eq!(spec.filter.label, ["bug"]);
191        assert_eq!(spec.sort.len(), 1);
192    }
193
194    #[test]
195    fn claim_owner_resolution() {
196        let mut flags = NextFlags::default();
197        assert_eq!(claim_owner(&flags).unwrap().as_str(), "cli");
198        flags.filter.owner = vec!["devin".parse().unwrap()];
199        assert_eq!(claim_owner(&flags).unwrap().as_str(), "devin");
200        flags.filter.owner = vec!["devin".parse().unwrap(), "human:x".parse().unwrap()];
201        assert!(claim_owner(&flags).is_err());
202    }
203
204    #[test]
205    fn pick_orders_pri_then_id() {
206        let items = vec![
207            Item::new(ItemId::new(ItemKind::Task, 1), "low").priority(myco_core::Priority::Low),
208            Item::new(ItemId::new(ItemKind::Task, 2), "crit")
209                .priority(myco_core::Priority::Critical),
210            Item::new(ItemId::new(ItemKind::Issue, 1), "hi").priority(myco_core::Priority::High),
211        ];
212        let scan = digest::Scan {
213            name: "p".into(),
214            items,
215            archived: Default::default(),
216        };
217        let spec = base_spec();
218        assert_eq!(
219            pick(&scan, &spec).unwrap().id,
220            ItemId::new(ItemKind::Task, 2)
221        );
222        // a done crit drops out of the open default
223        let mut scan2 = digest::Scan {
224            name: "p".into(),
225            items: scan.items.clone(),
226            archived: Default::default(),
227        };
228        scan2.items[1].status = Status::Done;
229        assert_eq!(
230            pick(&scan2, &spec).unwrap().id,
231            ItemId::new(ItemKind::Issue, 1)
232        );
233    }
