1//! `myco next` — the single highest-priority open item:
2//! "what should I work on" (I-020, falls out of P-006). It's a
3//! view eval with `limit = 1` — the filter flags are `view`'s
4//! [`FilterFlags`] merged over an open tasks+issues default.
5//! `--claim` is the only new surface: it mutates the emitted
6//! item to `claimed` + an owner in one per-verb commit.
7

10use clap::Args;
11use myco_core::{eval, Actor, Item, ItemKind, SortKey, Status, ViewFilter, ViewSpec};
12use myco_store::Store;
13use serde_json::json;

15use crate::{continuity, digest, itemcmd, items, ops, view};

17/// The `next` flag bundle — `view`'s filter flags plus `--sort`
18/// (the engine's `pri,id` when unset) and `--claim`. No
19/// `--limit`/`--group-by`/`--format`: the verb *is* limit 1
20/// rendered as one compact line.
21#[derive(Debug, Default, Args)]
22pub struct NextFlags {
23    #[command(flatten)]
24    pub(crate) filter: view::FilterFlags,
25    /// Sort keys — `pri,id,-created`; `-` flips a key's default
26    /// direction. Default `pri,id`.
27    #[arg(long, value_delimiter = ',')]
28    sort: Vec<SortKey>,
29    /// Claim the emitted item — status `claimed`, owner set to
30    /// the `--owner` actor (else `cli`); `--owner` names the
31    /// claimer here and doesn't filter. Mutating: one commit.
32    #[arg(long)]
33    pub(crate) claim: bool,
34}
35
36/// The default `next` spec — open tasks+issues, top item only.
37pub(crate) fn base_spec() -> ViewSpec {
38    ViewSpec {
39        filter: ViewFilter {
40            kind: vec![ItemKind::Task, ItemKind::Issue],
41            status: vec![Status::Open],
42            ..ViewFilter::default()
43        },
44        limit: Some(1),
45        ..ViewSpec::default()
46    }
47}
48
49/// Flags merged over [`base_spec`], validated — the same merge
50/// `myco view` performs over a named spec. Under `--claim` the
51/// `--owner` flag names the claimer, so it must not also
52/// restrict the pick to items that owner already holds.
53pub(crate) fn spec(flags: &NextFlags) -> Result<ViewSpec, String> {
54    let mut spec = base_spec();
55    flags.filter.apply(&mut spec.filter);
56    if flags.claim {
57        spec.filter.owner = Vec::new();
58    }
59    if !flags.sort.is_empty() {
60        spec.sort = flags.sort.clone();
61    }
62    spec.validate()?;
63    Ok(spec)
64}
65
66/// The highest-priority live item under `spec` — the pick
67/// behind the verb and `serve`'s `next` method. The universe is
68/// top-level items only — children ride with their parent (the
69/// `myco view` rule).
70pub(crate) fn pick<'a>(scan: &'a digest::Scan, spec: &ViewSpec) -> Option<&'a Item> {
71    let universe = scan.items.iter().filter(|i| !scan.archived.contains(&i.id));
72    eval(universe, spec)
73        .iter()
74        .find_map(|g| g.items.first())
75        .copied()
76}
77
78/// Who a `--claim` assigns — `--owner`'s single actor (under
79/// `--claim` it names the claimer, not a filter), else the
80/// `handoff`/`interrupt` fallback (`cli`). Several `--owner`s
81/// can't name one claimer, so that's an error.
82fn claim_owner(flags: &NextFlags) -> Result<Actor, String> {
83    match flags.filter.owner.as_slice() {
84        [] => Actor::new(continuity::DEFAULT_ACTOR).map_err(|e| e.to_string()),
85        [owner] => Ok(owner.clone()),
86        _ => Err("next --claim: pass one --owner (the claimer), not several".into()),
87    }
88}

90/// The one-line human output — `items::compact` + the `⚡`
91/// interrupted marker, same as a `list`-format view row.
92fn line(store: &impl Store, item: &Item) -> Result<String, String> {
93    let mut out = items::compact(item);
94    if continuity::is_interrupted(store, item.id)? {
95        out.push_str(" ⚡");
96    }
97    Ok(out)
98}

100/// The `next` verb — read-only unless `--claim`, which takes
101/// the mutating open path + per-verb commit (`chore: claim
102/// <ID>`; `--no-commit`/`$MYCO_NO_COMMIT` honored).
103pub fn run(
104    flags: NextFlags,
105    dir: Option<PathBuf>,
106    root: Option<PathBuf>,
107    json_out: bool,
108    no_commit: bool,
109) -> Result<i32, String> {
110    let spec = spec(&flags)?;
111    let owner = flags.claim.then(|| claim_owner(&flags)).transpose()?;
112    let (proj, store) = if flags.claim {
113        items::open(dir.as_deref(), root.as_deref())?
114    } else {
115        items::open_read(dir.as_deref(), root.as_deref())?
116    };
117    let scan = digest::scan(&store)?;
118    let Some(item) = pick(&scan, &spec).cloned() else {
119        if json_out {
120            ops::print_json(&json!({"item": null}));
121        } else {
122            println!("{}: nothing next — backlog is clear", scan.name);
123        }
124        return Ok(0);
125    };
126    let Some(owner) = owner else {
127        if json_out {
128            ops::print_json(&json!({"item": item}));
129        } else {
130            println!("{}", line(&store, &item)?);
131        }
132        return Ok(0);
133    };
134    let id = item.id;
135    let set = itemcmd::SetFlags {
136        // identity isn't a transition — an already-claimed item
137        // just re-stamps the owner
138        status: (item.status != Status::Claimed).then_some(Status::Claimed),
139        owner: Some(owner),
140        ..Default::default()
141    };
142    let (item, _location) = itemcmd::mutate_item(&store, id, &set)?;
143    let committed = items::commit(&proj.worktree, &format!("chore: claim {id}"), no_commit)?;
144    let human = match &item {
145        Some(i) => line(&store, i)?,
146        None => format!("claimed {id}"),
147    };
148    items::report(json!({"item": item}), committed, json_out, &human)
149}

151#[cfg(test)]
152#[allow(clippy::unwrap_used)]
153mod tests {
154    use super::*;
155    use myco_core::ItemId;
156
157    #[test]
158    fn default_spec_is_open_tasks_and_issues_limit_1() {
159        let spec = spec(&NextFlags::default()).unwrap();
160        assert_eq!(spec.limit, Some(1));
161        assert_eq!(spec.filter.kind, [ItemKind::Task, ItemKind::Issue]);
162        assert_eq!(spec.filter.status, [Status::Open]);
163        assert!(spec.sort.is_empty()); // engine default: pri,id
164    }
165
166    #[test]
167    fn flags_merge_over_the_default() {
168        let flags = NextFlags {
169            filter: view::FilterFlags {
170                kind: vec![ItemKind::Issue],
171                label: vec!["bug".into()],
172                ..Default::default()
173            },
174            sort: vec!["-id".parse().unwrap()],
175            claim: false,
176        };
177        let spec = spec(&flags).unwrap();
178        assert_eq!(spec.filter.kind, [ItemKind::Issue]);
179        // unset fields keep the base: status stays [open]
180        assert_eq!(spec.filter.status, [Status::Open]);
181        assert_eq!(spec.filter.label, ["bug"]);
182        assert_eq!(spec.sort.len(), 1);
183    }
184
185    #[test]
186    fn claim_owner_resolution() {
187        let mut flags = NextFlags::default();
188        assert_eq!(claim_owner(&flags).unwrap().as_str(), "cli");
189        flags.filter.owner = vec!["devin".parse().unwrap()];
190        assert_eq!(claim_owner(&flags).unwrap().as_str(), "devin");
191        flags.filter.owner = vec!["devin".parse().unwrap(), "human:x".parse().unwrap()];
192        assert!(claim_owner(&flags).is_err());
193    }
194
195    #[test]
196    fn pick_orders_pri_then_id() {
197        let items = vec![
198            Item::new(ItemId::new(ItemKind::Task, 1), "low").priority(myco_core::Priority::Low),
199            Item::new(ItemId::new(ItemKind::Task, 2), "crit")
200                .priority(myco_core::Priority::Critical),
201            Item::new(ItemId::new(ItemKind::Issue, 1), "hi").priority(myco_core::Priority::High),
202        ];
203        let scan = digest::Scan {
204            name: "p".into(),
205            items,
206            archived: Default::default(),
207        };
208        let spec = base_spec();
209        assert_eq!(
210            pick(&scan, &spec).unwrap().id,
211            ItemId::new(ItemKind::Task, 2)
212        );
213        // a done crit drops out of the open default
214        let mut scan2 = digest::Scan {
215            name: "p".into(),
216            items: scan.items.clone(),
217            archived: Default::default(),
218        };
219        scan2.items[1].status = Status::Done;
220        assert_eq!(
221            pick(&scan2, &spec).unwrap().id,
222            ItemId::new(ItemKind::Issue, 1)
223        );
224    }
