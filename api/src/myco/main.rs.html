1//! myco — project management, grown not built. A mycorrhizal
2//! network for your agents: tasks, signals, handoffs.
3//!
4//! Agent-first PM: markdown in `project/` is canonical and
5//! human-editable; every deterministic write goes through this CLI
6//! (auto-commit per verb, `--json` everywhere). The skill teaches
7//! agents the verbs; `myco manifest`/`myco schema` are the
8//! self-describing surface.
9//!
10//! Phase 6 wires continuity — `handoff`/`resume` over append-only
11//! `handoffs/<ID>.md` files (one per item, any kind); sepia hooks
12//! call them at session boundaries. Phase 7 lands the Jira-parity
13//! slice — `epic:`/`ms:` item metadata + grouping views,
14//! `absorb` importers (comment sweep, Jira CSV, GitHub dumps).

48/// myco — agent-first project management for a repo's `project/`
49/// worktree.
50#[derive(Parser)]
51#[command(name = "myco", version, about, long_about = None)]
52struct Cli {
53    /// Emit JSON (accepted everywhere; default output for machines).
54    #[arg(long, global = true)]
55    json: bool,
56
57    /// Skip the per-verb commit (also suppressed by $MYCO_NO_COMMIT).
58    #[arg(long, global = true)]
59    no_commit: bool,
60
61    /// Project worktree to act on, overriding discovery.
62    #[arg(long, global = true, value_name = "PATH")]
63    dir: Option<PathBuf>,
64
65    /// Repo root to act on, overriding discovery (the worktree is
66    /// `<root>/project`).
67    #[arg(long, global = true, value_name = "PATH")]
68    root: Option<PathBuf>,
69
70    #[command(subcommand)]
71    command: Cmd,
72}
73
74#[derive(Subcommand)]
75enum Cmd {
76    /// Capabilities + protocol version — agent self-discovery.
77    Manifest,
78    /// The file/state spec, machine-readable JSON Schema.
79    Schema,
80    /// Scaffold `project/`: orphan branch + worktree + .gitignore +
81    /// seed files + the managed AGENTS.md agent-awareness block.
82    /// Idempotent — existing worktree and AGENTS.md prose are
83    /// preserved, only the marked block is (re)written.
84    Init,
85    /// Validate schemas + state machine, repair file/index drift;
86    /// also validates the managed AGENTS.md block on the repo
87    /// root. Exit 1 with findings; `--fix` applies safe repairs +
88    /// commits.
89    Check {
90        /// Repair drift in place instead of only reporting.
91        #[arg(long)]
92        fix: bool,
93    },
94    /// Convert legacy ledgers to v2: `pm/` → `project/`, or
95    /// normalize files already on the `project` branch.
96    Migrate,
97    /// Stage + commit worktree changes (conventional message).
98    Commit {
99        /// Commit message (default: `chore: update project ledger`).
100        #[arg(short, long)]
101        message: Option<String>,
102    },
103    /// TODO.md tasks — add / done / defer / list.
104    Todo {
105        #[command(subcommand)]
106        cmd: todo::Cmd,
107    },
108    /// BACKLOG.md parked items — add / promote / list.
109    Backlog {
110        #[command(subcommand)]
111        cmd: backlog::Cmd,
112    },
113    /// Tracked issues, one file each under `issues/`.
114    Issue {
115        #[command(subcommand)]
116        cmd: issue::Cmd,
117    },
118    /// Design proposals under `proposals/` (`P-NNN`).
119    Proposal {
120        #[command(subcommand)]
121        cmd: proposal::Cmd,
122    },
123    /// Execution plans under `plans/` (`PL-NNN`).
124    Plan {
125        #[command(subcommand)]
126        cmd: plan::Cmd,
127    },
128    /// Architecture decision records under `adrs/` (`ADR-NNNN`).
129    Adr {
130        #[command(subcommand)]
131        cmd: adr::Cmd,
132    },
133    /// Kind-agnostic item verbs — show / set.
134    Item {
135        #[command(subcommand)]
136        cmd: itemcmd::Cmd,
137    },
138    /// Attach resumable state at session end — appends a dated
139    /// entry to `handoffs/<ID>.md` (any item kind, live or
140    /// archived; append-only).
141    Handoff {
142        /// Any item id (`T-NNN`, `I-NNN`, `P-NNN`, `ADR-NNNN`,
143        /// `PL-NNN`).
144        id: myco_core::ItemId,
145        /// Handoff note — done/next/open. `-` reads stdin.
146        #[arg(long, value_name = "TEXT|-")]
147        note: String,
148        /// Who wrote it (`devin`, `human:luis`, …; default `cli`).
149        #[arg(long)]
150        actor: Option<myco_core::Actor>,
151        /// Session provenance — a sepia session id, `cli:<pid>`,
152        /// any id string.
153        #[arg(long)]
154        session: Option<String>,
155    },
156    /// A session ended with work still claimed/in-progress/
157    /// blocked — revert the item to `open` + `interrupted`
158    /// handoff, one commit. Session-lifecycle hooks (sepia, CLI
159    /// session-end) call this; myco never detects sessions itself.
160    Interrupt {
161        /// Any item id (`T-NNN`, `I-NNN`, `P-NNN`, `ADR-NNNN`,
162        /// `PL-NNN`).
163        id: myco_core::ItemId,
164        /// Extra context appended to the `interrupted` note.
165        #[arg(long, value_name = "TEXT")]
166        note: Option<String>,
167        /// Who interrupted it (default `cli`).
168        #[arg(long)]
169        actor: Option<myco_core::Actor>,
170        /// Session provenance — a sepia session id, `cli:<pid>`,
171        /// any id string.
172        #[arg(long)]
173        session: Option<String>,
174    },
175    /// Minimal context block to restart work on an item — the
176    /// item, `blocks:` dep titles, latest handoffs, open
177    /// checklist. The single-item `status --prompt`.
178    Resume {
179        /// Any item id.
180        id: myco_core::ItemId,
181        /// Only the latest N handoff entries (default 10).
182        #[arg(long)]
183        last: Option<usize>,
184    },
185    /// Cheap digest of the work ledger. `--prompt` = token-sized.
186    Status {
187        /// Render for an agent prompt (minimal tokens).
188        #[arg(long)]
189        prompt: bool,
190    },
191    /// Kanban of the ledger — Backlog | Open | Claimed | In
192    /// Progress | Blocked. `--md` emits a markdown table.
193    Board {
194        /// Emit a markdown table (for docs/READMEs).
195        #[arg(long)]
196        md: bool,
197        /// Show every card — no row cap.
198        #[arg(long)]
199        all: bool,
200        /// Only items in this epic.
201        #[arg(long)]
202        epic: Option<String>,
203        /// Only items in this milestone.
204        #[arg(long)]
205        milestone: Option<String>,
206    },
207    /// Items grouped by epic — bare lists epics with counts, a
208    /// name renders that epic's board (`epic v2` ≡ `board
209    /// --epic v2`).
210    Epic {
211        /// Epic name — omit for the all-epics summary.
212        name: Option<String>,
213        /// Emit a markdown table (with a name).
214        #[arg(long)]
215        md: bool,
216        /// Show every card — no row cap (with a name).
217        #[arg(long)]
218        all: bool,
219    },
220    /// Items grouped by milestone — bare lists milestones with
221    /// counts, a name renders that milestone's board.
222    Milestone {
223        /// Milestone name — omit for the all-milestones summary.
224        name: Option<String>,
225        /// Emit a markdown table (with a name).
226        #[arg(long)]
227        md: bool,
228        /// Show every card — no row cap (with a name).
229        #[arg(long)]
230        all: bool,
231    },
232    /// Import work into the ledger — TODO/FIXME comments, Jira
233    /// CSV exports, GitHub issue dumps.
234    Absorb {
235        #[command(subcommand)]
236        cmd: absorb::Cmd,
237    },
238    /// Aging + velocity + blocked — how the ledger is trending.
239    Report,
240    /// The derived index: register repos, query/rebuild snapshots.
241    Index {
242        #[command(subcommand)]
243        cmd: indexcmd::Cmd,
244    },
245    /// Cross-repo views over the index (auto-rebuilds stale repos).
246    Org {
247        #[command(subcommand)]
248        cmd: org::Cmd,
249    },
250    /// Interactive board TUI (ratatui) — TODO/BACKLOG tabs, item
251    /// preview, writes through the same ops the verbs use.
252    #[cfg(feature = "tui")]
253    Ui,
254    /// Sequential request/response verb surface on stdio —
255    /// JSON-RPC 2.0, one line in / one line out. For embedding
256    /// (peregrin spawns one per repo); stdin IS the wire, so
257    /// no served method reads content from it.
258    Serve,
259    /// Print the CLI reference as markdown (docs pipeline).
260    #[command(hide = true, name = "gen-cli-docs")]
261    GenCliDocs,
262}

311fn main() {
312    // Restore conventional-Unix SIGPIPE behavior (die silently on
313    // EPIPE) — `println!` would otherwise panic when piped to a
314    // reader that closes early (e.g. `myco board | head`).
315    sigpipe::reset();
316    let cli = Cli::parse();
317    let json = cli.json;
318    let no_commit = cli.no_commit;
319    let dir = cli.dir.clone();
320    let root = cli.root.clone();
321    let result: Result<i32, String> = match cli.command {
322        Cmd::Manifest => {
323            ops::print_json(&manifest());
324            Ok(0)
325        }
326        Cmd::GenCliDocs => {
327            println!("{}", clap_markdown::help_markdown::<Cli>());
328            Ok(0)
329        }
330        Cmd::Init => init::run(dir, root, json, no_commit),
331        Cmd::Check { fix } => check::run(dir, root, fix, json, no_commit),
332        Cmd::Migrate => migrate::run(dir, root, json, no_commit),
333        Cmd::Commit { message } => commit::run(dir, root, message, json, no_commit),
334        Cmd::Todo { cmd } => todo::run(cmd, dir, root, json, no_commit),
335        Cmd::Backlog { cmd } => backlog::run(cmd, dir, root, json, no_commit),
336        Cmd::Issue { cmd } => issue::run(cmd, dir, root, json, no_commit),
337        Cmd::Proposal { cmd } => proposal::run(cmd, dir, root, json, no_commit),
338        Cmd::Plan { cmd } => plan::run(cmd, dir, root, json, no_commit),
339        Cmd::Adr { cmd } => adr::run(cmd, dir, root, json, no_commit),
340        Cmd::Item { cmd } => itemcmd::run(cmd, dir, root, json, no_commit),
341        Cmd::Handoff {
342            id,
343            note,
344            actor,
345            session,
346        } => continuity::handoff(id, &note, actor, session, dir, root, json, no_commit),
347        Cmd::Interrupt {
348            id,
349            note,
350            actor,
351            session,
352        } => continuity::interrupt(
353            id,
354            note.as_deref(),
355            actor,
356            session,
357            dir,
358            root,
359            json,
360            no_commit,
361        ),
362        Cmd::Resume { id, last } => continuity::resume(id, last, dir, root, json),
363        Cmd::Schema => schema::run(),
364        Cmd::Status { prompt } => status::run(dir, root, prompt, json),
365        Cmd::Board {
366            md,
367            all,
368            epic,
369            milestone,
370        } => board::run(dir, root, md, all, epic, milestone, json),
371        Cmd::Epic { name, md, all } => {
372            group::run(group::Group::Epic, name, md, all, dir, root, json)
373        }
374        Cmd::Milestone { name, md, all } => {
375            group::run(group::Group::Milestone, name, md, all, dir, root, json)
376        }
377        Cmd::Absorb { cmd } => absorb::run(cmd, dir, root, json, no_commit),
378        Cmd::Report => report::run(dir, root, json),
379        Cmd::Index { cmd } => indexcmd::run(cmd, json),
380        Cmd::Org { cmd } => org::run(cmd, json, no_commit),
381        #[cfg(feature = "tui")]
382        Cmd::Ui => myco_tui::run(dir.as_deref(), root.as_deref(), no_commit).map(|_| 0),
383        Cmd::Serve => serve::run(dir, root, no_commit),
384    };
385    match result {
386        Ok(code) => std::process::exit(code),
387        Err(e) => {
388            if json {
389                ops::print_json(&serde_json::json!({"error": e}));
390            } else {
391                eprintln!("myco: error: {e}");
392            }
393            std::process::exit(2);
394        }
395    }
