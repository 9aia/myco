1//! myco — project management, grown not built. A mycorrhizal
2//! network for your agents: tasks, signals, handoffs.
3//!
4//! Agent-first PM: markdown in `project/` is canonical and
5//! human-editable; every deterministic write goes through this CLI
6//! (auto-commit per verb, `--json` everywhere). The skill teaches
7//! agents the verbs; `myco manifest`/`myco schema` are the
8//! self-describing surface.
9//!
10//! Phase 6 wires continuity — `handoff`/`resume` over append-only
11//! `handoffs/<ID>.md` files (one per item, any kind); sepia hooks
12//! call them at session boundaries. Phase 7 lands the Jira-parity
13//! slice — `epic:`/`ms:` item metadata + grouping views,
14//! `absorb` importers (comment sweep, Jira CSV, GitHub dumps).
15//! P-007 makes the verb surface data: one [`registry`] table
16//! drives the clap tree, `manifest`, and serve/batch dispatch;
17//! `[verbs]` in `myco.toml` declares repo/user verbs.

58fn manifest() -> serde_json::Value {
59    serde_json::json!({
60        "name": "myco",
61        "version": env!("CARGO_PKG_VERSION"),
62        "protocol": 1,
63        "description": env!("CARGO_PKG_DESCRIPTION"),
64        "repository": env!("CARGO_PKG_REPOSITORY"),
65        "capabilities": {
66            "store": "markdown",
67            "index": "redb",
68            "protocol": 1,
69            "serve": {
70                "protocol": 1,
71                "transport": "stdio",
72                "wire": "jsonrpc-2.0",
73            },
74        },
75        "verbs": registry::manifest_verbs(),
76        "implemented": registry::manifest_implemented(),
77        "notes": "markdown in project/ is canonical; every verb is one \
78                  atomic commit on the `project` branch carrying an \
79                  `Actor:` trailer (the verb's actor, default `cli`); \
80                  the redb index under $MYCO_HOME|~/.local/share/myco is \
81                  a disposable derived cache",
82    })
83}

100/// The `ui` verb (feature `tui`). The index universe for
101/// `source = "index"` views is injected as a loader — the TUI
102/// re-runs it on every reload, so `load_fresh`'s staleness
103/// logic governs `:w` too.
104#[cfg(feature = "tui")]
105fn h_ui(ctx: &registry::Ctx, a: &registry::Args) -> Result<i32, String> {
106    let view = a.opt("view").map(str::to_string).or_else(|| {
107        // config `default_view` is the landing view — cwd falls
108        // back for index-sourced views outside a repo (`~/myco.toml`)
109        let root = ctx
110            .root
111            .clone()
112            .or_else(|| ctx.dir.clone())
113            .or_else(|| std::env::current_dir().ok())
114            .unwrap_or_else(|| std::path::PathBuf::from("."));
115        myco_ops::config::default_view_for(&root, ops::CONFIG)
116    });
117    let org: myco_tui::OrgLoader = Box::new(|| {
118        let (views, _warnings) = view::index_universe()?;
119        Ok(views
120            .into_iter()
121            .map(|v| {
122                let (items, archived) = v.snap.map(|s| (s.items, s.archived)).unwrap_or_default();
123                myco_tui::OrgRepo {
124                    name: v.name,
125                    worktree: v.worktree.into_std_path_buf(),
126                    items,
127                    archived,
128                    error: v.error,
129                }
130            })
131            .collect())
132    });
133    myco_tui::run(
134        ctx.dir.as_deref(),
135        ctx.root.as_deref(),
136        ctx.no_commit,
137        view,
138        Some(org),
139    )
140    .map(|_| 0)
141}

143fn main() {
144    // Restore conventional-Unix SIGPIPE behavior (die silently on
145    // EPIPE) — `println!` would otherwise panic when piped to a
146    // reader that closes early (e.g. `myco board | head`).
147    sigpipe::reset();
148    let matches = registry::command().get_matches();
149    let ctx = registry::Ctx {
150        json: matches.get_flag("json"),
151        no_commit: matches.get_flag("no_commit"),
152        dir: matches.get_one::<String>("dir").map(PathBuf::from),
153        root: matches.get_one::<String>("root").map(PathBuf::from),
154    };
155    match registry::dispatch(&ctx, &matches) {
156        Ok(code) => std::process::exit(code),
157        Err(e) => {
158            if ctx.json {
159                ops::print_json(&serde_json::json!({"error": e}));
160            } else {
161                eprintln!("myco: error: {e}");
162            }
163            std::process::exit(2);
164        }
165    }
166}
