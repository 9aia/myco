1//! myco — project management, grown not built. A mycorrhizal
2//! network for your agents: tasks, signals, handoffs.
3//!
4//! Agent-first PM: markdown in `project/` is canonical and
5//! human-editable; every deterministic write goes through this CLI
6//! (auto-commit per verb, `--json` everywhere). The skill teaches
7//! agents the verbs; `myco manifest`/`myco schema` are the
8//! self-describing surface.
9//!
10//! Phase 6 wires continuity — `handoff`/`resume` over append-only
11//! `handoffs/<ID>.md` files (one per item, any kind); sepia hooks
12//! call them at session boundaries. Phase 7 lands the Jira-parity
13//! slice — `epic:`/`ms:` item metadata + grouping views,
14//! `absorb` importers (comment sweep, Jira CSV, GitHub dumps).

47/// myco — agent-first project management for a repo's `project/`
48/// worktree.
49#[derive(Parser)]
50#[command(name = "myco", version, about, long_about = None)]
51struct Cli {
52    /// Emit JSON (accepted everywhere; default output for machines).
53    #[arg(long, global = true)]
54    json: bool,
55
56    /// Skip the per-verb commit (also suppressed by $MYCO_NO_COMMIT).
57    #[arg(long, global = true)]
58    no_commit: bool,
59
60    /// Project worktree to act on, overriding discovery.
61    #[arg(long, global = true, value_name = "PATH")]
62    dir: Option<PathBuf>,
63
64    /// Repo root to act on, overriding discovery (the worktree is
65    /// `<root>/project`).
66    #[arg(long, global = true, value_name = "PATH")]
67    root: Option<PathBuf>,
68
69    #[command(subcommand)]
70    command: Cmd,
71}
72
73#[derive(Subcommand)]
74enum Cmd {
75    /// Capabilities + protocol version — agent self-discovery.
76    Manifest,
77    /// The file/state spec, machine-readable JSON Schema.
78    Schema,
79    /// Scaffold `project/`: orphan branch + worktree + .gitignore +
80    /// seed files + the managed AGENTS.md agent-awareness block.
81    /// Idempotent — existing worktree and AGENTS.md prose are
82    /// preserved, only the marked block is (re)written.
83    Init,
84    /// Validate schemas + state machine, repair file/index drift;
85    /// also validates the managed AGENTS.md block on the repo
86    /// root. Exit 1 with findings; `--fix` applies safe repairs +
87    /// commits.
88    Check {
89        /// Repair drift in place instead of only reporting.
90        #[arg(long)]
91        fix: bool,
92    },
93    /// Convert legacy ledgers to v2: `pm/` → `project/`, or
94    /// normalize files already on the `project` branch.
95    Migrate,
96    /// Stage + commit worktree changes (conventional message).
97    Commit {
98        /// Commit message (default: `chore: update project ledger`).
99        #[arg(short, long)]
100        message: Option<String>,
101    },
102    /// TODO.md tasks — add / done / defer / list.
103    Todo {
104        #[command(subcommand)]
105        cmd: todo::Cmd,
106    },
107    /// BACKLOG.md parked items — add / promote / list.
108    Backlog {
109        #[command(subcommand)]
110        cmd: backlog::Cmd,
111    },
112    /// Tracked issues, one file each under `issues/`.
113    Issue {
114        #[command(subcommand)]
115        cmd: issue::Cmd,
116    },
117    /// Design proposals under `proposals/` (`P-NNN`).
118    Proposal {
119        #[command(subcommand)]
120        cmd: proposal::Cmd,
121    },
122    /// Execution plans under `plans/` (`PL-NNN`).
123    Plan {
124        #[command(subcommand)]
125        cmd: plan::Cmd,
126    },
127    /// Architecture decision records under `adrs/` (`ADR-NNNN`).
128    Adr {
129        #[command(subcommand)]
130        cmd: adr::Cmd,
131    },
132    /// Kind-agnostic item verbs — show / set.
133    Item {
134        #[command(subcommand)]
135        cmd: itemcmd::Cmd,
136    },
137    /// Attach resumable state at session end — appends a dated
138    /// entry to `handoffs/<ID>.md` (any item kind, live or
139    /// archived; append-only).
140    Handoff {
141        /// Any item id (`T-NNN`, `I-NNN`, `P-NNN`, `ADR-NNNN`,
142        /// `PL-NNN`).
143        id: myco_core::ItemId,
144        /// Handoff note — done/next/open. `-` reads stdin.
145        #[arg(long, value_name = "TEXT|-")]
146        note: String,
147        /// Who wrote it (`devin`, `human:luis`, …; default `cli`).
148        #[arg(long)]
149        actor: Option<myco_core::Actor>,
150        /// Session provenance — a sepia session id, `cli:<pid>`,
151        /// any id string.
152        #[arg(long)]
153        session: Option<String>,
154    },
155    /// A session ended with work still claimed/in-progress/
156    /// blocked — revert the item to `open` + `interrupted`
157    /// handoff, one commit. Session-lifecycle hooks (sepia, CLI
158    /// session-end) call this; myco never detects sessions itself.
159    Interrupt {
160        /// Any item id (`T-NNN`, `I-NNN`, `P-NNN`, `ADR-NNNN`,
161        /// `PL-NNN`).
162        id: myco_core::ItemId,
163        /// Extra context appended to the `interrupted` note.
164        #[arg(long, value_name = "TEXT")]
165        note: Option<String>,
166        /// Who interrupted it (default `cli`).
167        #[arg(long)]
168        actor: Option<myco_core::Actor>,
169        /// Session provenance — a sepia session id, `cli:<pid>`,
170        /// any id string.
171        #[arg(long)]
172        session: Option<String>,
173    },
174    /// Minimal context block to restart work on an item — the
175    /// item, `blocks:` dep titles, latest handoffs, open
176    /// checklist. The single-item `status --prompt`.
177    Resume {
178        /// Any item id.
179        id: myco_core::ItemId,
180        /// Only the latest N handoff entries (default 10).
181        #[arg(long)]
182        last: Option<usize>,
183    },
184    /// Cheap digest of the work ledger. `--prompt` = token-sized.
185    Status {
186        /// Render for an agent prompt (minimal tokens).
187        #[arg(long)]
188        prompt: bool,
189    },
190    /// Kanban of the ledger — Backlog | Open | Claimed | In
191    /// Progress | Blocked. `--md` emits a markdown table.
192    Board {
193        /// Emit a markdown table (for docs/READMEs).
194        #[arg(long)]
195        md: bool,
196        /// Show every card — no row cap.
197        #[arg(long)]
198        all: bool,
199        /// Only items in this epic.
200        #[arg(long)]
201        epic: Option<String>,
202        /// Only items in this milestone.
203        #[arg(long)]
204        milestone: Option<String>,
205    },
206    /// Items grouped by epic — bare lists epics with counts, a
207    /// name renders that epic's board (`epic v2` ≡ `board
208    /// --epic v2`).
209    Epic {
210        /// Epic name — omit for the all-epics summary.
211        name: Option<String>,
212        /// Emit a markdown table (with a name).
213        #[arg(long)]
214        md: bool,
215        /// Show every card — no row cap (with a name).
216        #[arg(long)]
217        all: bool,
218    },
219    /// Items grouped by milestone — bare lists milestones with
220    /// counts, a name renders that milestone's board.
221    Milestone {
222        /// Milestone name — omit for the all-milestones summary.
223        name: Option<String>,
224        /// Emit a markdown table (with a name).
225        #[arg(long)]
226        md: bool,
227        /// Show every card — no row cap (with a name).
228        #[arg(long)]
229        all: bool,
230    },
231    /// Import work into the ledger — TODO/FIXME comments, Jira
232    /// CSV exports, GitHub issue dumps.
233    Absorb {
234        #[command(subcommand)]
235        cmd: absorb::Cmd,
236    },
237    /// Aging + velocity + blocked — how the ledger is trending.
238    Report,
239    /// The derived index: register repos, query/rebuild snapshots.
240    Index {
241        #[command(subcommand)]
242        cmd: indexcmd::Cmd,
243    },
244    /// Cross-repo views over the index (auto-rebuilds stale repos).
245    Org {
246        #[command(subcommand)]
247        cmd: org::Cmd,
248    },
249    /// Interactive board TUI (ratatui) — TODO/BACKLOG tabs, item
250    /// preview, writes through the same ops the verbs use.
251    #[cfg(feature = "tui")]
252    Ui,
253    /// Print the CLI reference as markdown (docs pipeline).
254    #[command(hide = true, name = "gen-cli-docs")]
255    GenCliDocs,
256}

300fn main() {
301    // Restore conventional-Unix SIGPIPE behavior (die silently on
302    // EPIPE) — `println!` would otherwise panic when piped to a
303    // reader that closes early (e.g. `myco board | head`).
304    sigpipe::reset();
305    let cli = Cli::parse();
306    let json = cli.json;
307    let no_commit = cli.no_commit;
308    let dir = cli.dir.clone();
309    let root = cli.root.clone();
310    let result: Result<i32, String> = match cli.command {
311        Cmd::Manifest => {
312            ops::print_json(&manifest());
313            Ok(0)
314        }
315        Cmd::GenCliDocs => {
316            println!("{}", clap_markdown::help_markdown::<Cli>());
317            Ok(0)
318        }
319        Cmd::Init => init::run(dir, root, json, no_commit),
320        Cmd::Check { fix } => check::run(dir, root, fix, json, no_commit),
321        Cmd::Migrate => migrate::run(dir, root, json, no_commit),
322        Cmd::Commit { message } => commit::run(dir, root, message, json, no_commit),
323        Cmd::Todo { cmd } => todo::run(cmd, dir, root, json, no_commit),
324        Cmd::Backlog { cmd } => backlog::run(cmd, dir, root, json, no_commit),
325        Cmd::Issue { cmd } => issue::run(cmd, dir, root, json, no_commit),
326        Cmd::Proposal { cmd } => proposal::run(cmd, dir, root, json, no_commit),
327        Cmd::Plan { cmd } => plan::run(cmd, dir, root, json, no_commit),
328        Cmd::Adr { cmd } => adr::run(cmd, dir, root, json, no_commit),
329        Cmd::Item { cmd } => itemcmd::run(cmd, dir, root, json, no_commit),
330        Cmd::Handoff {
331            id,
332            note,
333            actor,
334            session,
335        } => continuity::handoff(id, &note, actor, session, dir, root, json, no_commit),
336        Cmd::Interrupt {
337            id,
338            note,
339            actor,
340            session,
341        } => continuity::interrupt(
342            id,
343            note.as_deref(),
344            actor,
345            session,
346            dir,
347            root,
348            json,
349            no_commit,
350        ),
351        Cmd::Resume { id, last } => continuity::resume(id, last, dir, root, json),
352        Cmd::Schema => schema::run(),
353        Cmd::Status { prompt } => status::run(dir, root, prompt, json),
354        Cmd::Board {
355            md,
356            all,
357            epic,
358            milestone,
359        } => board::run(dir, root, md, all, epic, milestone, json),
360        Cmd::Epic { name, md, all } => {
361            group::run(group::Group::Epic, name, md, all, dir, root, json)
362        }
363        Cmd::Milestone { name, md, all } => {
364            group::run(group::Group::Milestone, name, md, all, dir, root, json)
365        }
366        Cmd::Absorb { cmd } => absorb::run(cmd, dir, root, json, no_commit),
367        Cmd::Report => report::run(dir, root, json),
368        Cmd::Index { cmd } => indexcmd::run(cmd, json),
369        Cmd::Org { cmd } => org::run(cmd, json, no_commit),
370        #[cfg(feature = "tui")]
371        Cmd::Ui => myco_tui::run(dir.as_deref(), root.as_deref(), no_commit).map(|_| 0),
372    };
373    match result {
374        Ok(code) => std::process::exit(code),
375        Err(e) => {
376            if json {
377                ops::print_json(&serde_json::json!({"error": e}));
378            } else {
379                eprintln!("myco: error: {e}");
380            }
381            std::process::exit(2);
382        }
383    }
