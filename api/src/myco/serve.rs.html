1//! `myco serve` — a sequential request/response verb surface for
2//! embedding: an orchestrator (peregrin) spawns one long-lived
3//! process per repo instead of one spawn per verb. A single
4//! process serializes writes — no `index.lock` races — and the
5//! store is re-opened per request, so the served process always
6//! sees on-disk truth.
7//!
8//! Two abstractions keep the surface honest: [`ServeTransport`]
9//! moves one raw line in and out (stdio first; a unix socket can
10//! slot in untouched), and [`WireProtocol`] frames requests and
11//! responses (JSON-RPC 2.0 first — the default, not the spec).
12//! [`Server::dispatch`] is the verb surface itself and knows
13//! neither. stdin IS the wire — no served method reads content
14//! from it.
15

29/// JSON-RPC 2.0 error codes served on the wire.
30pub const PARSE_ERROR: i64 = -32700;
31/// The envelope decoded but isn't a callable request (also a
32/// missing `id` — notifications aren't served).
33pub const INVALID_REQUEST: i64 = -32600;
34/// The method isn't on the v1 surface.
35pub const METHOD_NOT_FOUND: i64 = -32601;
36/// `params` didn't decode into the method's shape.
37pub const INVALID_PARAMS: i64 = -32602;
38/// The verb itself failed.
39pub const VERB_ERROR: i64 = -32000;
40/// A status-machine rejection — `data` carries
41/// `{from, to, valid: []}`.
42pub const INVALID_TRANSITION: i64 = -32001;
43
44/// One decoded request — `id` is echoed back verbatim so callers
45/// correlate responses on a shared pipe.
46#[derive(Debug)]
47pub struct Request {
48    /// Wire-level correlation id.
49    pub id: Value,
50    /// The method name (`todo.add`, `status`, …).
51    pub method: String,
52    /// Method params (`null` when absent).
53    pub params: Value,
54}
55
56/// One response to encode.
57pub enum Response {
58    /// The method's payload.
59    Result {
60        /// Correlation id echoed from the request.
61        id: Value,
62        /// The result value.
63        result: Value,
64    },
65    /// A wire or dispatch failure.
66    Error {
67        /// Correlation id (`null` when the request never decoded).
68        id: Value,
69        /// One of the `-32xxx` codes.
70        code: i64,
71        /// Human/agent-readable message.
72        message: String,
73        /// Optional structured detail (transition targets, …).
74        data: Option<Value>,
75    },
76}
77
78/// A line that never decoded into a [`Request`].
79#[derive(Debug)]
80pub struct WireError {
81    /// The error code.
82    pub code: i64,
83    /// What went wrong.
84    pub message: String,
85}

87/// Request/response framing — the abstraction point. JSON-RPC
88/// 2.0 is the default impl; [`Server::serve`] doesn't know it.
89pub trait WireProtocol {
90    /// Decode one raw line into a request.
91    fn parse(&self, line: &str) -> Result<Request, WireError>;
92    /// Encode one response as a single line.
93    fn encode(&self, response: &Response) -> String;
94}
95
96/// JSON-RPC 2.0 — one request/response object per line.
97pub struct JsonRpc2;
98
99impl WireProtocol for JsonRpc2 {
100    fn parse(&self, line: &str) -> Result<Request, WireError> {
101        let v: Value = serde_json::from_str(line).map_err(|e| WireError {
102            code: PARSE_ERROR,
103            message: format!("parse error: {e}"),
104        })?;
105        let invalid = |message: &str| WireError {
106            code: INVALID_REQUEST,
107            message: message.to_string(),
108        };
109        let obj = v
110            .as_object()
111            .ok_or_else(|| invalid("request is not a JSON object"))?;
112        let method = obj
113            .get("method")
114            .and_then(Value::as_str)
115            .filter(|m| !m.is_empty())
116            .ok_or_else(|| invalid("missing or invalid `method`"))?
117            .to_string();
118        let id = obj
119            .get("id")
120            .filter(|id| !id.is_null())
121            .cloned()
122            .ok_or_else(|| invalid("missing `id` — notifications aren't served"))?;
123        let params = obj.get("params").cloned().unwrap_or(Value::Null);
124        Ok(Request { id, method, params })
125    }
126
127    fn encode(&self, response: &Response) -> String {
128        match response {
129            Response::Result { id, result } => {
130                json!({"jsonrpc": "2.0", "id": id, "result": result}).to_string()
131            }
132            Response::Error {
133                id,
134                code,
135                message,
136                data,
137            } => {
138                let mut e = json!({"code": code, "message": message});
139                if let Some(d) = data {
140                    e["data"] = d.clone();
141                }
142                json!({"jsonrpc": "2.0", "id": id, "error": e}).to_string()
143            }
144        }
145    }
146}

150/// One request line in, one response line out — the transport
151/// abstraction.
152pub trait ServeTransport {
153    /// The next raw request line — `Ok(None)` on clean EOF.
154    fn read_request(&mut self) -> Result<Option<String>, String>;
155    /// Write one raw response line (and flush).
156    fn write_response(&mut self, line: &str) -> Result<(), String>;
157}
158
159/// stdin/stdout — `BufRead` lines in, `writeln!` + flush out.
160pub struct StdioTransport<R, W> {
161    reader: R,
162    writer: W,
163}
164
165impl<R, W> StdioTransport<R, W> {
166    /// Wrap a reader/writer pair.
167    pub fn new(reader: R, writer: W) -> Self {
168        StdioTransport { reader, writer }
169    }
170}
171
172impl<R: BufRead, W: Write> ServeTransport for StdioTransport<R, W> {
173    fn read_request(&mut self) -> Result<Option<String>, String> {
174        let mut line = String::new();
175        let n = self
176            .reader
177            .read_line(&mut line)
178            .map_err(|e| format!("read request: {e}"))?;
179        if n == 0 {
180            return Ok(None);
181        }
182        Ok(Some(line.trim_end_matches(['\n', '\r']).to_string()))
183    }
184
185    fn write_response(&mut self, line: &str) -> Result<(), String> {
186        writeln!(self.writer, "{line}")
187            .and_then(|()| self.writer.flush())
188            .map_err(|e| format!("write response: {e}"))
189    }
190}

194/// `params` (or absent) → a typed struct — bad shape is -32602.
195fn parse_params<T: serde::de::DeserializeOwned>(v: &Value) -> Result<T, Fail> {
196    let v = if v.is_null() { json!({}) } else { v.clone() };
197    serde_json::from_value(v).map_err(|e| Fail::BadParams(format!("invalid params: {e}")))
198}

324/// Dispatch failure — [`Server::respond`] maps these to wire
325/// codes; dispatch itself knows no wire.
326#[derive(Debug)]
327enum Fail {
328    /// The method isn't served.
329    UnknownMethod,
330    /// Params didn't decode — -32602.
331    BadParams(String),
332    /// The verb failed — -32000.
333    Verb(String),
334    /// Status-machine rejection — -32001 + `data`.
335    Transition {
336        /// The item being set.
337        id: ItemId,
338        /// Current status.
339        from: Status,
340        /// Requested status.
341        to: Status,
342    },
343}
344
345impl From<String> for Fail {
346    fn from(s: String) -> Fail {
347        Fail::Verb(s)
348    }
349}

378/// The served surface — discovery flags + `--no-commit`, shared
379/// across requests. Nothing caches store state: every method
380/// re-opens the store.
381pub struct Server {
382    dir: Option<PathBuf>,
383    root: Option<PathBuf>,
384    no_commit: bool,
385}
386
387impl Server {
388    /// A server over the caller's discovery context.
389    pub fn new(dir: Option<PathBuf>, root: Option<PathBuf>, no_commit: bool) -> Self {
390        Server {
391            dir,
392            root,
393            no_commit,
394        }
395    }
396
397    fn open(&self) -> Result<(crate::discover::Project, myco_store_md::MdStore), Fail> {
398        items::open(self.dir.as_deref(), self.root.as_deref()).map_err(Fail::Verb)
399    }
400
401    fn commit(&self, wt: &std::path::Path, msg: &str) -> Result<bool, Fail> {
402        items::commit(wt, msg, self.no_commit).map_err(Fail::Verb)
403    }
404
405    /// The request loop — wire- and transport-agnostic: read one
406    /// request, write one response, until EOF (`Ok(0)`).
407    pub fn serve<T: ServeTransport, W: WireProtocol>(
408        &self,
409        transport: &mut T,
410        wire: &W,
411    ) -> Result<i32, String> {
412        while let Some(line) = transport.read_request()? {
413            let response = match wire.parse(&line) {
414                Ok(req) => self.respond(req),
415                Err(e) => Response::Error {
416                    id: Value::Null,
417                    code: e.code,
418                    message: e.message,
419                    data: None,
420                },
421            };
422            transport.write_response(&wire.encode(&response))?;
423        }
424        Ok(0)
425    }
426
427    fn respond(&self, req: Request) -> Response {
428        match self.dispatch(&req.method, &req.params) {
429            Ok(result) => Response::Result { id: req.id, result },
430            Err(fail) => {
431                let (code, message, data) = match fail {
432                    Fail::UnknownMethod => (
433                        METHOD_NOT_FOUND,
434                        format!("unknown method {:?}", req.method),
435                        None,
436                    ),
437                    Fail::BadParams(m) => (INVALID_PARAMS, m, None),
438                    Fail::Verb(m) => (VERB_ERROR, m, None),
439                    Fail::Transition { id, from, to } => (
440                        INVALID_TRANSITION,
441                        format!("{id}: invalid status transition {from} → {to}"),
442                        Some(json!({
443                            "from": from.token(),
444                            "to": to.token(),
445                            "valid": from
446                                .transitions()
447                                .iter()
448                                .map(|s| s.token())
449                                .collect::<Vec<_>>(),
450                        })),
451                    ),
452                };
453                Response::Error {
454                    id: req.id,
455                    code,
456                    message,
457                    data,
458                }
459            }
460        }
461    }
462
463    /// The verb surface — `method` + `params` → a JSON payload,
464    /// transport- and wire-agnostic.
465    fn dispatch(&self, method: &str, params: &Value) -> Result<Value, Fail> {
466        match method {
467            "initialize" => Ok(json!({
468                "protocol": 1,
469                "wire": "jsonrpc-2.0",
470                "transport": "stdio",
471                "server": {"name": "myco", "version": env!("CARGO_PKG_VERSION")},
472                "methods": METHODS,
473            })),
474            "manifest" => Ok(crate::manifest()),
475            "schema" => Ok(schema::document()),
476            "status" => {
477                let p: StatusParams = parse_params(params)?;
478                let (_proj, store) = self.open()?;
479                if p.prompt {
480                    status::prompt_json(&store).map_err(Fail::Verb)
481                } else {
482                    status::status_json(&store).map_err(Fail::Verb)
483                }
484            }
485            "board" => {
486                let p: BoardParams = parse_params(params)?;
487                let (_proj, store) = self.open()?;
488                let scan = digest::scan(&store).map_err(Fail::Verb)?;
489                let live: Vec<&myco_core::Item> = scan
490                    .live()
491                    .filter(|i| {
492                        p.epic
493                            .as_deref()
494                            .is_none_or(|e| i.epic.as_deref() == Some(e))
495                            && p.milestone
496                                .as_deref()
497                                .is_none_or(|m| i.milestone.as_deref() == Some(m))
498                    })
499                    .collect();
500                Ok(board::columns_json(live.into_iter(), None))
501            }
502            "item.show" => {
503                let p: IdParams = parse_params(params)?;
504                let (_proj, store) = self.open()?;
505                itemcmd::show_json(&store, p.id).map_err(Fail::Verb)
506            }
507            "todo.list" | "backlog.list" => {
508                let p: ListParams = parse_params(params)?;
509                let (_proj, store) = self.open()?;
510                let found = todo::list_items(&store, method == "backlog.list", &p.filter())?;
511                Ok(items_json(&found))
512            }
513            "issue.list" | "proposal.list" | "plan.list" | "adr.list" => {
514                let p: ListParams = parse_params(params)?;
515                let (_proj, store) = self.open()?;
516                let kind = match method {
517                    "issue.list" => ItemKind::Issue,
518                    "proposal.list" => ItemKind::Proposal,
519                    "plan.list" => ItemKind::Plan,
520                    _ => ItemKind::Adr,
521                };
522                let found = items::query_files(&store, kind, p.archived, &p.filter())
523                    .map_err(Fail::Verb)?;
524                Ok(items_json(&found))
525            }
526            "report" => {
527                let (proj, store) = self.open()?;
528                report::report_json(&store, &proj.worktree).map_err(Fail::Verb)
529            }
530            "resume" => {
531                let p: IdParams = parse_params(params)?;
532                let (_proj, store) = self.open()?;
533                continuity::resume_json(&store, p.id, p.last).map_err(Fail::Verb)
534            }
535            "check" => {
536                let (proj, _store) = self.open()?;
537                check::check_json(&proj).map_err(Fail::Verb)
538            }
539            "todo.add" => {
540                let p: TodoAddParams = parse_params(params)?;
541                let (proj, store) = self.open()?;
542                let (item, file) = todo::add_item(
543                    &store,
544                    &p.title,
545                    p.section.as_deref(),
546                    p.parent,
547                    &p.meta.into_meta(),
548                )
549                .map_err(Fail::Verb)?;
550                let committed = self.commit(&proj.worktree, &format!("chore: add {}", item.id))?;
551                Ok(json!({"item": item, "file": file, "parent": p.parent, "committed": committed}))
552            }
553            "todo.done" => {
554                let p: IdParams = parse_params(params)?;
555                let (proj, store) = self.open()?;
556                let item = todo::done_item(&store, p.id).map_err(Fail::Verb)?;
557                let committed = self.commit(&proj.worktree, &format!("chore: done {}", p.id))?;
558                Ok(json!({"item": item, "committed": committed}))
559            }
560            "todo.defer" => {
561                let p: IdParams = parse_params(params)?;
562                let (proj, store) = self.open()?;
563                let item = todo::defer_item(&store, p.id).map_err(Fail::Verb)?;
564                let committed = self.commit(&proj.worktree, &format!("chore: defer {}", p.id))?;
565                Ok(json!({"item": item, "file": "BACKLOG.md", "committed": committed}))
566            }
567            "backlog.add" => {
568                let p: BacklogAddParams = parse_params(params)?;
569                let (proj, store) = self.open()?;
570                let item =
571                    backlog::add_item(&store, &p.title, p.section.as_deref(), &p.meta.into_meta())
572                        .map_err(Fail::Verb)?;
573                let committed = self.commit(&proj.worktree, &format!("chore: add {}", item.id))?;
574                Ok(json!({"item": item, "file": "BACKLOG.md", "committed": committed}))
575            }
576            "backlog.promote" => {
577                let p: IdParams = parse_params(params)?;
578                let (proj, store) = self.open()?;
579                let item = backlog::promote_item(&store, p.id).map_err(Fail::Verb)?;
580                let committed = self.commit(&proj.worktree, &format!("chore: promote {}", p.id))?;
581                Ok(json!({"item": item, "file": "TODO.md", "committed": committed}))
582            }
583            "item.set" => {
584                let p: SetParams = parse_params(params)?;
585                let (proj, store) = self.open()?;
586                let flags = itemcmd::SetFlags {
587                    status: p.status,
588                    pri: p.pri,
589                    owner: p.owner,
590                    label: p.label,
591                    unlabel: p.unlabel,
592                    blocks: p.blocks,
593                    unblock: p.unblock,
594                    epic: p.epic,
595                    milestone: p.milestone,
596                };
597                if flags.is_noop() {
598                    return Err(Fail::Verb(
599                        "item set: nothing to change — pass at least one field".into(),
600                    ));
601                }
602                // pre-check the status step so the wire gets the
603                // typed -32001 + valid targets, not a string
604                if let Some(to) = flags.status {
605                    let Some((item, _, _)) = items::find_item(&store, p.id).map_err(Fail::Verb)?
606                    else {
607                        return Err(Fail::Verb(format!("{}: no such item", p.id)));
608                    };
609                    item.status
610                        .transition_to(to)
611                        .map_err(|_| Fail::Transition {
612                            id: p.id,
613                            from: item.status,
614                            to,
615                        })?;
616                }
617                let (item, location) =
618                    itemcmd::mutate_item(&store, p.id, &flags).map_err(Fail::Verb)?;
619                let committed = self.commit(&proj.worktree, &format!("chore: set {}", p.id))?;
620                Ok(json!({"item": item, "location": location, "committed": committed}))
621            }
622            "handoff" => {
623                let p: HandoffParams = parse_params(params)?;
624                if p.note == "-" {
625                    return Err(Fail::BadParams(
626                        "note \"-\" reads stdin — stdin is the wire; pass the note inline".into(),
627                    ));
628                }
629                let (proj, store) = self.open()?;
630                let (entry, rel) = continuity::record_handoff(
631                    &store,
632                    &proj.worktree,
633                    p.id,
634                    p.note,
635                    p.actor,
636                    p.session,
637                )
638                .map_err(Fail::Verb)?;
639                let committed = self.commit(&proj.worktree, &format!("chore: handoff {}", p.id))?;
640                Ok(
641                    json!({"id": p.id.to_string(), "path": rel, "entry": entry, "committed": committed}),
642                )
643            }
644            "interrupt" => {
645                let p: InterruptParams = parse_params(params)?;
646                let (proj, store) = self.open()?;
647                let entry =
648                    continuity::interrupt_item(&store, p.id, p.note.as_deref(), p.actor, p.session)
649                        .map_err(Fail::Verb)?;
650                let committed =
651                    self.commit(&proj.worktree, &format!("chore: interrupt {}", p.id))?;
652                Ok(
653                    json!({"id": p.id.to_string(), "status": "open", "entry": entry, "committed": committed}),
654                )
655            }
656            _ => Err(Fail::UnknownMethod),
657        }
658    }
659}

661/// `Vec<Item>` → a JSON array of item objects.
662fn items_json(items: &[myco_core::Item]) -> Value {
663    json!(items
664        .iter()
665        .map(|i| serde_json::to_value(i).unwrap_or(json!(null)))
666        .collect::<Vec<_>>())
667}

669/// The `serve` verb — stdio transport, JSON-RPC 2.0 wire, EOF
670/// exits 0.
671pub fn run(dir: Option<PathBuf>, root: Option<PathBuf>, no_commit: bool) -> Result<i32, String> {
672    let server = Server::new(dir, root, no_commit);
673    let stdin = std::io::stdin();
674    let stdout = std::io::stdout();
675    let mut transport = StdioTransport::new(stdin.lock(), stdout.lock());
676    server.serve(&mut transport, &JsonRpc2)
677}

699    #[test]
700    fn jsonrpc_parse_valid_and_errors() {
701        let w = JsonRpc2;
702        let r = w
703            .parse(r#"{"jsonrpc":"2.0","id":7,"method":"status","params":{"prompt":true}}"#)
704            .unwrap();
705        assert_eq!(r.id, json!(7));
706        assert_eq!(r.method, "status");
707        assert_eq!(r.params["prompt"], true);
708        // params optional
709        let r = w.parse(r#"{"id":"a","method":"manifest"}"#).unwrap();
710        assert_eq!(r.id, json!("a"));
711        assert!(r.params.is_null());
712        // not json → -32700
713        assert_eq!(w.parse("hello").unwrap_err().code, PARSE_ERROR);
714        // non-object → -32600
715        assert_eq!(w.parse("[1,2]").unwrap_err().code, INVALID_REQUEST);
716        // missing id → -32600 (notifications aren't served)
717        assert_eq!(
718            w.parse(r#"{"method":"status"}"#).unwrap_err().code,
719            INVALID_REQUEST
720        );
721        // missing method → -32600
722        assert_eq!(w.parse(r#"{"id":1}"#).unwrap_err().code, INVALID_REQUEST);
723    }

725    #[test]
726    fn jsonrpc_encode_roundtrip() {
727        let w = JsonRpc2;
728        let line = w.encode(&Response::Result {
729            id: json!("x"),
730            result: json!({"ok": true}),
731        });
732        let v: Value = serde_json::from_str(&line).unwrap();
733        assert_eq!(v["jsonrpc"], "2.0");
734        assert_eq!(v["id"], "x");
735        assert_eq!(v["result"]["ok"], true);
736        assert!(!line.contains('\n'));
737        // error with data
738        let line = w.encode(&Response::Error {
739            id: Value::Null,
740            code: INVALID_TRANSITION,
741            message: "nope".into(),
742            data: Some(json!({"valid": ["open"]})),
743        });
744        let v: Value = serde_json::from_str(&line).unwrap();
745        assert_eq!(v["id"], Value::Null);
746        assert_eq!(v["error"]["code"], INVALID_TRANSITION);
747        assert_eq!(v["error"]["data"]["valid"], json!(["open"]));
748    }

750    #[test]
751    fn serve_loop_over_memory_transport() {
752        let (_d, s) = fixture();
753        let input = b"{\"id\":1,\"method\":\"manifest\"}\ngarbage\n{\"id\":2,\"method\":\"todo.add\",\"params\":{\"title\":\"x\"}}\n";
754        let mut out = Vec::new();
755        {
756            let mut t = StdioTransport::new(&input[..], &mut out);
757            assert_eq!(s.serve(&mut t, &JsonRpc2).unwrap(), 0);
758        }
759        let lines: Vec<Value> = String::from_utf8(out)
760            .unwrap()
761            .lines()
762            .map(|l| serde_json::from_str(l).unwrap())
763            .collect();
764        assert_eq!(lines.len(), 3);
765        assert_eq!(lines[0]["id"], 1);
766        assert_eq!(lines[0]["result"]["name"], "myco");
767        // a parse error doesn't kill the loop
768        assert_eq!(lines[1]["error"]["code"], PARSE_ERROR);
769        assert_eq!(lines[1]["id"], Value::Null);
770        assert_eq!(lines[2]["result"]["item"]["id"], "T-001");
771        // EOF → clean exit (serve returned already)
772    }

774    #[test]
775    fn dispatch_reads_and_writes() {
776        let (_d, s) = fixture();
777        // reads on an empty ledger
778        let v = call(&s, "status", Value::Null).unwrap();
779        assert_eq!(v["repo"], "p");
780        assert_eq!(v["counts"]["tasks"]["open"], 0);
781        let v = call(&s, "check", Value::Null).unwrap();
782        assert!(v["errors"].is_array());
783        // write + read-back through fresh store opens
784        let v = call(&s, "todo.add", json!({"title": "first", "pri": "high"})).unwrap();
785        assert_eq!(v["item"]["id"], "T-001");
786        assert_eq!(v["committed"], false); // no_commit
787        let v = call(&s, "todo.list", Value::Null).unwrap();
788        assert_eq!(v[0]["id"], "T-001");
789        assert_eq!(v[0]["priority"], "high");
790        let v = call(&s, "status", json!({"prompt": true})).unwrap();
791        assert!(v["prompt"].as_str().unwrap().contains("T-001"));
792        // item.set — legal step, then an invalid one → -32001 shape
793        let v = call(
794            &s,
795            "item.set",
796            json!({"id": "T-001", "status": "in-progress"}),
797        )
798        .unwrap();
799        assert_eq!(v["item"]["status"], "in-progress");
800        match call(&s, "item.set", json!({"id": "T-001", "status": "claimed"})) {
801            Err(Fail::Transition { id, from, to }) => {
802                assert_eq!(id.to_string(), "T-001");
803                assert_eq!(from, Status::InProgress);
804                assert_eq!(to, Status::Claimed);
805            }
806            _ => panic!("expected transition failure"),
807        }
808        // interrupt a live item → open + marker
809        let v = call(&s, "interrupt", json!({"id": "T-001"})).unwrap();
810        assert_eq!(v["status"], "open");
811        // todo.done removes the line
812        call(
813            &s,
814            "item.set",
815            json!({"id": "T-001", "status": "in-progress"}),
816        )
817        .unwrap();
818        let v = call(&s, "todo.done", json!({"id": "T-001"})).unwrap();
819        assert_eq!(v["item"]["status"], "done");
820        assert_eq!(
821            call(&s, "todo.list", Value::Null)
822                .unwrap()
823                .as_array()
824                .unwrap()
825                .len(),
826            0
827        );
828        // unknown + bad params
829        assert!(matches!(
830            call(&s, "nope", Value::Null),
831            Err(Fail::UnknownMethod)
832        ));
833        assert!(matches!(
834            call(&s, "todo.add", json!({})),
835            Err(Fail::BadParams(_))
836        ));
837    }
