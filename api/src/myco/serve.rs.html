1//! `myco serve` — a sequential request/response verb surface for
2//! embedding: an orchestrator (peregrin) spawns one long-lived
3//! process per repo instead of one spawn per verb. A single
4//! process serializes writes — no `index.lock` races — and the
5//! store is re-opened per request, so the served process always
6//! sees on-disk truth.
7//!
8//! Two abstractions keep the surface honest: [`ServeTransport`]
9//! moves one raw line in and out (stdio first; a unix socket can
10//! slot in untouched), and [`WireProtocol`] frames requests and
11//! responses (JSON-RPC 2.0 first — the default, not the spec).
12//! [`Server::dispatch`] is the verb surface itself and knows
13//! neither. stdin IS the wire — no served method reads content
14//! from it. `myco batch` drives the identical surface one-shot
15//! through [`Server::handle_line`] — NDJSON in, NDJSON out.
16

33/// JSON-RPC 2.0 error codes served on the wire.
34pub const PARSE_ERROR: i64 = -32700;
35/// The envelope decoded but isn't a callable request (also a
36/// missing `id` — notifications aren't served).
37pub const INVALID_REQUEST: i64 = -32600;
38/// The method isn't on the v1 surface.
39pub const METHOD_NOT_FOUND: i64 = -32601;
40/// `params` didn't decode into the method's shape.
41pub const INVALID_PARAMS: i64 = -32602;
42/// The verb itself failed.
43pub const VERB_ERROR: i64 = -32000;
44/// A status-machine rejection — `data` carries
45/// `{from, to, valid: []}`.
46pub const INVALID_TRANSITION: i64 = -32001;
47
48/// One decoded request — `id` is echoed back verbatim so callers
49/// correlate responses on a shared pipe.
50#[derive(Debug)]
51pub struct Request {
52    /// Wire-level correlation id.
53    pub id: Value,
54    /// The method name (`todo.add`, `status`, …).
55    pub method: String,
56    /// Method params (`null` when absent).
57    pub params: Value,
58}
59
60/// One response to encode.
61pub enum Response {
62    /// The method's payload.
63    Result {
64        /// Correlation id echoed from the request.
65        id: Value,
66        /// The result value.
67        result: Value,
68    },
69    /// A wire or dispatch failure.
70    Error {
71        /// Correlation id (`null` when the request never decoded).
72        id: Value,
73        /// One of the `-32xxx` codes.
74        code: i64,
75        /// Human/agent-readable message.
76        message: String,
77        /// Optional structured detail (transition targets, …).
78        data: Option<Value>,
79    },
80}

82/// A line that never decoded into a [`Request`].
83#[derive(Debug)]
84pub struct WireError {
85    /// The error code.
86    pub code: i64,
87    /// What went wrong.
88    pub message: String,
89}

91/// Request/response framing — the abstraction point. JSON-RPC
92/// 2.0 is the default impl; [`Server::serve`] doesn't know it.
93pub trait WireProtocol {
94    /// Decode one raw line into a request.
95    fn parse(&self, line: &str) -> Result<Request, WireError>;
96    /// Encode one response as a single line.
97    fn encode(&self, response: &Response) -> String;
98}
99
100/// JSON-RPC 2.0 — one request/response object per line.
101pub struct JsonRpc2;
102
103impl WireProtocol for JsonRpc2 {
104    fn parse(&self, line: &str) -> Result<Request, WireError> {
105        let v: Value = serde_json::from_str(line).map_err(|e| WireError {
106            code: PARSE_ERROR,
107            message: format!("parse error: {e}"),
108        })?;
109        let invalid = |message: &str| WireError {
110            code: INVALID_REQUEST,
111            message: message.to_string(),
112        };
113        let obj = v
114            .as_object()
115            .ok_or_else(|| invalid("request is not a JSON object"))?;
116        let method = obj
117            .get("method")
118            .and_then(Value::as_str)
119            .filter(|m| !m.is_empty())
120            .ok_or_else(|| invalid("missing or invalid `method`"))?
121            .to_string();
122        let id = obj
123            .get("id")
124            .filter(|id| !id.is_null())
125            .cloned()
126            .ok_or_else(|| invalid("missing `id` — notifications aren't served"))?;
127        let params = obj.get("params").cloned().unwrap_or(Value::Null);
128        Ok(Request { id, method, params })
129    }
130
131    fn encode(&self, response: &Response) -> String {
132        match response {
133            Response::Result { id, result } => {
134                json!({"jsonrpc": "2.0", "id": id, "result": result}).to_string()
135            }
136            Response::Error {
137                id,
138                code,
139                message,
140                data,
141            } => {
142                let mut e = json!({"code": code, "message": message});
143                if let Some(d) = data {
144                    e["data"] = d.clone();
145                }
146                json!({"jsonrpc": "2.0", "id": id, "error": e}).to_string()
147            }
148        }
149    }
150}

154/// One request line in, one response line out — the transport
155/// abstraction.
156pub trait ServeTransport {
157    /// The next raw request line — `Ok(None)` on clean EOF.
158    fn read_request(&mut self) -> Result<Option<String>, String>;
159    /// Write one raw response line (and flush).
160    fn write_response(&mut self, line: &str) -> Result<(), String>;
161}
162
163/// stdin/stdout — `BufRead` lines in, `writeln!` + flush out.
164pub struct StdioTransport<R, W> {
165    reader: R,
166    writer: W,
167}
168
169impl<R, W> StdioTransport<R, W> {
170    /// Wrap a reader/writer pair.
171    pub fn new(reader: R, writer: W) -> Self {
172        StdioTransport { reader, writer }
173    }
174}
175
176impl<R: BufRead, W: Write> ServeTransport for StdioTransport<R, W> {
177    fn read_request(&mut self) -> Result<Option<String>, String> {
178        let mut line = String::new();
179        let n = self
180            .reader
181            .read_line(&mut line)
182            .map_err(|e| format!("read request: {e}"))?;
183        if n == 0 {
184            return Ok(None);
185        }
186        Ok(Some(line.trim_end_matches(['\n', '\r']).to_string()))
187    }
188
189    fn write_response(&mut self, line: &str) -> Result<(), String> {
190        writeln!(self.writer, "{line}")
191            .and_then(|()| self.writer.flush())
192            .map_err(|e| format!("write response: {e}"))
193    }
194}

198/// `params` (or absent) → a typed struct — bad shape is -32602.
199fn parse_params<T: serde::de::DeserializeOwned>(v: &Value) -> Result<T, Fail> {
200    let v = if v.is_null() { json!({}) } else { v.clone() };
201    serde_json::from_value(v).map_err(|e| Fail::BadParams(format!("invalid params: {e}")))
202}

371/// `pri_min`/`pri_max` wire values take the same lenient read as
372/// `[views]` TOML — `crit` and `critical` both parse.
373fn de_opt_priority<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Option<Priority>, D::Error> {
374    Option::<String>::deserialize(d)?
375        .map(|s| s.parse().map_err(serde::de::Error::custom))
376        .transpose()
377}

379impl SpecPatch {
380    /// Fold set fields over `base` — a set field replaces, an
381    /// unset one keeps the base's value (`Option` scalars can be
382    /// set but never cleared, matching the CLI flags).
383    fn apply(&self, base: &mut ViewSpec) {
384        if let Some(p) = &self.filter {
385            let f = &mut base.filter;
386            if let Some(v) = &p.kind {
387                f.kind = v.clone();
388            }
389            if let Some(v) = &p.status {
390                f.status = v.clone();
391            }
392            if let Some(v) = p.pri_min {
393                f.pri_min = Some(v);
394            }
395            if let Some(v) = p.pri_max {
396                f.pri_max = Some(v);
397            }
398            if let Some(v) = &p.label {
399                f.label = v.clone();
400            }
401            if let Some(v) = &p.owner {
402                f.owner = v.clone();
403            }
404            if let Some(v) = p.claimed {
405                f.claimed = Some(v);
406            }
407            if let Some(v) = &p.epic {
408                f.epic = Some(v.clone());
409            }
410            if let Some(v) = &p.milestone {
411                f.milestone = Some(v.clone());
412            }
413            if let Some(v) = p.superseded {
414                f.superseded = Some(v);
415            }
416        }
417        if let Some(g) = self.group_by {
418            base.group_by = Some(g);
419        }
420        if let Some(s) = &self.sort {
421            base.sort = s.clone();
422        }
423        if let Some(n) = self.limit {
424            base.limit = Some(n);
425        }
426        if let Some(fmt) = self.format {
427            base.format = fmt;
428        }
429    }
430}

434/// Dispatch failure — [`Server::respond`] maps these to wire
435/// codes; dispatch itself knows no wire.
436#[derive(Debug)]
437enum Fail {
438    /// The method isn't served.
439    UnknownMethod,
440    /// Params didn't decode — -32602.
441    BadParams(String),
442    /// The verb failed — -32000.
443    Verb(String),
444    /// Status-machine rejection — -32001 + `data`.
445    Transition {
446        /// The item being set.
447        id: ItemId,
448        /// Current status.
449        from: Status,
450        /// Requested status.
451        to: Status,
452    },
453}
454
455impl From<String> for Fail {
456    fn from(s: String) -> Fail {
457        Fail::Verb(s)
458    }
459}

489/// The served surface — discovery flags + `--no-commit`, shared
490/// across requests. Nothing caches store state: every method
491/// re-opens the store.
492pub struct Server {
493    dir: Option<PathBuf>,
494    root: Option<PathBuf>,
495    no_commit: bool,
496}
497
498impl Server {
499    /// A server over the caller's discovery context.
500    pub fn new(dir: Option<PathBuf>, root: Option<PathBuf>, no_commit: bool) -> Self {
501        Server {
502            dir,
503            root,
504            no_commit,
505        }
506    }
507
508    /// Exclusive-locked store — mutating methods.
509    fn open(&self) -> Result<(crate::discover::Project, items::LockedStore), Fail> {
510        items::open(self.dir.as_deref(), self.root.as_deref()).map_err(Fail::Verb)
511    }
512
513    /// Shared-locked store — read methods.
514    fn open_read(&self) -> Result<(crate::discover::Project, items::LockedStore), Fail> {
515        items::open_read(self.dir.as_deref(), self.root.as_deref()).map_err(Fail::Verb)
516    }
517
518    fn commit(&self, wt: &std::path::Path, msg: &str) -> Result<bool, Fail> {
519        items::commit(wt, msg, self.no_commit).map_err(Fail::Verb)
520    }
521
522    /// The request loop — wire- and transport-agnostic: read one
523    /// request, write one response, until EOF (`Ok(0)`).
524    pub fn serve<T: ServeTransport, W: WireProtocol>(
525        &self,
526        transport: &mut T,
527        wire: &W,
528    ) -> Result<i32, String> {
529        while let Some(line) = transport.read_request()? {
530            if let Some(response) = self.handle_line(wire, &line) {
531                transport.write_response(&response)?;
532            }
533        }
534        Ok(0)
535    }
536
537    /// One raw request line → one encoded response line — `None`
538    /// for blank lines (skipped, not an error). `myco batch`
539    /// drives this same path for one-shot NDJSON runs.
540    pub fn handle_line<W: WireProtocol>(&self, wire: &W, line: &str) -> Option<String> {
541        if line.trim().is_empty() {
542            return None;
543        }
544        let response = match wire.parse(line) {
545            Ok(req) => self.respond(req),
546            Err(e) => Response::Error {
547                id: Value::Null,
548                code: e.code,
549                message: e.message,
550                data: None,
551            },
552        };
553        Some(wire.encode(&response))
554    }
555
556    fn respond(&self, req: Request) -> Response {
557        match self.dispatch(&req.method, &req.params) {
558            Ok(result) => Response::Result { id: req.id, result },
559            Err(fail) => {
560                let (code, message, data) = match fail {
561                    Fail::UnknownMethod => (
562                        METHOD_NOT_FOUND,
563                        format!("unknown method {:?}", req.method),
564                        None,
565                    ),
566                    Fail::BadParams(m) => (INVALID_PARAMS, m, None),
567                    Fail::Verb(m) => (VERB_ERROR, m, None),
568                    Fail::Transition { id, from, to } => (
569                        INVALID_TRANSITION,
570                        format!("{id}: invalid status transition {from} → {to}"),
571                        Some(json!({
572                            "from": from.token(),
573                            "to": to.token(),
574                            "valid": from
575                                .transitions()
576                                .iter()
577                                .map(|s| s.token())
578                                .collect::<Vec<_>>(),
579                        })),
580                    ),
581                };
582                Response::Error {
583                    id: req.id,
584                    code,
585                    message,
586                    data,
587                }
588            }
589        }
590    }

592    /// The verb surface — `method` + `params` → a JSON payload,
593    /// transport- and wire-agnostic.
594    fn dispatch(&self, method: &str, params: &Value) -> Result<Value, Fail> {
595        match method {
596            "initialize" => Ok(json!({
597                "protocol": 1,
598                "wire": "jsonrpc-2.0",
599                "transport": "stdio",
600                "server": {"name": "myco", "version": env!("CARGO_PKG_VERSION")},
601                "methods": METHODS,
602            })),
603            "manifest" => Ok(crate::manifest()),
604            "schema" => Ok(schema::document()),
605            "status" => {
606                let p: StatusParams = parse_params(params)?;
607                let (_proj, store) = self.open_read()?;
608                if p.prompt {
609                    status::prompt_json(&store).map_err(Fail::Verb)
610                } else {
611                    status::status_json(&store).map_err(Fail::Verb)
612                }
613            }
614            "view" => {
615                let p: ViewParams = parse_params(params)?;
616                let (proj, store) = self.open_read()?;
617                match (p.name.as_deref(), p.spec.as_ref()) {
618                    // bare — enumerate the defined views
619                    (None, None) => {
620                        let (views, warnings) =
621                            myco_ops::view::named_views(&proj.root, crate::ops::CONFIG)
622                                .map_err(Fail::Verb)?;
623                        Ok(json!({
624                            "views": views
625                                .iter()
626                                .map(|v| json!({
627                                    "name": v.name,
628                                    "origin": match v.origin {
629                                        Origin::Builtin => "builtin",
630                                        Origin::Config => "config",
631                                    },
632                                    "spec": v.spec,
633                                }))
634                                .collect::<Vec<_>>(),
635                            "warnings": warnings,
636                        }))
637                    }
638                    (name, patch) => {
639                        let (mut spec, warnings) = match name {
640                            Some(n) => {
641                                let (v, w) =
642                                    myco_ops::view::resolve(&proj.root, n, crate::ops::CONFIG)
643                                        .map_err(Fail::Verb)?;
644                                (v.spec, w)
645                            }
646                            None => (ViewSpec::default(), Vec::new()),
647                        };
648                        if let Some(patch) = patch {
649                            patch.apply(&mut spec);
650                        }
651                        spec.validate().map_err(Fail::Verb)?;
652                        let scan = digest::scan(&store).map_err(Fail::Verb)?;
653                        // the universe: top-level items only —
654                        // children ride with their parent
655                        let universe = scan.items.iter().filter(|i| !scan.archived.contains(&i.id));
656                        let groups = eval(universe, &spec);
657                        let interrupted =
658                            continuity::interrupted_set(&store, scan.live()).map_err(Fail::Verb)?;
659                        Ok(json!({
660                            "view": name,
661                            "spec": spec,
662                            "groups": groups
663                                .iter()
664                                .map(|g| json!({"label": g.label, "items": g.items}))
665                                .collect::<Vec<_>>(),
666                            "interrupted": interrupted
667                                .iter()
668                                .map(|i| i.to_string())
669                                .collect::<Vec<_>>(),
670                            "warnings": warnings,
671                        }))
672                    }
673                }
674            }
675            "next" => {
676                let p: NextParams = parse_params(params)?;
677                let (_proj, store) = self.open_read()?;
678                let mut spec = next::base_spec();
679                if let Some(patch) = &p.spec {
680                    patch.apply(&mut spec);
681                }
682                spec.limit = Some(1);
683                spec.validate().map_err(Fail::Verb)?;
684                let scan = digest::scan(&store).map_err(Fail::Verb)?;
685                Ok(json!({"item": next::pick(&scan, &spec)}))
686            }
687            "item.show" => {
688                let p: IdParams = parse_params(params)?;
689                let (_proj, store) = self.open_read()?;
690                itemcmd::show_json(&store, p.id).map_err(Fail::Verb)
691            }
692            "todo.list" | "backlog.list" => {
693                let p: ListParams = parse_params(params)?;
694                let (_proj, store) = self.open_read()?;
695                let found = todo::list_items(&store, method == "backlog.list", &p.filter())?;
696                Ok(items_json(&found))
697            }
698            "issue.list" | "proposal.list" | "plan.list" | "adr.list" => {
699                let p: ListParams = parse_params(params)?;
700                let (_proj, store) = self.open_read()?;
701                let kind = match method {
702                    "issue.list" => ItemKind::Issue,
703                    "proposal.list" => ItemKind::Proposal,
704                    "plan.list" => ItemKind::Plan,
705                    _ => ItemKind::Adr,
706                };
707                let found = items::query_files(&store, kind, p.archived, &p.filter())
708                    .map_err(Fail::Verb)?;
709                Ok(items::marked_json(&found))
710            }
711            "report" => {
712                let (proj, store) = self.open_read()?;
713                report::report_json(&store, &proj.worktree).map_err(Fail::Verb)
714            }
715            "resume" => {
716                let p: IdParams = parse_params(params)?;
717                let (_proj, store) = self.open_read()?;
718                continuity::resume_json(&store, p.id, p.last).map_err(Fail::Verb)
719            }
720            "check" => {
721                let (proj, _store) = self.open_read()?;
722                check::check_json(&proj).map_err(Fail::Verb)
723            }
724            "todo.add" => {
725                let p: TodoAddParams = parse_params(params)?;
726                let (proj, store) = self.open()?;
727                let (item, file) = todo::add_item(
728                    &store,
729                    &p.title,
730                    p.section.as_deref(),
731                    p.parent,
732                    &p.meta.into_meta(),
733                )
734                .map_err(Fail::Verb)?;
735                let committed = self.commit(&proj.worktree, &format!("chore: add {}", item.id))?;
736                Ok(json!({"item": item, "file": file, "parent": p.parent, "committed": committed}))
737            }
738            "todo.done" => {
739                let p: IdParams = parse_params(params)?;
740                let (proj, store) = self.open()?;
741                let item = todo::done_item(&store, p.id).map_err(Fail::Verb)?;
742                let committed = self.commit(&proj.worktree, &format!("chore: done {}", p.id))?;
743                Ok(json!({"item": item, "committed": committed}))
744            }
745            "todo.defer" => {
746                let p: IdParams = parse_params(params)?;
747                let (proj, store) = self.open()?;
748                let item = todo::defer_item(&store, p.id).map_err(Fail::Verb)?;
749                let committed = self.commit(&proj.worktree, &format!("chore: defer {}", p.id))?;
750                Ok(json!({"item": item, "file": "BACKLOG.md", "committed": committed}))
751            }
752            "backlog.add" => {
753                let p: BacklogAddParams = parse_params(params)?;
754                let (proj, store) = self.open()?;
755                let item =
756                    backlog::add_item(&store, &p.title, p.section.as_deref(), &p.meta.into_meta())
757                        .map_err(Fail::Verb)?;
758                let committed = self.commit(&proj.worktree, &format!("chore: add {}", item.id))?;
759                Ok(json!({"item": item, "file": "BACKLOG.md", "committed": committed}))
760            }
761            "backlog.promote" => {
762                let p: IdParams = parse_params(params)?;
763                let (proj, store) = self.open()?;
764                let item = backlog::promote_item(&store, p.id).map_err(Fail::Verb)?;
765                let committed = self.commit(&proj.worktree, &format!("chore: promote {}", p.id))?;
766                Ok(json!({"item": item, "file": "TODO.md", "committed": committed}))
767            }
768            "item.set" => {
769                let p: SetParams = parse_params(params)?;
770                let (proj, store) = self.open()?;
771                let flags = itemcmd::SetFlags {
772                    status: p.status,
773                    pri: p.pri,
774                    owner: p.owner,
775                    label: p.label,
776                    unlabel: p.unlabel,
777                    blocks: p.blocks,
778                    unblock: p.unblock,
779                    epic: p.epic,
780                    milestone: p.milestone,
781                };
782                if flags.is_noop() {
783                    return Err(Fail::Verb(
784                        "item set: nothing to change — pass at least one field".into(),
785                    ));
786                }
787                // pre-check the status step so the wire gets the
788                // typed -32001 + valid targets, not a string
789                if let Some(to) = flags.status {
790                    let Some((item, _, _)) = items::find_item(&store, p.id).map_err(Fail::Verb)?
791                    else {
792                        return Err(Fail::Verb(format!("{}: no such item", p.id)));
793                    };
794                    item.status
795                        .transition_to(to)
796                        .map_err(|_| Fail::Transition {
797                            id: p.id,
798                            from: item.status,
799                            to,
800                        })?;
801                }
802                let (item, location) =
803                    itemcmd::mutate_item(&store, p.id, &flags).map_err(Fail::Verb)?;
804                let committed = self.commit(&proj.worktree, &format!("chore: set {}", p.id))?;
805                Ok(json!({"item": item, "location": location, "committed": committed}))
806            }
807            "handoff" => {
808                let p: HandoffParams = parse_params(params)?;
809                if p.note == "-" {
810                    return Err(Fail::BadParams(
811                        "note \"-\" reads stdin — stdin is the wire; pass the note inline".into(),
812                    ));
813                }
814                let (proj, store) = self.open()?;
815                let (entry, rel) = continuity::record_handoff(
816                    &store,
817                    &proj.worktree,
818                    p.id,
819                    p.note,
820                    p.actor,
821                    p.session,
822                )
823                .map_err(Fail::Verb)?;
824                let committed = self.commit(&proj.worktree, &format!("chore: handoff {}", p.id))?;
825                Ok(
826                    json!({"id": p.id.to_string(), "path": rel, "entry": entry, "committed": committed}),
827                )
828            }
829            "interrupt" => {
830                let p: InterruptParams = parse_params(params)?;
831                let (proj, store) = self.open()?;
832                let entry =
833                    continuity::interrupt_item(&store, p.id, p.note.as_deref(), p.actor, p.session)
834                        .map_err(Fail::Verb)?;
835                let committed =
836                    self.commit(&proj.worktree, &format!("chore: interrupt {}", p.id))?;
837                Ok(
838                    json!({"id": p.id.to_string(), "status": "open", "entry": entry, "committed": committed}),
839                )
840            }
841            _ => Err(Fail::UnknownMethod),
842        }
843    }
844}

846/// `Vec<Item>` → a JSON array of item objects.
847fn items_json(items: &[myco_core::Item]) -> Value {
848    json!(items
849        .iter()
850        .map(|i| serde_json::to_value(i).unwrap_or(json!(null)))
851        .collect::<Vec<_>>())
852}

854/// The `serve` verb — stdio transport, JSON-RPC 2.0 wire, EOF
855/// exits 0.
856pub fn run(dir: Option<PathBuf>, root: Option<PathBuf>, no_commit: bool) -> Result<i32, String> {
857    let server = Server::new(dir, root, no_commit);
858    let stdin = std::io::stdin();
859    let stdout = std::io::stdout();
860    let mut transport = StdioTransport::new(stdin.lock(), stdout.lock());
861    server.serve(&mut transport, &JsonRpc2)
862}
