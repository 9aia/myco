1//! `myco workspace list|use|show|clear` — the multi-repo
2//! selector (P-016): `[workspace] repos` glob patterns in user
3//! `myco.toml`, matched against universe repo names. Logic lives
4//! in `myco_ops::workspace`; these handlers own argv →
5//! JSON/human output and the index handle resolution needs.

15fn cwd() -> Result<PathBuf, String> {

27/// The shared payload builders — cli prints them, serve returns
28/// them verbatim.
29pub fn list_json() -> Result<Value, String> {
30    let cwd = cwd()?;
31    let index = open_index()?;
32    let repos = myco_ops::universe::repos(&cwd, as_index(&index), CONFIG)?;
33    let targets: Vec<Value> = repos
34        .iter()
35        .map(|u| {
36            json!({
37                "name": u.entry.name,
38                "source": u.source.as_str(),
39                "ledger": match u.entry.ledger {
40                    myco_index::LedgerMode::Org => "org",
41                    myco_index::LedgerMode::None => "none",
42                    myco_index::LedgerMode::Local => "local",
43                },
44                "worktree": u.entry.worktree_path().map(|w| w.as_std_path().to_path_buf()),
45            })
46        })
47        .collect();
48    Ok(json!({"repos": targets}))
49}

51pub fn show_json() -> Result<Value, String> {
52    let cwd = cwd()?;
53    let globs = workspace::configured(CONFIG);
54    let index = open_index()?;
55    let resolved = workspace::resolve(&cwd, as_index(&index), CONFIG);
56    Ok(json!({
57        "globs": globs,
58        "resolved": match &resolved {
59            Ok((repos, warnings)) => json!({
60                "repos": repos.iter().map(|u| json!({
61                    "name": u.entry.name,
62                    "source": u.source.as_str(),
63                    "worktree": u.entry.worktree_path().map(|w| w.as_std_path().to_path_buf()),
64                })).collect::<Vec<_>>(),
65                "warnings": warnings,
66            }),
67            Err(e) => json!({"error": e}),
68        },
69    }))
70}

72pub fn use_json(globs: &[String]) -> Result<Value, String> {
73    if globs.is_empty() {
74        return Err(
75            "workspace use needs at least one glob — `myco workspace clear` empties".into(),
76        );
77    }
78    let cwd = cwd()?;
79    let index = open_index()?;
80    // validate now — a typo'd glob fails loudly at `use` time
81    let (_repos, warnings) = workspace::resolve_globs(&cwd, as_index(&index), CONFIG, globs)?;
82    let path = workspace::write_config(globs)?;
83    Ok(json!({
84        "config": path,
85        "repos": globs,
86        "warnings": warnings,
87    }))
88}

90pub fn clear_json() -> Result<Value, String> {
91    let path = workspace::write_config(&[])?;
92    Ok(json!({"config": path, "repos": []}))
93}

95fn print_value(ctx: &Ctx, v: &Value) {
96    if ctx.json {
97        println!("{}", serde_json::to_string_pretty(v).unwrap_or_default());
98        return;
99    }
100    match v {
101        Value::Object(map) if map.contains_key("repos") && map.contains_key("globs") => {
102            // show
103            if let Some(g) = v["globs"].as_array() {
104                println!(
105                    "globs: {}",
106                    g.iter()
107                        .filter_map(|x| x.as_str())
108                        .collect::<Vec<_>>()
109                        .join(", ")
110                );
111            } else {
112                println!("globs: (none — `myco workspace use <glob>…`)");
113            }
114            if let Some(r) = v["resolved"]["repos"].as_array() {
115                for u in r {
116                    println!(
117                        "{}\t{}",
118                        u["name"].as_str().unwrap_or("?"),
119                        u["source"].as_str().unwrap_or("")
120                    );
121                }
122            }
123            for w in v["resolved"]["warnings"].as_array().into_iter().flatten() {
124                eprintln!("warning: {}", w.as_str().unwrap_or(""));
125            }
126            if let Some(e) = v["resolved"]["error"].as_str() {
127                eprintln!("error: {e}");
128            }
129        }
130        Value::Object(map) if map.contains_key("repos") => {
131            // list / use / clear — one name per line for list;
132            // use/clear print the written file
133            if v.get("config").is_some() {
134                println!("{}", serde_json::to_string_pretty(v).unwrap_or_default());
135            } else {
136                for u in v["repos"].as_array().into_iter().flatten() {
137                    let name = u["name"].as_str().unwrap_or("?");
138                    let src = u["source"].as_str().unwrap_or("");
139                    let wt = u["worktree"]
140                        .as_str()
141                        .map(|w| format!("  {w}"))
142                        .unwrap_or_default();
143                    println!("{name}\t{src}{wt}");
144                }
145            }
146        }
147        _ => println!("{}", serde_json::to_string_pretty(v).unwrap_or_default()),
148    }
149}

151pub fn h_list(ctx: &Ctx, _a: &Args) -> Result<i32, String> {
152    let v = list_json()?;
153    print_value(ctx, &v);
154    Ok(0)
155}
156
157pub fn h_show(ctx: &Ctx, _a: &Args) -> Result<i32, String> {
158    let v = show_json()?;
159    print_value(ctx, &v);
160    Ok(0)
161}
162
163pub fn h_use(ctx: &Ctx, a: &Args) -> Result<i32, String> {
164    let globs = a.many("repos");
165    let v = use_json(&globs)?;
166    print_value(ctx, &v);
167    Ok(0)
168}
169
170pub fn h_clear(ctx: &Ctx, _a: &Args) -> Result<i32, String> {
171    let v = clear_json()?;
172    print_value(ctx, &v);
173    Ok(0)
