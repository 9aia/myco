1//! `myco index add|remove|list|rebuild` — the derived-cache
2//! registry verbs.
3//!
4//! `add` validates the repo has a `project/` worktree (same rules
5//! as discovery), registers it, and takes a first snapshot.
6//! `rebuild` re-scans via `MdStore` and re-keys the snapshot on the
7//! project-branch HEAD — freshness afterwards is the trivial
8//! `stored head == current head` check `org` does before trusting
9//! a snapshot.
10
11use std::path::{Path, PathBuf};

13use camino::Utf8PathBuf;
14use clap::Subcommand;
15use myco_index::{Index, RepoEntry, RepoSnapshot};
16use myco_index_redb::RedbIndex;
17use serde_json::json;

19use crate::{digest, discover, ops};

21#[derive(Subcommand)]
22pub enum Cmd {
23    /// Register a repo — `<path>` is a repo root (or its
24    /// `project/` worktree). Takes an initial snapshot.
25    Add {
26        /// Repo root (or project worktree) to register.
27        path: PathBuf,
28    },
29    /// Unregister a repo and drop its snapshot.
30    Remove {
31        /// Registered repo root (or project worktree).
32        path: PathBuf,
33    },
34    /// List registered repos with snapshot freshness.
35    List,
36    /// Re-scan a registered repo (all of them without `<path>`).
37    Rebuild {
38        /// Registered repo root (or project worktree).
39        path: Option<PathBuf>,
40    },
41}
42
43/// Open the default index (`$MYCO_HOME` / `~/.local/share/myco`).
44pub fn open_index() -> Result<RedbIndex, String> {
45    RedbIndex::open().map_err(|e| e.to_string())
46}
47
48/// RFC 3339 now — `scanned_at`.
49pub fn now() -> String {
50    time::OffsetDateTime::now_utc()
51        .format(&time::format_description::well_known::Rfc3339)
52        .unwrap_or_else(|_| "unknown".into())
53}
54
55/// `<path>` → the repo root + project worktree. Accepts a repo
56/// root (`<path>/project` must qualify) or the worktree itself.
57/// `--dir` discovery doesn't qualify its path, so the check is
58/// repeated here — `index add` on a plain dir must fail.
59pub fn repo_at(path: &Path) -> Result<discover::Project, String> {
60    if let Ok(p) = discover::discover(None, Some(path)) {
61        return Ok(p);
62    }
63    match discover::discover(Some(path), None) {
64        Ok(p) if discover::qualifies(&p.worktree) => Ok(p),
65        _ => Err(format!(
66            "{}: not a myco repo — needs a project/ worktree",
67            path.display()
68        )),
69    }
70}

72/// Scan a repo's worktree into a snapshot.
73pub fn scan_repo(proj: &discover::Project) -> Result<RepoSnapshot, String> {
74    let store = ops::store_at(&proj.worktree)?;
75    let scan = digest::scan(&store)?;
76    let path = Utf8PathBuf::from_path_buf(proj.root.clone())
77        .map_err(|p| format!("{}: non-utf8 path", p.display()))?;
78    Ok(RepoSnapshot {
79        path,
80        name: scan.name,
81        head_commit: myco_git::head_commit(&proj.worktree).unwrap_or_default(),
82        scanned_at: now(),
83        items: scan.items,
84        archived: scan.archived,
85    })
86}

88/// Freshness check — `None` when the repo isn't registered or its
89/// worktree is gone. Used by `index list` and `org`'s
90/// auto-rebuild.
91pub fn current_head(repo_root: &Path) -> Option<String> {
92    myco_git::head_commit(&repo_root.join(discover::DIR))
93}
94
95/// Re-scan + store the snapshot for a registered repo root.
96pub fn rebuild_one(index: &impl Index, root: &Path) -> Result<RepoSnapshot, String> {
97    let proj = repo_at(root)?;
98    let snap = scan_repo(&proj)?;
99    index.save_snapshot(&snap).map_err(|e| e.to_string())?;
100    Ok(snap)
101}
102
103fn registered_root(index: &impl Index, path: &Path) -> Result<PathBuf, String> {
104    let proj = repo_at(path)?;
105    let canon = proj
106        .root
107        .canonicalize()
108        .map_err(|e| format!("{}: {e}", proj.root.display()))?;
109    let key = Utf8PathBuf::from_path_buf(canon.clone())
110        .map_err(|p| format!("{}: non-utf8 path", p.display()))?;
111    if index
112        .repos()
113        .map_err(|e| e.to_string())?
114        .iter()
115        .any(|r| r.path == key)
116    {
117        Ok(canon)
118    } else {
119        Err(format!(
120            "{}: not registered — `myco index add` it first",
121            canon.display()
122        ))
123    }
124}

126fn snap_json(s: &RepoSnapshot) -> serde_json::Value {
127    json!({
128        "path": s.path,
129        "name": s.name,
130        "head_commit": s.head_commit,
131        "scanned_at": s.scanned_at,
132        "items": s.items.len(),
133        "archived": s.archived.len(),
134    })
135}

137/// The `index` verb group.
138pub fn run(cmd: Cmd, json_out: bool) -> Result<i32, String> {
139    let index = open_index()?;
140    match cmd {
141        Cmd::Add { path } => {
142            let proj = repo_at(&path)?;
143            let snap = scan_repo(&proj)?;
144            let entry = RepoEntry::new(snap.path.clone(), snap.name.clone());
145            index.register(&entry).map_err(|e| e.to_string())?;
146            index.save_snapshot(&snap).map_err(|e| e.to_string())?;
147            if json_out {
148                ops::print_json(&json!({"registered": entry, "snapshot": snap_json(&snap)}));
149            } else {
150                println!(
151                    "registered {} ({}) — {} items @ {}",
152                    snap.name,
153                    snap.path,
154                    snap.items.len(),
155                    &snap.head_commit[..snap.head_commit.len().min(7)],
156                );
157            }
158            Ok(0)
159        }
160        Cmd::Remove { path } => {
161            // resolve via discovery, else the raw path — a repo
162            // whose worktree vanished must stay removable
163            let root = match repo_at(&path) {
164                Ok(p) => p.root,
165                Err(_) => path.canonicalize().unwrap_or(path.clone()),
166            };
167            let root = root
168                .canonicalize()
169                .map_err(|e| format!("{}: {e}", root.display()))?;
170            let key = Utf8PathBuf::from_path_buf(root)
171                .map_err(|p| format!("{}: non-utf8 path", p.display()))?;
172            if index.remove(&key).map_err(|e| e.to_string())? {
173                if json_out {
174                    ops::print_json(&json!({"removed": key}));
175                } else {
176                    println!("removed {key}");
177                }
178                Ok(0)
179            } else {
180                Err(format!("{key}: not registered"))
181            }
182        }
183        Cmd::List => {
184            let repos = index.repos().map_err(|e| e.to_string())?;
185            let mut rows = Vec::new();
186            let mut out = Vec::new();
187            for r in &repos {
188                let snap = index.snapshot(&r.path).map_err(|e| e.to_string())?;
189                let head = current_head(r.path.as_std_path());
190                let stale = match (&snap, &head) {
191                    (Some(s), Some(h)) => s.head_commit != *h,
192                    (Some(_), None) => true, // worktree gone
193                    (None, _) => true,       // never scanned
194                };
195                out.push(json!({
196                    "path": r.path,
197                    "name": r.name,
198                    "head_commit": snap.as_ref().map(|s| s.head_commit.clone()),
199                    "scanned_at": snap.as_ref().map(|s| s.scanned_at.clone()),
200                    "items": snap.as_ref().map(|s| s.items.len()),
201                    "stale": stale,
202                }));
203                rows.push(format!(
204                    "{}\t{}\t{}\t{}",
205                    r.path,
206                    r.name,
207                    snap.as_ref()
208                        .map(|s| s.items.len().to_string())
209                        .unwrap_or_else(|| "-".into()),
210                    if stale { "stale" } else { "fresh" },
211                ));
212            }
213            if json_out {
214                ops::print_json(&json!(out));
215            } else if rows.is_empty() {
216                println!("no repos registered — `myco index add <path>`");
217            } else {
218                for r in rows {
219                    println!("{r}");
220                }
221            }
222            Ok(0)
223        }
224        Cmd::Rebuild { path } => {
225            let roots: Vec<PathBuf> = match path {
226                Some(p) => vec![registered_root(&index, &p)?],
227                None => index
228                    .repos()
229                    .map_err(|e| e.to_string())?
230                    .into_iter()
231                    .map(|r| r.path.into_std_path_buf())
232                    .collect(),
233            };
234            if roots.is_empty() {
235                if json_out {
236                    ops::print_json(&json!([]));
237                } else {
238                    println!("no repos registered — `myco index add <path>`");
239                }
240                return Ok(0);
241            }
242            let mut results = Vec::new();
243            let mut failures = Vec::new();
244            for root in &roots {
245                match rebuild_one(&index, root) {
246                    Ok(s) => {
247                        if !json_out {
248                            println!(
249                                "rebuilt {} — {} items @ {}",
250                                s.name,
251                                s.items.len(),
252                                &s.head_commit[..s.head_commit.len().min(7)],
253                            );
254                        }
255                        results.push(
256                            json!({"path": s.path, "status": "rebuilt", "snapshot": snap_json(&s)}),
257                        );
258                    }
259                    Err(e) => {
260                        failures.push(format!("{}: {e}", root.display()));
261                        results.push(
262                            json!({"path": root.to_string_lossy(), "status": "error", "error": e}),
263                        );
264                    }
265                }
266            }
267            if json_out {
268                ops::print_json(&json!(results));
269            }
270            if failures.is_empty() {
271                Ok(0)
272            } else {
273                Err(format!(
274                    "{} rebuild(s) failed: {}",
275                    failures.len(),
276                    failures.join("; ")
277                ))
278            }
279        }
280    }
281}
