2//! from `[views]` config + built-ins, or an ad-hoc spec from
3//! flags. One engine: filter → group → sort → limit → format.
4//! Read-only — the write path stays `item set`/`todo`/`issue`.

19/// The filter flags — each maps onto the same `ViewFilter`
20/// field `filter = { … }` uses. Shared by `view` (merged over
21/// the named spec's filter) and `next` (merged over its
22/// open-item default): a set flag replaces that field, unset
23/// fields keep the base spec's value.
24#[derive(Debug, Default, Args)]
25pub struct FilterFlags {
26    /// Item kinds — task|issue|proposal|adr|plan
27    /// (repeatable/comma-separated).
28    #[arg(long, value_delimiter = ',')]
29    pub(crate) kind: Vec<ItemKind>,
30    /// Statuses — open|claimed|in-progress|blocked|done|deferred
31    /// (repeatable/comma-separated).
32    #[arg(long, value_delimiter = ',')]
33    pub(crate) status: Vec<Status>,
34    /// Inclusive lower severity bound — low|med|high|crit.
35    #[arg(long)]
36    pub(crate) pri_min: Option<Priority>,
37    /// Inclusive upper severity bound.
38    #[arg(long)]
39    pub(crate) pri_max: Option<Priority>,
40    /// `#`-less label — item carries any of these
41    /// (repeatable/comma-separated).
42    #[arg(long, value_delimiter = ',')]
43    pub(crate) label: Vec<String>,
44    /// Owner actor — item's owner is any of these
45    /// (repeatable/comma-separated).
46    #[arg(long, value_delimiter = ',')]
47    pub(crate) owner: Vec<Actor>,
48    /// Items with an owner.
49    #[arg(long, conflicts_with = "unclaimed")]
50    pub(crate) claimed: bool,
51    /// Items with no owner.
52    #[arg(long)]
53    pub(crate) unclaimed: bool,
54    /// Items in this epic.
55    #[arg(long)]
56    pub(crate) epic: Option<String>,
57    /// Items in this milestone.
58    #[arg(long)]
59    pub(crate) milestone: Option<String>,
60    /// Items not superseded by another (ADRs).
61    #[arg(long, conflicts_with = "superseded")]
62    pub(crate) live_only: bool,
63    /// Items superseded by another (ADRs).
64    #[arg(long)]
65    pub(crate) superseded: bool,
66}
67
68impl FilterFlags {
69    /// Fold every set flag into `f`; unset fields keep the
70    /// base filter's value.
71    pub(crate) fn apply(&self, f: &mut ViewFilter) {
72        if !self.kind.is_empty() {
73            f.kind = self.kind.clone();
74        }
75        if !self.status.is_empty() {
76            f.status = self.status.clone();
77        }
78        if let Some(p) = self.pri_min {
79            f.pri_min = Some(p);
80        }
81        if let Some(p) = self.pri_max {
82            f.pri_max = Some(p);
83        }
84        if !self.label.is_empty() {
85            f.label = self.label.clone();
86        }
87        if !self.owner.is_empty() {
88            f.owner = self.owner.clone();
89        }
90        if self.claimed {
91            f.claimed = Some(true);
92        }
93        if self.unclaimed {
94            f.claimed = Some(false);
95        }
96        if let Some(e) = &self.epic {
97            f.epic = Some(e.clone());
98        }
99        if let Some(m) = &self.milestone {
100            f.milestone = Some(m.clone());
101        }
102        if self.live_only {
103            f.superseded = Some(false);
104        }
105        if self.superseded {
106            f.superseded = Some(true);
107        }
108    }
109
110    /// Any flag set?
111    pub(crate) fn any_set(&self) -> bool {
112        !self.kind.is_empty()
113            || !self.status.is_empty()
114            || self.pri_min.is_some()
115            || self.pri_max.is_some()
116            || !self.label.is_empty()
117            || !self.owner.is_empty()
118            || self.claimed
119            || self.unclaimed
120            || self.epic.is_some()
121            || self.milestone.is_some()
122            || self.live_only
123            || self.superseded
124    }
125}

127/// The ad-hoc spec flags — [`FilterFlags`] plus the
128/// group/sort/limit/format spec fields; with NAME they merge
129/// over the named spec (a set flag replaces that field, unset
130/// fields keep the named spec's value).
131#[derive(Debug, Default, Args)]
132pub struct ViewFlags {
133    #[command(flatten)]
134    pub(crate) filter: FilterFlags,
135    /// Group axis — status|priority|kind|owner|epic|milestone.
136    #[arg(long)]
137    group_by: Option<GroupAxis>,
138    /// Sort keys — `pri,id,-created`; `-` flips a key's default
139    /// direction.
140    #[arg(long, value_delimiter = ',')]
141    sort: Vec<SortKey>,
142    /// Keep the first N items after sorting.
143    #[arg(long, value_name = "N")]
144    limit: Option<u32>,
145    /// Renderer — list|board.
146    #[arg(long)]
147    format: Option<ViewFormat>,
148}
149
150impl ViewFlags {
151    /// Fold every set flag over `spec`; unset fields keep the
152    /// named spec's value.
153    fn apply(&self, spec: &mut ViewSpec) {
154        self.filter.apply(&mut spec.filter);
155        if let Some(g) = self.group_by {
156            spec.group_by = Some(g);
157        }
158        if !self.sort.is_empty() {
159            spec.sort = self.sort.clone();
160        }
161        if let Some(n) = self.limit {
162            spec.limit = Some(n);
163        }
164        if let Some(fmt) = self.format {
165            spec.format = fmt;
166        }
167    }
168
169    /// Any flag set? Bare `myco view` (no name, no flags) lists.
170    fn any_set(&self) -> bool {
171        self.filter.any_set()
172            || self.group_by.is_some()
173            || !self.sort.is_empty()
174            || self.limit.is_some()
175            || self.format.is_some()
176    }
177}

179/// One-line spec digest for `--list` — only set fields.
180fn spec_digest(s: &ViewSpec) -> String {
181    let f = &s.filter;
182    let mut parts = Vec::new();
183    let names = |v: &[ItemKind]| v.iter().map(|k| k.name()).collect::<Vec<_>>().join(",");
184    if !f.kind.is_empty() {
185        parts.push(format!("kind=[{}]", names(&f.kind)));
186    }
187    if !f.status.is_empty() {
188        parts.push(format!(
189            "status=[{}]",
190            f.status
191                .iter()
192                .map(|s| s.token())
193                .collect::<Vec<_>>()
194                .join(",")
195        ));
196    }
197    if let Some(p) = f.pri_min {
198        parts.push(format!("pri_min={}", p.name()));
199    }
200    if let Some(p) = f.pri_max {
201        parts.push(format!("pri_max={}", p.name()));
202    }
203    if !f.label.is_empty() {
204        parts.push(format!("label=[{}]", f.label.join(",")));
205    }
206    if !f.owner.is_empty() {
207        parts.push(format!(
208            "owner=[{}]",
209            f.owner
210                .iter()
211                .map(|a| a.as_str())
212                .collect::<Vec<_>>()
213                .join(",")
214        ));
215    }
216    if let Some(c) = f.claimed {
217        parts.push(format!("claimed={c}"));
218    }
219    if let Some(e) = &f.epic {
220        parts.push(format!("epic={e}"));
221    }
222    if let Some(m) = &f.milestone {
223        parts.push(format!("milestone={m}"));
224    }
225    if let Some(sup) = f.superseded {
226        parts.push(format!("superseded={sup}"));
227    }
228    let mut out = if parts.is_empty() {
229        "filter:{}".to_string()
230    } else {
231        format!("filter:{{{}}}", parts.join(" "))
232    };
233    if let Some(g) = s.group_by {
234        out.push_str(&format!(" group_by:{}", g.name()));
235    }
236    if !s.sort.is_empty() {
237        out.push_str(&format!(
238            " sort:{}",
239            s.sort
240                .iter()
241                .map(|k| k.to_string())
242                .collect::<Vec<_>>()
243                .join(",")
244        ));
245    }
246    if let Some(n) = s.limit {
247        out.push_str(&format!(" limit:{n}"));
248    }
249    out.push_str(&format!(" format:{}", s.format.name()));
250    out
251}

253/// The `--list` output.
254fn list_views(views: &[NamedView], json_out: bool) {
255    if json_out {
256        ops::print_json(&json!({
257            "views": views.iter().map(|v| json!({
258                "name": v.name,
259                "origin": match v.origin {
260                    Origin::Builtin => "builtin",
261                    Origin::Config => "config",
262                },
263                "spec": v.spec,
264            })).collect::<Vec<_>>(),
265        }));
266        return;
267    }
268    for v in views {
269        let origin = match v.origin {
270            Origin::Builtin => "builtin",
271            Origin::Config => "config",
272        };
273        println!("{:<10} {:<7} {}", v.name, origin, spec_digest(&v.spec));
274    }
275}

277/// `list` — compact lines; `<label> (n)` headers when the spec
278/// groups; children ride indented under their parent.
279pub(crate) fn render_list(groups: &[ViewGroup], interrupted: &BTreeSet<ItemId>) -> String {
280    fn push_item(
281        out: &mut String,
282        item: &Item,
283        depth: usize,
284        base: &str,
285        interrupted: &BTreeSet<ItemId>,
286    ) {
287        out.push_str(base);
288        out.push_str(&"  ".repeat(depth));
289        out.push_str(&items::compact(item));
290        if interrupted.contains(&item.id) {
291            out.push_str(" ⚡");
292        }
293        out.push('\n');
294        for c in &item.children {
295            push_item(out, c, depth + 1, base, interrupted);
296        }
297    }
298    let mut out = String::new();
299    for g in groups {
300        if g.items.is_empty() {
301            continue;
302        }
303        if let Some(label) = &g.label {
304            out.push_str(&format!("{label} ({})\n", g.items.len()));
305        }
306        let base = if g.label.is_some() { "  " } else { "" };
307        for i in &g.items {
308            push_item(&mut out, i, 0, base, interrupted);
309        }
310    }
311    out.trim_end_matches('\n').to_string()
312}

314/// `board` — a status-grouped spec earns the lane chrome
315/// (capitalized Backlog | Open | … lanes via
316/// [`board::status_cards`]); any other grouping renders generic
317/// columns headed by the group labels.
318pub(crate) fn render_board(
319    spec: &ViewSpec,
320    groups: &[ViewGroup],
321    interrupted: &BTreeSet<ItemId>,
322    md: bool,
323    all: bool,
324) -> String {
325    if board::is_status_board(spec) {
326        let (headers, cards) = board::status_cards(groups, None, interrupted);
327        return board::render_columns(&headers, &cards, md, board::Fit::terminal(all));
328    }
329    let headers: Vec<String> = groups
330        .iter()
331        .map(|g| g.label.clone().unwrap_or_default())
332        .collect();
333    let columns: Vec<Vec<String>> = groups
334        .iter()
335        .map(|g| {
336            g.items
337                .iter()
338                .map(|i| board::card(i, None, interrupted.contains(&i.id)))
339                .collect()
340        })
341        .collect();
342    board::render_columns(&headers, &columns, md, board::Fit::terminal(all))
343}

356/// Print resolution warnings to stderr — a config view shadowing
357/// a built-in preset warns on every resolution.
358pub(crate) fn warn(warnings: &[String]) {
359    for w in warnings {
360        eprintln!("myco: warning: {w}");
361    }
362}

364/// Pick the renderer `spec.format` names — the shared tail of
365/// every eval-driven read.
366pub(crate) fn render_groups(
367    spec: &ViewSpec,
368    groups: &[ViewGroup],
369    interrupted: &BTreeSet<ItemId>,
370    md: bool,
371    all: bool,
372) -> String {
373    match spec.format {
374        ViewFormat::Board => render_board(spec, groups, interrupted, md, all),
375        ViewFormat::List => render_list(groups, interrupted),
376    }
377}

379/// The `--json` payload — `{view, spec, groups, interrupted}`.
380/// A status-axis board takes the five-lane columns map
381/// ([`board::groups_json`]) instead — the shape `myco board
382/// --json` always emitted.
383pub(crate) fn view_json(
384    name: Option<&str>,
385    spec: &ViewSpec,
386    groups: &[ViewGroup],
387    interrupted: &BTreeSet<ItemId>,
388) -> serde_json::Value {
389    json!({
390        "view": name,
391        "spec": spec,
392        "groups": groups.iter().map(|g| json!({
393            "label": g.label,
394            "items": g.items,
395        })).collect::<Vec<_>>(),
396        "interrupted": interrupted.iter().map(|i| i.to_string()).collect::<Vec<_>>(),
397    })
398}

400/// The `view` verb — `NAME` resolves builtin presets then config
401/// views; no NAME builds the spec from flags alone; bare `myco
402/// view` (or `--list`) enumerates the defined views.
403pub fn run(
404    name: Option<String>,
405    flags: ViewFlags,
406    out: Out,
407    dir: Option<PathBuf>,
408    root: Option<PathBuf>,
409    json_out: bool,
410) -> Result<i32, String> {
411    let (proj, store) = items::open_read(dir.as_deref(), root.as_deref())?;
412
413    if out.list || (name.is_none() && !flags.any_set()) {
414        let (views, _) = myco_ops::view::named_views_lenient(&proj.root, ops::CONFIG)?;
415        list_views(&views, json_out);
416        return Ok(0);
417    }
418
419    let (mut spec, warnings) = match &name {
420        Some(n) => {
421            let (v, w) = myco_ops::view::resolve_lenient(&proj.root, n, ops::CONFIG)?;
422            (v.spec, w)
423        }
424        None => (ViewSpec::default(), Vec::new()),
425    };
426    flags.apply(&mut spec);
427    spec.validate()?;
428    warn(&warnings);
429
430    // the universe: top-level items only — children ride with
431    // their parent (they can't match a filter on their own)
432    let scan = digest::scan(&store)?;
433    let universe = scan.items.iter().filter(|i| !scan.archived.contains(&i.id));
434    let groups = eval(universe, &spec);
435    let interrupted = continuity::interrupted_set(&store, scan.live())?;
436
437    if json_out {
438        let payload = if board::is_status_board(&spec) {
439            board::groups_json(&groups, &interrupted)
440        } else {
441            view_json(name.as_deref(), &spec, &groups, &interrupted)
442        };
443        ops::print_json(&payload);
444        return Ok(0);
445    }
446
447    let rendered = render_groups(&spec, &groups, &interrupted, out.md, out.all);
448    if groups.iter().all(|g| g.items.is_empty()) {
449        println!("{}: empty view — no items match", scan.name);
450    } else {
451        ops::print_paged(&rendered);
452    }
453    Ok(0)
454}

458mod tests {
459    use super::*;
460
461    #[test]
462    fn repo_config_shadow_redefines_the_named_board() {
463        // `[views.board]` in the repo's myco.toml replaces the
464        // preset `myco view board` resolves — the spec-level proof
465        // that config shadowing reaches the named view, and that a
466        // non-status spec drops the lane chrome.
467        let dir = tempfile::tempdir().unwrap();
468        std::fs::write(
469            dir.path().join("myco.toml"),
470            "[views.board]\nfilter = { kind = [\"issue\"] }\n",
471        )
472        .unwrap();
473        temp_env::with_vars(
474            [
475                ("MYCO_CONFIG", None::<&str>),
476                (
477                    "HOME",
478                    Some(dir.path().join("nohome").to_string_lossy().as_ref()),
479                ),
480                ("XDG_CONFIG_HOME", None::<&str>),
481            ],
482            || {
483                let (v, warnings) =
484                    myco_ops::view::resolve_lenient(dir.path(), "board", ops::CONFIG).unwrap();
485                assert_eq!(warnings.len(), 1);
486                let spec = v.spec;
487                assert_eq!(spec.filter.kind, vec![ItemKind::Issue]);
488                // the shadowed spec isn't a status-axis board →
489                // the generic renderers/json, not the lane chrome
490                assert!(!board::is_status_board(&spec));
491                // flag merge still applies over the shadowed spec
492                let mut scoped = spec.clone();
493                ViewFlags {
494                    filter: FilterFlags {
495                        epic: Some("v2".into()),
496                        ..Default::default()
497                    },
498                    ..Default::default()
499                }
500                .apply(&mut scoped);
501                assert_eq!(scoped.filter.epic.as_deref(), Some("v2"));
502                // and eval honours the shadow — tasks drop out of
503                // every group, so the rendered board changes
504                let items = vec![
505                    Item::new(ItemId::new(ItemKind::Task, 1), "t"),
506                    Item::new(ItemId::new(ItemKind::Issue, 1), "i"),
507                ];
508                let ids: Vec<String> = eval(&items, &spec)
509                    .iter()
510                    .flat_map(|g| g.items.iter().map(|i| i.id.to_string()))
511                    .collect();
512                assert_eq!(ids, ["I-001"]);
513            },
514        );
515    }
