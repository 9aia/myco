1//! `myco handoff` / `myco resume` — the continuity pair. At
2//! session end `handoff` appends a `## <today> · <actor>` entry
3//! plus note to `handoffs/<ID>.md` (append-only, one file per
4//! item, uniform across kinds). At session start `resume` emits
5//! the minimal restart context — the item, its `blocks:` dep
6//! titles, the latest handoff entries, the open checklist — sized
7//! for a prompt, the single-item counterpart of `status --prompt`.
8//!
9//! sepia owns session lifecycle; it calls these verbs from hooks.
10//! `--session` is provenance only: a sepia session id,
11//! `cli:<pid>`, whatever — myco records it, never interprets it.
12

21/// `--actor` fallback — the local CLI wrote it.
22const DEFAULT_ACTOR: &str = "cli";
23
24/// Handoff entries shown by default (`--last` overrides).
25const DEFAULT_LAST: usize = 10;
26
27/// `--note` inline, or `-` for stdin.
28fn read_note(note: &str) -> Result<String, String> {
29    let note = if note == "-" {
30        use std::io::Read;
31        let mut s = String::new();
32        std::io::stdin()
33            .read_to_string(&mut s)
34            .map_err(|e| format!("read --note from stdin: {e}"))?;
35        s
36    } else {
37        note.to_string()
38    };
39    let note = note.trim().to_string();
40    if note.is_empty() {
41        return Err("handoff: --note is empty".into());
42    }
43    Ok(note)
44}

46/// The `handoff` verb — the item must exist somewhere in the
47/// ledger (any kind, live or archived).
48#[allow(clippy::too_many_arguments)]
49pub fn handoff(
50    id: ItemId,
51    note: &str,
52    actor: Option<Actor>,
53    session: Option<String>,
54    dir: Option<PathBuf>,
55    root: Option<PathBuf>,
56    json_out: bool,
57    no_commit: bool,
58) -> Result<i32, String> {
59    let (proj, store) = items::open(dir.as_deref(), root.as_deref())?;
60    if items::find_item(&store, id)?.is_none() {
61        return Err(format!("{id}: no such item — handoffs attach to items"));
62    }
63    if let Some(s) = &session {
64        if !is_valid_session(s) {
65            return Err(format!(
66                "invalid --session {s:?} — no empty, no `·`, no newlines"
67            ));
68        }
69    }
70    let entry = HandoffEntry {
71        date: items::today(),
72        actor: match actor {
73            Some(a) => a,
74            None => Actor::new(DEFAULT_ACTOR).map_err(|e| e.to_string())?,
75        },
76        session,
77        note: read_note(note)?,
78    };
79    store
80        .append_handoff(id, &entry)
81        .map_err(|e| e.to_string())?;
82    // the file lands wherever it already lived
83    let live = format!("handoffs/{id}.md");
84    let rel = if proj.worktree.join(&live).exists() {
85        live
86    } else {
87        format!("handoffs/archive/{id}.md")
88    };
89    let committed = items::commit(&proj.worktree, &format!("chore: handoff {id}"), no_commit)?;
90    items::report(
91        json!({"id": id.to_string(), "path": rel, "entry": entry}),
92        committed,
93        json_out,
94        &format!("handoff {id} → {rel}"),
95    )
96}

98/// Unchecked `- [ ]` checkbox items in a markdown body — the open
99/// checklist `resume` surfaces (issue bodies, plan phases).
100fn open_checklist(body: &str) -> Vec<String> {
101    body.lines()
102        .filter_map(|l| {
103            let t = l.trim_start();
104            for m in ["- [ ]", "* [ ]", "+ [ ]"] {
105                if let Some(rest) = t.strip_prefix(m) {
106                    if rest.is_empty() || rest.starts_with(char::is_whitespace) {
107                        return Some(rest.trim().to_string());
108                    }
109                }
110            }
111            None
112        })
113        .collect()
114}

116/// The `resume` verb — `--json` is
117/// `{item, location, blocks_titles, handoffs[], open_checklist}`;
118/// human output is a compact prompt-ready block.
119pub fn resume(
120    id: ItemId,
121    last: Option<usize>,
122    dir: Option<PathBuf>,
123    root: Option<PathBuf>,
124    json_out: bool,
125) -> Result<i32, String> {
126    let (_proj, store) = items::open(dir.as_deref(), root.as_deref())?;
127    let Some((item, location, body)) = items::find_item(&store, id)? else {
128        return Err(format!("{id}: no such item"));
129    };
130    let mut blocks_titles = Vec::new();
131    for b in &item.blocks {
132        let title = items::find_item(&store, *b)?.map(|(i, _, _)| i.title);
133        blocks_titles.push(json!({"id": b.to_string(), "title": title}));
134    }
135    let entries = store.read_handoffs(id).map_err(|e| e.to_string())?;
136    let total = entries.len();
137    let n = last.unwrap_or(DEFAULT_LAST);
138    let shown: Vec<&HandoffEntry> = entries
139        .iter()
140        .rev()
141        .take(n)
142        .collect::<Vec<_>>()
143        .into_iter()
144        .rev()
145        .collect();
146    let checklist = body.as_deref().map(open_checklist).unwrap_or_default();
147    if json_out {
148        ops::print_json(&json!({
149            "item": item,
150            "location": location,
151            "blocks_titles": blocks_titles,
152            "handoffs": shown,
153            "handoffs_total": total,
154            "open_checklist": checklist,
155        }));
156        return Ok(0);
157    }
158    // prompt-ready block — compact item line first; sections only
159    // when they carry context
160    println!("{}", items::compact(&item));
161    println!("in: {location}");
162    if !blocks_titles.is_empty() {
163        println!("blocks:");
164        for b in &blocks_titles {
165            println!(
166                "  {} {}",
167                b["id"].as_str().unwrap_or_default(),
168                b["title"].as_str().unwrap_or("(missing)")
169            );
170        }
171    }
172    if !shown.is_empty() {
173        println!("handoffs:");
174        for e in &shown {
175            let mut head = format!("{} · {}", e.date, e.actor);
176            if let Some(s) = &e.session {
177                head.push_str(&format!(" · session:{s}"));
178            }
179            println!("{head}");
180            for line in e.note.lines() {
181                println!("  {line}");
182            }
183            println!();
184        }
185        if total > shown.len() {
186            println!("… +{} earlier entries", total - shown.len());
187        }
188    }
189    if !checklist.is_empty() {
190        println!("open checklist:");
191        for c in &checklist {
192            println!("  - [ ] {c}");
193        }
194    }
195    Ok(0)
196}

199#[allow(clippy::unwrap_used)]
200mod tests {

203    #[test]
204    fn checklist_extraction() {
205        let body = "## Checklist\n\n- [ ] do a\n  - [x] done\n* [ ] star\n+ [ ] plus\n- [ ]not-a-box\n- plain\n\n- [ ]\n";
206        assert_eq!(open_checklist(body), vec!["do a", "star", "plus", ""]);
207        assert!(open_checklist("no items\n").is_empty());
