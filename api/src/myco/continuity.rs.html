1//! `myco handoff` / `myco resume` / `myco interrupt` — the
2//! continuity trio. At session end `handoff` appends a
3//! `## <today> · <actor>` entry plus note to `handoffs/<ID>.md`
4//! (append-only, one file per item, uniform across kinds). At
5//! session start `resume` emits the minimal restart context — the
6//! item, its `blocks:` dep titles, the latest handoff entries,
7//! the open checklist — sized for a prompt, the single-item
8//! counterpart of `status --prompt`. `interrupt` is the abnormal
9//! end: a work-state item (claimed/in-progress/blocked) reverts
10//! to `open` and gets an `interrupted` handoff in one commit.
11//!
12//! sepia owns session lifecycle; it calls these verbs from hooks.
13//! `--session` is provenance only: a sepia session id,
14//! `cli:<pid>`, whatever — myco records it, never interprets it.
15

25/// `--actor` fallback — the local CLI wrote it.
26const DEFAULT_ACTOR: &str = "cli";
27
28/// Handoff entries shown by default (`--last` overrides).
29const DEFAULT_LAST: usize = 10;
30
31/// The note prefix `interrupt` writes — also the derivation rule:
32/// an item "was interrupted" when its latest handoff note starts
33/// with this word. No dedicated status — the event lives in the
34/// append-only log; status/board/report surfaces derive the flag.
35pub const INTERRUPT_NOTE: &str = "interrupted";
36
37/// Did the item's latest handoff mark it interrupted?
38pub fn is_interrupted(store: &impl Store, id: ItemId) -> bool {
39    store
40        .read_handoffs(id)
41        .ok()
42        .and_then(|entries| entries.last().map(|e| e.note.clone()))
43        .is_some_and(|n| n.trim_start().starts_with(INTERRUPT_NOTE))
44}
45
46/// The interrupted subset of `items` — for status/board markers.
47pub fn interrupted_set<'a>(
48    store: &impl Store,
49    items: impl Iterator<Item = &'a myco_core::Item>,
50) -> BTreeSet<ItemId> {
51    items
52        .filter(|i| is_interrupted(store, i.id))
53        .map(|i| i.id)
54        .collect()
55}

57/// `--note` inline, or `-` for stdin.
58fn read_note(note: &str) -> Result<String, String> {
59    let note = if note == "-" {
60        use std::io::Read;
61        let mut s = String::new();
62        std::io::stdin()
63            .read_to_string(&mut s)
64            .map_err(|e| format!("read --note from stdin: {e}"))?;
65        s
66    } else {
67        note.to_string()
68    };
69    let note = note.trim().to_string();
70    if note.is_empty() {
71        return Err("handoff: --note is empty".into());
72    }
73    Ok(note)
74}

76/// The `handoff` verb — the item must exist somewhere in the
77/// ledger (any kind, live or archived).
78#[allow(clippy::too_many_arguments)]
79pub fn handoff(
80    id: ItemId,
81    note: &str,
82    actor: Option<Actor>,
83    session: Option<String>,
84    dir: Option<PathBuf>,
85    root: Option<PathBuf>,
86    json_out: bool,
87    no_commit: bool,
88) -> Result<i32, String> {
89    let (proj, store) = items::open(dir.as_deref(), root.as_deref())?;
90    if items::find_item(&store, id)?.is_none() {
91        return Err(format!("{id}: no such item — handoffs attach to items"));
92    }
93    if let Some(s) = &session {
94        if !is_valid_session(s) {
95            return Err(format!(
96                "invalid --session {s:?} — no empty, no `·`, no newlines"
97            ));
98        }
99    }
100    let entry = HandoffEntry {
101        date: items::today(),
102        actor: match actor {
103            Some(a) => a,
104            None => Actor::new(DEFAULT_ACTOR).map_err(|e| e.to_string())?,
105        },
106        session,
107        note: read_note(note)?,
108    };
109    store
110        .append_handoff(id, &entry)
111        .map_err(|e| e.to_string())?;
112    // the file lands wherever it already lived
113    let live = format!("handoffs/{id}.md");
114    let rel = if proj.worktree.join(&live).exists() {
115        live
116    } else {
117        format!("handoffs/archive/{id}.md")
118    };
119    let committed = items::commit(&proj.worktree, &format!("chore: handoff {id}"), no_commit)?;
120    items::report(
121        json!({"id": id.to_string(), "path": rel, "entry": entry}),
122        committed,
123        json_out,
124        &format!("handoff {id} → {rel}"),
125    )
126}

128/// The `interrupt` verb — a session ended with work still
129/// claimed/in-progress/blocked. Reverts the item to `open` and
130/// appends an `interrupted` handoff entry in one atomic commit.
131/// Called by whoever owns session lifecycle (sepia hooks, CLI
132/// session-end hooks) — myco never detects sessions itself.
133#[allow(clippy::too_many_arguments)]
134pub fn interrupt(
135    id: ItemId,
136    note: Option<&str>,
137    actor: Option<Actor>,
138    session: Option<String>,
139    dir: Option<PathBuf>,
140    root: Option<PathBuf>,
141    json_out: bool,
142    no_commit: bool,
143) -> Result<i32, String> {
144    let (proj, store) = items::open(dir.as_deref(), root.as_deref())?;
145    let Some((item, _, _)) = items::find_item(&store, id)? else {
146        return Err(format!("{id}: no such item"));
147    };
148    match item.status {
149        Status::Claimed | Status::InProgress | Status::Blocked => {}
150        s => {
151            return Err(format!(
152                "{id}: status is {s} — nothing to interrupt \
153                 (only claimed/in-progress/blocked revert to open)"
154            ));
155        }
156    }
157    if let Some(s) = &session {
158        if !is_valid_session(s) {
159            return Err(format!(
160                "invalid --session {s:?} — no empty, no `·`, no newlines"
161            ));
162        }
163    }
164    // revert through the shared mutation path (relocation rules
165    // included), then the marker handoff — one commit covers both
166    crate::itemcmd::mutate_item(&store, id, &crate::itemcmd::SetFlags::status(Status::Open))?;
167    let note = match note.map(str::trim).filter(|n| !n.is_empty()) {
168        Some(extra) => format!("{INTERRUPT_NOTE} — {extra}"),
169        None => INTERRUPT_NOTE.to_string(),
170    };
171    let entry = HandoffEntry {
172        date: items::today(),
173        actor: match actor {
174            Some(a) => a,
175            None => Actor::new(DEFAULT_ACTOR).map_err(|e| e.to_string())?,
176        },
177        session,
178        note,
179    };
180    store
181        .append_handoff(id, &entry)
182        .map_err(|e| e.to_string())?;
183    let committed = items::commit(&proj.worktree, &format!("chore: interrupt {id}"), no_commit)?;
184    items::report(
185        json!({"id": id.to_string(), "status": "open", "entry": entry}),
186        committed,
187        json_out,
188        &format!("interrupted {id} → open"),
189    )
190}

192/// Unchecked `- [ ]` checkbox items in a markdown body — the open
193/// checklist `resume` surfaces (issue bodies, plan phases).
194fn open_checklist(body: &str) -> Vec<String> {
195    body.lines()
196        .filter_map(|l| {
197            let t = l.trim_start();
198            for m in ["- [ ]", "* [ ]", "+ [ ]"] {
199                if let Some(rest) = t.strip_prefix(m) {
200                    if rest.is_empty() || rest.starts_with(char::is_whitespace) {
201                        return Some(rest.trim().to_string());
202                    }
203                }
204            }
205            None
206        })
207        .collect()
208}

210/// The `resume` verb — `--json` is
211/// `{item, location, blocks_titles, handoffs[], open_checklist}`;
212/// human output is a compact prompt-ready block.
213pub fn resume(
214    id: ItemId,
215    last: Option<usize>,
216    dir: Option<PathBuf>,
217    root: Option<PathBuf>,
218    json_out: bool,
219) -> Result<i32, String> {
220    let (_proj, store) = items::open(dir.as_deref(), root.as_deref())?;
221    let Some((item, location, body)) = items::find_item(&store, id)? else {
222        return Err(format!("{id}: no such item"));
223    };
224    let mut blocks_titles = Vec::new();
225    for b in &item.blocks {
226        let title = items::find_item(&store, *b)?.map(|(i, _, _)| i.title);
227        blocks_titles.push(json!({"id": b.to_string(), "title": title}));
228    }
229    let entries = store.read_handoffs(id).map_err(|e| e.to_string())?;
230    let total = entries.len();
231    let n = last.unwrap_or(DEFAULT_LAST);
232    let shown: Vec<&HandoffEntry> = entries
233        .iter()
234        .rev()
235        .take(n)
236        .collect::<Vec<_>>()
237        .into_iter()
238        .rev()
239        .collect();
240    let checklist = body.as_deref().map(open_checklist).unwrap_or_default();
241    if json_out {
242        ops::print_json(&json!({
243            "item": item,
244            "location": location,
245            "blocks_titles": blocks_titles,
246            "handoffs": shown,
247            "handoffs_total": total,
248            "open_checklist": checklist,
249        }));
250        return Ok(0);
251    }
252    // prompt-ready block — compact item line first; sections only
253    // when they carry context
254    println!("{}", items::compact(&item));
255    println!("in: {location}");
256    if !blocks_titles.is_empty() {
257        println!("blocks:");
258        for b in &blocks_titles {
259            println!(
260                "  {} {}",
261                b["id"].as_str().unwrap_or_default(),
262                b["title"].as_str().unwrap_or("(missing)")
263            );
264        }
265    }
266    if !shown.is_empty() {
267        println!("handoffs:");
268        for e in &shown {
269            let mut head = format!("{} · {}", e.date, e.actor);
270            if let Some(s) = &e.session {
271                head.push_str(&format!(" · session:{s}"));
272            }
273            println!("{head}");
274            for line in e.note.lines() {
275                println!("  {line}");
276            }
277            println!();
278        }
279        if total > shown.len() {
280            println!("… +{} earlier entries", total - shown.len());
281        }
282    }
283    if !checklist.is_empty() {
284        println!("open checklist:");
285        for c in &checklist {
286            println!("  - [ ] {c}");
287        }
288    }
289    Ok(0)
290}

298    fn checklist_extraction() {
