1//! links — the md-grammar half of the link index (P-005).
2//!
3//! [`extract_file`] classifies a `project/` file by path and
4//! produces its `refs::extract` edges; `index add`/`rebuild` feed
5//! it to `myco_ops::links::sync`. Kept in the composition root
6//! because it's the file spec speaking — ops stays grammar-free,
7//! and phase-3 consumers (`item show`, `serve`) can share this
8//! same read side.
9
10use camino::Utf8Path;
11use myco_core::{refs, Edge, Item, ItemId, ItemKind};
12use myco_store_md::{
13    item_dir, parse_backlog, parse_handoff_file, parse_handoff_filename, parse_item_file,
14    parse_todo, HANDOFFS_DIR,
15};
16
17/// The edges one `project/` file declares — classified by the md
18/// grammar:
19///
20/// - `TODO.md` / `BACKLOG.md` — one extract per task item
21///   (frontmatter `blocks:` + title mentions), children included.
22/// - `handoffs/[archive/]<ID>.md` — the filename `handoff`
23///   self-edge plus mentions inside entry notes.
24/// - `<dir>/[archive/]<slug>.md` — frontmatter `blocks:`/
25///   `supersedes:`/`superseded_by:` + body mentions.
26/// - anything else (`index.md`, notes) — zero edges; the file is
27///   still fingerprinted so `sweep` tracks it.
28pub fn extract_file(rel: &Utf8Path, src: &str) -> Result<Vec<Edge>, String> {
29    let mut edges = Vec::new();
30    match rel.as_str() {
31        "TODO.md" => {
32            let doc = parse_todo(src).map_err(|e| format!("{rel}: {e}"))?;
33            for i in doc.items() {
34                push_item(&mut edges, i);
35            }
36        }
37        "BACKLOG.md" => {
38            let doc = parse_backlog(src).map_err(|e| format!("{rel}: {e}"))?;
39            for i in doc.items() {
40                push_item(&mut edges, i);
41            }
42        }
43        _ => {
44            if let Some(id) = handoff_id(rel) {
45                edges.push(Edge::handoff(id));
46                let (_, entries) = parse_handoff_file(src, rel.as_str())
47                    .map_err(|e| format!("{rel}: {e}"))?;
48                let notes = entries
49                    .iter()
50                    .map(|e| e.note.as_str())
51                    .collect::<Vec<_>>()
52                    .join("\n");
53                edges.extend(refs::extract(&notes, None, id));
54            } else if let Some(kind) = item_kind(rel) {
55                let f = parse_item_file(src, kind, rel.as_str())
56                    .map_err(|e| format!("{rel}: {e}"))?;
57                edges.extend(refs::extract(&f.body, Some(&f.meta), f.meta.id));
58            }
59        }
60    }
61    Ok(edges)
62}

64/// One ledger item's edges — the whole subtree, since children
65/// are full items.
66fn push_item(edges: &mut Vec<Edge>, item: &Item) {
67    for i in item.subtree() {
68        edges.extend(refs::extract(&i.title, Some(i), i.id));
69    }
70}

72/// `handoffs/[archive/]<ID>.md` → the item id, when the file is
73/// under `handoffs/` and its name parses as an id.
74fn handoff_id(rel: &Utf8Path) -> Option<ItemId> {
75    if rel.components().next()?.as_str() != HANDOFFS_DIR {
76        return None;
77    }
78    parse_handoff_filename(rel.file_name()?)
79}
80
81/// The item kind a path's first component names — covers both
82/// `<dir>/x.md` and `<dir>/archive/x.md`.
83fn item_kind(rel: &Utf8Path) -> Option<ItemKind> {
84    let dir = rel.components().next()?.as_str();
85    [
86        ItemKind::Issue,
87        ItemKind::Proposal,
88        ItemKind::Plan,
89        ItemKind::Adr,
90    ]
91    .into_iter()
92    .find(|k| item_dir(*k) == Some(dir))
93}

95#[cfg(test)]
96#[allow(clippy::unwrap_used)]
97mod tests {
98    use super::*;
99    use myco_core::EdgeKind;
100
101    fn tid(n: u32) -> ItemId {
102        ItemId::new(ItemKind::Task, n)
103    }
104
105    #[test]
106    fn todo_items_contribute_blocks_and_mentions() {
107        let src = "# TODO — x\n\n- [ ] T-001 alpha\n- [ ] T-002 beta, see T-001 · blocks:T-003\n- [ ] T-003 gamma\n";
108        let edges = extract_file(Utf8Path::new("TODO.md"), src).unwrap();
109        assert!(
110            edges.contains(&Edge::new(tid(2), tid(1), EdgeKind::References)),
111            "{edges:?}"
112        );
113        assert!(
114            edges.contains(&Edge::new(tid(2), tid(3), EdgeKind::Blocks)),
115            "{edges:?}"
116        );
117    }
118
119    #[test]
120    fn handoff_file_is_a_self_edge_plus_mentions() {
121        let src = "+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\n## 2026-10-29 · devin\nDone: x. Next: I-003.\n";
122        let edges = extract_file(Utf8Path::new("handoffs/T-001.md"), src).unwrap();
123        assert!(edges.contains(&Edge::handoff(tid(1))), "{edges:?}");
124        assert!(
125            edges.contains(&Edge::new(
126                tid(1),
127                ItemId::new(ItemKind::Issue, 3),
128                EdgeKind::References
129            )),
130            "{edges:?}"
131        );
132    }
133
134    #[test]
135    fn item_file_edges_come_from_meta_and_body() {
136        let src = "+++\nid = \"I-003\"\ntitle = \"leak\"\nstatus = \"open\"\ncreated = \"2026-01-01\"\n+++\n\n# I-003 leak\n\nRelated to T-007.\n";
137        let edges = extract_file(Utf8Path::new("issues/issue-0003-leak.md"), src).unwrap();
138        assert!(
139            edges.contains(&Edge::new(
140                ItemId::new(ItemKind::Issue, 3),
141                tid(7),
142                EdgeKind::References
143            )),
144            "{edges:?}"
145        );
146        // archive scope classifies the same
147        let edges =
148            extract_file(Utf8Path::new("issues/archive/issue-0003-leak.md"), src).unwrap();
149        assert!(!edges.is_empty());
150    }
151
152    #[test]
153    fn unrecognized_files_yield_no_edges() {
154        assert!(extract_file(Utf8Path::new("index.md"), "# index\nT-001").is_ok());
155        let edges = extract_file(Utf8Path::new("index.md"), "# index\nT-001").unwrap();
156        assert!(edges.is_empty());
157        // a T-NNN filename outside handoffs/ is not a handoff edge
158        let edges = extract_file(Utf8Path::new("notes/T-001.md"), "hi").unwrap();
159        assert!(edges.is_empty());
160    }
161}
