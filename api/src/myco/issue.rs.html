1//! `myco issue new|list|resolve|archive|comment` — one md file
2//! per issue under `issues/` (archived files in `issues/archive/`).
3
4use std::path::PathBuf;
5
6use clap::Subcommand;
7use myco_core::{ItemId, ItemKind, Status};
8use myco_store::Store;
9use serde_json::json;
10
11use crate::items;
12use crate::templates;
13
14#[derive(Subcommand)]
15pub enum Cmd {
16    /// File an issue — allocs an I-id, writes
17    /// `issues/issue-NNNN-<slug>.md` with frontmatter + template.
18    New {
19        /// One-line title (drives the filename slug).
20        title: String,
21        #[command(flatten)]
22        body: items::BodyFlags,
23        #[command(flatten)]
24        meta: items::MetaFlags,
25    },
26    /// List issues (live files; `--archived` includes
27    /// `issues/archive/`).
28    List {
29        /// Include archived issues.
30        #[arg(long)]
31        archived: bool,
32        #[command(flatten)]
33        list: items::ListFlags,
34    },
35    /// Close an issue — walks the state machine to `done`;
36    /// `--resolution` appends text under `## Resolution`.
37    Resolve {
38        /// Issue id (`I-NNN`).
39        id: ItemId,
40        /// Resolution text appended to the body.
41        #[arg(long)]
42        resolution: Option<String>,
43    },
44    /// Shelve a done issue — moves the file to `issues/archive/`.
45    Archive {
46        /// Issue id (`I-NNN`).
47        id: ItemId,
48    },
49    /// Append a dated note under `## Comments` (created when
50    /// missing).
51    Comment {
52        /// Issue id (`I-NNN`).
53        id: ItemId,
54        /// Comment text.
55        text: String,
56    },
57}
58
59/// The `issue` verb group.
60pub fn run(
61    cmd: Cmd,
62    dir: Option<PathBuf>,
63    root: Option<PathBuf>,
64    json_out: bool,
65    no_commit: bool,
66) -> Result<i32, String> {
67    let (proj, store) = items::open(dir.as_deref(), root.as_deref())?;
68    match cmd {
69        Cmd::New { title, body, meta } => items::new_file(
70            &proj,
71            &store,
72            &items::NewItem {
73                kind: ItemKind::Issue,
74                title,
75                body: body.read()?.as_deref(),
76                meta: &meta.to_meta(),
77                template: templates::issue,
78                word: "issue",
79                json_out,
80                no_commit,
81            },
82        ),
83        Cmd::List { archived, list } => {
84            items::list_files(&store, ItemKind::Issue, archived, &list.filter(), json_out)
85        }
86        Cmd::Resolve { id, resolution } => {
87            items::require_kind(id, ItemKind::Issue, "issue resolve")?;
88            let mut file = items::item_for_edit(&store, id)?;
89            items::walk_to(&mut file.meta, Status::Done)?;
90            if let Some(res) = &resolution {
91                templates::append_to_section(&mut file.body, "## Resolution", res);
92            }
93            let item = file.meta.clone();
94            store.save_item(&file).map_err(|e| e.to_string())?;
95            let committed =
96                items::commit(&proj.worktree, &format!("chore: resolve {id}"), no_commit)?;
97            items::report(
98                json!({"item": item}),
99                committed,
100                json_out,
101                &format!("resolved {id}"),
102            )
103        }
104        Cmd::Archive { id } => {
105            items::require_kind(id, ItemKind::Issue, "issue archive")?;
106            let file = items::item_for_edit(&store, id)?;
107            if file.meta.status != Status::Done {
108                return Err(format!(
109                    "{id} must be done before archiving (status: {}) — resolve it first",
110                    file.meta.status
111                ));
112            }
113            let rel = items::item_rel_path(&file.meta).replacen('/', "/archive/", 1);
114            store.archive_item(id).map_err(|e| e.to_string())?;
115            let committed =
116                items::commit(&proj.worktree, &format!("chore: archive {id}"), no_commit)?;
117            items::report(
118                json!({"item": file.meta, "archived_to": rel}),
119                committed,
120                json_out,
121                &format!("archived {id} → {rel}"),
122            )
123        }
124        Cmd::Comment { id, text } => {
125            items::require_kind(id, ItemKind::Issue, "issue comment")?;
126            let mut file = items::item_for_edit(&store, id)?;
127            let line = format!("- {} {}", myco_core::date::render(items::today()), text);
128            templates::append_to_section(&mut file.body, "## Comments", &line);
129            store.save_item(&file).map_err(|e| e.to_string())?;
130            let committed =
131                items::commit(&proj.worktree, &format!("chore: comment {id}"), no_commit)?;
132            items::report(
133                json!({"item": file.meta}),
134                committed,
135                json_out,
136                &format!("commented on {id}"),
137            )
138        }
139    }
140}
