1//! `myco board [--md]` — kanban of the current repo's ledger:
2//! Backlog | Open | Claimed | In Progress | Blocked columns (done
3//! items leave the ledger, so there's no Done column). Human
4//! output is a `tabled` grid; `--md` is the same board as a
5//! markdown table for embedding in docs; `--json` is the columns
6//! map.
7

17/// Column order — a `deferred` item is a backlog item wherever it
18/// lives, and `done` has no column (done leaves the ledger).
19pub const COLUMNS: [(&str, Option<Status>); 5] = [
20    ("backlog", Some(Status::Deferred)),
21    ("open", Some(Status::Open)),
22    ("claimed", Some(Status::Claimed)),
23    ("in-progress", Some(Status::InProgress)),
24    ("blocked", Some(Status::Blocked)),
25];
26
27/// One board row — five card cells, any may be empty.
28#[derive(Tabled)]
29struct Row {
30    #[tabled(rename = "Backlog")]
31    backlog: String,
32    #[tabled(rename = "Open")]
33    open: String,
34    #[tabled(rename = "Claimed")]
35    claimed: String,
36    #[tabled(rename = "In Progress")]
37    in_progress: String,
38    #[tabled(rename = "Blocked")]
39    blocked: String,
40}
41
42/// Card text — `T-012 the title`, with an optional prefix
43/// (`myco:`) for org boards.
44fn card(item: &Item, prefix: Option<&str>) -> String {
45    let id = match prefix {
46        Some(p) => format!("{p}:{}", item.id),
47        None => item.id.to_string(),
48    };
49    if item.title.is_empty() {
50        id
51    } else {
52        format!("{id} {}", item.title)
53    }
54}

56/// Bucket live items into the five columns (in-column order is by
57/// priority then id — the most urgent cards float to the top).
58fn columns<'a>(items: impl Iterator<Item = &'a Item>) -> [Vec<&'a Item>; 5] {
59    let mut cols: [Vec<&'a Item>; 5] = Default::default();
60    for i in items {
61        let idx = COLUMNS.iter().position(|(_, s)| *s == Some(i.status));
62        if let Some(idx) = idx {
63            cols[idx].push(i);
64        }
65    }
66    for col in &mut cols {
67        col.sort_by(|a, b| digest::pri_then_id(a, b));
68    }
69    cols
70}

72/// Render the board. `prefix` puts `repo:` on every card (org).
73pub fn render<'a>(items: impl Iterator<Item = &'a Item>, prefix: Option<&str>, md: bool) -> String {
74    let cols = columns(items);
75    let cards: [Vec<String>; 5] =
76        std::array::from_fn(|c| cols[c].iter().map(|i| card(i, prefix)).collect());
77    render_cards(&cards, md)
78}

80/// Render five columns of card strings — shared with `org board`,
81/// which builds mixed-repo cards itself.
82pub fn render_cards(cards: &[Vec<String>; 5], md: bool) -> String {
83    let height = cards.iter().map(Vec::len).max().unwrap_or(0);
84    let mut rows = Vec::with_capacity(height);
85    for r in 0..height {
86        let cell = |c: usize| cards[c].get(r).cloned().unwrap_or_default();
87        rows.push(Row {
88            backlog: cell(0),
89            open: cell(1),
90            claimed: cell(2),
91            in_progress: cell(3),
92            blocked: cell(4),
93        });
94    }
95    let mut table = Table::new(rows);
96    if md {
97        table.with(Style::markdown());
98    }
99    table.to_string()
100}

102/// Bucket `(label, item)` pairs into the five columns, sorting
103/// each by priority — org cards carry their repo prefix already.
104pub fn cards_by_column<'a>(items: impl Iterator<Item = (String, &'a Item)>) -> [Vec<String>; 5] {
105    let mut cols: [Vec<(&'a Item, String)>; 5] = Default::default();
106    for (label, i) in items {
107        if let Some(idx) = COLUMNS.iter().position(|(_, s)| *s == Some(i.status)) {
108            cols[idx].push((i, label));
109        }
110    }
111    std::array::from_fn(|c| {
112        let mut col = std::mem::take(&mut cols[c]);
113        col.sort_by(|(a, _), (b, _)| digest::pri_then_id(a, b));
114        col.into_iter().map(|(_, l)| l).collect()
115    })
116}

118/// The `--json` columns map — column name → full items.
119pub fn columns_json<'a>(
120    items: impl Iterator<Item = &'a Item>,
121    prefix: Option<&str>,
122) -> serde_json::Value {
123    let cols = columns(items);
124    let mut map = serde_json::Map::new();
125    for (i, (name, _)) in COLUMNS.iter().enumerate() {
126        map.insert(
127            (*name).to_string(),
128            json!(cols[i]
129                .iter()
130                .map(|it| {
131                    let mut v = serde_json::to_value(it).unwrap_or(json!(null));
132                    if let (Some(p), Some(obj)) = (prefix, v.as_object_mut()) {
133                        obj.insert("repo".into(), json!(p));
134                    }
135                    v
136                })
137                .collect::<Vec<_>>()),
138        );
139    }
140    json!(map)
141}

143/// The `board` verb. `--epic`/`--milestone` scope the columns to
144/// items in that group (both set = AND).
145pub fn run(
146    dir: Option<PathBuf>,
147    root: Option<PathBuf>,
148    md: bool,
149    epic: Option<String>,
150    milestone: Option<String>,
151    json_out: bool,
152) -> Result<i32, String> {
153    let (_proj, store) = items::open(dir.as_deref(), root.as_deref())?;
154    let scan = digest::scan(&store)?;
155    let live: Vec<&Item> = scan
156        .live()
157        .filter(|i| {
158            epic.as_deref().is_none_or(|e| i.epic.as_deref() == Some(e))
159                && milestone
160                    .as_deref()
161                    .is_none_or(|m| i.milestone.as_deref() == Some(m))
162        })
163        .collect();
164    if json_out {
165        ops::print_json(&columns_json(live.into_iter(), None));
166        return Ok(0);
167    }
168    if live.is_empty() {
169        println!("{}: empty board — `myco todo add` something", scan.name);
170    } else {
171        println!("{}", render(live.into_iter(), None, md));
172    }
173    Ok(0)
174}

176#[cfg(test)]
177#[allow(clippy::unwrap_used)]
178mod tests {
179    use super::*;
180    use myco_core::{ItemId, ItemKind, Priority};

182    fn items() -> Vec<Item> {
183        vec![
184            Item::new(ItemId::new(ItemKind::Task, 1), "parked").status(Status::Deferred),
185            Item::new(ItemId::new(ItemKind::Task, 2), "plain"),
186            Item::new(ItemId::new(ItemKind::Task, 3), "urgent").priority(Priority::Critical),
187            Item::new(ItemId::new(ItemKind::Issue, 1), "owned").status(Status::Claimed),
188            Item::new(ItemId::new(ItemKind::Task, 4), "wip").status(Status::InProgress),
189            Item::new(ItemId::new(ItemKind::Task, 5), "stuck").status(Status::Blocked),
190            Item::new(ItemId::new(ItemKind::Task, 6), "gone").status(Status::Done),
191        ]
192    }
193
194    #[test]
195    fn columns_bucket_and_sort() {
196        let v = items();
197        let cols = columns(v.iter());
198        assert_eq!(cols[0][0].id, ItemId::new(ItemKind::Task, 1));
199        // open column: crit before med
200        assert_eq!(
201            cols[1].iter().map(|i| i.id.number()).collect::<Vec<_>>(),
202            [3, 2]
203        );
204        assert_eq!(cols[2][0].id, ItemId::new(ItemKind::Issue, 1));
205        assert_eq!(cols[3][0].id, ItemId::new(ItemKind::Task, 4));
206        assert_eq!(cols[4][0].id, ItemId::new(ItemKind::Task, 5));
207        // done items land nowhere
208        assert!(cols.iter().flatten().all(|i| i.status != Status::Done));
209    }
210
211    #[test]
212    fn render_grid_and_md() {
213        let v = items();
214        let grid = render(v.iter(), None, false);
215        assert!(grid.contains("Backlog"));
216        assert!(grid.contains("In Progress"));
217        assert!(grid.contains("T-004 wip"));
218        let md = render(v.iter(), Some("myco"), true);
219        assert!(md.starts_with('|'));
220        assert!(md.contains("myco:T-001 parked"));
221        assert!(md.contains("myco:I-001 owned"));
222        // markdown table: header + separator + rows
223        assert!(md.lines().any(|l| l.contains("---")));
224    }
225
226    #[test]
227    fn card_bare_id_for_empty_title() {
228        let i = Item::new(ItemId::new(ItemKind::Task, 9), "");
229        assert_eq!(card(&i, None), "T-009");
230        assert_eq!(card(&i, Some("myco")), "myco:T-009");
231    }
232
233    #[test]
234    fn columns_json_map() {
235        let v = items();
236        let j = columns_json(v.iter(), Some("myco"));
237        assert_eq!(j["open"].as_array().unwrap().len(), 2);
238        assert_eq!(j["open"][0]["id"], "T-003");
239        assert_eq!(j["open"][0]["repo"], "myco");
240        assert!(j["blocked"].as_array().unwrap()[0]["id"] == "T-005");
241        assert!(j["backlog"].as_array().unwrap().len() == 1);
242    }
