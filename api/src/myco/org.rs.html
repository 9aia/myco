1//! `myco org status|board` — cross-repo views over the index.
2//!
3//! Every registered repo's snapshot is checked for freshness
4//! (stored `head_commit` vs the worktree's current HEAD); a stale
5//! or missing snapshot is rebuilt on the spot — org reads never
6//! serve old data silently. Repos whose worktree vanished keep
7//! their last snapshot and report an error inline.
8

18#[derive(Subcommand)]
19pub enum Cmd {
20    /// Per-repo counts + org totals + items grouped by owner.
21    Status {
22        /// Show repos with no live items too.
23        #[arg(long)]
24        all: bool,
25    },
26    /// Org kanban — repo-prefixed cards over every live item.
27    Board {
28        /// Emit a markdown table (for docs/READMEs).
29        #[arg(long)]
30        md: bool,
31        /// Show every card — no row cap.
32        #[arg(long)]
33        all: bool,
34        /// Only this repo's items (name or registered path).
35        #[arg(long)]
36        repo: Option<String>,
37    },
38    /// Re-run `init` on every registered repo — refreshes the
39    /// worktree scaffold and root files (.gitignore, the AGENTS.md
40    /// agent-awareness block). The rollout path for new scaffolding.
41    Init,
42}

44/// One repo's snapshot load outcome for org views.
45pub struct RepoView {
46    /// The snapshot (may be the last good one on rescan failure).
47    pub snap: Option<RepoSnapshot>,
48    /// Rescan/load error, when the fresh snapshot couldn't be had.
49    pub error: Option<String>,
50    /// The snapshot was rebuilt during this read.
51    pub rebuilt: bool,
52}
53
54/// A scan-shaped adapter so `digest::counts` runs on snapshots.
55fn as_scan(s: &RepoSnapshot) -> digest::Scan {
56    digest::Scan {
57        name: s.name.clone(),
58        items: s.items.clone(),
59        archived: s.archived.clone(),
60    }
61}

63/// Load every registered repo's snapshot, auto-rebuilding stale
64/// ones (head mismatch or missing). The per-repo `git`/fs work is
65/// parallel — redb opens lock the file, so index I/O stays serial.
66pub fn load_fresh(index: &impl Index) -> Result<Vec<RepoView>, String> {
67    let repos = index.repos().map_err(|e| e.to_string())?;
68    let roots: Vec<std::path::PathBuf> = repos
69        .iter()
70        .map(|r| r.path.as_std_path().to_path_buf())
71        .collect();
72    let heads = indexcmd::par_map(&roots, |r| indexcmd::current_head(r));
73    let mut cached = Vec::with_capacity(repos.len());
74    for r in &repos {
75        cached.push(index.snapshot(&r.path).map_err(|e| e.to_string())?);
76    }
77    let fresh =
78        |i: usize| matches!((&cached[i], &heads[i]), (Some(s), Some(h)) if s.head_commit == *h);
79    let stale: Vec<usize> = (0..repos.len()).filter(|i| !fresh(*i)).collect();
80    let scans = indexcmd::par_map(&stale, |&i| indexcmd::scan_one(&roots[i]));
81    let mut rebuilt: std::collections::BTreeMap<usize, Result<RepoSnapshot, String>> =
82        stale.iter().copied().zip(scans).collect();
83    let mut out = Vec::with_capacity(repos.len());
84    for (i, c) in cached.iter().enumerate() {
85        if fresh(i) {
86            out.push(RepoView {
87                snap: c.clone(),
88                error: None,
89                rebuilt: false,
90            });
91            continue;
92        }
93        match rebuilt
94            .remove(&i)
95            .unwrap_or_else(|| Err("rescan lost".into()))
96        {
97            Ok(snap) => {
98                index.save_snapshot(&snap).map_err(|e| e.to_string())?;
99                out.push(RepoView {
100                    snap: Some(snap),
101                    error: None,
102                    rebuilt: true,
103                });
104            }
105            Err(e) => out.push(RepoView {
106                snap: cached[i].clone(),
107                error: Some(e),
108                rebuilt: false,
109            }),
110        }
111    }
112    Ok(out)
113}

115/// `owner → item refs` across all snapshots — active items only.
116fn by_owner<'a>(snaps: impl Iterator<Item = &'a RepoSnapshot>) -> BTreeMap<String, Vec<String>> {
117    let mut m: BTreeMap<String, Vec<String>> = BTreeMap::new();
118    for s in snaps {
119        for i in s
120            .live_items()
121            .filter(|i| i.status != Status::Done && i.status != Status::Deferred)
122        {
123            if let Some(o) = &i.owner {
124                m.entry(o.as_str().to_string())
125                    .or_default()
126                    .push(format!("{}:{} {}", s.name, i.id, i.title));
127            }
128        }
129    }
130    for v in m.values_mut() {
131        v.sort();
132    }
133    m
134}

136/// Cap on items listed per owner in `org status` — the section is
137/// a pointer to where work sits, not a full listing.
138const OWNER_CAP: usize = 8;
139
140fn status(views: &[RepoView], all: bool, json_out: bool) -> Result<i32, String> {
141    let mut total = digest::Counts::default();
142    let mut repos_json = Vec::new();
143    let mut snaps = Vec::new();
144    for v in views {
145        let Some(s) = &v.snap else {
146            repos_json.push(json!({"error": v.error}));
147            continue;
148        };
149        snaps.push(s);
150        let c = digest::counts(&as_scan(s));
151        total.add(&c);
152        repos_json.push(json!({
153            "name": s.name,
154            "path": s.path,
155            "head_commit": s.head_commit,
156            "rebuilt": v.rebuilt,
157            "error": v.error,
158            "counts": c.to_json(),
159        }));
160    }
161    let owners = by_owner(snaps.iter().copied());
162    if json_out {
163        ops::print_json(&json!({
164            "repos": repos_json,
165            "totals": total.to_json(),
166            "owners": owners,
167        }));
168        return Ok(0);
169    }
170    let mut out = format!("org status — {} repo(s)\n", views.len());
171    let mut idle = 0usize;
172    for v in views {
173        // a repo with a clean snapshot but no live items is idle —
174        // hundreds of registered repos bury the interesting ones;
175        // errored repos always surface
176        let idle_repo = v.error.is_none()
177            && v.snap
178                .as_ref()
179                .is_some_and(|s| s.live_items().next().is_none());
180        if idle_repo && !all {
181            idle += 1;
182            continue;
183        }
184        match &v.snap {
185            Some(s) => {
186                let c = digest::counts(&as_scan(s));
187                let head = if s.head_commit.is_empty() {
188                    "gone".to_string()
189                } else {
190                    s.head_commit[..s.head_commit.len().min(7)].to_string()
191                };
192                let mut line = format!("{:<16} {}  @{head}", s.name, c.oneline());
193                if v.rebuilt {
194                    line.push_str("  (rebuilt)");
195                }
196                if let Some(e) = &v.error {
197                    line.push_str(&format!("  ⚠ {e}"));
198                }
199                out.push_str(&line);
200                out.push('\n');
201            }
202            None => out.push_str(&format!(
203                "{:<16} ⚠ {}\n",
204                "?",
205                v.error.as_deref().unwrap_or("unavailable")
206            )),
207        }
208    }
209    if idle > 0 {
210        out.push_str(&format!("… +{idle} idle repos (`--all` to show)\n"));
211    }
212    out.push_str(&format!("totals: {}\n", total.oneline()));
213    if !owners.is_empty() {
214        out.push_str("by owner:\n");
215        for (o, items) in &owners {
216            let more = items.len().saturating_sub(OWNER_CAP);
217            let shown: Vec<&str> = items.iter().take(OWNER_CAP).map(String::as_str).collect();
218            let suffix = if more > 0 {
219                format!(", … +{more} more")
220            } else {
221                String::new()
222            };
223            out.push_str(&format!("  {o}: {}{}\n", shown.join(", "), suffix));
224        }
225    }
226    ops::print_paged(out.trim_end());
227    Ok(0)
228}

230fn board_view(
231    views: &[RepoView],
232    md: bool,
233    all: bool,
234    repo: Option<String>,
235    json_out: bool,
236) -> Result<i32, String> {
237    // `--repo` matches a snapshot by name or registered path
238    let wanted = |s: &RepoSnapshot| repo.as_deref().is_none_or(|r| s.name == r || s.path == r);
239    if repo.is_some() && !views.iter().any(|v| v.snap.as_ref().is_some_and(&wanted)) {
240        return Err(format!(
241            "no repo {:?} in the index — `myco index list`",
242            repo.unwrap_or_default()
243        ));
244    }
245    if json_out {
246        // columns map with a `repo` tag on each item
247        let mut merged = serde_json::Map::new();
248        for (name, _) in board::COLUMNS {
249            merged.insert(name.to_string(), json!([]));
250        }
251        for v in views {
252            let Some(s) = &v.snap else { continue };
253            if !wanted(s) {
254                continue;
255            }
256            let cols = board::columns_json(s.live_items(), Some(&s.name));
257            for (name, _) in board::COLUMNS {
258                let mut dst = merged[name].as_array().cloned().unwrap_or_default();
259                dst.extend(cols[name].as_array().cloned().unwrap_or_default());
260                merged.insert(name.to_string(), json!(dst));
261            }
262        }
263        ops::print_json(&serde_json::Value::Object(merged));
264        return Ok(0);
265    }
266    // human board: one merged column set, repo-prefixed cards
267    let tagged = views.iter().flat_map(|v| {
268        v.snap.iter().filter(|s| wanted(s)).flat_map(|s| {
269            s.live_items()
270                .map(move |i| (format!("{}:{}", s.name, i.id), i))
271        })
272    });
273    let cards = board::cards_by_column(tagged.map(|(id, i)| {
274        let label = if i.title.is_empty() {
275            id
276        } else {
277            format!("{id} {}", i.title)
278        };
279        (label, i)
280    }));
281    if cards.iter().all(Vec::is_empty) {
282        println!("org board — nothing live across {} repo(s)", views.len());
283    } else {
284        ops::print_paged(&board::render_cards(&cards, md, board::Fit::terminal(all)));
285    }
286    Ok(0)
287}

289/// `init` over every registered repo — per-repo outcomes, errors
290/// collected (one bad repo doesn't stop the rollout).
291fn init_all(index: &impl Index, json_out: bool, no_commit: bool) -> Result<i32, String> {
292    let repos = index.repos().map_err(|e| e.to_string())?;
293    let mut out = Vec::new();
294    let mut failures = 0usize;
295    for r in &repos {
296        let p = r.path.as_std_path().to_path_buf();
297        match crate::init::ensure(None, Some(&p), no_commit) {
298            Ok(o) => out.push(json!({
299                "path": r.path,
300                "files_created": o.files_created,
301                "gitignore_updated": o.gitignore_updated,
302                "agents_md_updated": o.agents_md_updated,
303                "committed": o.committed,
304            })),
305            Err(e) => {
306                failures += 1;
307                out.push(json!({"path": r.path, "error": e}));
308            }
309        }
310    }
311    if json_out {
312        ops::print_json(&json!({"repos": out, "failures": failures}));
313        return Ok(if failures == 0 { 0 } else { 1 });
314    }
315    for (r, o) in repos.iter().zip(&out) {
316        if let Some(e) = o.get("error").and_then(|v| v.as_str()) {
317            println!("{:<24} ⚠ {e}", r.path.as_str());
318        } else {
319            let mut notes = Vec::new();
320            if o["gitignore_updated"].as_bool().unwrap_or(false) {
321                notes.push(".gitignore");
322            }
323            if o["agents_md_updated"].as_bool().unwrap_or(false) {
324                notes.push("AGENTS.md");
325            }
326            let n = o["files_created"].as_array().map_or(0, Vec::len);
327            if n > 0 {
328                notes.push("scaffold");
329            }
330            println!(
331                "{:<24} {}",
332                r.path.as_str(),
333                if notes.is_empty() {
334                    "already current".to_string()
335                } else {
336                    format!("updated {}", notes.join(" + "))
337                }
338            );
339        }
340    }
341    Ok(if failures == 0 { 0 } else { 1 })
342}

344/// The `org` verb group.
345pub fn run(cmd: Cmd, json_out: bool, no_commit: bool) -> Result<i32, String> {
346    let index = indexcmd::open_index()?;
347    if let Cmd::Init = cmd {
348        return init_all(&index, json_out, no_commit);
349    }
350    let views = load_fresh(&index)?;
351    match cmd {
352        Cmd::Status { all } => status(&views, all, json_out),
353        Cmd::Board { md, all, repo } => board_view(&views, md, all, repo, json_out),
354        Cmd::Init => unreachable!(),
355    }
