1//! `myco epic [name]` / `myco milestone [name]` — grouping views
2//! over item metadata, driven by the named views of the same
3//! names (`[views.epic]`/`[views.milestone]` config shadows the
4//! presets). Bare lists every group with its item count; a name
5//! renders that group's board (`epic v2` ≡ `board --epic v2`).
6
7use std::path::PathBuf;
8
9use myco_core::{eval, UNGROUPED};
10use serde_json::json;
11
12use crate::{board, digest, items, ops, view};
13
14/// Which named view the group verb resolves.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub enum Group {
17    /// `item.epic` — `myco epic` resolves the `epic` view.
18    Epic,
19    /// `item.milestone` — `myco milestone` resolves `milestone`.
20    Milestone,
21}
22
23impl Group {
24    /// The verb/view/field name in output (`epic`, `milestone`).
25    pub fn name(self) -> &'static str {
26        match self {
27            Group::Epic => "epic",
28            Group::Milestone => "milestone",
29        }
30    }
31}
32
33/// The shared `epic`/`milestone` verb body. `all` lifts the board
34/// row cap when a name renders a group board.
35pub fn run(
36    group: Group,
37    name: Option<String>,
38    md: bool,
39    all: bool,
40    dir: Option<PathBuf>,
41    root: Option<PathBuf>,
42    json_out: bool,
43) -> Result<i32, String> {
44    let (proj, store) = items::open_read(dir.as_deref(), root.as_deref())?;
45    let scan = digest::scan(&store)?;
46    match name {
47        None => {
48            // the named view's buckets, summarized to
49            // `name  count` lines — empty and `ungrouped`
50            // buckets aren't groups
51            let spec = view::scoped_spec(&proj.root, group.name(), None, None)?;
52            let groups = eval(scan.live(), &spec);
53            let counts: Vec<(&str, usize)> = groups
54                .iter()
55                .filter(|g| !g.items.is_empty())
56                .filter_map(|g| {
57                    g.label
58                        .as_deref()
59                        .filter(|l| *l != UNGROUPED)
60                        .map(|l| (l, g.items.len()))
61                })
62                .collect();
63            if json_out {
64                ops::print_json(&json!({
65                    "group": group.name(),
66                    "groups": counts
67                        .iter()
68                        .map(|(name, count)| json!({"name": name, "count": count}))
69                        .collect::<Vec<_>>(),
70                }));
71            } else if counts.is_empty() {
72                println!(
73                    "{}: no {}s — set one via `myco item set <id> --{} <name>`",
74                    scan.name,
75                    group.name(),
76                    group.name()
77                );
78            } else {
79                for (name, count) in counts {
80                    println!("{name}  {count}");
81                }
82            }
83            Ok(0)
84        }
85        Some(name) => {
86            if !myco_core::is_valid_group_name(&name) {
87                return Err(format!("invalid {} name {name:?}", group.name()));
88            }
89            // `epic v2` ≡ `board --epic v2` — the board view
90            // scoped to this group
91            let (epic, milestone) = match group {
92                Group::Epic => (Some(name.clone()), None),
93                Group::Milestone => (None, Some(name.clone())),
94            };
95            let spec = view::scoped_spec(&proj.root, "board", epic, milestone)?;
96            let groups = eval(scan.live(), &spec);
97            let interrupted = crate::continuity::interrupted_set(
98                &store,
99                groups.iter().flat_map(|g| g.items.iter().copied()),
100            )?;
101            board::emit(&board::Emit {
102                repo: &scan.name,
103                view: "board",
104                spec: &spec,
105                groups: &groups,
106                interrupted: &interrupted,
107                md,
108                all,
109                json_out,
110                empty_msg: &format!("no items in {} {name:?}", group.name()),
111            })
112        }
113    }
114}
