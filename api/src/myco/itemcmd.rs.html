1//! `myco item show|set` — the kind-agnostic item verbs: locate an
2//! item anywhere in the ledger; mutate metadata, with `--status`
3//! going through the transition machine only.
4
5use std::path::PathBuf;
6
7use clap::Subcommand;
8use myco_core::{is_valid_label, Actor, Item, ItemId, Priority, Status};
9use myco_store::Store;
10use serde_json::json;
11
12use crate::items;

14#[derive(Subcommand)]
15pub enum Cmd {
16    /// Dump one item — frontmatter fields plus the body for
17    /// file-backed kinds (`--json` = the full Item + location +
18    /// body).
19    Show {
20        /// Any item id (`T-NNN`, `I-NNN`, `P-NNN`, `ADR-NNNN`,
21        /// `PL-NNN`) — live or archived.
22        id: ItemId,
23    },
24    /// Mutate an item in place. `--status` is the only field routed
25    /// through the state machine — single steps only, invalid ones
26    /// error with the valid targets. Status changes on tasks move
27    /// the line between TODO.md and BACKLOG.md (`done` deletes it).
28    Set {
29        /// Any item id.
30        id: ItemId,
31        /// New status — one legal step from the current one.
32        #[arg(long)]
33        status: Option<Status>,
34        /// Priority: low|med|high|crit.
35        #[arg(long, visible_alias = "priority")]
36        pri: Option<Priority>,
37        /// Owner actor id.
38        #[arg(long)]
39        owner: Option<Actor>,
40        /// Add (`+x` or `x`) or remove (`-x`) a label — repeatable.
41        #[arg(long, allow_hyphen_values = true)]
42        label: Vec<String>,
43        /// Remove a label — repeatable.
44        #[arg(long)]
45        unlabel: Vec<String>,
46        /// Add (`+T-1` or `T-1`) or remove (`-T-1`) a blocks dep —
47        /// repeatable.
48        #[arg(long, allow_hyphen_values = true)]
49        blocks: Vec<String>,
50        /// Remove a blocks dep — repeatable.
51        #[arg(long)]
52        unblock: Vec<ItemId>,
53        /// Epic name — `""` clears the epic.
54        #[arg(long)]
55        epic: Option<String>,
56        /// Milestone name — `""` clears the milestone.
57        #[arg(long)]
58        milestone: Option<String>,
59    },
60}
61
62/// The set-verb flag bundle — applied to an `Item` after the
63/// status transition check.
64struct SetFlags {
65    status: Option<Status>,
66    pri: Option<Priority>,
67    owner: Option<Actor>,
68    label: Vec<String>,
69    unlabel: Vec<String>,
70    blocks: Vec<String>,
71    unblock: Vec<ItemId>,
72    epic: Option<String>,
73    milestone: Option<String>,
74}
75
76impl SetFlags {
77    fn is_noop(&self) -> bool {
78        self.status.is_none()
79            && self.pri.is_none()
80            && self.owner.is_none()
81            && self.label.is_empty()
82            && self.unlabel.is_empty()
83            && self.blocks.is_empty()
84            && self.unblock.is_empty()
85            && self.epic.is_none()
86            && self.milestone.is_none()
87    }
88}
89
90/// `+x` → add `x`, `-x` → remove `x`, `x` → add — the `item set`
91/// add/remove convention.
92fn signed(s: &str) -> (bool, &str) {
93    match s.strip_prefix('-') {
94        Some(rest) => (false, rest),
95        None => (true, s.strip_prefix('+').unwrap_or(s)),
96    }
97}
98
99/// Apply every set flag to `item` — status first (machine-checked),
100/// then the plain metadata.
101fn apply_set(item: &mut Item, flags: &SetFlags) -> Result<(), String> {
102    if let Some(to) = flags.status {
103        let from = item.status;
104        item.status = from.transition_to(to).map_err(|_| {
105            format!(
106                "{id}: invalid status transition {from} → {to} (valid from {from}: {t})",
107                id = item.id,
108                t = items::targets(from)
109            )
110        })?;
111    }
112    if let Some(p) = flags.pri {
113        item.priority = p;
114    }
115    if let Some(o) = &flags.owner {
116        item.owner = Some(o.clone());
117    }
118    for raw in &flags.label {
119        let (add, name) = signed(raw);
120        if !is_valid_label(name) {
121            return Err(format!("invalid label {name:?}"));
122        }
123        if add {
124            item.labels.insert(name.to_string());
125        } else {
126            item.labels.remove(name);
127        }
128    }
129    for name in &flags.unlabel {
130        item.labels.remove(name.as_str());
131    }
132    for raw in &flags.blocks {
133        let (add, s) = signed(raw);
134        let id: ItemId = s
135            .parse()
136            .map_err(|e: myco_core::ParseItemIdError| e.to_string())?;
137        if add {
138            item.blocks.insert(id);
139        } else {
140            item.blocks.remove(&id);
141        }
142    }
143    for id in &flags.unblock {
144        item.blocks.remove(id);
145    }
146    for (what, flag, slot) in [
147        ("epic", &flags.epic, &mut item.epic),
148        ("milestone", &flags.milestone, &mut item.milestone),
149    ] {
150        if let Some(name) = flag {
151            if name.is_empty() {
152                *slot = None; // `--epic ""` clears
153            } else if !myco_core::is_valid_group_name(name) {
154                return Err(format!("invalid {what} name {name:?}"));
155            } else {
156                *slot = Some(name.clone());
157            }
158        }
159    }
160    Ok(())
161}

163/// The `item` verb group.
164pub fn run(
165    cmd: Cmd,
166    dir: Option<PathBuf>,
167    root: Option<PathBuf>,
168    json_out: bool,
169    no_commit: bool,
170) -> Result<i32, String> {
171    let (proj, store) = items::open(dir.as_deref(), root.as_deref())?;
172    match cmd {
173        Cmd::Show { id } => match items::find_item(&store, id)? {
174            Some((item, location, body)) => {
175                let handoffs = store.read_handoffs(id).map_err(|e| e.to_string())?.len();
176                if json_out {
177                    crate::ops::print_json(
178                        &json!({"item": item, "location": location, "handoffs": handoffs, "body": body}),
179                    );
180                } else {
181                    println!("{}", items::compact(&item));
182                    println!("  {location}");
183                    println!("  handoffs: {handoffs}");
184                    if let Some(body) = &body {
185                        if !body.is_empty() {
186                            println!();
187                            println!("{body}");
188                        }
189                    }
190                }
191                Ok(0)
192            }
193            None => Err(format!("{id}: no such item")),
194        },
195        Cmd::Set {
196            id,
197            status,
198            pri,
199            owner,
200            label,
201            unlabel,
202            blocks,
203            unblock,
204            epic,
205            milestone,
206        } => {
207            let flags = SetFlags {
208                status,
209                pri,
210                owner,
211                label,
212                unlabel,
213                blocks,
214                unblock,
215                epic,
216                milestone,
217            };
218            if flags.is_noop() {
219                return Err("item set: nothing to change — pass at least one flag".into());
220            }
221            let mut ledger = items::Ledger::load(&store)?;
222            if let Some(loc) = ledger.locate(id) {
223                // task item: mutate in place; a status change may
224                // relocate the line — done deletes the subtree,
225                // deferred detaches the node into BACKLOG.md
226                // (children park with it), an active status moves
227                // backlog nodes into TODO.md (children unpark).
228                // Nested children that keep their doc stay nested.
229                let status_change = flags.status.is_some();
230                apply_set(ledger.item_mut(&loc), &flags)?;
231                if status_change {
232                    match ledger.item(&loc).status {
233                        Status::Done => {
234                            ledger.take(&loc);
235                        }
236                        Status::Deferred => {
237                            let mut item = ledger.take(&loc);
238                            items::walk_descendants_to(&mut item, Status::Deferred)?;
239                            ledger.put_backlog(item, None);
240                        }
241                        _ if loc.backlog => {
242                            let mut item = ledger.take(&loc);
243                            items::walk_descendants_to(&mut item, Status::Open)?;
244                            ledger.put_todo(item, None);
245                        }
246                        _ if !loc.is_child() => {
247                            // top-level items re-land at the
248                            // section tail — same as before
249                            let item = ledger.take(&loc);
250                            ledger.put_todo(item, None);
251                        }
252                        _ => {} // a TODO child stays nested
253                    }
254                }
255                let item = ledger.locate(id).map(|l| ledger.item(&l).clone());
256                ledger.save(&store)?;
257                let committed =
258                    items::commit(&proj.worktree, &format!("chore: set {id}"), no_commit)?;
259                let human = item
260                    .as_ref()
261                    .map(|i| format!("set {}", items::compact(i)))
262                    .unwrap_or_else(|| format!("set {id} done — line removed"));
263                items::report(
264                    json!({"item": item, "location": "tasks"}),
265                    committed,
266                    json_out,
267                    &human,
268                )
269            } else {
270                let mut file = items::item_for_edit(&store, id)?;
271                apply_set(&mut file.meta, &flags)?;
272                let item = file.meta.clone();
273                store.save_item(&file).map_err(|e| e.to_string())?;
274                let committed =
275                    items::commit(&proj.worktree, &format!("chore: set {id}"), no_commit)?;
276                items::report(
277                    json!({"item": item, "location": items::item_rel_path(&item)}),
278                    committed,
279                    json_out,
280                    &format!("set {}", items::compact(&item)),
281                )
282            }
283        }
284    }
