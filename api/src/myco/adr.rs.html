1//! `myco adr new|list|supersede` — architecture decision records
2//! in `adrs/NNNN-<slug>.md` (Context / Decision / Consequences).
3
4use std::path::PathBuf;
5
6use clap::Subcommand;
7use myco_core::{ItemId, ItemKind, Status};
8use myco_store::Store;
9use serde_json::json;
10
11use crate::items;
12use crate::templates;
13
14#[derive(Subcommand)]
15pub enum Cmd {
16    /// File an ADR — allocs an ADR-id, writes the file with
17    /// frontmatter (status `open` = "proposed") + template.
18    New {
19        /// One-line title (drives the filename slug).
20        title: String,
21        #[command(flatten)]
22        body: items::BodyFlags,
23        #[command(flatten)]
24        meta: items::MetaFlags,
25    },
26    /// List ADRs.
27    List {
28        #[command(flatten)]
29        list: items::ListFlags,
30    },
31    /// Mark an ADR superseded by another — the old file gets
32    /// `superseded_by:` + status `done`, the replacement gets a
33    /// reciprocal `supersedes:` key. Both must already exist.
34    Supersede {
35        /// The ADR being replaced (`ADR-NNNN`).
36        id: ItemId,
37        /// The ADR replacing it (`ADR-NNNN`).
38        #[arg(long)]
39        by: ItemId,
40    },
41}
42
43/// The `adr` verb group.
44pub fn run(
45    cmd: Cmd,
46    dir: Option<PathBuf>,
47    root: Option<PathBuf>,
48    json_out: bool,
49    no_commit: bool,
50) -> Result<i32, String> {
51    let (proj, store) = items::open(dir.as_deref(), root.as_deref())?;
52    match cmd {
53        Cmd::New { title, body, meta } => items::new_file(
54            &proj,
55            &store,
56            &items::NewItem {
57                kind: ItemKind::Adr,
58                title,
59                body: body.read()?.as_deref(),
60                meta: &meta.to_meta(),
61                template: templates::adr,
62                word: "adr",
63                json_out,
64                no_commit,
65            },
66        ),
67        Cmd::List { list } => {
68            items::list_files(&store, ItemKind::Adr, false, &list.filter(), json_out)
69        }
70        Cmd::Supersede { id, by } => {
71            items::require_kind(id, ItemKind::Adr, "adr supersede")?;
72            items::require_kind(by, ItemKind::Adr, "adr supersede --by")?;
73            if id == by {
74                return Err(format!("{id} can't supersede itself"));
75            }
76            let mut old = items::item_for_edit(&store, id)?;
77            let mut new = items::item_for_edit(&store, by)?;
78            items::walk_to(&mut old.meta, Status::Done)?;
79            old.meta.superseded_by = Some(by);
80            new.meta.supersedes = Some(id);
81            store.save_item(&old).map_err(|e| e.to_string())?;
82            store.save_item(&new).map_err(|e| e.to_string())?;
83            let committed = items::commit(
84                &proj.worktree,
85                &format!("chore: supersede {id} by {by}"),
86                no_commit,
87            )?;
88            items::report(
89                json!({"superseded": old.meta, "by": new.meta}),
90                committed,
91                json_out,
92                &format!("{id} superseded by {by}"),
93            )
94        }
95    }
96}
