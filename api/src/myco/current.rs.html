26fn mode_str(m: myco_core::CurrentMode) -> &'static str {
27    match m {
28        myco_core::CurrentMode::Always => "always",
29        myco_core::CurrentMode::Backup => "backup",
30    }

37/// The shared payload builders — cli prints them, serve returns
38/// them verbatim.
39pub fn list_json() -> Result<Value, String> {
40    let cwd = cwd()?;
41    let index = open_index()?;
42    let targets = current::list_targets(&cwd, as_index(&index), CONFIG)?;
43    Ok(json!({"targets": targets}))
44}

46pub fn show_json() -> Result<Value, String> {
47    let cwd = cwd()?;
48    let env = current::env_selector();
49    let cfg_sel = current::config_selector(CONFIG);
50    let pointer = current::pointer_selector(&cwd);
51    let resolved = crate::discover::discover(None, None);
52    Ok(json!({
53        "env": env.as_ref().map(|s| sel_json("env", s)),
54        "config": cfg_sel.as_ref().map(|s| sel_json("config", s)),
55        "pointer": pointer.map(|(f, s)| json!({
56            "file": f, "ledger": s.ledger,
57        })),
58        "resolved": match &resolved {
59            Ok(p) => json!({"root": p.root, "worktree": p.worktree}),
60            Err(e) => json!({"error": e}),
61        },
62    }))
63}

65pub fn use_json(target: &str, mode: &str, here: bool) -> Result<Value, String> {
66    let cwd = cwd()?;
67    let mode_enum = match mode {
68        "always" => myco_core::CurrentMode::Always,
69        "backup" => myco_core::CurrentMode::Backup,
70        _ => return Err(format!("mode must be `always` or `backup`, got `{mode}`")),
71    };
72    // `repo` is inert — nothing to validate; every other target
73    // resolves *now* so `use` fails loudly on a bad pin.
74    if target != "repo" {
75        let index = open_index().ok();
76        let idx: Option<&dyn myco_index::Index> = index.as_ref().map(as_index);
77        let sel = Selector {
78            ledger: target.to_string(),
79            mode: mode_enum,
80        };
81        current::resolve(&sel, &cwd, idx, CONFIG)?;
82    }
83    if here {
84        // `repo` in a pointer file is inert — clear instead
85        let path = current::write_pointer(&cwd, (target != "repo").then_some(target))?;
86        return Ok(json!({"pointer": path, "ledger": target}));
87    }
88    let path = current::write_config(target, mode_str(mode_enum))?;
89    Ok(json!({"config": path, "ledger": target, "mode": mode_str(mode_enum)}))
90}

92pub fn clear_json(here: bool) -> Result<Value, String> {
93    let cwd = cwd()?;
94    if here {
95        let path = current::write_pointer(&cwd, None)?;
96        return Ok(json!({"pointer": path, "cleared": true}));
97    }
98    let mode = current::config_selector(CONFIG)
99        .map(|s| mode_str(s.mode))
100        .unwrap_or("backup");
101    let path = current::write_config("repo", mode)?;
102    Ok(json!({"config": path, "ledger": "repo"}))
103}

105fn print_value(ctx: &Ctx, v: &Value) {
106    if ctx.json {
107        println!("{}", serde_json::to_string_pretty(v).unwrap_or_default());
108    } else {
109        match v {
110            Value::Object(map) if map.contains_key("targets") => {
111                for t in v["targets"].as_array().into_iter().flatten() {
112                    let name = t["target"].as_str().unwrap_or("?");
113                    let kind = t["kind"].as_str().unwrap_or("");
114                    let extra = t["worktree"]
115                        .as_str()
116                        .map(|w| format!("  {w}"))
117                        .or_else(|| t["note"].as_str().map(|n| format!("  {n}")))
118                        .unwrap_or_default();
119                    println!("{name}\t{kind}{extra}");
120                }
121            }
122            _ => println!("{}", serde_json::to_string_pretty(v).unwrap_or_default()),
123        }
124    }
125}

127pub fn h_list(ctx: &Ctx, _a: &Args) -> Result<i32, String> {
128    let v = list_json()?;
129    print_value(ctx, &v);
130    Ok(0)
131}

133pub fn h_show(ctx: &Ctx, _a: &Args) -> Result<i32, String> {
134    let v = show_json()?;
135    print_value(ctx, &v);
136    Ok(0)
137}

139pub fn h_use(ctx: &Ctx, a: &Args) -> Result<i32, String> {
140    let target = a.text("target")?;
141    let mode = a.opt("mode").unwrap_or("backup");
142    let v = use_json(target, mode, a.flag("here"))?;
143    print_value(ctx, &v);
144    Ok(0)
145}
146
147pub fn h_clear(ctx: &Ctx, a: &Args) -> Result<i32, String> {
148    let v = clear_json(a.flag("here"))?;
149    print_value(ctx, &v);
150    Ok(0)
151}
