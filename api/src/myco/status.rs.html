1//! `myco status [--prompt]` — the ledger digest; THE agent-context
2//! verb. Human output is compact counts + the in-progress and
3//! blocked lists; `--prompt` is a token-sized block (repo, counts
4//! one-liner, then items grouped in-progress → blocked → claimed →
5//! open-by-priority, one per line, no prose); `--json` is the
6//! structured digest — with `--prompt` it's `{"prompt": "<block>"}`.
7

16/// Hard cap on `--prompt` lines — the block must stay token-cheap.
17pub const PROMPT_MAX_LINES: usize = 60;
18
19/// The four active-status buckets via the view engine —
20/// status-grouped, `pri,id`-sorted (the engine default). One
21/// `eval` replaces the per-status `in_status` scans the digest
22/// paths used to hand-roll; deferred/done items don't reach any
23/// bucket.
24fn active_groups(scan: &digest::Scan) -> Vec<ViewGroup<'_>> {
25    eval(
26        scan.live(),
27        &ViewSpec {
28            filter: ViewFilter {
29                status: vec![
30                    Status::InProgress,
31                    Status::Blocked,
32                    Status::Claimed,
33                    Status::Open,
34                ],
35                ..ViewFilter::default()
36            },
37            group_by: Some(GroupAxis::Status),
38            ..ViewSpec::default()
39        },
40        myco_core::kind::prelude_table(),
41    )
42}
43
44/// One status's slice of [`active_groups`] — `[]` when empty.
45fn in_group<'a>(groups: &'a [ViewGroup<'a>], status: Status) -> &'a [Scoped<'a>] {
46    groups
47        .iter()
48        .find(|g| g.label.as_deref() == Some(status.token()))
49        .map(|g| g.items.as_slice())
50        .unwrap_or(&[])
51}

53/// `<id> <pri> <title> [⚡]` — one prompt line, no prose; `⚡`
54/// marks an interrupted item (latest handoff is an interrupt).
55fn prompt_item(i: &Item, interrupted: bool) -> String {
56    format!(
57        "{} {} {}{}",
58        i.id,
59        i.priority.token(),
60        i.title,
61        if interrupted { " ⚡" } else { "" }
62    )
63}

65/// The `--prompt` block, hard-capped at [`PROMPT_MAX_LINES`].
66/// Sections render in spec order (in-progress → blocked →
67/// claimed → open); when the block overflows, items are cut from
68/// the *last* non-empty group first — open (always largest)
69/// absorbs the budget, then claimed/blocked, with in-progress
70/// dropped last. A truncated group keeps a `… +N more <name>`
71/// marker; a fully cut group disappears (the counts line still
72/// summarizes it).
73pub fn prompt_block(scan: &digest::Scan, interrupted: &BTreeSet<ItemId>) -> String {
74    let c = digest::counts(scan);
75    let evaled = active_groups(scan);
76    // every bucket is already pri,id-sorted by the engine — the
77    // open group used to be sorted by hand here
78    let mut groups: Vec<(&str, Vec<Scoped>, usize)> = [
79        ("in-progress", Status::InProgress),
80        ("blocked", Status::Blocked),
81        ("claimed", Status::Claimed),
82        ("open", Status::Open),
83    ]
84    .into_iter()
85    .map(|(name, s)| (name, in_group(&evaled, s).to_vec(), 0))
86    .collect();
87
88    // render cost: repo line + per non-empty group (header +
89    // shown items + marker when truncated)
90    let cost = |groups: &[(&str, Vec<Scoped>, usize)]| {
91        1 + groups
92            .iter()
93            .map(|(_, shown, hidden)| {
94                if shown.is_empty() {
95                    0
96                } else {
97                    1 + shown.len() + usize::from(*hidden > 0)
98                }
99            })
100            .sum::<usize>()
101    };
102    // shrink tail-first until the whole block fits
103    while cost(&groups) > PROMPT_MAX_LINES {
104        match groups
105            .iter_mut()
106            .rev()
107            .find(|(_, shown, _)| !shown.is_empty())
108        {
109            Some((_, shown, hidden)) => {
110                shown.pop();
111                *hidden += 1;
112            }
113            None => break,
114        }
115    }
116
117    let mut out = vec![format!("repo:{} {}", scan.name, c.oneline())];
118    for (name, shown, hidden) in &groups {
119        if shown.is_empty() {
120            continue;
121        }
122        out.push(format!("{name}:"));
123        out.extend(
124            shown
125                .iter()
126                .map(|i| prompt_item(i.item, interrupted.contains(&i.item.id))),
127        );
128        if *hidden > 0 {
129            out.push(format!("… +{hidden} more {name}"));
130        }
131    }
132    debug_assert!(out.len() <= PROMPT_MAX_LINES);
133    out.join("\n")
134}

136/// The `--json` digest — counts + the active groups as full items.
137pub fn digest_json(scan: &digest::Scan) -> serde_json::Value {
138    let c = digest::counts(scan);
139    let groups = active_groups(scan);
140    let group = |s: Status| {
141        in_group(&groups, s)
142            .iter()
143            .map(|i| serde_json::to_value(i.item).unwrap_or(json!(null)))
144            .collect::<Vec<_>>()
145    };
146    json!({
147        "repo": scan.name,
148        "counts": c.to_json(),
149        "in-progress": group(Status::InProgress),
150        "blocked": group(Status::Blocked),
151        "claimed": group(Status::Claimed),
152        "open": group(Status::Open),
153    })
154}

156/// The status digest + `interrupted` id list — the `--json`
157/// payload, shared with `myco serve`.
158pub fn status_json(store: &impl Store) -> Result<serde_json::Value, String> {
159    let scan = digest::scan(store)?;
160    let interrupted = crate::continuity::interrupted_set(store, scan.live())?;
161    let mut v = digest_json(&scan);
162    v["interrupted"] = json!(interrupted
163        .iter()
164        .map(|i| i.to_string())
165        .collect::<Vec<_>>());
166    Ok(v)
167}

169/// The `--prompt` block payload — `{"prompt": "<block>"}` for
170/// `myco serve`, where the digest isn't a useful shape.
171pub fn prompt_json(store: &impl Store) -> Result<serde_json::Value, String> {
172    let scan = digest::scan(store)?;
173    let interrupted = crate::continuity::interrupted_set(store, scan.live())?;
174    Ok(json!({"prompt": prompt_block(&scan, &interrupted)}))
175}

177/// The `status` verb.
178pub fn h(ctx: &crate::registry::Ctx, a: &crate::registry::Args) -> Result<i32, String> {
179    let prompt = a.flag("prompt");
180    let json_out = ctx.json;
181    let (_proj, store) = items::open_read(ctx.dir.as_deref(), ctx.root.as_deref())?;
182    let scan = digest::scan(&store)?;
183    let interrupted = crate::continuity::interrupted_set(&store, scan.live())?;
184    if prompt {
185        if json_out {
186            ops::print_json(&json!({"prompt": prompt_block(&scan, &interrupted)}));
187        } else {
188            println!("{}", prompt_block(&scan, &interrupted));
189        }
190        return Ok(0);
191    }
192    if json_out {
193        let mut v = digest_json(&scan);
194        v["interrupted"] = json!(interrupted
195            .iter()
196            .map(|i| i.to_string())
197            .collect::<Vec<_>>());
198        ops::print_json(&v);
199        return Ok(0);
200    }
201    // human: counts + the two groups an agent asks about first
202    let c = digest::counts(&scan);
203    let groups = active_groups(&scan);
204    println!("{} — project ledger", scan.name);
205    println!("{}", c.oneline());
206    for (name, status) in [
207        ("in progress", Status::InProgress),
208        ("blocked", Status::Blocked),
209    ] {
210        let list = in_group(&groups, status);
211        if !list.is_empty() {
212            println!("{name}:");
213            for i in list {
214                println!(
215                    "  {}{}",
216                    items::compact(i.item),
217                    if interrupted.contains(&i.item.id) {
218                        " ⚡"
219                    } else {
220                        ""
221                    }
222                );
223            }
224        }
225    }
226    Ok(0)
227}

244    #[test]
245    fn prompt_block_groups_and_caps() {
246        let mut v = vec![
247            Item::new(ItemId::new(ItemKind::Task, 2), "work").status(Status::InProgress),
248            Item::new(ItemId::new(ItemKind::Issue, 1), "stuck").status(Status::Blocked),
249            Item::new(ItemId::new(ItemKind::Task, 5), "owned").status(Status::Claimed),
250            Item::new(ItemId::new(ItemKind::Task, 1), "hi").priority(Priority::Critical),
251            Item::new(ItemId::new(ItemKind::Task, 3), "mid"),
252            Item::new(ItemId::new(ItemKind::Task, 4), "done").status(Status::Done),
253            Item::new(ItemId::new(ItemKind::Task, 6), "parked").status(Status::Deferred),
254        ];
255        let block = prompt_block(&scan_with(v.clone()), &BTreeSet::new());
256        let lines: Vec<&str> = block.lines().collect();
257        assert_eq!(
258            lines[0],
259            "repo:myco tasks 2o/1c/1ip/0b backlog 1 issues 1o p 0o pl 0o adr 0o"
260        );
261        let whole = lines.join("\n");
262        let ip = lines.iter().position(|l| *l == "in-progress:").unwrap();
263        let bl = lines.iter().position(|l| *l == "blocked:").unwrap();
264        let cl = lines.iter().position(|l| *l == "claimed:").unwrap();
265        let op = lines.iter().position(|l| *l == "open:").unwrap();
266        assert!(ip < bl && bl < cl && cl < op);
267        // open group is crit-first
268        assert!(whole.contains("T-001 crit hi"));
269        assert!(lines.len() <= PROMPT_MAX_LINES);
270        // done/deferred never appear
271        assert!(!whole.contains("done") && !whole.contains("parked"));
272
273        // overflow: 70 open items → capped at 60 lines with a marker
274        v.clear();
275        for n in 1..=70 {
276            v.push(Item::new(
277                ItemId::new(ItemKind::Task, n),
278                format!("task {n}"),
279            ));
280        }
281        let block = prompt_block(&scan_with(v), &BTreeSet::new());
282        let lines: Vec<&str> = block.lines().collect();
283        assert!(lines.len() <= PROMPT_MAX_LINES, "{}", lines.len());
284        assert!(block.contains("… +"), "{}", block);
285    }

287    #[test]
288    fn prompt_block_caps_the_whole_block() {
289        // 30 in-progress + 30 blocked + 20 claimed + 20 open —
290        // >60 non-open lines must not overflow the cap; open is
291        // cut first, claimed next, in-progress last
292        let mut v = Vec::new();
293        for n in 1..=30 {
294            v.push(
295                Item::new(ItemId::new(ItemKind::Task, n), format!("wip {n}"))
296                    .status(Status::InProgress),
297            );
298        }
299        for n in 31..=60 {
300            v.push(
301                Item::new(ItemId::new(ItemKind::Task, n), format!("stuck {n}"))
302                    .status(Status::Blocked),
303            );
304        }
305        for n in 61..=80 {
306            v.push(
307                Item::new(ItemId::new(ItemKind::Task, n), format!("owned {n}"))
308                    .status(Status::Claimed),
309            );
310        }
311        for n in 81..=100 {
312            v.push(Item::new(
313                ItemId::new(ItemKind::Task, n),
314                format!("open {n}"),
315            ));
316        }
317        let block = prompt_block(&scan_with(v), &BTreeSet::new());
318        let lines: Vec<&str> = block.lines().collect();
319        assert!(lines.len() <= PROMPT_MAX_LINES, "{}", lines.len());
320        // in-progress is the last section cut — survives whole
321        assert!(block.contains("in-progress:"), "{block}");
322        assert!(block.contains("T-030"), "{block}");
323        // open + claimed were cut entirely (tail-first)
324        assert!(!block.contains("open:"), "{block}");
325        assert!(!block.contains("claimed:"), "{block}");
326        assert!(!block.contains("T-081"), "{block}");
327        // blocked shows a truncated tail + marker
328        assert!(block.contains("… +"), "{block}");
329    }
