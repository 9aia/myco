1//! `myco plan new|list` — execution plans in
2//! `plans/pl-NNNN-<slug>.md` with a phase checklist.
3
4use std::path::PathBuf;
5
6use clap::Subcommand;
7use myco_core::ItemKind;
8
9use crate::items;
10use crate::templates;
11
12#[derive(Subcommand)]
13pub enum Cmd {
14    /// File a plan — allocs a PL-id, writes the plan file with
15    /// frontmatter + a phase-checklist template.
16    New {
17        /// One-line title (drives the filename slug).
18        title: String,
19        #[command(flatten)]
20        body: items::BodyFlags,
21        #[command(flatten)]
22        meta: items::MetaFlags,
23    },
24    /// List plans.
25    List {
26        #[command(flatten)]
27        list: items::ListFlags,
28    },
29}
30
31/// The `plan` verb group.
32pub fn run(
33    cmd: Cmd,
34    dir: Option<PathBuf>,
35    root: Option<PathBuf>,
36    json_out: bool,
37    no_commit: bool,
38) -> Result<i32, String> {
39    let (proj, store) = items::open(dir.as_deref(), root.as_deref())?;
40    match cmd {
41        Cmd::New { title, body, meta } => items::new_file(
42            &proj,
43            &store,
44            &items::NewItem {
45                kind: ItemKind::Plan,
46                title,
47                body: &body,
48                meta: &meta,
49                template: templates::plan,
50                word: "plan",
51                json_out,
52                no_commit,
53            },
54        ),
55        Cmd::List { list } => items::list_files(&store, ItemKind::Plan, false, &list, json_out),
56    }
57}
