1//! Shared plumbing for the item verbs — the clap flag structs
2//! (`--pri/--owner/…`, `--status/--label/--limit`, `--body/--text`)
3//! that parse into `myco_ops::items` / `myco-core` types. The ops
4//! layer itself is re-exported so call sites read `items::X` for
5//! either.

11/// `--pri/--owner/--label/--blocks/--epic/--milestone` — the
12/// metadata flags shared by every `add`/`new` verb.
13#[derive(Debug, Default, clap::Args)]
14pub struct MetaFlags {
15    /// Priority: low|med|high|crit.
16    #[arg(long, visible_alias = "priority")]
17    pub pri: Option<myco_core::Priority>,
18    /// Owner actor id (`devin`, `human:luis`, …).
19    #[arg(long)]
20    pub owner: Option<myco_core::Actor>,
21    /// `#label` — repeatable.
22    #[arg(long)]
23    pub label: Vec<String>,
24    /// Ids this item blocks — comma-separated or repeatable.
25    #[arg(long, value_delimiter = ',')]
26    pub blocks: Vec<myco_core::ItemId>,
27    /// Epic name — the delivery grouping this item belongs to
28    /// (`[A-Za-z0-9_.-]+`).
29    #[arg(long)]
30    pub epic: Option<String>,
31    /// Milestone name — the target this item ships under
32    /// (`[A-Za-z0-9_.-]+`).
33    #[arg(long)]
34    pub milestone: Option<String>,
35}
36
37impl MetaFlags {
38    /// The parsed form — `NewItemMeta` for `new_file`/`add` paths.
39    pub fn to_meta(&self) -> NewItemMeta {
40        NewItemMeta {
41            priority: self.pri,
42            owner: self.owner.clone(),
43            labels: self.label.clone(),
44            blocks: self.blocks.clone(),
45            epic: self.epic.clone(),
46            milestone: self.milestone.clone(),
47        }
48    }
49}

51/// `--status/--owner/--label` list filters, shared by every `list`.
52#[derive(Debug, Default, clap::Args)]
53pub struct ListFlags {
54    /// Keep only items with this status.
55    #[arg(long)]
56    pub status: Option<myco_core::Status>,
57    /// Keep only items owned by this actor.
58    #[arg(long)]
59    pub owner: Option<String>,
60    /// Keep only items carrying all of these labels (repeatable).
61    #[arg(long)]
62    pub label: Vec<String>,
63    /// Keep only the first N items after filtering.
64    #[arg(long, value_name = "N")]
65    pub limit: Option<usize>,
66}
67
68impl ListFlags {
69    /// The parsed form — `Filter` for `list_files`.
70    pub fn filter(&self) -> Filter {
71        Filter {
72            status: self.status,
73            owner: self.owner.clone(),
74            labels: self.label.clone(),
75            limit: self.limit,
76        }
77    }
78
79    /// Does `item` pass the filters?
80    pub fn matches(&self, item: &Item) -> bool {
81        self.filter().matches(item)
82    }
83
84    /// `--limit N` as a `.take()` bound.
85    pub fn take(&self) -> usize {
86        self.limit.unwrap_or(usize::MAX)
87    }
88}

90/// A body-source flag pair — `--body FILE|-` reads a file or stdin,
91/// `--text` is inline. Mutually exclusive.
92#[derive(Debug, Default, clap::Args)]
93pub struct BodyFlags {
94    /// Body markdown from a file, or `-` for stdin.
95    #[arg(long, conflicts_with = "text", value_name = "FILE|-")]
96    pub body: Option<String>,
97    /// Body markdown inline.
98    #[arg(long)]
99    pub text: Option<String>,
100}
101
102impl BodyFlags {
103    /// The supplied body, or `None` → the caller uses the template.
104    pub fn read(&self) -> Result<Option<String>, String> {
105        if let Some(t) = &self.text {
106            return Ok(Some(t.clone()));
107        }
108        match self.body.as_deref() {
109            None => Ok(None),
110            Some("-") => {
111                use std::io::Read;
112                let mut s = String::new();
113                std::io::stdin()
114                    .read_to_string(&mut s)
115                    .map_err(|e| format!("read --body from stdin: {e}"))?;
116                Ok(Some(s))
117            }
118            Some(path) => std::fs::read_to_string(path)
119                .map(Some)
120                .map_err(|e| format!("read --body {path}: {e}")),
121        }
122    }
123}
