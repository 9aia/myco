1//! The policy decision point — an *external* engine (peregrin's
2//! job, never cuora's) that `cuora serve` consults before every
3//! vault-touching op — and before `list`/`check`, which
4//! materialize too (record decrypts, scoped-token gets, vault
5//! probes).
6//!
7//! One driver: `exec`. The configured command runs via `sh -c`
8//! once per decision; the [`PdpRequest`] goes in on stdin as one
9//! JSON line, the verdict comes back on stdout —
10//! `{"allow": bool, "reason"?: string, "obligations"?: [string]}`.
11//! Spawn failure, non-zero exit, missing `allow`, unparseable
12//! output, a timeout, or an oversized verdict are all
13//! [`Decision::Unreachable`] — what that means is the caller's
14//! `[serve.pdp] allow_on_error` flag.
15//!
16//! Hard bounds keep a misbehaving PDP from taking the broker
17//! with it, all without leaving `std`:
18//!
19//! - **Deadline** — the child is polled with `try_wait` against
20//!   `[serve.pdp] timeout_ms` (default 5 s) and killed past it;
21//!   a hung decider can never freeze the serve loop.
22//! - **Bounded verdict** — stdout is read through
23//!   `Read::take(MAX_VERDICT_BYTES + 1)`; a flooding `yes`-style
24//!   child fills its pipe, blocks, and dies on the deadline
25//!   instead of sinking memory.
26//! - **Non-blocking stdin** — the request is written on a helper
27//!   thread so a >64 KiB request can't deadlock the loop against
28//!   a child that never reads.
29//! - **Scrubbed env** — the child inherits only the
30//!   `cuora-shell` allowlist (same scrub `cuora run` applies);
31//!   `CUORA_CF_TOKEN`, `CLOUDFLARE_API_TOKEN`, and every other
32//!   parent credential stay out.
33//!
34//! The exec driver keeps the binary dependency-light and sync —
35//! no HTTP client, no async runtime, no daemon protocol. A PDP
36//! that wants a persistent socket can put `socat`/`nc` in its
37//! exec line; the wire stays the same.
38
39use std::io::{Read, Write};
40use std::process::{Command, Stdio};
41use std::time::{Duration, Instant};
42
43use cuora_core::PdpConfig;
44use serde::Serialize;

46/// The most a PDP verdict may be — 64 KiB. A verdict is a small
47/// JSON object; a child that floods stdout is broken, not
48/// chatty, and must not sink memory.
49const MAX_VERDICT_BYTES: u64 = 64 * 1024;
50
51/// One consult's deadline when `[serve.pdp] timeout_ms` is unset.
52const DEFAULT_TIMEOUT: Duration = Duration::from_secs(5);
53
54/// Grace to drain the verdict pipe after the child exits — a
55/// grandchild inheriting stdout could otherwise hold the pipe
56/// open forever.
57const DRAIN_GRACE: Duration = Duration::from_secs(2);
58
59/// `try_wait` poll interval inside the consult deadline.
60const POLL: Duration = Duration::from_millis(5);
61
62/// What serve asks the PDP — one materialization decision.
63#[derive(Debug, Serialize)]
64pub struct PdpRequest<'a> {
65    /// The claimed actor (self-asserted, advisory — attestation is
66    /// where real proof would ride).
67    pub actor: &'a str,
68    /// The requester's opaque identity claim, passed through
69    /// verbatim. cuora never interprets it; the PDP validates (or
70    /// ignores) it. Real signing is peregrin's side — a later ADR.
71    #[serde(skip_serializing_if = "Option::is_none")]
72    pub attestation: Option<&'a str>,
73    /// The op being decided (`get`/`put`/`rm`/`rotate`/`list`/
74    /// `check`).
75    pub op: &'a str,
76    /// The env the op resolves under (pin-aware).
77    pub env: &'a str,
78    /// The resolved ref path (env pin stripped) — absent for the
79    /// ref-less `list`/`check` consults.
80    #[serde(rename = "ref", skip_serializing_if = "Option::is_none")]
81    pub rref: Option<&'a str>,
82}
83
84/// What the PDP answered.
85#[derive(Debug, Clone, PartialEq, Eq)]
86pub enum Decision {
87    /// The op may run; optional reason/obligations ride along.
88    Allow {
89        /// PDP-supplied rationale.
90        reason: Option<String>,
91        /// Conditions on the allow (`allow-with-conditions`).
92        obligations: Option<Vec<String>>,
93    },
94    /// The op is refused.
95    Deny {
96        /// PDP-supplied rationale.
97        reason: Option<String>,
98    },
99    /// No verdict was obtained — spawn failed, non-zero exit,
100    /// timeout, oversized/unparseable output. The string is
101    /// diagnostics, never policy input.
102    Unreachable(String),
103}

105/// The exec-driver PDP — one `sh -c` spawn per decision.
106pub struct ExecPdp {
107    cmd: String,
108    allow_on_error: bool,
109    timeout: Duration,
110}
111
112impl ExecPdp {
113    /// From `[serve.pdp]` config.
114    pub fn new(cfg: &PdpConfig) -> Self {
115        ExecPdp {
116            cmd: cfg.exec.clone(),
117            allow_on_error: cfg.allow_on_error,
118            timeout: cfg
119                .timeout_ms
120                .map(Duration::from_millis)
121                .unwrap_or(DEFAULT_TIMEOUT),
122        }
123    }
124
125    /// `[serve.pdp] allow_on_error`.
126    pub fn allow_on_error(&self) -> bool {
127        self.allow_on_error
128    }
129
130    /// Ask the PDP. Never panics, never hangs — every failure
131    /// shape (spawn, timeout, flood, garbage) folds into
132    /// [`Decision::Unreachable`].
133    pub fn decide(&self, req: &PdpRequest) -> Decision {
134        let mut command = Command::new("sh");
135        command
136            .arg("-c")
137            .arg(&self.cmd)
138            .stdin(Stdio::piped())
139            .stdout(Stdio::piped())
140            // The PDP's stderr is not the audit wire — null it so a
141            // chatty decider can't pollute serve's event stream.
142            .stderr(Stdio::null());
143        // Same scrub `cuora run` applies — the PDP decides policy,
144        // it doesn't need the broker's credentials to do it.
145        cuora_shell::scrub_env(&mut command, &[]);
146        let mut child = match command.spawn() {
147            Ok(c) => c,
148            Err(e) => return Decision::Unreachable(format!("pdp spawn failed: {e}")),
149        };
150        // Serializing this POD request cannot fail — the expect
151        // documents the invariant, it doesn't guard a real path.
152        #[allow(clippy::unwrap_used)]
153        let line = serde_json::to_string(req).expect("pdp request serializes");
154        // The request can exceed the pipe buffer (a 64 KiB
155        // attestation serializes bigger) — write on a helper
156        // thread so a child that never reads can't deadlock the
157        // consult. Killing the child unblocks the writer via EPIPE.
158        let mut stdin = child.stdin.take();
159        std::thread::spawn(move || {
160            if let Some(mut s) = stdin.take() {
161                let _ = writeln!(s, "{line}");
162            }
163            // stdin drops here → the child sees EOF.
164        });
165        // Read the verdict bounded, also on a helper thread: a
166        // grandchild inheriting stdout could hold the pipe open
167        // past the child's exit and stall the drain.
168        let mut stdout = child.stdout.take();
169        let (tx, rx) = std::sync::mpsc::channel();
170        std::thread::spawn(move || {
171            let mut buf = Vec::new();
172            if let Some(out) = stdout.take() {
173                let _ = out.take(MAX_VERDICT_BYTES + 1).read_to_end(&mut buf);
174            }
175            let _ = tx.send(buf);
176        });
177        // wait_timeout semantics without a new dependency: poll
178        // try_wait against the deadline, kill on expiry. A
179        // flooding child that out-writes the bounded reader blocks
180        // on its own pipe and dies here — memory stays bounded.
181        let deadline = Instant::now() + self.timeout;
182        let status = loop {
183            match child.try_wait() {
184                Ok(Some(s)) => break s,
185                Ok(None) if Instant::now() >= deadline => {
186                    let _ = child.kill();
187                    let _ = child.wait();
188                    return Decision::Unreachable(format!(
189                        "pdp timed out after {}ms",
190                        self.timeout.as_millis()
191                    ));
192                }
193                Ok(None) => std::thread::sleep(POLL),
194                Err(e) => return Decision::Unreachable(format!("pdp wait failed: {e}")),
195            }
196        };
197        // The child exited so its own stdout is closed — a short
198        // grace covers the reader finishing; a leaked grandchild
199        // holding the pipe open must not stall the consult.
200        let out = match rx.recv_timeout(DRAIN_GRACE) {
201            Ok(buf) => buf,
202            Err(_) => {
203                return Decision::Unreachable("pdp stdout never drained".into());
204            }
205        };
206        // Size before status: a flooding child that out-writes the
207        // bounded reader dies on SIGPIPE — the oversized verdict is
208        // the root cause and the more useful verdict either way.
209        if out.len() as u64 > MAX_VERDICT_BYTES {
210            return Decision::Unreachable(format!("pdp verdict exceeds {MAX_VERDICT_BYTES} bytes"));
211        }
212        if !status.success() {
213            return Decision::Unreachable(format!("pdp exited {status}"));
214        }
215        parse_verdict(&out)
216    }
217}
218
219/// The verdict shape the PDP owes on stdout.
220#[derive(Debug, serde::Deserialize)]
221struct Verdict {
222    allow: bool,
223    reason: Option<String>,
224    obligations: Option<Vec<String>>,
225}
226
227fn parse_verdict(stdout: &[u8]) -> Decision {
228    match serde_json::from_slice::<Verdict>(stdout) {
229        Ok(v) if v.allow => Decision::Allow {
230            reason: v.reason,
231            obligations: v.obligations,
232        },
233        Ok(v) => Decision::Deny { reason: v.reason },
234        Err(e) => Decision::Unreachable(format!("pdp verdict unparseable: {e}")),
235    }
236}

262    #[test]
263    fn allow_deny_and_obligations_parse() {
264        let d = pdp(
265            "printf '%s' '{\"allow\":true,\"reason\":\"ok\",\"obligations\":[\"redact\"]}'",
266            false,
267        )
268        .decide(&req());
269        assert_eq!(
270            d,
271            Decision::Allow {
272                reason: Some("ok".into()),
273                obligations: Some(vec!["redact".into()])
274            }
275        );
276        let d = pdp(
277            "printf '%s' '{\"allow\":false,\"reason\":\"no prod\"}'",
278            false,
279        )
280        .decide(&req());
281        assert_eq!(
282            d,
283            Decision::Deny {
284                reason: Some("no prod".into())
285            }
286        );
287        // a bare allow carries no extras
288        let d = pdp("printf '%s' '{\"allow\":true}'", false).decide(&req());
289        assert_eq!(
290            d,
291            Decision::Allow {
292                reason: None,
293                obligations: None
294            }
295        );
296    }

298    #[test]
299    fn every_failure_shape_is_unreachable() {
300        // unspawnable command

302            pdp("/nonexistent-pdp-binary-xyz", false).decide(&req()),
303            Decision::Unreachable(_)

305        // non-zero exit

307            pdp("exit 3", false).decide(&req()),
308            Decision::Unreachable(_)

310        // garbage on stdout

313            Decision::Unreachable(_)

323            Decision::Allow {

390    #[test]
391    fn a_flooding_pdp_stays_bounded_and_dies() {
392        // `yes` writes forever — the bounded reader stops, the
393        // child blocks on a full pipe, the deadline kills it.
394        let p = ExecPdp::new(&PdpConfig {
395            exec: "yes".into(),
396            allow_on_error: false,
397            timeout_ms: Some(500),
398        });
399        let start = Instant::now();
400        assert!(matches!(p.decide(&req()), Decision::Unreachable(_)));
401        assert!(start.elapsed() < Duration::from_secs(10));
402        // an oversized-but-finite verdict is unreachable too
403        let p = ExecPdp::new(&PdpConfig {
404            exec: "head -c 100000 /dev/zero".into(),
405            allow_on_error: false,
406            timeout_ms: Some(5000),
407        });
408        match p.decide(&req()) {
409            Decision::Unreachable(why) => assert!(why.contains("exceeds"), "{why}"),
410            d => panic!("expected unreachable, got {d:?}"),
411        }
412    }
