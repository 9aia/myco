1//! cuora-serve — `cuora serve`: the ndjson request/response surface
2//! orchestrators (peregrin) embed (P-005).
3//!
4//! One JSON request per **stdin** line, one JSON response per
5//! **stdout** line; the audit wire stays on **stderr** (plus OTLP
6//! span export when `OTEL_EXPORTER_OTLP_ENDPOINT` is set) — three
7//! separate fds, so a reply never rides the event stream
8//! and an event never answers a request. A thin front over
9//! `cuora-ops` — every vault access goes through
10//! [`Ops`], so env-pin resolution, `[envs]` routing, and the
11//! audit events are the same ones the CLI emits.
12//!
13//! Protocol `cuora-serve/1`:
14//!
15//! ```json
16//! {"id": 1, "op": "get", "ref": "cuora://billing/stripe",
17//!  "env": "prod", "actor": "agent:peregrin", "attestation": "…"}
18//! ```
19//!
20//! → `{"id": 1, "ok": true, "result": {…}}` or
21//! `{"id": 1, "ok": false, "error": {"code": "…", "message": "…"}}`.
22//! `id` is echoed verbatim — any JSON value, absent → `null`.
23//!
24//! Ops: `get`/`put`/`rm`/`rotate`/`list`/`check` (each consults
25//! the configured `[serve.pdp]` first — `list` decrypts record
26//! metadata and `check` probes vaults including scoped-token
27//! gets, so they are materializations too), `manifest` (this
28//! protocol's self-description), and `shutdown` (clean exit).
29//! `run`/`env`/`deploy`/`init` stay single-shot CLI verbs —
30//! serve covers the port peregrin needs in-process.
31//!
32//! Hard bounds on the wire: a request line over
33//! [`MAX_LINE_BYTES`] (4 MiB) answers `bad_request` and the
34//! reader resyncs to the next line; `actor` (256 B),
35//! `attestation` (64 KiB), `ref` (1 KiB), `value` (64 KiB), and
36//! `id` (4 KiB serialized) are each capped. Blank or
37//! whitespace-only lines get **no reply** — keep-alive whitespace
38//! is legal and unanswered.
39//!
40//! Policy: requests carry *advisory* actor claims — `actor` is the
41//! attribution string audit records, `attestation` an opaque blob
42//! passed to the PDP verbatim. cuora verifies neither; signing is
43//! the PDP's side (P-005 is the boundary contract, not the
44//! enforcement). A configured PDP that denies — or that can't be
45//! reached under the default fail-closed posture — turns into a
46//! `denied`/`pdp_unreachable` error, a `denied` audit event, and
47//! no vault access. `manifest` and `shutdown` are protocol
48//! meta-ops: they skip guardrails and the PDP consult by design
49//! (gating session lifecycle is the embedder's spawn decision,
50//! not a materialization).
51//!
52//! Values cross this wire deliberately: `get` returns them,
53//! `put`/`rotate` take them — the parent owns the channel exactly
54//! like it owns `cuora get`'s stdout.

67/// The most one request line may be — 4 MiB. Legitimate requests
68/// are bytes-to-kilobytes; an unbounded `read_line` lets a broken
69/// or hostile embedder exhaust memory, so an oversized line is a
70/// `bad_request` and the reader resyncs to the next line.
71pub const MAX_LINE_BYTES: u64 = 4 * 1024 * 1024;
72
73/// Field caps — sane sizes consistent with core's 64-char env
74/// cap; every oversized field is a `bad_request`, never a
75/// consult or a vault touch.
76const MAX_ACTOR_BYTES: usize = 256;
77const MAX_ATTESTATION_BYTES: usize = 64 * 1024;
78const MAX_REF_BYTES: usize = 1024;
79const MAX_VALUE_BYTES: usize = 64 * 1024;
80const MAX_ID_BYTES: usize = 4 * 1024;
81
82/// Serve the ndjson loop on `reader`/`writer` against the loaded
83/// [`Ops`] — the `cuora serve` verb's whole job. Runs until EOF or
84/// a `shutdown` request. A malformed, non-UTF-8, or oversized
85/// line is a `bad_request` response, never a crash and never the
86/// end of the loop; only a real stdout/reader I/O failure
87/// propagates.
88pub fn run(ops: &Ops, reader: &mut impl BufRead, writer: &mut impl Write) -> io::Result<()> {
89    let pdp = ops
90        .config()
91        .serve
92        .as_ref()
93        .and_then(|s| s.pdp.as_ref())
94        .map(ExecPdp::new);
95    let mut buf = Vec::new();
96    loop {
97        buf.clear();
98        // Bounded read — `take` caps how much a single line may
99        // hold; `read_until` on the take stops at the cap even
100        // when no newline ever arrives, and unlike `read_line`
101        // it reports raw bytes so UTF-8 validation is ours.
102        match reader
103            .by_ref()
104            .take(MAX_LINE_BYTES + 1)
105            .read_until(b'\n', &mut buf)
106        {
107            // EOF — the parent closed the wire; flush buffered
108            // audit events to OTLP before the clean exit.
109            Ok(0) => {
110                cuora_ops::audit::flush();
111                return Ok(());
112            }
113            Ok(_) => {}
114            Err(e) => return Err(e),
115        }
116        if buf.len() as u64 > MAX_LINE_BYTES {
117            // Oversized — swallow the remainder of the line so the
118            // next read starts at a real request boundary.
119            discard_to_newline(reader)?;
120            write_reply(
121                writer,
122                &err_reply(
123                    serde_json::Value::Null,
124                    "bad_request",
125                    format!("request line exceeds the {MAX_LINE_BYTES}-byte limit"),
126                ),
127            )?;
128            continue;
129        }
130        // Not UTF-8 — a bad line like any other; keep serving. The
131        // whole line was already consumed through its newline, so
132        // no resync is needed here.
133        let line = match String::from_utf8(std::mem::take(&mut buf)) {
134            Ok(l) => l,
135            Err(_) => {
136                write_reply(
137                    writer,
138                    &err_reply(
139                        serde_json::Value::Null,
140                        "bad_request",
141                        "line is not valid UTF-8",
142                    ),
143                )?;
144                continue;
145            }
146        };
147        // Blank/whitespace lines get no reply (L-1 — documented in
148        // the manifest and protocol reference).
149        if line.trim().is_empty() {
150            continue;
151        }
152        let reply = dispatch(ops, pdp.as_ref(), &line);
153        let done = reply.shutdown;
154        write_reply(writer, &reply)?;
155        // Long-running surface — export buffered audit events per
156        // request rather than accumulating until shutdown.
157        cuora_ops::audit::flush();
158        if done {
159            return Ok(());
160        }
161    }
162}

164/// Consume bytes until (and including) the next `\n`, or EOF —
165/// the resync step after an oversized line. Bounded: the discard
166/// buffer is the reader's own fill buffer.
167fn discard_to_newline(reader: &mut impl BufRead) -> io::Result<()> {
168    loop {
169        let buf = reader.fill_buf()?;
170        if buf.is_empty() {
171            return Ok(());
172        }
173        match buf.iter().position(|b| *b == b'\n') {
174            Some(i) => {
175                reader.consume(i + 1);
176                return Ok(());
177            }
178            None => {
179                let n = buf.len();
180                reader.consume(n);
181            }
182        }
183    }
184}

186/// One incoming line, validated loosely — field-level errors are
187/// `bad_request`s at dispatch, not parse failures here. No `Debug`
188/// derive: `value` is plaintext on the wire and must never print.
189#[derive(Deserialize)]
190struct Request {
191    /// Echo handle — any JSON value, echoed verbatim.
192    #[serde(default)]
193    id: serde_json::Value,
194    /// The verb name.
195    op: String,
196    /// `cuora://` ref (`get`/`put`/`rm`/`rotate`).
197    #[serde(default, rename = "ref")]
198    rref: Option<String>,
199    /// Per-request env override (else the loaded context; an
200    /// `cuora://<env>/…` pin still wins).
201    #[serde(default)]
202    env: Option<String>,
203    /// The new value (`put`/`rotate`) — on the wire, not in args.
204    #[serde(default)]
205    value: Option<String>,
206    /// Claimed actor id — advisory attribution, recorded on audit.
207    #[serde(default)]
208    actor: Option<String>,
209    /// Opaque identity claim for the PDP — passthrough, never
210    /// interpreted.
211    #[serde(default)]
212    attestation: Option<String>,
213}
214
215/// The per-request context threaded through dispatch — the
216/// validated claims (actor/env) plus the wire fields the PDP
217/// consult needs.
218struct Ctx<'a> {
219    id: &'a serde_json::Value,
220    actor: Option<Actor>,
221    env: Option<Env>,
222    attestation: Option<&'a str>,
223}

225/// One outgoing line.
226struct Reply {
227    id: serde_json::Value,
228    ok: bool,
229    result: Option<serde_json::Value>,
230    /// The PDP verdict, when one was consulted — the requester
231    /// sees the same `policy:` the audit event records.
232    policy: Option<PolicyVerdict>,
233    code: Option<&'static str>,
234    message: Option<String>,
235    /// Set on `shutdown` — respond, then break the loop.
236    shutdown: bool,
237}

239fn ok_reply(id: serde_json::Value, result: serde_json::Value) -> Reply {
240    Reply {
241        id,
242        ok: true,
243        result: Some(result),
244        policy: None,
245        code: None,
246        message: None,
247        shutdown: false,
248    }
249}

251fn err_reply(id: serde_json::Value, code: &'static str, message: impl Into<String>) -> Reply {
252    Reply {
253        id,
254        ok: false,
255        result: None,
256        policy: None,
257        code: Some(code),
258        message: Some(message.into()),
259        shutdown: false,
260    }
261}

263fn write_reply(writer: &mut impl Write, reply: &Reply) -> io::Result<()> {
264    let mut v = serde_json::json!({
265        "id": reply.id,
266        "ok": reply.ok,
267    });
268    if let Some(r) = &reply.result {
269        v["result"] = r.clone();
270    }
271    if let Some(p) = &reply.policy {
272        v["policy"] = serde_json::to_value(p).map_err(io::Error::other)?;
273    }
274    if let (Some(code), Some(message)) = (reply.code, &reply.message) {
275        v["error"] = serde_json::json!({"code": code, "message": message});
276    }
277    // A serde_json::Value always serializes — the expect documents
278    // the invariant rather than guarding a reachable panic.
279    #[allow(clippy::unwrap_used)]
280    let line = serde_json::to_string(&v).expect("reply serializes");
281    writeln!(writer, "{line}")
282}

284/// Route one request line — everything that can fail per-request
285/// folds into the reply; nothing here panics.
286fn dispatch(ops: &Ops, pdp: Option<&ExecPdp>, line: &str) -> Reply {
287    let req: Request = match serde_json::from_str(line) {
288        Ok(r) => r,
289        Err(e) => {
290            // Try to salvage just the id so the caller can
291            // correlate; a totally broken line gets null.
292            let id = serde_json::from_str::<serde_json::Value>(line)
293                .ok()
294                .and_then(|v| v.get("id").cloned())
295                .unwrap_or(serde_json::Value::Null);
296            return err_reply(id, "bad_request", format!("malformed request: {e}"));
297        }
298    };
299    let id = req.id.clone();
300    // Field caps before anything else — an oversized field is a
301    // bad_request, never a consult and never a vault touch.
302    for (field, value, cap) in [
303        ("actor", req.actor.as_deref(), MAX_ACTOR_BYTES),
304        (
305            "attestation",
306            req.attestation.as_deref(),
307            MAX_ATTESTATION_BYTES,
308        ),
309        ("ref", req.rref.as_deref(), MAX_REF_BYTES),
310        ("value", req.value.as_deref(), MAX_VALUE_BYTES),
311    ] {
312        if let Some(v) = value {
313            if v.len() > cap {
314                return err_reply(
315                    id,
316                    "bad_request",
317                    format!("field {field:?} exceeds the {cap}-byte limit"),
318                );
319            }
320        }
321    }
322    // `id` is any JSON value echoed back verbatim — cap its
323    // serialized size so a giant id can't inflate every reply.
324    if serde_json::to_string(&req.id)
325        .map(|s| s.len() > MAX_ID_BYTES)
326        .unwrap_or(true)
327    {
328        return err_reply(
329            id,
330            "bad_request",
331            format!("field \"id\" exceeds the {MAX_ID_BYTES}-byte limit"),
332        );
333    }
334    // Per-request claims, validated up front — bad values are
335    // bad_request, not silent fallback.
336    let actor = match req.actor.as_deref().map(Actor::new) {
337        Some(Ok(a)) => Some(a),
338        Some(Err(e)) => return err_reply(id, "bad_request", e.to_string()),
339        None => None,
340    };
341    let env = match req.env.as_deref().map(Env::new) {
342        Some(Ok(e)) => Some(e),
343        Some(Err(e)) => return err_reply(id, "bad_request", e.to_string()),
344        None => None,
345    };
346
347    let ctx = Ctx {
348        id: &id,
349        actor,
350        env,
351        attestation: req.attestation.as_deref(),
352    };
353    match req.op.as_str() {
354        "manifest" => ok_reply(id, manifest()),
355        "shutdown" => {
356            let mut r = ok_reply(id.clone(), serde_json::json!({"shutdown": true}));
357            r.shutdown = true;
358            let scoped = ops.scoped(ctx.actor, ctx.env, None);
359            scoped.emit(Op::Serve, None, Outcome::Ok);
360            r
361        }
362        "list" => refless(ops, pdp, ctx, Op::List, |s| {
363            let (env, _) = s.list(None)?;
364            let (_, records) = s.list_records(None)?;
365            let arr: Vec<serde_json::Value> = records
366                .iter()
367                .map(|(r, rec)| record_json(&env, r, rec))
368                .collect();
369            s.emit(Op::List, None, Outcome::Ok);
370            Ok(serde_json::json!({"env": env.as_str(), "refs": arr}))
371        }),
372        "check" => refless(ops, pdp, ctx, Op::Check, |s| {
373            let report = s.check();
374            // L-6: a check that found problems isn't an `ok`
375            // on the audit wire — the outcome mirrors the
376            // probe results, like the CLI's exit code.
377            s.emit(
378                Op::Check,
379                None,
380                if report.problems.is_empty() {
381                    Outcome::Ok
382                } else {
383                    Outcome::Error
384                },
385            );
386            Ok(check_json(&report))
387        }),
388        "get" | "put" | "rm" | "rotate" => vault_op(ops, pdp, ctx, &req),
389        other => {
390            let msg = match other.parse::<Op>() {
391                // A real cuora verb that serve deliberately doesn't
392                // cover — name the CLI path rather than "unknown".
393                Ok(op) => format!("op '{op}' is not served — use the single-shot CLI"),
394                Err(_) => format!("unknown op {other:?}"),
395            };
396            err_reply(id, "unknown_op", msg)
397        }
398    }
399}

401/// The ops that don't take a ref — `list`/`check`. Guardrails
402/// still apply, and the PDP is consulted too: both *do*
403/// materialize (`list` decrypts every record's metadata,
404/// `check` probes vaults — including scoped-token `get`s and
405/// outbound signed HTTP against cf vaults). The verdict attaches
406/// to the scoped handle before the op runs, so the incidental
407/// materializations inside carry the policy context.
408fn refless(
409    ops: &Ops,
410    pdp: Option<&ExecPdp>,
411    ctx: Ctx<'_>,
412    op: Op,
413    f: impl FnOnce(&Ops) -> Result<serde_json::Value, OpsError>,
414) -> Reply {
415    let mut scoped = ops.scoped(ctx.actor, ctx.env, None);
416    if let Some(reply) = guardrail_reply(&scoped, ctx.id, op, None) {
417        return reply;
418    }
419    let resolved_env = scoped.env().clone();
420    let verdict = match consult(
421        pdp,
422        &mut scoped,
423        ctx.id,
424        ctx.attestation,
425        op,
426        &resolved_env,
427        None,
428    ) {
429        Ok(v) => v,
430        Err(reply) => return *reply,
431    };
432    scoped.set_policy(verdict.clone());
433    match f(&scoped) {
434        Ok(result) => {
435            let mut reply = ok_reply(ctx.id.clone(), result);
436            reply.policy = verdict;
437            reply
438        }
439        Err(e) => {
440            scoped.emit(op, None, Outcome::Error);
441            let mut reply = err_reply(ctx.id.clone(), "op_error", e.to_string());
442            reply.policy = verdict;
443            reply
444        }
445    }
446}

448/// Advisory guardrails — same check the CLI applies. A denial is
449/// a `denied` error plus the usual denied audit event.
450fn guardrail_reply(
451    scoped: &Ops,
452    id: &serde_json::Value,
453    op: Op,
454    rref: Option<&str>,
455) -> Option<Reply> {
456    let g = scoped.config().guardrails.as_ref()?;
457    if g.allows(op) {
458        return None;
459    }
460    let norm = rref.and_then(|s| s.parse::<Ref>().ok().map(|r| scoped.resolve(&r).1.path()));
461    scoped.emit(op, norm.as_deref().or(rref), Outcome::Denied);
462    Some(err_reply(
463        id.clone(),
464        "denied",
465        format!("op '{op}' denied by guardrails (advisory)"),
466    ))
467}

469/// The shared PDP consult — every op that can materialize goes
470/// through it (`get`/`put`/`rm`/`rotate`, plus the ref-less
471/// `list`/`check`). `Ok(verdict)` means the op may run and the
472/// verdict should ride the reply's `policy` field; `Err(reply)`
473/// is the finished denial response (boxed to keep `Result`
474/// small) — the consult already set the policy and emitted the
475/// `denied` audit event on `scoped`.
476fn consult(
477    pdp: Option<&ExecPdp>,
478    scoped: &mut Ops,
479    id: &serde_json::Value,
480    attestation: Option<&str>,
481    op: Op,
482    env: &Env,
483    rref: Option<&str>,
484) -> Result<Option<PolicyVerdict>, Box<Reply>> {
485    let Some(pdp) = pdp else { return Ok(None) };
486    match pdp.decide(&PdpRequest {
487        actor: scoped.actor().as_str(),
488        attestation,
489        op: op.as_str(),
490        env: env.as_str(),
491        rref,
492    }) {
493        Decision::Allow {
494            reason,
495            obligations,
496        } => Ok(Some(PolicyVerdict {
497            allow: true,
498            reason,
499            obligations,
500        })),
501        Decision::Deny { reason } => {
502            let verdict = Some(PolicyVerdict {
503                allow: false,
504                reason: reason.clone(),
505                obligations: None,
506            });
507            scoped.set_policy(verdict.clone());
508            scoped.emit(op, rref, Outcome::Denied);
509            let mut reply = err_reply(
510                id.clone(),
511                "denied",
512                reason
513                    .map(|r| format!("denied by policy: {r}"))
514                    .unwrap_or_else(|| "denied by policy".to_string()),
515            );
516            reply.policy = verdict;
517            Err(Box::new(reply))
518        }
519        Decision::Unreachable(why) => {
520            if !pdp.allow_on_error() {
521                // Fail closed — an unreachable policy engine must
522                // not silently mean "allowed".
523                let verdict = Some(PolicyVerdict {
524                    allow: false,
525                    reason: Some(why.clone()),
526                    obligations: None,
527                });
528                scoped.set_policy(verdict.clone());
529                scoped.emit(op, rref, Outcome::Denied);
530                let mut reply = err_reply(
531                    id.clone(),
532                    "pdp_unreachable",
533                    format!("policy decision point unreachable: {why}"),
534                );
535                reply.policy = verdict;
536                return Err(Box::new(reply));
537            }
538            // allow_on_error — the op runs and the audit event
539            // records the degraded verdict.
540            Ok(Some(PolicyVerdict {
541                allow: true,
542                reason: Some(format!(
543                    "pdp unreachable — allowed by allow_on_error: {why}"
544                )),
545                obligations: None,
546            }))
547        }
548    }
549}

551/// The vault-touching ops — parse, validate, guardrails, resolve,
552/// PDP, execute. Validation finishes *before* the consult (a
553/// doomed request never spawns a PDP), and the consult happens
554/// *before* the vault is opened; the verdict attaches to the
555/// scoped handle so every audit event the op emits carries the
556/// policy context.
557fn vault_op(ops: &Ops, pdp: Option<&ExecPdp>, ctx: Ctx<'_>, req: &Request) -> Reply {
558    let mut scoped = ops.scoped(ctx.actor, ctx.env, None);
559    let raw = match req.rref.as_deref() {
560        Some(r) => r,
561        None => {
562            return err_reply(
563                ctx.id.clone(),
564                "bad_request",
565                format!("op '{}' needs a ref", req.op),
566            )
567        }
568    };
569    let parsed = match raw.parse::<Ref>() {
570        Ok(r) => r,
571        Err(e) => return err_reply(ctx.id.clone(), "bad_request", e.to_string()),
572    };
573    let op = match req.op.parse::<Op>() {
574        Ok(o) => o,
575        // The dispatch table only routes get/put/rm/rotate here —
576        // the parse is total by construction.
577        Err(e) => return err_reply(ctx.id.clone(), "bad_request", e.to_string()),
578    };
579    // `put`/`rotate` are doomed without a value — reject before
580    // the consult so a malformed request never spawns a PDP.
581    if matches!(op, Op::Put | Op::Rotate) {
582        if let Err((code, msg)) = req_secret(req) {
583            return err_reply(ctx.id.clone(), code, msg);
584        }
585    }
586    if let Some(reply) = guardrail_reply(&scoped, ctx.id, op, Some(raw)) {
587        return reply;
588    }
589    let (resolved_env, resolved_ref) = scoped.resolve(&parsed);
590    let resolved_path = resolved_ref.path();
591    let verdict = match consult(
592        pdp,
593        &mut scoped,
594        ctx.id,
595        ctx.attestation,
596        op,
597        &resolved_env,
598        Some(&resolved_path),
599    ) {
600        Ok(v) => v,
601        Err(reply) => return *reply,
602    };
603    scoped.set_policy(verdict.clone());
604
605    match execute(&scoped, op, &parsed, &resolved_env, &resolved_ref, req) {
606        Ok(v) => {
607            let mut reply = ok_reply(ctx.id.clone(), v);
608            reply.policy = verdict;
609            reply
610        }
611        Err((code, msg)) => {
612            scoped.emit(op, Some(&resolved_ref.path()), Outcome::Error);
613            let mut reply = err_reply(ctx.id.clone(), code, msg);
614            reply.policy = verdict;
615            reply
616        }
617    }
618}

620/// Dispatch the four vault ops on the *unstripped* parsed ref —
621/// `scoped.resolve` already decided env for the PDP context, but
622/// the ops calls resolve once themselves, so the pin semantics
623/// stay single-application (a stripped ref could re-pin wrongly).
624/// `env`/`r` are only for response shaping.
625fn execute(
626    scoped: &Ops,
627    op: Op,
628    parsed: &Ref,
629    env: &Env,
630    r: &Ref,
631    req: &Request,
632) -> Result<serde_json::Value, (&'static str, String)> {
633    match op {
634        Op::Get => match scoped.get(parsed) {
635            Ok(Some(rec)) => {
636                let mut v = record_json(env, r, &rec);
637                v["value"] = rec.value.expose().into();
638                Ok(v)
639            }
640            Ok(None) => Err(("not_found", format!("{parsed} is not stored in env {env}"))),
641            Err(e) => Err(ops_err(&e)),
642        },
643        Op::Put => {
644            let secret = req_secret(req)?;
645            match scoped.put(parsed, secret) {
646                Ok(rec) => Ok(record_json(env, r, &rec)),
647                Err(e) => Err(ops_err(&e)),
648            }
649        }
650        Op::Rm => match scoped.rm(parsed) {
651            Ok(true) => Ok(serde_json::json!({
652                "removed": true,
653                "ref": parsed.to_string(),
654                "env": env.as_str(),
655            })),
656            Ok(false) => Err(("not_found", format!("{parsed} is not stored in env {env}"))),
657            Err(e) => Err(ops_err(&e)),
658        },
659        Op::Rotate => {
660            let secret = req_secret(req)?;
661            match scoped.rotate(parsed, secret) {
662                Ok(rec) => Ok(record_json(env, r, &rec)),
663                Err(e) => Err(ops_err(&e)),
664            }
665        }
666        // vault_op only routes the four ref ops here.
667        other => Err(("op_error", format!("op {other} is not a vault op"))),
668    }
669}
670
671/// `put`/`rotate` need a non-empty `value` — same rule as the
672/// CLI's empty-secret refusal.
673fn req_secret(req: &Request) -> Result<Secret, (&'static str, String)> {
674    match req.value.as_deref() {
675        Some(v) if !v.is_empty() => Ok(Secret::new(v)),
676        _ => Err((
677            "bad_request",
678            format!("op '{}' needs a non-empty value", req.op),
679        )),
680    }
681}
682
683/// Map an [`OpsError`] to a wire `(code, message)`.
684fn ops_err(e: &OpsError) -> (&'static str, String) {
685    match e {
686        OpsError::Vault(VaultError::NotFound { .. }) | OpsError::NotStored { .. } => {
687            ("not_found", e.to_string())
688        }
689        OpsError::Ref(_) | OpsError::Env(_) | OpsError::Actor(_) => ("bad_request", e.to_string()),
690        _ => ("op_error", e.to_string()),
691    }
692}

694/// The canonical record shape — same fields as `record_json` in
695/// the CLI: resolved ref + env + record metadata. `value` is added
696/// by `get` only.
697fn record_json(env: &Env, r: &Ref, rec: &SecretRecord) -> serde_json::Value {
698    serde_json::json!({
699        "ref": r.to_string(),
700        "env": env.as_str(),
701        "version": rec.version,
702        "created_at": rec.created_at.to_string(),
703        "rotated_at": rec.rotated_at.map(|t| t.to_string()),
704    })
705}

707/// The `check` result — mirrors `cuora check --json`, plus `ok`.
708fn check_json(report: &cuora_ops::CheckReport) -> serde_json::Value {
709    let secrets: serde_json::Map<String, serde_json::Value> = report
710        .secrets
711        .iter()
712        .map(|s| {
713            (
714                s.var.clone(),
715                serde_json::json!({
716                    "ref": s.rref,
717                    "env": s.env,
718                    "resolves": s.resolves,
719                }),
720            )
721        })
722        .collect();
723    let vaults: Vec<serde_json::Value> = report
724        .vaults
725        .iter()
726        .map(|v| {
727            serde_json::json!({
728                "name": v.name,
729                "kind": v.kind,
730                "path": v.path,
731                "ok": v.ok,
732                "error": v.error,
733            })
734        })
735        .collect();
736    let guardrails = report.guardrails.as_ref().map(|g| {
737        serde_json::json!({
738            "default": match g.default {
739                cuora_core::GuardDefault::Allow => "allow",
740                cuora_core::GuardDefault::Deny => "deny",
741            },
742            "allow": g.allow.iter().map(|o| o.as_str()).collect::<Vec<_>>(),
743            "deny": g.deny.iter().map(|o| o.as_str()).collect::<Vec<_>>(),
744            "deny_adhoc": g.deny_adhoc,
745        })
746    });
747    serde_json::json!({
748        "ok": report.problems.is_empty(),
749        "config": report.config_path,
750        "env": report.env.as_str(),
751        "vault": report.vault,
752        "vaults": vaults,
753        "secrets": secrets,
754        "guardrails": guardrails,
755        "errors": report.problems,
756    })
757}

859    /// A tiny PDP script — writes `body` to stdout, or `cat`s the
860    /// request into `capture` first. Exec driver runs `sh -c` so a
861    /// bare path works as the command.
862    fn pdp_script(home: &tempfile::TempDir, capture: bool, body: &str) -> String {
863        let path = home.path().join("pdp.sh");
864        let first = if capture {
865            format!("cat > {}\n", home.path().join("pdp-req.json").display())
866        } else {
867            "cat >/dev/null\n".to_string()
868        };
869        std::fs::write(&path, format!("#!/bin/sh\n{first}{body}\n")).unwrap();
870        // exec runs `sh -c <cmd>` — `sh <path>` needs no chmod.
871        format!("sh {}", path.display())
872    }

874    /// Run the loop over `input` bytes, return the decoded replies.
875    fn serve(ops: &Ops, input: &[u8]) -> Vec<serde_json::Value> {
876        let mut out = Vec::new();
877        run(ops, &mut &input[..], &mut out).unwrap();
878        String::from_utf8(out)
879            .unwrap()
880            .lines()
881            .map(|l| serde_json::from_str(l).unwrap())
882            .collect()
883    }

885    fn replies(ops: &Ops, input: &str) -> Vec<serde_json::Value> {
886        serve(ops, input.as_bytes())
887    }

1211    fn field_caps_are_bad_request() {
1212        let (_home, ops) = fixture("");
1213        let big = |n: usize| "x".repeat(n);
1214        let cases = [
1215            // actor > 256B
1216            format!("{{\"op\":\"list\",\"actor\":\"agent:{}\"}}", big(300)),
1217            // attestation > 64KiB
1218            format!(
1219                "{{\"op\":\"get\",\"ref\":\"cuora://a/b\",\"attestation\":\"{}\"}}",
1220                big(70 * 1024)
1221            ),
1222            // ref > 1KiB
1223            format!("{{\"op\":\"get\",\"ref\":\"cuora://{}\"}}", big(2000)),
1224            // value > 64KiB
1225            format!(
1226                "{{\"op\":\"put\",\"ref\":\"cuora://a/b\",\"value\":\"{}\"}}",
1227                big(70 * 1024)
1228            ),
1229            // id > 4KiB serialized
1230            format!("{{\"id\":\"{}\",\"op\":\"manifest\"}}", big(5000)),
1231        ];
1232        for line in cases {
1233            let out = replies(&ops, &format!("{line}\n"));
1234            assert_eq!(out.len(), 1);
1235            assert_eq!(out[0]["error"]["code"], "bad_request", "{line:.80}…");
1236        }
1237        // and a legit-sized request still works afterwards
1238        let out = replies(&ops, "{\"op\":\"manifest\"}\n");
1239        assert_eq!(out[0]["ok"], true);
1240    }
