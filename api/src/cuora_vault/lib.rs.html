1//! cuora-vault — the readable-store port.
2//!
3//! A [`Vault`] is a store of [`SecretRecord`]s keyed by
4//! `(env, ref-path)`. Every impl stores opaque ciphertext — stores
5//! never see plaintext (ADR-0001); impls are `cuora-vault-age`
6//! (local file, the default), `cuora-vault-cf` (self-hosted
7//! Cloudflare Worker + D1), and `cuora-vault-cached` (read-through
8//! encrypted local cache in front of any backend, P-007). The
9//! trait is sync by design — no async runtime in the binary; an
10//! HTTP-backed impl uses a blocking client.
11//!
12//! The contract below is conformance-tested: every impl runs the
13//! `cuora-testkit` suite and must agree on semantics —
14//!
15//! - `put` — upsert. New ref → `version = 1`, `created_at = now`,
16//!   `rotated_at = None`. Existing ref → value replaced, `version`
17//!   bumps, `created_at` preserved, `rotated_at` untouched.
18//! - `get` — `Ok(None)` when the ref isn't stored in that env.
19//! - `rm` — `Ok(true)` removed, `Ok(false)` it wasn't there.
20//! - `list` — the refs stored in that env, sorted.
21//! - `rotate` — deliberate rotation: `Err(NotFound)` on a missing
22//!   ref, else new value + `version` bump + `rotated_at = now`.
23//! - envs are isolated keyspaces — same ref path in `dev` and
24//!   `prod` are independent records.
25
26use cuora_core::{Env, Ref, Secret};
27use time::OffsetDateTime;
28
29/// A stored secret plus its bookkeeping.
30#[derive(Debug, Clone)]
31pub struct SecretRecord {
32    /// The value — redacting [`Secret`]; `.expose()` is the only
33    /// way out, at edges only.
34    pub value: Secret,
35    /// Write generation: `1` on first put, +1 per put/rotate.
36    pub version: u64,
37    /// When the record was first stored.
38    pub created_at: OffsetDateTime,
39    /// Last deliberate rotation (`rotate`), if ever.
40    pub rotated_at: Option<OffsetDateTime>,
41}
42
43/// A readable secret store.
44pub trait Vault: Send + Sync {
45    /// Fetch the record for `rref` in `env`, or `None`.
46    fn get(&self, env: &Env, rref: &Ref) -> Result<Option<SecretRecord>, VaultError>;
47
48    /// Upsert `value` at `rref` in `env`; returns the stored record.
49    fn put(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError>;
50
51    /// Remove `rref` from `env`; `true` when something was removed.
52    fn rm(&self, env: &Env, rref: &Ref) -> Result<bool, VaultError>;
53
54    /// All refs stored in `env`, sorted.
55    fn list(&self, env: &Env) -> Result<Vec<Ref>, VaultError>;
56
57    /// Rotate `rref` in `env` to a new value — `version` bump +
58    /// `rotated_at`. [`VaultError::NotFound`] on a missing ref.
59    fn rotate(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError>;
60}
61
62/// Vault port errors. Impls map their own failure modes onto these
63/// variants so callers can match without knowing the backend.
64#[derive(Debug, thiserror::Error)]
65pub enum VaultError {
66    /// The ref isn't stored in that env (rotate of a missing ref).
67    #[error("no secret at cuora://{rref} in env {env}")]
68    NotFound {
69        /// The bare ref path.
70        rref: String,
71        /// The env name.
72        env: String,
73    },
74    /// Filesystem / network I/O failure.
75    #[error("vault i/o: {0}")]
76    Io(String),
77    /// Encryption/decryption failure (bad identity, corrupt file).
78    #[error("vault crypto: {0}")]
79    Crypto(String),
80    /// Misconfigured backend (bad path, missing identity file).
81    #[error("vault config: {0}")]
82    Config(String),
83    /// Anything else the backend wants to say.
84    #[error("vault backend: {0}")]
85    Backend(String),
86}
87
88impl From<std::io::Error> for VaultError {
89    fn from(e: std::io::Error) -> Self {
90        VaultError::Io(e.to_string())
91    }
92}
