1//! Engine ops — `cli.<op>` wire queries dispatch into the `kuda`
2//! engine lib through this binary's own `run` verb: each call
3//! re-execs `tamias-driver-kuda run <argv>` with the request's repo
4//! as cwd, so `kuda::dispatch` writes to the child's captured stdout
5//! and never touches the serve loop's ndjson stream (the engine's
6//! emit() exits on broken pipe — one more reason ops stay out of
7//! the serve process).
8//!
9//! `ops` answers the manifest table below as JSON. Long-running
10//! surfaces (`watch`, `mcp`, `acp`) are deliberately *not* ops —
11//! they never return, so they stay out of the request/response wire.
12//! They live in the manifest's `surfaces` list instead: surface
13//! drivers (tamias-driver-cli / -mcp, the pavo-driver-kit `surface`
14//! layer) expose them as streaming subcommands —
15//! `tamias-driver-kuda cli watch` ≡ `kuda watch`.
16
17use pavo_protocol::wire::{QueryResult, Response};
18use serde_json::{json, Map, Value};
19use std::path::Path;

21/// One op param, typed the way `pavo-driver-kit`'s surface layer
22/// spells it: `arg` = positional, `opt` = `--name [v]`, `flag` =
23/// `--name` bool. (Bare-string params in a manifest read as `opt` —
24/// typed entries are what let generated surfaces not guess.)
25struct Param {
26    name: &'static str,
27    kind: &'static str,
28}
29
30const fn arg(name: &'static str) -> Param {
31    Param { name, kind: "arg" }
32}
33const fn opt(name: &'static str) -> Param {
34    Param { name, kind: "opt" }
35}
36const fn flag(name: &'static str) -> Param {
37    Param { name, kind: "flag" }
38}
39
40/// One engine verb in the ops manifest.
41struct Op {
42    /// `cli.<name>` wire query / `kuda <name>` argv verb.
43    name: &'static str,
44    /// One-line doc — mirrors the clap subcommand's help.
45    doc: &'static str,
46    /// Wire params the op reads (positional args + flags); `workspace`
47    /// and `config` are the engine's globals, accepted by every op.
48    params: &'static [Param],
49    /// `true` when the op writes `.kuda/` — mutators belong behind a
50    /// deliberate trigger, not ambient view polling.
51    mutates: bool,
52}
53

106/// The `ops` query/`ops` argv verb payload — self-describing so
107/// hosts and agents discover the engine surface without argv help.
108/// `params` are typed (`{name, kind: arg|opt|flag}`) — the contract
109/// surface drivers consume to render args without guessing.
110pub fn manifest() -> Value {
111    json!({
112        "engine": "kuda",
113        "invoke": "query `cli.<op>` with the listed params, or argv `tamias-driver-kuda run <kuda argv>`",
114        "ops": OPS.iter().map(|o| json!({
115            "query": format!("cli.{}", o.name),
116            "doc": o.doc,
117            "params": o.params.iter().map(|p| json!({
118                "name": p.name,
119                "kind": p.kind,
120            })).collect::<Vec<_>>(),
121            "mutates": o.mutates,
122        })).collect::<Vec<_>>(),
123        "surfaces": ["watch", "mcp", "acp"],
124    })
125}

127/// Is `query` an engine op (`cli.<op>`)? Returns the op name.
128pub fn op_of(query: &str) -> Option<&'static str> {
129    query
130        .strip_prefix("cli.")
131        .and_then(|n| OPS.iter().find(|o| o.name == n).map(|o| o.name))
132}
133
134fn bool_param(params: &Map<String, Value>, key: &str) -> bool {
135    params.get(key).and_then(Value::as_bool).unwrap_or(false)
136}
137
138fn str_param<'a>(params: &'a Map<String, Value>, keys: &[&str]) -> Option<&'a str> {
139    keys.iter()
140        .find_map(|k| params.get(*k).and_then(Value::as_str))
141}

143/// `cli.<op>` params → kuda argv (no argv[0] — `run` prepends it).
144/// Globals go first; clap accepts them either side of the verb.
145fn argv(op: &str, params: &Map<String, Value>) -> Vec<String> {
146    let mut v: Vec<String> = Vec::new();
147    if bool_param(params, "workspace") {
148        v.push("--workspace".into());
149    }
150    if let Some(c) = str_param(params, &["config"]) {
151        v.extend(["--config".into(), c.to_owned()]);
152    }
153    v.push(op.into());
154    match op {
155        "sync" => {
156            if bool_param(params, "full") {
157                v.push("--full".into());
158            }
159        }
160        "card" => {
161            if let Some(m) = str_param(params, &["module", "name"]) {
162                v.push(m.into());
163            }
164        }
165        "find" => {
166            if let Some(s) = str_param(params, &["symbol", "name", "pattern", "query"]) {
167                v.push(s.into());
168            }
169        }
170        "dependents" | "refs" => {
171            if let Some(n) = str_param(params, &["name", "module", "symbol"]) {
172                v.push(n.into());
173            }
174            if op == "refs" && bool_param(params, "docs") {
175                v.push("--docs".into());
176            }
177        }
178        _ => {}
179    }
180    v
181}

183/// Run one engine op against `root`: re-exec `run` in a child whose
184/// cwd is the request's repo, and hand back
185/// `{op, argv, code, stdout, stderr}` as the query payload.
186/// `ok:false` is reserved for spawn failures — a nonzero `code` is
187/// the engine's own exit status (e.g. `find` miss), honest data.
188pub fn run_op(root: &Path, op: &str, params: &Map<String, Value>) -> Response {
189    let args = argv(op, params);
190    let out = std::env::current_exe().and_then(|exe| {
191        std::process::Command::new(exe)
192            .arg("run")
193            .args(&args)
194            .current_dir(root)
195            .output()
196    });
197    let data = match out {
198        Ok(o) => json!({
199            "op": op,
200            "argv": args,
201            "code": o.status.code().unwrap_or(-1),
202            "stdout": String::from_utf8_lossy(&o.stdout),
203            "stderr": String::from_utf8_lossy(&o.stderr),
204        }),
205        Err(e) => return Response::err(format!("run {op}: {e}")),
206    };
207    Response::ok_typed(&QueryResult {
208        query: format!("cli.{op}"),
209        data,
210    })
211}

213#[cfg(test)]
214mod tests {
215    #![allow(clippy::unwrap_used)]
216    use super::*;
217    use serde_json::json;
218
219    #[test]
220    fn op_of_accepts_cli_prefix_only() {
221        assert_eq!(op_of("cli.sync"), Some("sync"));
222        assert_eq!(op_of("cli.doctor"), Some("doctor"));
223        assert_eq!(op_of("cli.watch"), None); // surface, not an op
224        assert_eq!(op_of("find"), None); // view query, not an op
225        assert_eq!(op_of("cli.bogus"), None);
226    }
227
228    #[test]
229    fn argv_maps_params_to_kuda_flags() {
230        let mut p = Map::new();
231        assert_eq!(argv("status", &p), vec!["status"]);
232        p.insert("full".into(), json!(true));
233        p.insert("workspace".into(), json!(true));
234        assert_eq!(argv("sync", &p), vec!["--workspace", "sync", "--full"]);
235        let mut p = Map::new();
236        p.insert("module".into(), json!("kuda-store"));
237        assert_eq!(argv("card", &p), vec!["card", "kuda-store"]);
238        let mut p = Map::new();
239        p.insert("name".into(), json!("X"));
240        p.insert("docs".into(), json!(true));
241        assert_eq!(argv("refs", &p), vec!["refs", "X", "--docs"]);
242    }
243
244    #[test]
245    fn manifest_lists_every_op_as_cli_query() {
246        let m = manifest();
247        let ops = m["ops"].as_array().unwrap();
248        assert_eq!(ops.len(), OPS.len());
249        for (i, o) in OPS.iter().enumerate() {
250            assert_eq!(ops[i]["query"], json!(format!("cli.{}", o.name)));
251            assert_eq!(ops[i]["mutates"], json!(o.mutates));
252            // params ride typed ({name, kind: arg|opt|flag}) — the
253            // contract surface drivers render args from.
254            for p in ops[i]["params"].as_array().unwrap() {
255                assert!(["arg", "opt", "flag"].contains(&p["kind"].as_str().unwrap()));
256            }
257        }
258        assert_eq!(m["surfaces"], json!(["watch", "mcp", "acp"]));
259    }
