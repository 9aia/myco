1//! Driver manifest + ViewNode trees — the static half of the pavo
2//! contract, built with `pavo-driver-kit` constructors over
3//! `pavo-protocol` types. Keep in sync with
4//! `pavo/crates/pavo-protocol/src/{manifest,view_node,wire}.rs`.
5
6use pavo_driver_kit as kit;
7use pavo_protocol::{Contribution, DriverManifest, FormFieldKind, QueryBinding, ViewNode};
8
9/// One `query` wire binding — `{"name":n,"op":"query"}`; the host
10/// calls the driver's `query` op with the binding name.
11fn binding(name: &str) -> QueryBinding {
12    kit::binding(name, "query")
13}
14
15/// A `ViewSpec` builder shorthand — id/title/route/description plus
16/// the query bindings the tree references, and the declarative root.
17fn spec(
18    id: &str,
19    title: &str,
20    route: &str,
21    description: &str,
22    queries: &[&str],
23    root: ViewNode,
24) -> kit::View {
25    let mut v = kit::view(id, title, route, root).description(description);
26    for q in queries {
27        v = v.query(binding(q));
28    }
29    v
30}
31
32/// `table` node — declared columns, query-fed rows (empty inline
33/// `rows`); `sortable` only where the current view had it.
34fn table(
35    id: Option<&str>,
36    title: &str,
37    query: &str,
38    columns: &[(&str, &str)],
39    sortable: bool,
40) -> ViewNode {
41    let mut base = kit::base().title(title).query(query);
42    if let Some(i) = id {
43        base = base.id(i);
44    }
45    let cols = columns
46        .iter()
47        .map(|(k, t)| kit::table_column(k, t))
48        .collect();
49    if sortable {
50        kit::sortable_table(base, cols, Vec::new())
51    } else {
52        kit::table(base, cols, Vec::new())
53    }
54}

56/// `kv` node — title + query binding, pairs filled by the host.
57fn kv(title: &str, query: &str) -> ViewNode {
58    kit::kv(kit::base().title(title).query(query), Vec::new())
59}
60
61/// `column` layout container.
62fn col(children: Vec<ViewNode>) -> ViewNode {
63    kit::column(kit::base(), children)
64}
65
66/// `graph` node — query-fed (`graph` binding); `layout: "dot"` is a
67/// renderer hint the kit constructor leaves unset, so patch it in.
68fn graph_node() -> ViewNode {
69    let mut g = kit::graph(
70        kit::base().title("Dependency graph").query("graph"),
71        Vec::new(),
72        Vec::new(),
73    );
74    if let ViewNode::Graph(ref mut node) = g {
75        node.layout = Some("dot".to_string());
76    }
77    g
78}

80/// The `find` view's search form — `submit_op: "query"` with a
81/// "Search" label (the kit's `form_submit` leaves `submit_label`
82/// unset; patch it like `graph`'s layout).
83fn search_form() -> ViewNode {
84    let mut pattern = kit::field_kind("pattern", "Pattern", FormFieldKind::Text);
85    pattern.hint = Some("substring match; empty lists recent symbols".to_string());
86    let exact = kit::field_kind("exact", "Exact only", FormFieldKind::Bool);
87    let mut form = kit::form_submit(kit::base().title("Search"), vec![pattern, exact], "query");
88    if let ViewNode::Form(ref mut node) = form {
89        node.submit_label = Some("Search".to_string());
90    }
91    form
92}

94/// A chrome-slot [`Contribution`] with explicit priority/icon — the
95/// kit's `contribution()` defaults priority to 100 and no icon.
96fn contrib(id: &str, title: &str, route: &str, priority: i32, icon: Option<&str>) -> Contribution {
97    let mut c = kit::contribution(id, title, route);
98    c.priority = priority;
99    c.icon = icon.map(str::to_string);
100    c
101}

103/// The `manifest` verb body — also served over the wire `manifest`

259    #[test]
260    fn manifest_advertises_pavo_and_declares_views() {
261        let m = serde_json::to_value(manifest()).unwrap();
262        assert_eq!(m["name"], "kuda");
263        assert!(m["verbs"].as_array().unwrap().iter().any(|v| v == "serve"));
264        assert!(m["capabilities"]
265            .as_array()
266            .unwrap()
267            .iter()
268            .any(|c| c == "pavo"));
269        let views = m["views"].as_array().unwrap();
270        assert!(!views.is_empty());
271        for v in views {
272            assert!(v["root"]["kind"].is_string(), "view {} has root", v["id"]);
273            for q in v["queries"].as_array().unwrap() {
274                assert_eq!(q["op"], "query");
275                assert!(q["name"].is_string());
276            }
277        }
278    }
