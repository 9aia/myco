1//! Repo context — resolve which kuda-indexed repo this request is
2//! about, load its config, cards, and L2 store.
3//!
4//! Resolution order for the repo root:
5//! 1. `root` / `dir` / `path` param on the wire request (JSON string).
6//! 2. `$KUDA_DRIVER_ROOT` env var.
7//! 3. Walk up from the process cwd to the nearest dir containing
8//!    `.kuda/` or `kuda.toml` — pavo spawns drivers with the user's
9//!    cwd, so `pavo view kuda:graph` inside repo X reads X's index.
10//! 4. The cwd itself (queries then report "not indexed" honestly).
11
12use kuda_core::{Config, ModuleCard, RepoCard};
13use kuda_store_sqlite::SqliteStore;
14use kuda_vcs::Vcs;
15use serde_json::{Map, Value};
16use std::path::{Path, PathBuf};
17
18/// Everything a query needs for one repo — opened lazily per wire
19/// request (sqlite open is cheap; cards parse on demand).
20pub struct Ctx {
21    /// Repo root the context was resolved against.
22    pub root: PathBuf,
23    /// Parsed `kuda.toml` (defaults when absent/unparseable).
24    pub cfg: Config,
25    /// `.kuda/index.yaml` L0 card, when present and parseable.
26    pub repo_card: Option<RepoCard>,
27    /// L2 store — `None` when `.kuda/index.sqlite` doesn't exist
28    /// (never created: the view/query side stays read-only;
29    /// `cli.sync` mutates through the `run` re-exec, not `Ctx`).
30    pub store: Option<SqliteStore>,
31    /// `git rev-parse HEAD` at open time (empty when not a repo).
32    pub head: String,
33}
34
35impl Ctx {
36    /// Open a context for `root`. Never fails hard — missing pieces
37    /// surface as `None`/empty so views degrade to error payloads.
38    pub fn open(root: PathBuf) -> Self {
39        let cfg = load_config(&root);
40        let cards_dir = cards_dir(&root, &cfg);
41        let repo_card = std::fs::read_to_string(cards_dir.join("index.yaml"))
42            .ok()
43            .and_then(|t| kuda_cards::parse_repo_card(&t).ok());
44        let store_path = root.join(&cfg.store.path);
45        let store = store_path
46            .is_file()
47            .then(|| SqliteStore::open(&store_path).ok())
48            .flatten();
49        let head = kuda_git::GitVcs.head_sha(&root).unwrap_or_default();
50        Self {
51            root,
52            cfg,
53            repo_card,
54            store,
55            head,
56        }
57    }
58
59    /// `.kuda/modules/` dir for this context.
60    pub fn modules_dir(&self) -> PathBuf {
61        cards_dir(&self.root, &self.cfg).join("modules")
62    }
63
64    /// All parseable module cards, sorted by name. Unparseable files
65    /// are skipped (they surface as drift "error" rows only when the
66    /// file itself can't be read as YAML at all — parse failures are
67    /// counted separately by callers that care).
68    pub fn module_cards(&self) -> Vec<(String, ModuleCard)> {
69        let mut out = Vec::new();
70        let Ok(entries) = std::fs::read_dir(self.modules_dir()) else {
71            return out;
72        };
73        for e in entries.flatten() {
74            let p = e.path();
75            if p.extension()
76                .map(|x| x == "yaml" || x == "yml")
77                .unwrap_or(false)
78            {
79                if let Ok(text) = std::fs::read_to_string(&p) {
80                    if let Ok(card) = kuda_cards::parse_module_card(&text) {
81                        out.push((card.name.clone(), card));
82                    }
83                }
84            }
85        }
86        out.sort_by(|a, b| a.0.cmp(&b.0));
87        out
88    }
89
90    /// True when the context has anything kuda-shaped at all.
91    pub fn indexed(&self) -> bool {
92        self.repo_card.is_some() || self.store.is_some() || self.modules_dir().is_dir()
93    }
94}

96/// Resolve the repo root for one request. `params` are the merged
97/// view/query params; string values under `root`/`dir`/`path` win.
98pub fn resolve_root(params: &Map<String, Value>) -> PathBuf {
99    for key in ["root", "dir", "path"] {
100        if let Some(s) = params.get(key).and_then(Value::as_str) {
101            let p = PathBuf::from(s);
102            return std::fs::canonicalize(&p).unwrap_or(p);
103        }
104    }
105    if let Ok(r) = std::env::var("KUDA_DRIVER_ROOT") {
106        if !r.is_empty() {
107            let p = PathBuf::from(r);
108            return std::fs::canonicalize(&p).unwrap_or(p);
109        }
110    }
111    let cwd = std::env::current_dir().unwrap_or_else(|_| PathBuf::from("."));
112    // Walk up: nearest ancestor with `.kuda/` or `kuda.toml`.
113    for dir in cwd.ancestors() {
114        if dir.join(".kuda").is_dir() || dir.join("kuda.toml").is_file() {
115            return dir.to_path_buf();
116        }
117    }
118    cwd
119}

121/// Parse `<root>/kuda.toml` into a `Config`; defaults on absence or
122/// parse failure (the driver never refuses to serve).
123fn load_config(root: &Path) -> Config {
124    let p = root.join("kuda.toml");
125    let Ok(text) = std::fs::read_to_string(&p) else {
126        return Config::default();
127    };
128    toml::from_str(&text).unwrap_or_else(|e| {
129        eprintln!("kuda-driver: parse {}: {e} — defaults in use", p.display());
130        Config::default()
131    })
132}
133
134fn cards_dir(root: &Path, cfg: &Config) -> PathBuf {
135    if cfg.cards.dir.is_absolute() {
136        cfg.cards.dir.clone()
137    } else {
138        root.join(&cfg.cards.dir)
139    }
140}

142/// `kuda status` drift for one card — recompute the git-blob hash of
143/// every provenance file and compare. Mirrors `status()` in
144/// `kuda-cli`: stale on any hash mismatch or a file-count change.
145pub fn card_drift(root: &Path, card: &ModuleCard) -> &'static str {
146    let hashes: Vec<String> = card
147        .provenance
148        .files
149        .iter()
150        .map(|f| blob_hash(&root.join(f)))
151        .collect();
152    let stale = card
153        .provenance
154        .hashes
155        .iter()
156        .zip(&hashes)
157        .any(|(a, b)| a != b)
158        || card.provenance.files.len() != hashes.len();
159    if stale {
160        "drifted"
161    } else {
162        "fresh"
163    }
164}

166/// git blob hash — identical values to `git hash-object`
167/// ("blob <len>\0" + bytes), same recipe as `kuda status`.
168fn blob_hash(path: &Path) -> String {
169    use sha1::Digest;
170    let Ok(bytes) = std::fs::read(path) else {
171        return String::new();
172    };
173    let mut h = sha1::Sha1::new();
174    h.update(format!("blob {}\0", bytes.len()).as_bytes());
175    h.update(&bytes);
176    h.finalize().iter().map(|b| format!("{b:02x}")).collect()
177}

179/// Filename-safe module name → card file stem (same rule as
180/// `kuda card`: keep alnum, `-`, `_`, `.`; fold the rest to `_`).
181pub fn safe(name: &str) -> String {
182    name.chars()
183        .map(|c| {
184            if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.') {
185                c
186            } else {
187                '_'
188            }
189        })
190        .collect()
191}

193#[cfg(test)]
194mod tests {
195    #![allow(clippy::unwrap_used)]
196    use super::*;
197    use serde_json::json;
198
199    #[test]
200    fn safe_folds_unsafe_chars() {
201        assert_eq!(safe("kuda-core"), "kuda-core");
202        assert_eq!(safe("a/b c"), "a_b_c");
203    }
204
205    #[test]
206    fn resolve_root_prefers_params() {
207        let tmp = std::env::temp_dir().join(format!("kuda-drv-{}", std::process::id()));
208        std::fs::create_dir_all(&tmp).unwrap();
209        let mut params = Map::new();
210        params.insert("root".into(), json!(tmp.to_string_lossy()));
211        let r = resolve_root(&params);
212        assert_eq!(r, std::fs::canonicalize(&tmp).unwrap());
213        let _ = std::fs::remove_dir_all(&tmp);
214    }
215
216    #[test]
217    fn unindexed_ctx_reports_not_indexed() {
218        let tmp = std::env::temp_dir().join(format!("kuda-drv-empty-{}", std::process::id()));
219        std::fs::create_dir_all(&tmp).unwrap();
220        let ctx = Ctx::open(tmp.clone());
221        assert!(!ctx.indexed());
222        assert!(ctx.repo_card.is_none());
223        assert!(ctx.store.is_none());
224        let q = crate::queries::run_query(&ctx, "status", &Map::new());
225        assert!(q.get("error").is_some(), "{q}");
226        let _ = std::fs::remove_dir_all(&tmp);
227    }
