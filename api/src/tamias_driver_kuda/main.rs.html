1//! tamias-driver-kuda — pavo view driver for kuda.
2//!
3//! Answers the pavo wire ops (`manifest`, `views`, `view`, `query`)
4//! as ndjson JSON-RPC over stdio via `pavo-driver-kit`'s serve loop
5//! — the same `{"op":…}` / `{"ok":…,"result"|"error":…}` shape every
6//! sibling `serve` speaks (see `pavo/crates/pavo-protocol/src/wire.rs`).
7//! Also answers the discovery verbs `manifest` and `capabilities` as
8//! plain argv, plus `ops` (the engine ops manifest) and `run`
9//! (engine passthrough).
10//!
11//! Optional surfaces: features `cli`/`mcp` embed pavo-driver-kit's
12//! surface layer — `cli <op> [args]` is a generated CLI over the ops
13//! manifest (`watch`/`mcp`/`acp` ride as streaming subcommands),
14//! `mcp` serves the ops as MCP tools on stdio. Both are equivalent
15//! to the standalone `tamias-driver-cli`/`tamias-driver-mcp` surface
16//! drivers, minus the PATH/discovery hop.
17//!
18//! Repo context: views/queries are read-only — the driver never
19//! opens `.kuda/index.sqlite` for writing. Engine ops are the
20//! exception by design: `cli.<op>` wire queries re-exec the `run`
21//! verb in the request's repo and let `kuda::dispatch` own the
22//! outcome (`cli.sync` mutates `.kuda/`). The repo resolves from
23//! the `root`/`dir`/`path` request param, `$KUDA_DRIVER_ROOT`, or
24//! by walking up from the cwd (pavo inherits the user's cwd, so
25//! `pavo view kuda:graph` inside repo X reads X's `.kuda/`).
26//!
27//! Install: `cargo install --path crates/kuda-driver` places
28//! `tamias-driver-kuda` on PATH — pavo discovers it there or under
29//! `<tamias-home>/drivers/`.

45fn main() {
46    let args: Vec<String> = std::env::args().skip(1).collect();
47    let code = match args.first().map(String::as_str) {
48        Some("manifest") => {
49            println!("{}", kit::manifest_json(&views::manifest()));
50            0
51        }
52        Some("capabilities") => {
53            let mut v = kit::capabilities_value(&views::manifest());
54            v["driver"] = json!("kuda");
55            println!("{v}");
56            0
57        }
58        Some("ops") => {
59            println!("{}", ops::manifest());
60            0
61        }
62        // Engine passthrough — `tamias-driver-kuda run sync --full`
63        // is `kuda sync --full`. Also the child side of `cli.*` wire
64        // queries (ops::run_op re-execs this same verb).
65        Some("run") => {
66            let argv = std::iter::once("kuda".to_string()).chain(args.iter().skip(1).cloned());
67            std::process::exit(kuda::dispatch(&kuda::parse_from(argv)));
68        }
69        // Embedded surface verbs (feature-gated, optional): the
70        // pavo-driver-kit surface layer renders this driver's own
71        // ops manifest — `cli` = generated clap CLI (op → subcommand,
72        // `cli watch` streams), `mcp` = ops → MCP tools over stdio.
73        // The standalone tamias-driver-cli/-mcp binaries do the same
74        // over process dispatch; these verbs need nothing on PATH.
75        #[cfg(feature = "cli")]
76        Some("cli") => {
77            let m = kit::surface::OpsManifest::from_value(&ops::manifest()).unwrap_or_default();
78            let argv: Vec<String> = std::iter::once("kuda".to_string())
79                .chain(args.iter().skip(1).cloned())
80                .collect();
81            kit::surface::cli::run("kuda", &m, &argv, &mut surface::Engine)
82        }
83        #[cfg(feature = "mcp")]
84        Some("mcp") => {
85            let m = kit::surface::OpsManifest::from_value(&ops::manifest()).unwrap_or_default();
86            match kit::surface::mcp::serve_stdio(&m, &mut surface::Engine) {
87                Ok(()) => 0,
88                Err(e) => {
89                    eprintln!("tamias-driver-kuda mcp: {e}");
90                    1
91                }
92            }
93        }
94        Some("serve") | None => serve(),
95        _ => {
96            eprintln!(
97                "usage: tamias-driver-kuda manifest|capabilities|ops|run <argv>|serve\n       (feature `cli` adds `cli <op> [args]`; feature `mcp` adds `mcp`)"
98            );
99            1
100        }
101    };
102    std::process::exit(code);
103}

105/// One parsed wire [`Request`] → response. The kit's serve loop owns
106/// the ndjson plumbing and turns parse failures into `ok:false`;
107/// ops this driver doesn't implement (`render`, `submit`, `jobs`,
108/// `notifications`) get the same `ok:false` convention every
109/// non-provider driver answers.
110fn dispatch(req: &Request, manifest: &DriverManifest) -> Response {
111    match req {
112        Request::Manifest => Response::ok_typed(manifest),
113        Request::Views => Response::ok_typed(&views_summaries(manifest)),
114        Request::View { view, params } => view_result(manifest, view, params),
115        Request::Query { query, params } => {
116            if query == "ops" {
117                Response::ok_typed(&QueryResult {
118                    query: query.clone(),
119                    data: ops::manifest(),
120                })
121            } else if let Some(op) = ops::op_of(query) {
122                ops::run_op(&resolve_root(params), op, params)
123            } else {
124                query_result(query, params)
125            }
126        }
127        Request::Render { .. } => Response::err("unsupported op: render"),
128        Request::Submit { .. } => Response::err("unsupported op: submit"),
129        Request::Jobs => Response::err("unsupported op: jobs"),
130        Request::Notifications => Response::err("unsupported op: notifications"),
131    }
132}

134/// `Vec<ViewSummary>` — id/title/route/description/queries, no trees.
135fn views_summaries(manifest: &DriverManifest) -> Vec<ViewSummary> {
136    manifest.views.iter().map(ViewSummary::from).collect()
137}
138
139/// `view` — resolve one ViewSpec, run every query binding against a
140/// fresh `Ctx` for the request's repo, return `ViewResult`.
141fn view_result(manifest: &DriverManifest, id: &str, params: &Map<String, Value>) -> Response {
142    let Some(spec) = manifest.views.iter().find(|v| v.id == id) else {
143        return Response::err(format!("unknown view: {id}"));
144    };
145    let spec = spec.clone();
146    let ctx = Ctx::open(resolve_root(params));
147    let mut data = Map::new();
148    for b in &spec.queries {
149        // static binding params first, caller params override
150        let mut merged = b.params.clone();
151        for (k, v) in params {
152            merged.insert(k.clone(), v.clone());
153        }
154        data.insert(b.name.clone(), run_query(&ctx, &b.name, &merged));
155    }
156    Response::ok_typed(&ViewResult { view: spec, data })
157}
158
159/// `query` — one named query with caller params.
160fn query_result(name: &str, params: &Map<String, Value>) -> Response {
161    let ctx = Ctx::open(resolve_root(params));
162    let data = run_query(&ctx, name, params);
163    Response::ok_typed(&QueryResult {
164        query: name.to_string(),
165        data,
166    })
167}
168
169fn serve() -> i32 {
170    let manifest = views::manifest();
171    match kit::serve_stdio(|req| dispatch(req, &manifest)) {
172        Ok(()) => 0,
173        Err(e) => {
174            eprintln!("tamias-driver-kuda: {e}");
175            1
176        }
177    }
178}
