1//! The op vocabulary — the verb names guardrails, audit events, and
2//! (later) the `serve` wire all share. Lowercase serde/parse forms
3//! match the CLI subcommands.
4
5use std::fmt;
6use std::str::FromStr;
7
8/// An cuora verb.
9#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
10#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
11#[cfg_attr(feature = "schema", schemars(rename_all = "lowercase"))]
12#[derive(serde::Serialize, serde::Deserialize)]
13#[serde(rename_all = "lowercase")]
14pub enum Op {
15    /// Bootstrap identity + vault.
16    Init,
17    /// Store a secret.
18    Put,
19    /// Read a secret's value.
20    Get,
21    /// List refs in an env.
22    List,
23    /// Remove a secret.
24    Rm,
25    /// Rotate a secret (new value, version bump).
26    Rotate,
27    /// Inject declared secrets into a child process.
28    Run,
29    /// Print `export` lines for `eval`.
30    Env,
31    /// Validate config + declared surface.
32    Check,
33    /// Push a stored value to a write-only deploy target.
34    Deploy,
35    /// Rewrite stored records for a wire/config upgrade.
36    Migrate,
37    /// ndjson verb dispatch (later ADR).
38    Serve,
39    /// Agent self-discovery.
40    Manifest,
41    /// Machine-readable schemas.
42    Schema,
43}
44
45impl Op {
46    /// All ops, in declaration order.
47    pub const ALL: [Op; 14] = [
48        Op::Init,
49        Op::Put,
50        Op::Get,
51        Op::List,
52        Op::Rm,
53        Op::Rotate,
54        Op::Run,
55        Op::Env,
56        Op::Check,
57        Op::Deploy,
58        Op::Migrate,
59        Op::Serve,
60        Op::Manifest,
61        Op::Schema,
62    ];
63}
64
65/// Failure to parse an [`Op`] name.
66#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
67#[error("invalid op {0:?}")]
68pub struct ParseOpError(pub String);
69
70impl Op {
71    /// The CLI verb name.
72    pub fn as_str(&self) -> &'static str {
73        match self {
74            Op::Init => "init",
75            Op::Put => "put",
76            Op::Get => "get",
77            Op::List => "list",
78            Op::Rm => "rm",
79            Op::Rotate => "rotate",
80            Op::Run => "run",
81            Op::Env => "env",
82            Op::Check => "check",
83            Op::Deploy => "deploy",
84            Op::Migrate => "migrate",
85            Op::Serve => "serve",
86            Op::Manifest => "manifest",
87            Op::Schema => "schema",
88        }
89    }
90}
91
92impl FromStr for Op {
93    type Err = ParseOpError;
94
95    fn from_str(s: &str) -> Result<Self, Self::Err> {
96        Op::ALL
97            .iter()
98            .copied()
99            .find(|op| op.as_str() == s)
100            .ok_or_else(|| ParseOpError(s.to_string()))
101    }
102}
103
104impl fmt::Display for Op {
105    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
106        f.write_str(self.as_str())
107    }
108}

110#[cfg(test)]
111#[allow(clippy::unwrap_used)]
112mod tests {
113    use super::*;
114
115    #[test]
116    fn names_roundtrip() {
117        for op in Op::ALL {
118            assert_eq!(op.as_str().parse::<Op>().unwrap(), op);
119            assert_eq!(op.to_string(), op.as_str());
120        }
121        assert!("nope".parse::<Op>().is_err());
122    }
123
124    #[test]
125    fn serde_is_lowercase() {
126        assert_eq!(serde_json::to_string(&Op::Rotate).unwrap(), "\"rotate\"");
127        assert_eq!(serde_json::from_str::<Op>("\"run\"").unwrap(), Op::Run);
128        assert!(serde_json::from_str::<Op>("\"Run\"").is_err());
129    }
130
131    #[test]
132    fn all_covers_schema() {
133        assert!(Op::ALL.contains(&Op::Schema));
134        assert_eq!(Op::Schema.as_str(), "schema");
135    }
136}
