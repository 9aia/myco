1//! Drawing — tab bar on top, board list filling the frame, status
2//! bar at the bottom. Preview mode swaps the list for the item
3//! detail pane (pm's `Enter` full-pane preview). Everything degrades
4//! per frame: rows truncate with `…` on narrow screens, chrome bars
5//! yield to the list on short ones, and `?` folds into columns.

15const ACCENT: Color = Color::Indexed(69); // pm's nav blue
16const MUTED: Color = Color::Indexed(241);

18/// Chrome budget — which fixed rows the screen can afford, so the
19/// list keeps a real viewport on short terminals. `(tabs, status)`:
20/// tabs cost a row at height ≥3, the status bar only at height ≥5
21/// (at 4 rows it yields to a three-row viewport).
22fn chrome(height: u16) -> (bool, bool) {
23    (height >= 3, height >= 5)
24}
25
26/// One full frame. Also records the tab/board geometry on `m` so
27/// the next mouse event can hit-test against where rows landed.
28pub fn draw(f: &mut Frame, m: &mut Model) {
29    let area = f.area();
30    let (show_tabs, show_status) = chrome(area.height);
31    let mut constraints = Vec::with_capacity(3);
32    if show_tabs {
33        constraints.push(Constraint::Length(1));
34    }
35    constraints.push(Constraint::Min(0));
36    if show_status {
37        constraints.push(Constraint::Length(1));
38    }
39    let chunks = Layout::vertical(constraints).split(area);
40
41    let mut next = 0;
42    if show_tabs {
43        draw_tabs(f, m, chunks[next]);
44        m.tabs_area = Some(chunks[next]);
45        next += 1;
46    } else {
47        m.tabs_area = None;
48    }
49    match m.mode {
50        Mode::Preview => draw_preview(f, m, chunks[next]),
51        _ => draw_board(f, m, chunks[next]),
52    }
53    if show_status {
54        draw_status(f, m, chunks[next + 1]);
55    }
56    if m.mode == Mode::Help {
57        draw_help(f, area);
58    }
59}

61/// Tab captions for `width` columns — full names when they fit,
62/// initials when they don't (the `1`/`2`/`h`/`l` binds still work).
63fn tab_labels(width: usize) -> Vec<String> {
64    let full: Vec<String> = Board::ALL
65        .iter()
66        .map(|b| format!(" {} ", b.label()))
67        .collect();
68    // leading space + one separator per tab
69    let need: usize = full.iter().map(|s| s.chars().count()).sum::<usize>() + full.len() + 1;
70    if need <= width {
71        full
72    } else {
73        Board::ALL
74            .iter()
75            .map(|b| b.label()[..1].to_string())
76            .collect()
77    }
78}

80fn draw_tabs(f: &mut Frame, m: &Model, area: Rect) {
81    let mut spans = vec![Span::raw(" ")];
82    for (b, label) in Board::ALL.iter().zip(tab_labels(area.width as usize)) {
83        let style = if *b == m.board {
84            Style::default().fg(Color::White).bg(ACCENT)
85        } else {
86            Style::default().fg(MUTED)
87        };
88        spans.push(Span::styled(label, style));
89        spans.push(Span::raw(" "));
90    }
91    f.render_widget(Paragraph::new(Line::from(spans)), area);
92}

94/// Click hit-test for the tab row: which tab does column `x` land
95/// on in a `width`-wide bar? The leading and separator columns
96/// belong to no tab. Mirrors the span layout `draw_tabs` emits.
97pub fn tab_hit(width: u16, x: u16) -> Option<usize> {
98    let mut col = 1u16; // leading space
99    for (i, label) in tab_labels(width as usize).iter().enumerate() {
100        let w = label.chars().count() as u16;
101        if (col..col + w).contains(&x) {
102            return Some(i);
103        }
104        col += w + 1; // the separator after each tab
105    }
106    None
107}

109/// Fit `spans` into `width` columns; when they overflow, keep the
110/// longest prefix that leaves one column for a muted `…`. Widths are
111/// char counts — wide glyphs may still let ratatui clip the tail.
112fn ellipsize<'a>(spans: Vec<Span<'a>>, width: usize) -> Vec<Span<'a>> {
113    let total: usize = spans.iter().map(|s| s.content.chars().count()).sum();
114    if total <= width {
115        return spans;
116    }
117    if width == 0 {
118        return Vec::new();
119    }
120    let keep = width - 1; // reserve a column for the ellipsis
121    let mut out = Vec::new();
122    let mut used = 0;
123    for span in spans {
124        let n = span.content.chars().count();
125        if used + n <= keep {
126            used += n;
127            out.push(span);
128        } else {
129            let take = keep - used;
130            if take > 0 {
131                let cut: String = span.content.chars().take(take).collect();
132                out.push(Span::styled(cut, span.style));
133            }
134            break;
135        }
136    }
137    out.push(Span::styled("…", Style::default().fg(MUTED)));
138    out
139}

141fn draw_board(f: &mut Frame, m: &mut Model, area: Rect) {
142    // the highlight symbol eats two columns of every row
143    let row_w = (area.width as usize).saturating_sub(2);
144    let items: Vec<ListItem> = m
145        .rows
146        .iter()
147        .map(|r| match r {
148            Row::Header(name) => ListItem::new(Line::from(ellipsize(
149                vec![Span::styled(
150                    format!("## {name}"),
151                    Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
152                )],
153                row_w,
154            ))),
155            Row::Item { item, depth } => {
156                let indent = "  ".repeat(*depth);
157                let mut spans = vec![Span::raw(indent)];
158                if m.show_ids {
159                    spans.push(Span::styled(
160                        item.id.to_string(),
161                        Style::default().fg(ACCENT),
162                    ));
163                    spans.push(Span::raw(" "));
164                }
165                spans.push(Span::raw(item.title.clone()));
166                let mut meta = format!(" {}", item.status);
167                if !matches!(item.priority, myco_core::Priority::Medium) {
168                    meta += &format!(" pri:{}", item.priority.token());
169                }
170                for l in &item.labels {
171                    meta += &format!(" #{l}");
172                }
173                spans.push(Span::styled(meta, Style::default().fg(MUTED)));
174                ListItem::new(Line::from(ellipsize(spans, row_w)))
175            }
176        })
177        .collect();
178
179    let list = List::new(items)
180        .highlight_style(Style::default().bg(Color::Indexed(238)))
181        .highlight_symbol("▸ ");
182    let mut state = ListState::default().with_selected(Some(m.sel));
183    f.render_stateful_widget(list, area, &mut state);
184    // remember where the rows landed for mouse hit-testing
185    m.board_area = area;
186    m.viewport_top = state.offset();
187}

189fn draw_preview(f: &mut Frame, m: &Model, area: Rect) {
190    let Some(item) = m.selected() else {
191        return;
192    };
193    let mut lines = vec![
194        Line::from(Span::styled(
195            item.id.to_string(),
196            Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
197        )),
198        Line::from(Span::styled(
199            item.title.clone(),
200            Style::default().add_modifier(Modifier::BOLD),
201        )),
202        Line::from(""),
203        Line::from(vec![
204            Span::styled("status ", Style::default().fg(MUTED)),
205            Span::raw(item.status.to_string()),
206        ]),
207        Line::from(vec![
208            Span::styled("pri    ", Style::default().fg(MUTED)),
209            Span::raw(item.priority.token().to_string()),
210        ]),
211    ];
212    if let Some(o) = &item.owner {
213        lines.push(Line::from(vec![
214            Span::styled("owner  ", Style::default().fg(MUTED)),
215            Span::raw(o.to_string()),
216        ]));
217    }
218    if !item.labels.is_empty() {
219        lines.push(Line::from(vec![
220            Span::styled("labels ", Style::default().fg(MUTED)),
221            Span::raw(
222                item.labels
223                    .iter()
224                    .map(|l| format!("#{l}"))
225                    .collect::<Vec<_>>()
226                    .join(" "),
227            ),
228        ]));
229    }
230    for (key, val) in [("epic   ", &item.epic), ("ms     ", &item.milestone)] {
231        if let Some(v) = val {
232            lines.push(Line::from(vec![
233                Span::styled(key, Style::default().fg(MUTED)),
234                Span::raw(v.clone()),
235            ]));
236        }
237    }
238    if let Some(d) = item.created {
239        lines.push(Line::from(vec![
240            Span::styled("created", Style::default().fg(MUTED)),
241            Span::raw(format!(" {d}")),
242        ]));
243    }
244    if !item.children.is_empty() {
245        lines.push(Line::from(""));
246        lines.push(Line::from(Span::styled(
247            "children",
248            Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
249        )));
250        for c in &item.children {
251            lines.push(Line::from(format!("  {}", myco_ops::items::compact(c))));
252        }
253    }
254    if let Some(note) = &item.note {
255        lines.push(Line::from(""));
256        lines.push(Line::from(Span::styled(
257            "note",
258            Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
259        )));
260        lines.push(Line::from(note.clone()));
261    }
262    // File-backed items render their body (pm's note preview).
263    if let Some(body) = m.file_body(item.id) {
264        lines.push(Line::from(""));
265        for l in body.lines() {
266            let style = if l.starts_with('#') {
267                Style::default().fg(ACCENT).add_modifier(Modifier::BOLD)
268            } else {
269                Style::default()
270            };
271            lines.push(Line::from(Span::styled(l.to_string(), style)));
272        }
273    }
274    // wrap rather than clip long titles/notes on narrow screens
275    f.render_widget(Paragraph::new(lines).wrap(Wrap { trim: false }), area);
276}

278/// Normal-mode hint sets, widest first — `pick_hints` takes the
279/// first that fits so narrow terminals degrade instead of cropping.
280const HINTS: [&str; 3] = [
281    "j/k move · 1/2/3 boards · space done · D defer · p promote · alt+↑/↓ reorder · a add · / filter · : cmd · enter preview · E edit · R reload · ? help · q quit",
282    "j/k move · space done · a add · / filter · : cmd · enter · E edit · ? help · q quit",
283    "? help · q quit",
284];
285
286/// The hint line for `mode` that fits `width` columns (including the
287/// leading space). The minimal set always wins as the fallback.
288fn pick_hints(mode: Mode, width: usize) -> &'static str {
289    match mode {
290        Mode::Add => "enter add · esc cancel",
291        Mode::Filter => "enter keep · esc clear",
292        Mode::Command => "enter run · esc cancel",
293        Mode::Preview => "esc back",
294        Mode::Help => "esc/?/q close",
295        Mode::Normal => HINTS
296            .iter()
297            .copied()
298            .find(|h| h.chars().count() < width)
299            .unwrap_or(HINTS[HINTS.len() - 1]),
300    }
301}

303fn draw_status(f: &mut Frame, m: &Model, area: Rect) {
304    if matches!(m.mode, Mode::Add | Mode::Filter | Mode::Command) {
305        let (prompt, buf) = match m.mode {
306            Mode::Add => ("add> ", &m.input),
307            Mode::Filter => ("/ ", &m.filter),
308            _ => (":", &m.cmdline),
309        };
310        let line = Line::from(vec![
311            Span::styled(prompt, Style::default().fg(ACCENT)),
312            Span::raw(buf.clone()),
313            Span::styled("█", Style::default().fg(ACCENT)),
314        ]);
315        f.render_widget(Paragraph::new(line), area);
316        return;
317    }
318    // The status message eats columns first; hints get what remains.
319    let status_w = if m.status.is_empty() {
320        0
321    } else {
322        m.status.chars().count() + 2
323    };
324    let hints = pick_hints(m.mode, (area.width as usize).saturating_sub(status_w));
325    let line = if m.status.is_empty() {
326        Line::from(Span::styled(
327            format!(" {hints}"),
328            Style::default().fg(MUTED),
329        ))
330    } else {
331        Line::from(vec![
332            Span::styled(format!(" {} ", m.status), Style::default().fg(Color::White)),
333            Span::styled(format!(" {hints}"), Style::default().fg(MUTED)),
334        ])
335    };
336    f.render_widget(Paragraph::new(line), area);
337}

339/// A `width`×`height` rect centered in `area`, clamped to fit.
340fn centered(area: Rect, width: u16, height: u16) -> Rect {
341    let width = width.min(area.width);
342    let height = height.min(area.height);
343    Rect {
344        x: area.x + (area.width - width) / 2,
345        y: area.y + (area.height - height) / 2,
346        width,
347        height,
348    }
349}

351/// The `?` overlay — every bind, grouped like the status bar hints.
352fn help_lines() -> Vec<Line<'static>> {
353    fn group(name: &'static str) -> Line<'static> {
354        Line::from(Span::styled(
355            name,
356            Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
357        ))
358    }
359    fn bind(keys: &'static str, desc: &'static str) -> Line<'static> {
360        Line::from(vec![
361            Span::raw("  "),
362            Span::styled(format!("{keys:<12}"), Style::default().fg(ACCENT)),
363            Span::styled(desc, Style::default().fg(MUTED)),
364        ])
365    }
366    // No blank separators — the whole list must fit a 24-row screen.
367    vec![
368        group("nav"),
369        bind("j/k, ↑/↓", "move"),
370        bind("gg / G", "first / last"),
371        bind("enter", "preview item"),
372        group("tabs"),
373        bind("1 / 2", "TODO / BACKLOG"),
374        bind("h/l, ←/→", "switch board"),
375        group("items"),
376        bind("space, dd", "mark done"),
377        bind("D", "defer → BACKLOG"),
378        bind("p", "promote → TODO"),
379        bind("alt+j/k, ↑/↓", "reorder"),
380        group("edit"),
381        bind("a", "add item"),
382        bind("E", "open in $EDITOR"),
383        bind("R", "reload"),
384        group("mouse"),
385        bind("click, wheel", "select, scroll"),
386        group("misc"),
387        bind(": / ?", "command line / help"),
388        bind("q, ctrl+c", "quit"),
389        Line::from(Span::styled(
390            "esc / ? / q to close",
391            Style::default().fg(MUTED),
392        )),
393    ]
394}

396/// Break `lines` into side-by-side columns: as many as the screen
397/// height needs (one when everything fits vertically), capped by
398/// what the width affords. `inner` is the box content area.
399fn help_columns(lines: &[Line<'static>], area: Rect) -> Vec<Vec<Line<'static>>> {
400    let inner_h = area.height.saturating_sub(2).max(1) as usize;
401    let inner_w = area.width.saturating_sub(2).max(1) as usize;
402    let col_w = lines.iter().map(Line::width).max().unwrap_or(0) + 3;
403    let fits = (inner_w / col_w.max(1)).max(1); // columns the width allows
404    let needs = lines.len().div_ceil(inner_h).max(1); // columns the height wants
405    let n = fits.min(needs);
406    let per = lines.len().div_ceil(n);
407    lines.chunks(per).map(<[Line]>::to_vec).collect()
408}

410fn draw_help(f: &mut Frame, area: Rect) {
411    let cols = help_columns(&help_lines(), area);
412    let col_ws: Vec<u16> = cols
413        .iter()
414        .map(|c| c.iter().map(Line::width).max().unwrap_or(0) as u16 + 3)
415        .collect();
416    let width = col_ws.iter().sum::<u16>() + 2;
417    let height = cols.iter().map(Vec::len).max().unwrap_or(0) as u16 + 2;
418    let popup = centered(area, width, height);
419    f.render_widget(Clear, popup);
420    let block = Block::default()
421        .borders(Borders::ALL)
422        .border_style(Style::default().fg(ACCENT))
423        .title(Span::styled(" keys ", Style::default().fg(ACCENT)));
424    let inner = block.inner(popup);
425    f.render_widget(block, popup);
426    let rects = Layout::horizontal(col_ws.iter().map(|w| Constraint::Length(*w))).split(inner);
427    for (lines, rect) in cols.iter().zip(rects.iter()) {
428        f.render_widget(Paragraph::new(lines.clone()), *rect);
429    }
430}
