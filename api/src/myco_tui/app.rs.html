1//! The TUI model — boards, cursor, modes, and the ledger ops every
2//! key maps to. Mirrors pm's design: two board tabs over
3//! `myco_ops::items::Ledger`, a flattened row list, and writes that
4//! end in `items::commit`.

16/// Which ledger doc the board tab shows.
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub enum Board {
19    /// `TODO.md` — committed work.
20    Todo,
21    /// `BACKLOG.md` — deferred items.
22    Backlog,
23    /// File-backed kinds — `issues/`/`proposals/`/`plans/`/`adrs/`,
24    /// grouped under kind headers. Read-only for now: task verbs
25    /// (`done`/`defer`/…) don't apply here.
26    Files,
27}
28
29impl Board {
30    /// Tab label.
31    pub fn label(self) -> &'static str {
32        match self {
33            Board::Todo => "TODO",
34            Board::Backlog => "BACKLOG",
35            Board::Files => "ITEMS",
36        }
37    }
38
39    /// Tab order for `h`/`l` and the `1`/`2`/`3` jumps.
40    pub const ALL: [Board; 3] = [Board::Todo, Board::Backlog, Board::Files];
41}

43/// UI mode — like pm's: the board handles keys differently per mode.
44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub enum Mode {
46    /// Browsing.
47    Normal,
48    /// Full-pane item preview (`Enter`, `esc` back).
49    Preview,
50    /// `a` — typing a new item title; `Enter` commits, `esc` drops.
51    Add,
52    /// `/` — live filter over the current board's items; `Enter`
53    /// keeps the filter, `esc` clears it.
54    Filter,
55    /// `:` — ex-style command line (`:q` `:w` `:e` `:<id>` `:go <id>`
56    /// `:set id`); `Enter` runs it, `esc` drops the line.
57    Command,
58    /// `?` — keybind overlay; `esc`/`?`/`q` closes.
59    Help,
60}

62/// A flattened display row — section headers aren't selectable.
63#[derive(Debug, Clone)]
64pub enum Row {
65    /// `## <section>` header line.
66    Header(String),
67    /// An item line; `depth` indents children under their parent.
68    Item { item: Box<Item>, depth: usize },
69}

71/// The whole TUI state.
72pub struct Model {
73    proj: discover::Project,
74    store: MdStore,
75    ledger: Ledger,
76    no_commit: bool,

80    pub rows: Vec<Row>,

93    /// `:set noid` (vim aliases `numbers`/`nonumbers` accepted).
94    pub show_ids: bool,

96    pub status: String,
97    /// Editor to open on the next loop turn (`E`).
98    pub pending_editor: Option<PathBuf>,
99    /// File-backed items for the `Files` board.
100    files: Vec<ItemFile>,
101    /// Multi-key prefix: `g` armed for `gg`, `d` for `dd`.
102    pending: Option<char>,
103    /// Board-list rect from the last frame — `render::draw` writes
104    /// it so mouse clicks and wheel hits map to rows.
105    pub board_area: Rect,
106    /// First row index visible in `board_area` — the scroll offset
107    /// the `List` widget chose to keep `sel` on screen.

109    /// Tab-bar rect when the frame is tall enough to draw one —
110    /// for click-to-switch hit-testing.
111    pub tabs_area: Option<Rect>,

115    /// Discovery + first ledger load. `dir`/`root` override like the
116    /// CLI globals.
117    pub fn open(dir: Option<&Path>, root: Option<&Path>, no_commit: bool) -> Result<Self, String> {
118        let (proj, store) = items::open(dir, root)?;
119        let ledger = items::load_ledger(&store)?;
120        let mut m = Model {
121            proj,
122            store,
123            ledger,
124            no_commit,
125            board: Board::Todo,
126            rows: Vec::new(),
127            sel: 0,
128            mode: Mode::Normal,
129            input: String::new(),
130            filter: String::new(),
131            cmdline: String::new(),
132            show_ids: true,
133            status: String::new(),
134            pending_editor: None,
135            files: Vec::new(),
136            pending: None,
137            board_area: Rect::default(),
138            viewport_top: 0,
139            tabs_area: None,
140        };
141        m.load_files();
142        m.rebuild_rows();
143        m.restore_session();
144        Ok(m)
145    }

147    /// Pull file-backed items for the `Files` board.
148    fn load_files(&mut self) {
149        self.files.clear();
150        for kind in [
151            ItemKind::Issue,
152            ItemKind::Proposal,
153            ItemKind::Plan,
154            ItemKind::Adr,
155        ] {
156            if let Ok(mut v) = self.store.list_items(kind) {
157                self.files.append(&mut v);
158            }
159        }
160    }

162    /// Body of a file-backed item (preview on the `Files` board).
163    pub fn file_body(&self, id: ItemId) -> Option<&str> {
164        self.files
165            .iter()
166            .find(|f| f.meta.id == id)
167            .map(|f| f.body.as_str())
168    }

170    /// Reload both docs from disk (`R`, post-editor).
171    pub fn reload(&mut self) {
172        let sel_id = self.selected_id();
173        match items::load_ledger(&self.store) {
174            Ok(ledger) => {
175                self.ledger = ledger;
176                self.rebuild_rows();
177                self.reselect(sel_id);
178            }
179            Err(e) => self.status = format!("reload: {e}"),
180        }
181    }

183    /// Flatten the active board's sections into rows.
184    fn rebuild_rows(&mut self) {
185        self.rows.clear();
186        match self.board {
187            Board::Files => {
188                for (kind, label) in [
189                    (ItemKind::Issue, "issues"),
190                    (ItemKind::Proposal, "proposals"),
191                    (ItemKind::Plan, "plans"),
192                    (ItemKind::Adr, "adrs"),
193                ] {
194                    let group: Vec<Item> = self
195                        .files
196                        .iter()
197                        .filter(|f| f.meta.id.kind() == kind)
198                        .map(|f| f.meta.clone())
199                        .collect();
200                    if group.is_empty() {
201                        continue;
202                    }
203                    self.rows.push(Row::Header(label.to_string()));
204                    push_items(&group, 0, &mut self.rows);
205                }
206            }
207            _ => {
208                let sections = match self.board {
209                    Board::Todo => &self.ledger.todo.sections,
210                    Board::Backlog => &self.ledger.backlog.sections,
211                    Board::Files => unreachable!(),
212                };
213                for s in sections {
214                    if let Some(name) = &s.name {
215                        self.rows.push(Row::Header(name.clone()));
216                    }
217                    push_items(&s.items, 0, &mut self.rows);
218                }
219            }
220        }
221        self.apply_filter();
222        self.sel = self.first_item_row().unwrap_or(0);
223    }
224
225    /// `/` filter — keep only item rows matching the query
226    /// (title/id/labels, case-insensitive); headers drop out.
227    fn apply_filter(&mut self) {
228        if self.filter.is_empty() {
229            return;
230        }
231        let q = self.filter.to_lowercase();
232        self.rows.retain(|r| {
233            matches!(r, Row::Item { item, .. }
234                if item.title.to_lowercase().contains(&q)
235                    || item.id.to_string().to_lowercase().contains(&q)
236                    || item.labels.iter().any(|l| l.to_lowercase().contains(&q)))
237        });
238    }
239
240    /// After a mutation: land the cursor back on `id` (or the
241    /// nearest item row).
242    fn reselect(&mut self, id: Option<ItemId>) {
243        if let Some(id) = id {
244            if let Some(i) = self
245                .rows
246                .iter()
247                .position(|r| matches!(r, Row::Item { item, .. } if item.id == id))
248            {
249                self.sel = i;
250                return;
251            }
252        }
253        self.sel = self
254            .first_item_row()
255            .unwrap_or(0)
256            .min(self.rows.len().saturating_sub(1));
257    }
258
259    fn first_item_row(&self) -> Option<usize> {
260        self.rows.iter().position(|r| matches!(r, Row::Item { .. }))
261    }

263    /// The id under the cursor.
264    fn selected_id(&self) -> Option<ItemId> {
265        self.selected().map(|i| i.id)
266    }

268    /// The item under the cursor.
269    pub fn selected(&self) -> Option<&Item> {
270        match self.rows.get(self.sel) {
271            Some(Row::Item { item, .. }) => Some(item),
272            _ => None,
273        }
274    }

276    /// The file `E` opens — the board doc, or the selected item's
277    /// file on the `Files` board.
278    fn board_path(&self) -> PathBuf {
279        match self.board {
280            Board::Todo => self.proj.worktree.join("TODO.md"),
281            Board::Backlog => self.proj.worktree.join("BACKLOG.md"),
282            Board::Files => self
283                .selected()
284                .map(|i| self.proj.worktree.join(items::item_rel_path(i)))
285                .unwrap_or_else(|| self.proj.worktree.clone()),
286        }

289    /// Handle one key. Returns `true` to quit.
290    pub fn key(&mut self, k: KeyEvent) -> Result<bool, String> {
291        match self.mode {
292            Mode::Preview => return self.key_preview(k),
293            Mode::Add => return self.key_add(k),
294            Mode::Filter => return self.key_filter(k),
295            Mode::Command => return self.key_command(k),
296            Mode::Help => return self.key_help(k),
297            Mode::Normal => {}
298        }
299        let alt = k.modifiers.contains(KeyModifiers::ALT);
300        let was_g = self.pending == Some('g');
301        let was_d = self.pending == Some('d');
302        self.pending = None;
303        match k.code {
304            KeyCode::Char('c') if k.modifiers.contains(KeyModifiers::CONTROL) => return Ok(true),
305            KeyCode::Char('q') => return Ok(true),
306            KeyCode::Char('1') => self.switch_board(0),
307            KeyCode::Char('2') => self.switch_board(1),
308            KeyCode::Char('3') => self.switch_board(2),
309            KeyCode::Char('h') | KeyCode::Left => self.step_board(-1),
310            KeyCode::Char('l') | KeyCode::Right => self.step_board(1),
311            KeyCode::Char('j') | KeyCode::Down if alt => self.reorder(1),
312            KeyCode::Char('k') | KeyCode::Up if alt => self.reorder(-1),
313            KeyCode::Char('j') | KeyCode::Down => self.move_sel(1),
314            KeyCode::Char('k') | KeyCode::Up => self.move_sel(-1),
315            KeyCode::Char('g') if was_g => self.sel = self.first_item_row().unwrap_or(0),
316            KeyCode::Char('g') => self.pending = Some('g'),
317            KeyCode::Char('G') => self.sel = self.last_item_row().unwrap_or(0),
318            KeyCode::Char('d') if was_d => self.done_selected()?,
319            KeyCode::Char('d') => self.pending = Some('d'),
320            KeyCode::Char('D') => self.defer_selected()?,
321            KeyCode::Char('p') => self.promote_selected()?,
322            KeyCode::Char(' ') => self.done_selected()?,
323            KeyCode::Char('a') => {
324                if self.board == Board::Files {
325                    self.status = "add is a TODO/BACKLOG op — `myco <kind> new` for files".into();
326                } else {
327                    self.mode = Mode::Add;
328                    self.input.clear();
329                }
330            }
331            KeyCode::Char('/') => self.mode = Mode::Filter,
332            KeyCode::Char(':') => {
333                self.mode = Mode::Command;
334                self.cmdline.clear();
335            }
336            KeyCode::Char('F') => {
337                self.filter.clear();
338                self.rebuild_rows();
339            }
340            KeyCode::Enter => {
341                if self.selected().is_some() {
342                    self.mode = Mode::Preview;
343                }
344            }
345            KeyCode::Char('E') => self.pending_editor = Some(self.board_path()),
346            KeyCode::Char('R') => self.reload(),
347            KeyCode::Char('?') => self.mode = Mode::Help,
348            _ => {}
349        }
350        Ok(false)
351    }

353    fn key_preview(&mut self, k: KeyEvent) -> Result<bool, String> {
354        match k.code {
355            KeyCode::Esc | KeyCode::Enter | KeyCode::Char('q') => self.mode = Mode::Normal,
356            _ => {}
357        }
358        Ok(false)
359    }

361    /// `/` filter — chars narrow rows live; `Enter` keeps the
362    /// filter set, `esc` clears it.
363    fn key_filter(&mut self, k: KeyEvent) -> Result<bool, String> {
364        match k.code {
365            KeyCode::Esc => {
366                self.filter.clear();
367                self.mode = Mode::Normal;
368                self.rebuild_rows();
369            }
370            KeyCode::Enter => self.mode = Mode::Normal,
371            KeyCode::Backspace => {
372                self.filter.pop();
373                self.rebuild_rows();
374            }
375            KeyCode::Char(c) => {
376                self.filter.push(c);
377                self.rebuild_rows();
378            }
379            _ => {}
380        }
381        Ok(false)
382    }

384    /// `:` command line — chars append, `Enter` runs `exec_command`,
385    /// `esc` drops the line.
386    fn key_command(&mut self, k: KeyEvent) -> Result<bool, String> {
387        match k.code {
388            KeyCode::Esc => self.mode = Mode::Normal,
389            KeyCode::Enter => {
390                let line = std::mem::take(&mut self.cmdline);
391                self.mode = Mode::Normal;
392                return self.exec_command(&line);
393            }
394            KeyCode::Backspace => {
395                self.cmdline.pop();
396            }
397            KeyCode::Char(c) => self.cmdline.push(c),
398            _ => {}
399        }
400        Ok(false)
401    }

403    /// Run one `:` line — first word is the command, the rest its
404    /// args. v1 keeps the set small: `q`/`q!` quit, `w` reloads
405    /// from disk, `e` opens the board doc in `$EDITOR`, `go <id>`
406    /// and a bare `<id>` jump the cursor, `set id`/`noid` toggles
407    /// the id column (`numbers`/`nonumbers` also accepted).
408    fn exec_command(&mut self, line: &str) -> Result<bool, String> {
409        let line = line.trim();
410        let mut parts = line.splitn(2, char::is_whitespace);
411        let cmd = parts.next().unwrap_or("");
412        let arg = parts.next().unwrap_or("").trim();
413        match cmd {
414            "" => {}
415            "q" | "q!" => return Ok(true),
416            "w" => self.reload(),
417            "e" => self.pending_editor = Some(self.board_path()),
418            "go" => match arg.parse::<ItemId>() {
419                Ok(id) => self.jump_to(id),
420                Err(_) => self.status = format!(":go expects an id, got {arg:?}"),
421            },
422            "set" => match arg {
423                "id" | "numbers" => self.show_ids = true,
424                "noid" | "nonumbers" => self.show_ids = false,
425                _ => self.status = ":set takes id / noid".into(),
426            },
427            _ => match cmd.parse::<ItemId>() {
428                Ok(id) => self.jump_to(id),
429                Err(_) => self.status = format!("unknown command :{cmd}"),
430            },
431        }
432        Ok(false)
433    }

435    /// `:<id>` / `:go <id>` — land the cursor on the item when it's
436    /// a row on this board.
437    fn jump_to(&mut self, id: ItemId) {
438        if self
439            .rows
440            .iter()
441            .any(|r| matches!(r, Row::Item { item, .. } if item.id == id))
442        {
443            self.reselect(Some(id));
444        } else {
445            self.status = format!("{id}: not on this board");
446        }
447    }

458    fn key_add(&mut self, k: KeyEvent) -> Result<bool, String> {
459        match k.code {
460            KeyCode::Esc => self.mode = Mode::Normal,
461            KeyCode::Enter => {
462                let title = self.input.trim().to_string();
463                self.mode = Mode::Normal;
464                if !title.is_empty() {
465                    self.add_item(&title)?;
466                }
467            }
468            KeyCode::Backspace => {
469                self.input.pop();
470            }
471            KeyCode::Char(c) => self.input.push(c),
472            _ => {}
473        }
474        Ok(false)
475    }

477    /// Mouse input (Normal mode only — acting on clicks under an
478    /// overlay or mid-`a`/`/` edit would be surprising): the wheel
479    /// moves the cursor over the board list, a left click picks a
480    /// board tab on the tab row or selects the item row under it.
481    pub fn mouse(&mut self, ev: MouseEvent) {
482        if self.mode != Mode::Normal {
483            return;
484        }
485        let pos = (ev.column, ev.row).into();
486        match ev.kind {
487            MouseEventKind::ScrollUp | MouseEventKind::ScrollDown => {
488                if self.board_area.contains(pos) {
489                    let delta = if ev.kind == MouseEventKind::ScrollUp {
490                        -1
491                    } else {
492                        1
493                    };
494                    self.move_sel(delta);
495                }
496            }
497            MouseEventKind::Down(MouseButton::Left) => {
498                if let Some(tabs) = self.tabs_area {
499                    if ev.row == tabs.y {
500                        if let Some(i) =
501                            crate::render::tab_hit(tabs.width, ev.column.saturating_sub(tabs.x))
502                        {
503                            self.switch_board(i);
504                        }
505                        return;
506                    }
507                }
508                if self.board_area.contains(pos) {
509                    let row = self.viewport_top + usize::from(ev.row - self.board_area.y);
510                    self.select_row(row);
511                }
512            }
513            _ => {}
514        }
515    }

523    /// j/k — skip header rows, wrap nothing (pm stops at edges).
524    fn move_sel(&mut self, delta: isize) {
525        let mut i = self.sel as isize;
526        loop {
527            i += delta;
528            if i < 0 || i as usize >= self.rows.len() {
529                break;
530            }
531            if matches!(self.rows[i as usize], Row::Item { .. }) {
532                self.sel = i as usize;
533                break;
534            }
535        }
536    }

538    /// Mouse click — land the cursor on `row`, but only on items:
539    /// headers aren't selectable and misses keep `sel` put.
540    pub fn select_row(&mut self, row: usize) {
541        if matches!(self.rows.get(row), Some(Row::Item { .. })) {
542            self.sel = row;
543        }
544    }

546    fn switch_board(&mut self, idx: usize) {

551    fn step_board(&mut self, delta: isize) {
552        let idx = (Board::ALL
553            .iter()
554            .position(|b| *b == self.board)
555            .unwrap_or(0) as isize
556            + delta)
557            .rem_euclid(Board::ALL.len() as isize) as usize;
558        self.switch_board(idx);
559    }

561    /// `Space`/`dd` — the `todo done` verb: walk to done, remove the
562    /// line (and its subtree), one commit.
563    fn done_selected(&mut self) -> Result<(), String> {
564        let id = match self.selected_id() {
565            Some(id) => id,
566            None => return Ok(()),
567        };
568        let loc = self
569            .ledger
570            .locate(id)
571            .ok_or_else(|| format!("{id}: no such task"))?;
572        if loc.backlog {
573            self.status = format!("{id} is deferred — `p` promotes it first");
574            return Ok(());
575        }
576        let mut item = self.ledger.take(&loc);
577        items::walk_to(&mut item, Status::Done)?;
578        items::save_ledger(&self.store, &self.ledger)?;
579        items::commit(
580            &self.proj.worktree,
581            &format!("chore: done {id}"),
582            self.no_commit,
583        )?;
584        self.status = format!("done {id} — {}", item.title);
585        self.rebuild_rows();
586        self.reselect(None);
587        Ok(())
588    }

590    /// `D` — `todo defer`: park the subtree into BACKLOG.md.
591    fn defer_selected(&mut self) -> Result<(), String> {
592        let id = match self.selected_id() {
593            Some(id) => id,
594            None => return Ok(()),
595        };
596        let loc = self
597            .ledger
598            .locate(id)
599            .ok_or_else(|| format!("{id}: no such task"))?;
600        if loc.backlog {
601            self.status = format!("{id} is already deferred");
602            return Ok(());
603        }
604        let mut item = self.ledger.take(&loc);
605        items::walk_to(&mut item, Status::Deferred)?;
606        items::walk_descendants_to(&mut item, Status::Deferred)?;
607        self.ledger.put_backlog(item, None);
608        items::save_ledger(&self.store, &self.ledger)?;
609        items::commit(
610            &self.proj.worktree,
611            &format!("chore: defer {id}"),
612            self.no_commit,
613        )?;
614        self.status = format!("deferred {id} → BACKLOG.md");
615        self.rebuild_rows();
616        self.reselect(None);
617        Ok(())
618    }

620    /// `p` — `backlog promote`: unpark the subtree into TODO.md.
621    fn promote_selected(&mut self) -> Result<(), String> {
622        let id = match self.selected_id() {
623            Some(id) => id,
624            None => return Ok(()),
625        };
626        let loc = self
627            .ledger
628            .locate(id)
629            .ok_or_else(|| format!("{id}: no such task"))?;
630        if !loc.backlog {
631            self.status = format!("{id} isn't in BACKLOG.md");
632            return Ok(());
633        }
634        let mut item = self.ledger.take(&loc);
635        items::walk_to(&mut item, Status::Open)?;
636        items::walk_descendants_to(&mut item, Status::Open)?;
637        self.ledger.put_todo(item, None);
638        items::save_ledger(&self.store, &self.ledger)?;
639        items::commit(
640            &self.proj.worktree,
641            &format!("chore: promote {id}"),
642            self.no_commit,
643        )?;
644        self.status = format!("promoted {id} → TODO.md");
645        self.rebuild_rows();
646        self.reselect(Some(id));
647        Ok(())
648    }

650    /// `Alt+↑/↓` — sibling reorder (pm's backlog priority move).
651    fn reorder(&mut self, delta: isize) {
652        let Some(id) = self.selected_id() else {
653            return;
654        };
655        let Some(mut loc) = self.ledger.locate(id) else {
656            return;
657        };
658        if self.ledger.move_within(&mut loc, delta) {
659            if let Err(e) = items::save_ledger(&self.store, &self.ledger).and_then(|_| {
660                items::commit(
661                    &self.proj.worktree,
662                    &format!("chore: reorder {id}"),
663                    self.no_commit,
664                )
665                .map(|_| ())
666            }) {
667                self.status = format!("reorder {id}: {e}");
668                self.reload();
669            } else {
670                self.status = format!("reordered {id}");
671                self.rebuild_rows();
672                self.reselect(Some(id));
673            }
674        }
675    }
676
677    /// `a` Enter — `todo add`/`backlog add` for the active board.
678    fn add_item(&mut self, title: &str) -> Result<(), String> {
679        let id = self
680            .store
681            .alloc_id(ItemKind::Task)
682            .map_err(|e| e.to_string())?;
683        if self.board == Board::Files {
684            return Err("add is a TODO/BACKLOG op".into());
685        }
686        let item = match self.board {
687            Board::Todo => Item::new(id, title).created(items::today()),
688            _ => Item::new(id, title)
689                .status(Status::Deferred)
690                .created(items::today()),
691        };
692        match self.board {
693            Board::Todo => self.ledger.put_todo(item.clone(), None),
694            _ => self.ledger.put_backlog(item.clone(), None),
695        }
696        items::save_ledger(&self.store, &self.ledger)?;
697        items::commit(
698            &self.proj.worktree,
699            &format!("chore: add {id}"),
700            self.no_commit,
701        )?;
702        self.status = format!("added {}", items::compact(&item));
703        self.rebuild_rows();
704        self.reselect(Some(id));
705        Ok(())
706    }
707}

709/// `$MYCO_HOME` → `$XDG_DATA_HOME/myco` → `~/.local/share/myco` —
710/// same convention as the index.
711fn data_dir() -> PathBuf {
712    if let Some(d) = std::env::var_os("MYCO_HOME") {
713        return PathBuf::from(d);
714    }
715    if let Some(d) = std::env::var_os("XDG_DATA_HOME") {
716        return PathBuf::from(d).join("myco");
717    }
718    PathBuf::from(std::env::var_os("HOME").unwrap_or_default()).join(".local/share/myco")
719}

721/// Per-repo session file — `<data_dir>/ui/<hash-of-root>.toml`.
722fn session_path(root: &Path) -> PathBuf {
723    use std::collections::hash_map::DefaultHasher;
724    use std::hash::{Hash, Hasher};
725    let mut h = DefaultHasher::new();
726    root.hash(&mut h);
727    data_dir()
728        .join("ui")
729        .join(format!("{:016x}.toml", h.finish()))
730}

732impl Model {
733    /// Persist board + selected item for the next launch (`q`).
734    pub fn save_session(&self) {
735        let board = match self.board {
736            Board::Todo => "todo",
737            Board::Backlog => "backlog",
738            Board::Files => "files",
739        };
740        let item = self
741            .selected_id()
742            .map(|i| i.to_string())
743            .unwrap_or_default();
744        let p = session_path(&self.proj.root);
745        if let Some(parent) = p.parent() {
746            let _ = std::fs::create_dir_all(parent);
747        }
748        let _ = std::fs::write(p, format!("board = \"{board}\"\nitem = \"{item}\"\n"));
749    }
750
751    /// Reopen on the last board + item (best effort).
752    fn restore_session(&mut self) {
753        let Ok(src) = std::fs::read_to_string(session_path(&self.proj.root)) else {
754            return;
755        };
756        let get = |key: &str| {
757            src.lines()
758                .find_map(|l| l.split_once('=').filter(|(k, _)| k.trim() == key))
759                .map(|(_, v)| v.trim().trim_matches('"').to_string())
760        };
761        self.board = match get("board").as_deref() {
762            Some("backlog") => Board::Backlog,
763            Some("files") => Board::Files,
764            _ => Board::Todo,
765        };
766        self.rebuild_rows();
767        if let Some(id) = get("item").and_then(|s| s.parse::<ItemId>().ok()) {
768            self.reselect(Some(id));
769        }
770    }
771}

773fn push_items(items: &[Item], depth: usize, rows: &mut Vec<Row>) {
774    for item in items {
775        rows.push(Row::Item {
776            item: Box::new(item.clone()),
777            depth,
778        });
779        push_items(&item.children, depth + 1, rows);
780    }
781}
