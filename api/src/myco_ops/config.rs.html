1//! Config plumbing — ops owns the *policy* (which file is the
2//! repo layer, merged vs repo-only reads); the file format is
3//! injected by the composition root (`myco` wires TOML), the
4//! same story as `items::open_with`'s `mk` naming the store
5//! impl. Only `myco_core::Config` crosses the seam.
6
7use std::path::Path;
8
9/// The two config reads ops needs. `layered` runs the impl's
10/// layered search — `Some(path)` pins the repo layer, `None`
11/// its default probe (env → cwd → user layer); `file` reads
12/// exactly one file with no search and no merge — `myco check`
13/// lints the repo layer alone so findings are identical for
14/// every user.
15///
16/// Plain fn pointers keep ops free of `myco-config` impls and
17/// make the loader `Copy` — front-ends bind one `const`.
18#[derive(Clone, Copy)]
19pub struct ConfigLoader {
20    /// Layered load — `Some(path)` pins the repo layer, `None`
21    /// runs the impl's default search.
22    pub layered: fn(Option<&Path>) -> Result<myco_core::Config, String>,
23    /// Single-file load — no search, no merge.
24    pub file: fn(&Path) -> Result<myco_core::Config, String>,
25}
26
27impl ConfigLoader {
28    /// `(self.layered)(repo_path)` — reads better at call sites.
29    pub fn load(&self, repo_path: Option<&Path>) -> Result<myco_core::Config, String> {
30        (self.layered)(repo_path)
31    }
32
33    /// `(self.file)(path)` — reads better at call sites.
34    pub fn load_file(&self, path: &Path) -> Result<myco_core::Config, String> {
35        (self.file)(path)
36    }
37}
38
39/// The layer's `default_view` — same path policy as
40/// [`kinds_for`]: repo `myco.toml` (or `$MYCO_CONFIG`) if present,
41/// the impl's default probe (user layer) otherwise. Missing
42/// config, not error.
43pub fn default_view_for(root: &Path, cfg: ConfigLoader) -> Option<String> {
44    let path = std::env::var("MYCO_CONFIG")
45        .map(std::path::PathBuf::from)
46        .unwrap_or_else(|_| root.join("myco.toml"));
47    cfg.load(path.exists().then_some(&path))
48        .ok()
49        .and_then(|c| c.default_view)
50}
51
52/// The kind table for the repo rooted at `root` — the repo-layer
53/// `myco.toml` (or `$MYCO_CONFIG`), same path policy as
54/// `discover`'s `project_dir` read. The prelude on absence *or*
55/// error — a broken config already fails the verbs that read it
56/// strictly, and `check` lints `[[kinds]]` on its own.
57pub fn kinds_for(root: &Path, cfg: ConfigLoader) -> myco_core::KindTable {
58    let path = std::env::var("MYCO_CONFIG")
59        .map(std::path::PathBuf::from)
60        .unwrap_or_else(|_| root.join("myco.toml"));
61    cfg.load(Some(&path))
62        .ok()
63        .and_then(|c| c.kind_table().ok())
64        .unwrap_or_else(myco_core::KindTable::prelude)
65}

67/// The real TOML impl for tests — most ops tests exercise actual
68/// `myco.toml` files (config-driven `project_dir`, `[views]`).
69#[cfg(test)]
70pub(crate) fn test_loader() -> ConfigLoader {
71    use myco_config::ConfigSource;
72    ConfigLoader {
73        layered: |p| {
74            myco_config_toml::TomlSource {
75                path: p.map(|p| p.to_string_lossy().into_owned()),
76            }
77            .load()
78        },
79        file: myco_config_toml::load_file,
80    }
81}

83#[cfg(test)]
84#[allow(clippy::unwrap_used)]
85mod tests {
86    use super::*;
87
88    #[test]
89    fn default_view_for_reads_the_repo_layer() {
90        let dir = tempfile::tempdir().unwrap();
91        std::fs::write(dir.path().join("myco.toml"), "default_view = \"board\"\n").unwrap();
92        temp_env::with_var("MYCO_CONFIG", None::<&str>, || {
93            assert_eq!(
94                default_view_for(dir.path(), test_loader()).as_deref(),
95                Some("board")
96            );
97        });
98    }
99
100    #[test]
101    fn default_view_for_missing_config_is_none() {
102        let dir = tempfile::tempdir().unwrap();
103        // $MYCO_CONFIG naming a missing file → the impl's default
104        // probe also misses → None, not an error
105        temp_env::with_var("MYCO_CONFIG", Some("/nonexistent/myco.toml"), || {
106            assert_eq!(default_view_for(dir.path(), test_loader()), None);
107        });
108    }
