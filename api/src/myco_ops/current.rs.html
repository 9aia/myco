1//! The current-ledger selector (P-015/P-016) — the user-level,
2//! mutable "which ledger does myco open" pointer.
3//!
4//! Three sources feed discovery, all resolved against the same
5//! target vocabulary (`repo` | `org` | `<universe name>` |
6//! `</abs/path>` | `name:<repo>`):
7//!
8//! - `MYCO_LEDGER` / `MYCO_LEDGER_SCOPE` — per-invocation env.
9//! - `[current]` in the user-layer `myco.toml` — `mode = always`
10//!   beats cwd discovery, `mode = backup` answers only when
11//!   discovery finds nothing. `MYCO_CONFIG` may *read* the table
12//!   (it wins over the user file); writes always land in the
13//!   canonical user file. Repo `./myco.toml` is never consulted —
14//!   a repo must not pin the user's ledger.
15//! - `.myco.toml` — a per-directory pointer (`ledger = "…"`),
16//!   found walking up from cwd, honored only outside git repos
17//!   (a committed pointer would be repo-controlled injection).
18//!
19//! Writes go through `toml_edit` so comments/formatting survive.
20

29/// A resolved selector — which ledger + where it sits.
30#[derive(Debug, Clone)]
31pub struct Selector {
32    /// The raw target string (`repo`/`org`/name/path/`name:`).
33    pub ledger: String,
34    /// `always` (before discovery) or `backup` (after, before org).
35    pub mode: CurrentMode,
36}
37
38/// `MYCO_LEDGER` (+ `MYCO_LEDGER_SCOPE`, default `always`).
39/// `repo`/empty disables the env override for the invocation.
40pub fn env_selector() -> Option<Selector> {
41    let ledger = std::env::var("MYCO_LEDGER").ok()?;
42    let ledger = ledger.trim();
43    if ledger.is_empty() || ledger == "repo" {
44        return None;
45    }
46    let mode = match std::env::var("MYCO_LEDGER_SCOPE").as_deref() {
47        Ok("backup") => CurrentMode::Backup,
48        _ => CurrentMode::Always,
49    };
50    Some(Selector {
51        ledger: ledger.to_string(),
52        mode,
53    })
54}

56/// The persisted `[current]` table — `MYCO_CONFIG` (when set and
57/// readable) wins over the user-layer file; repo `./myco.toml`
58/// is never a source.
59pub fn config_selector(cfg: ConfigLoader) -> Option<Selector> {
60    let path = crate::config::user_state_path()?;
61    let c = cfg.load_file(&path).ok()?;
62    c.current.map(|cur| Selector {
63        ledger: cur.ledger,
64        mode: cur.mode,
65    })
66}

68/// The `.myco.toml` directory pointer — upward search from `cwd`,
69/// nearest wins, files inside git repos skipped (repo-controlled
70/// injection). Returns the file path + selector; the pointer has
71/// fallback semantics only (no mode).
72pub fn pointer_selector(cwd: &Path) -> Option<(PathBuf, Selector)> {
73    for d in cwd.ancestors() {
74        let f = d.join(".myco.toml");
75        if !f.is_file() {
76            continue;
77        }
78        // inside a git worktree → repo-controlled, skip and keep
79        // walking (a parent plain dir's pointer may still apply)
80        if crate::vcs::repo_root(d).is_ok() {
81            continue;
82        }
83        if let Some(ledger) = read_pointer(&f) {
84            return Some((
85                f,
86                Selector {
87                    ledger,
88                    mode: CurrentMode::Backup,
89                },
90            ));
91        }
92    }
93    None
94}

96fn read_pointer(path: &Path) -> Option<String> {
97    let text = std::fs::read_to_string(path).ok()?;
98    let doc = text.parse::<toml_edit::DocumentMut>().ok()?;
99    let ledger = doc.get("ledger")?.as_str()?.trim();
100    (!ledger.is_empty() && ledger != "repo").then(|| ledger.to_string())
101}

103/// Resolve a selector to a [`Project`] — the target vocabulary
104/// from P-016. `repo` never reaches here (callers filter it).
105pub fn resolve(
106    sel: &Selector,
107    cwd: &Path,
108    index: Option<&dyn Index>,
109    cfg: ConfigLoader,
110) -> Result<Project, String> {
111    resolve_target(&sel.ledger, cwd, index, cfg).map_err(|e| {
112        format!(
113            "current ledger '{}': {e}\n  \
114             escape a stale pin: MYCO_LEDGER=repo <cmd>  \
115             (or `myco current clear`)",
116            sel.ledger
117        )
118    })
119}

121/// The target vocabulary — `repo`/`org`/`name:`/abs-path/name.
122fn resolve_target(
123    target: &str,
124    cwd: &Path,
125    index: Option<&dyn Index>,
126    cfg: ConfigLoader,
127) -> Result<Project, String> {
128    let target = target.trim();
129    if target == "repo" {
130        return Err("`repo` disables the selector — nothing to resolve".into());
131    }
132    if target == "org" {
133        return org_project(cwd, index, cfg);
134    }
135    if let Some(name) = target.strip_prefix("name:") {
136        return named_project(name, cwd, index, cfg);
137    }
138    if Path::new(target).is_absolute() {
139        return path_project(Path::new(target));
140    }
141    named_project(target, cwd, index, cfg)
142}

144/// `org` — the index-designated org ledger (Lasius-source org
145/// preferred, mirroring `discover`'s org fallback).
146fn org_project(
147    cwd: &Path,
148    index: Option<&dyn Index>,
149    cfg: ConfigLoader,
150) -> Result<Project, String> {
151    let index = index.ok_or("target `org` needs the repo index")?;
152    let repos = crate::universe::repos(cwd, index, cfg)?;
153    let org = repos
154        .iter()
155        .find(|u| {
156            u.source == crate::universe::RepoSource::Lasius && u.entry.ledger == LedgerMode::Org
157        })
158        .or_else(|| repos.iter().find(|u| u.entry.ledger == LedgerMode::Org))
159        .ok_or("no org ledger designated — `myco index add <repo> --org`")?;
160    project_from_entry(org)
161}

163/// `<name>` / `name:<name>` — exactly one universe match required.
164fn named_project(
165    name: &str,
166    cwd: &Path,
167    index: Option<&dyn Index>,
168    cfg: ConfigLoader,
169) -> Result<Project, String> {
170    if name.is_empty() {
171        return Err("empty repo name".into());
172    }
173    let index = index.ok_or_else(|| format!("target `{name}` needs the repo index"))?;
174    let repos = crate::universe::repos(cwd, index, cfg)?;
175    let matches: Vec<_> = repos.iter().filter(|u| u.entry.name == name).collect();
176    match matches.as_slice() {
177        [] => Err(format!("no universe repo named `{name}`")),
178        [u] => project_from_entry(u),
179        many => Err(format!(
180            "ambiguous name `{name}` — matches: {}",
181            many.iter()
182                .map(|u| u.entry.name.as_str())
183                .collect::<Vec<_>>()
184                .join(", ")
185        )),
186    }
187}

189/// A universe entry → its ledger worktree, revalidated live.
190fn project_from_entry(u: &crate::universe::UniverseRepo) -> Result<Project, String> {
191    let wt = u
192        .entry
193        .worktree_path()
194        .ok_or_else(|| format!("`{}` has no ledger worktree", u.entry.name))?;
195    let wt = wt.as_std_path();
196    if !discover::qualifies(wt) {
197        return Err(format!(
198            "`{}` worktree {} no longer qualifies — re-run discovery",
199            u.entry.name,
200            wt.display()
201        ));
202    }
203    Ok(Project {
204        root: u.entry.path.as_std_path().to_path_buf(),
205        worktree: wt.canonicalize().unwrap_or_else(|_| wt.to_path_buf()),
206    })
207}

209/// `</abs/path>` — a ledger worktree directly, or a repo root
210/// resolved like `--root` (`<root>/project`).
211fn path_project(p: &Path) -> Result<Project, String> {
212    let p = p.canonicalize().unwrap_or_else(|_| p.to_path_buf());
213    if discover::qualifies(&p) {
214        let root = crate::vcs::main_worktree(&p)
215            .or_else(|| p.parent().map(Path::to_path_buf))
216            .unwrap_or_else(|| p.clone());
217        return Ok(Project { root, worktree: p });
218    }
219    let wt = p.join(discover::DIR);
220    if discover::qualifies(&wt) {
221        return Ok(Project {
222            root: p,
223            worktree: wt,
224        });
225    }
226    Err(format!(
227        "{} is not a myco ledger worktree or repo root",
228        p.display()
229    ))
230}

232/// `current list` — every ledger-capable universe repo plus the
233/// `repo`/`org` pseudo-targets. Needs the index.
234pub fn list_targets(
235    cwd: &Path,
236    index: &dyn Index,
237    cfg: ConfigLoader,
238) -> Result<Vec<serde_json::Value>, String> {
239    let repos = crate::universe::repos(cwd, index, cfg)?;
240    let mut out = vec![
241        serde_json::json!({"target": "repo", "kind": "selector", "note": "cwd discovery (default)"}),
242        serde_json::json!({"target": "org", "kind": "selector", "note": "index-designated org ledger"}),
243    ];
244    for u in repos {
245        let Some(wt) = u.entry.worktree_path() else {
246            continue;
247        };
248        if !discover::qualifies(wt.as_std_path()) {
249            continue;
250        }
251        out.push(serde_json::json!({
252            "target": u.entry.name,
253            "kind": match u.source {
254                crate::universe::RepoSource::Lasius => "lasius",
255                crate::universe::RepoSource::Registry => "registry",
256            },
257            "ledger": match u.entry.ledger {
258                LedgerMode::Org => "org",
259                LedgerMode::None => "none",
260                LedgerMode::Local => "local",
261            },
262            "worktree": wt.as_std_path(),
263        }));
264    }
265    Ok(out)
266}

268/// `current use|clear` — write `[current]` to the canonical
269/// user-layer `myco.toml` via `toml_edit` (comments preserved).
270/// `MYCO_CONFIG` is ignored — env is a read override only.
271pub fn write_config(ledger: &str, mode: &str) -> Result<PathBuf, String> {
272    let path = crate::config::user_config_write_path()
273        .ok_or("no user config dir ($XDG_CONFIG_HOME or $HOME)".to_string())?;
274    if let Some(parent) = path.parent() {
275        std::fs::create_dir_all(parent).map_err(|e| format!("create {}: {e}", parent.display()))?;
276    }
277    let text = std::fs::read_to_string(&path).unwrap_or_default();
278    let mut doc = text
279        .parse::<toml_edit::DocumentMut>()
280        .map_err(|e| format!("{}: {e}", path.display()))?;
281    if doc.get("current").is_none() {
282        doc["current"] = toml_edit::Item::Table(toml_edit::Table::new());
283    }
284    let cur = doc["current"]
285        .as_table_mut()
286        .ok_or("`current` in myco.toml is not a table")?;
287    cur["ledger"] = toml_edit::value(ledger);
288    cur["mode"] = toml_edit::value(mode);
289    std::fs::write(&path, doc.to_string()).map_err(|e| format!("write {}: {e}", path.display()))?;
290    Ok(path)
291}

293/// `current use <target> --here` / `clear --here` — the
294/// `.myco.toml` directory pointer in `dir` (fallback semantics
295/// only — no mode key).
296pub fn write_pointer(dir: &Path, ledger: Option<&str>) -> Result<PathBuf, String> {
297    let path = dir.join(".myco.toml");
298    match ledger {
299        Some(l) => {
300            let text = format!("ledger = \"{}\"\n", l.replace('"', "\\\""));
301            std::fs::write(&path, text).map_err(|e| format!("write {}: {e}", path.display()))?;
302        }
303        None => {
304            if path.exists() {
305                std::fs::remove_file(&path)
306                    .map_err(|e| format!("remove {}: {e}", path.display()))?;
307            }
308        }
309    }
310    Ok(path)
311}

355    #[test]
356    fn pointer_ignores_git_dirs() {
357        // a .myco.toml inside a git repo is skipped; one in a
358        // plain parent dir still answers
359        let base = tempfile::tempdir().unwrap();
360        let repo = base.path().join("repo");
361        std::fs::create_dir_all(&repo).unwrap();
362        let st = std::process::Command::new("git")
363            .args(["init", "-q"])
364            .arg(&repo)
365            .status()
366            .unwrap();
367        assert!(st.success());
368        std::fs::write(repo.join(".myco.toml"), "ledger = \"org\"\n").unwrap();
369        std::fs::write(base.path().join(".myco.toml"), "ledger = \"myrepo\"\n").unwrap();
370        let inside = repo.join("sub");
371        std::fs::create_dir_all(&inside).unwrap();
372        let (f, s) = pointer_selector(&inside).unwrap();
373        assert_eq!(f, base.path().join(".myco.toml"));
374        assert_eq!(s.ledger, "myrepo");
375        assert_eq!(s.mode, CurrentMode::Backup);
376    }

378    #[test]
379    fn write_config_roundtrip_and_clear() {
380        let xdg = tempfile::tempdir().unwrap();
381        temp_env::with_vars(
382            [
383                ("XDG_CONFIG_HOME", Some(xdg.path().to_str().unwrap())),
384                ("HOME", None::<&str>),
385                ("MYCO_CONFIG", None::<&str>),
386            ],
387            || {
388                let p = write_config("org", "always").unwrap();
389                assert_eq!(p, xdg.path().join("myco/myco.toml"));
390                // pre-existing comments survive the toml_edit write
391                let mut text = std::fs::read_to_string(&p).unwrap();
392                text = format!("# my prefs\n{text}");
393                std::fs::write(&p, &text).unwrap();
394                write_config("myrepo", "backup").unwrap();
395                let out = std::fs::read_to_string(&p).unwrap();
396                assert!(out.starts_with("# my prefs"), "{out}");
397                assert!(out.contains("ledger = \"myrepo\""), "{out}");
398                assert!(out.contains("mode = \"backup\""), "{out}");
399
400                let sel = config_selector(crate::config::test_loader()).unwrap();
401                assert_eq!(sel.ledger, "myrepo");
402                assert_eq!(sel.mode, CurrentMode::Backup);
403
404                // clear → ledger = "repo" (inert)
405                write_config("repo", "backup").unwrap();
406                let sel = config_selector(crate::config::test_loader()).unwrap();
407                assert_eq!(sel.ledger, "repo");
408            },
409        );
