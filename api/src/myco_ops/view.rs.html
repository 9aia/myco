1//! Named views — built-in presets plus `[views]` from merged
2//! config layers (P-006). Resolution is I/O (config files), so it
3//! lives here; evaluation is pure and stays in `myco_core::view`.
4
5use std::collections::BTreeMap;
6use std::path::Path;
7
8use myco_config::ConfigSource;
9use myco_config_toml::TomlSource;
10use myco_core::{GroupAxis, Status, ViewFilter, ViewFormat, ViewSpec};

12/// Where a named view came from — config shadows builtin.
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub enum Origin {
15    /// A preset compiled into the binary (`board`, `epic`, …).
16    Builtin,
17    /// A `[views.<name>]` entry from a config layer.
18    Config,
19}
20
21/// A resolved named view.
22#[derive(Debug, Clone)]
23pub struct NamedView {
24    /// The `[views.<name>]` / preset name.
25    pub name: String,
26    /// The spec (already merged — a config view fully replaces a
27    /// same-named builtin).
28    pub spec: ViewSpec,
29    /// Builtin preset or config-defined.
30    pub origin: Origin,
31}
32
33/// The built-in presets — the read verbs' shapes as data. Default
34/// `sort` (empty) means the engine's `pri,id`.
35pub fn builtins() -> Vec<NamedView> {
36    let live = ViewFilter {
37        status: vec![
38            Status::Open,
39            Status::Claimed,
40            Status::InProgress,
41            Status::Blocked,
42            Status::Deferred,
43        ],
44        ..ViewFilter::default()
45    };
46    let spec = |filter: ViewFilter, group_by, format| ViewSpec {
47        filter,
48        group_by,
49        sort: vec![],
50        limit: None,
51        format,
52    };
53    vec![
54        NamedView {
55            name: "board".into(),
56            spec: spec(live.clone(), Some(GroupAxis::Status), ViewFormat::Board),
57            origin: Origin::Builtin,
58        },
59        NamedView {
60            name: "open".into(),
61            spec: spec(
62                ViewFilter {
63                    status: vec![Status::Open],
64                    ..ViewFilter::default()
65                },
66                None,
67                ViewFormat::List,
68            ),
69            origin: Origin::Builtin,
70        },
71        NamedView {
72            name: "epic".into(),
73            spec: spec(live.clone(), Some(GroupAxis::Epic), ViewFormat::Board),
74            origin: Origin::Builtin,
75        },
76        NamedView {
77            name: "milestone".into(),
78            spec: spec(live, Some(GroupAxis::Milestone), ViewFormat::Board),
79            origin: Origin::Builtin,
80        },
81    ]
82}

84/// `[views]` for a repo — the repo layer is `<root>/myco.toml` when
85/// it exists (else the default `./myco.toml` probe), merged with
86/// the user layer by `TomlSource`. "No config anywhere" is empty,
87/// not an error — builtins still resolve. `strict`: a config path
88/// that names a missing file errors (an explicit `$MYCO_CONFIG` is
89/// "the user pointed at it" — `myco view` reports the miss);
90/// lenient callers degrade to the presets — a read verb shouldn't
91/// brick on a config-file absence.
92fn config_views(root: &Path, strict: bool) -> Result<BTreeMap<String, ViewSpec>, String> {
93    let repo_cfg = root.join("myco.toml");
94    let src = TomlSource {
95        path: repo_cfg
96            .exists()
97            .then(|| repo_cfg.to_string_lossy().into_owned()),
98    };
99    match src.load() {
100        Ok(c) => Ok(c.views),
101        // the documented search-miss — absence isn't a failure
102        Err(e) if e.contains("no myco.toml found") => Ok(BTreeMap::new()),
103        Err(e) if !strict && e.contains("not found") => Ok(BTreeMap::new()),
104        Err(e) => Err(e),
105    }
106}

108/// The warning a config-vs-builtin name collision earns.
109fn shadow_warning(name: &str) -> String {
110    format!("view \"{name}\" shadows the built-in preset")
111}
112
113/// Builtin presets merged with the repo's config views — config
114/// wins a name collision and the returned warnings name it.
115/// Specs are *not* validated here — `myco check` lints them all.
116fn merged_views(root: &Path, strict: bool) -> Result<(Vec<NamedView>, Vec<String>), String> {
117    let mut warnings = Vec::new();
118    let mut out = builtins();
119    for (name, spec) in config_views(root, strict)? {
120        if let Some(v) = out.iter_mut().find(|v| v.name == name) {
121            warnings.push(shadow_warning(&name));
122            v.spec = spec;
123            v.origin = Origin::Config;
124        } else {
125            out.push(NamedView {
126                name,
127                spec,
128                origin: Origin::Config,
129            });
130        }
131    }
132    Ok((out, warnings))
133}
134
135/// [`merged_views`] with strict config handling — `myco view`,
136/// `myco check`, `myco serve`.
137pub fn named_views(root: &Path) -> Result<(Vec<NamedView>, Vec<String>), String> {
138    merged_views(root, true)
139}
140
141/// [`merged_views`] tolerant of a *missing* config — for the read
142/// verbs that now resolve named views (`board`, `epic`, …):
143/// `$MYCO_CONFIG` naming a missing file degrades to the presets;
144/// an unparseable or invalid config still errors.
145pub fn named_views_lenient(root: &Path) -> Result<(Vec<NamedView>, Vec<String>), String> {
146    merged_views(root, false)
147}

149/// Lint the repo-layer config only — deterministic, so `myco
150/// check` reports the same findings for every user and CI.
151/// `Ok` = per-view validation errors (`(name, error)`); `Err` =
152/// the file is unreadable/unparseable. Missing file is `Ok(vec![])`.
153/// Personal (user-layer) views are intentionally not linted here.
154pub fn validate_repo_views(root: &Path) -> Result<Vec<(String, String)>, String> {
155    let path = root.join("myco.toml");
156    if !path.exists() {
157        return Ok(vec![]);
158    }
159    let cfg = myco_config_toml::load_file(&path)?;
160    Ok(cfg
161        .views
162        .iter()
163        .filter_map(|(name, spec)| spec.validate().err().map(|e| (name.clone(), e)))
164        .collect())
165}

167/// Resolve one view by name — config shadows builtin (warns).
168/// Unknown names error listing the defined set. The spec is
169/// validated before return. Warnings are scoped to the resolved
170/// view — a verb loading `board` shouldn't hear about a shadowed
171/// `[views.epic]`.
172pub fn resolve(root: &Path, name: &str) -> Result<(NamedView, Vec<String>), String> {
173    resolve_in(root, name, true)
174}
175
176/// [`resolve`] tolerant of a *missing* config — the read-verb
177/// entry point (`board`, `epic <name>`, `milestone <name>`).
178pub fn resolve_lenient(root: &Path, name: &str) -> Result<(NamedView, Vec<String>), String> {
179    resolve_in(root, name, false)
180}
181
182fn resolve_in(root: &Path, name: &str, strict: bool) -> Result<(NamedView, Vec<String>), String> {
183    let (views, _) = merged_views(root, strict)?;
184    let names: Vec<String> = views.iter().map(|v| v.name.clone()).collect();
185    let view = views
186        .into_iter()
187        .find(|v| v.name == name)
188        .ok_or_else(|| format!("unknown view \"{name}\" — defined: {}", names.join(", ")))?;
189    view.spec.validate()?;
190    let warnings = if view.origin == Origin::Config && builtins().iter().any(|b| b.name == name) {
191        vec![shadow_warning(name)]
192    } else {
193        Vec::new()
194    };
195    Ok((view, warnings))
196}

198#[cfg(test)]
199#[allow(clippy::unwrap_used)]
200mod tests {
201    use super::*;
202    use myco_core::ItemKind;
203
204    #[test]
205    fn builtins_are_valid() {
206        let names: Vec<_> = builtins().iter().map(|v| v.name.clone()).collect();
207        assert_eq!(names, ["board", "open", "epic", "milestone"]);
208        for v in builtins() {
209            v.spec.validate().unwrap();
210        }
211        let board = &builtins()[0].spec;
212        assert_eq!(board.group_by, Some(GroupAxis::Status));
213        assert_eq!(board.format, ViewFormat::Board);
214        assert!(!board.filter.status.contains(&Status::Done));
215    }
216
217    #[test]
218    fn resolve_merges_config_and_warns_on_shadow() {
219        let dir = tempfile::tempdir().unwrap();
220        std::fs::write(
221            dir.path().join("myco.toml"),
222            "[views.board]\nfilter = { kind = [\"issue\"] }\n[views.mine]\nfilter = { pri_min = \"high\" }\n",
223        )
224        .unwrap();
225        temp_env::with_var("MYCO_CONFIG", None::<&str>, || {
226            let (views, warnings) = named_views(dir.path()).unwrap();
227            assert_eq!(warnings.len(), 1);
228            assert!(warnings[0].contains("shadows"));
229            // config's board replaces the builtin
230            let board = views.iter().find(|v| v.name == "board").unwrap();
231            assert_eq!(board.origin, Origin::Config);
232            assert_eq!(board.spec.filter.kind, vec![ItemKind::Issue]);
233            // repo + builtin names coexist
234            assert!(views.iter().any(|v| v.name == "mine"));
235            // unknown name errors listing defined views
236            let err = resolve(dir.path(), "nope").unwrap_err();
237            assert!(err.contains("mine") && err.contains("board"), "{err}");
238        });
239    }
240
241    #[test]
242    fn no_config_still_resolves_builtins() {
243        let dir = tempfile::tempdir().unwrap();
244        temp_env::with_vars(
245            [
246                ("MYCO_CONFIG", None::<&str>),
247                (
248                    "XDG_CONFIG_HOME",
249                    Some(dir.path().join("none").to_string_lossy().as_ref()),
250                ),
251                ("HOME", Some(dir.path().to_string_lossy().as_ref())),
252            ],
253            || {
254                let (v, _) = resolve(dir.path(), "board").unwrap();
255                assert_eq!(v.origin, Origin::Builtin);
256            },
257        );
258    }
