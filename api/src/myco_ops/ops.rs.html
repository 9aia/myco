1//! Shared plumbing for the lifecycle verbs — the `project/`
2//! scaffold and the auto-commit-per-verb machinery.
3//!
4//! Every mutating verb ends in one atomic commit on the `project`
5//! branch (`--no-commit` or `$MYCO_NO_COMMIT` opts out, e.g. for
6//! tests and dry runs). Commits happen only when the worktree
7//! actually changed — a no-op verb leaves no noise in the audit
8//! trail.
9

17/// Dirs `myco init` / `check --fix` scaffold (with `.gitkeep`).
18const SCAFFOLD_DIRS: [&str; 5] = ["issues", "proposals", "adrs", "plans", "handoffs"];

20/// `MdStore` rooted at a worktree path.
21pub fn store_at(worktree: &Path) -> Result<MdStore, String> {
22    let root = Utf8PathBuf::from_path_buf(worktree.to_path_buf())
23        .map_err(|p| format!("{}: non-utf8 path", p.display()))?;
24    Ok(MdStore::new(root))
25}

27/// Name of the lock file — inside the repo's shared git dir
28/// (`<main>/.git/myco.lock`) so it never lands in a ledger commit
29/// (`git add -A` can't see it) and covers the `project/` worktree
30/// of every checkout of the repo.
31const LOCK_FILE: &str = "myco.lock";
32
33/// RAII guard for a worktree lock — an OS advisory lock (`flock`)
34/// on the repo's lock file. The lock releases when the guard
35/// drops (or the process dies — the OS cleans up).
36#[derive(Debug)]
37pub struct WorktreeLock {
38    _file: std::fs::File,
39}
40
41/// The file the lock lives on: `<git-common-dir>/myco.lock` — the
42/// common dir is shared by a repo's worktrees and exists before
43/// `myco init` creates the worktree, so one path covers init and
44/// every verb. A dir outside any repo falls back to a dotfile
45/// inside itself (nothing can autocommit there anyway, and the
46/// ledger spec ignores dotfiles).
47fn lock_path(dir: &Path) -> Result<PathBuf, String> {
48    match myco_git::common_dir(dir) {
49        Ok(git) => Ok(git.join(LOCK_FILE)),
50        Err(_) => {
51            if dir.is_dir() {
52                Ok(dir.join(".myco.lock"))
53            } else {
54                Err(format!("{} is not a directory", dir.display()))
55            }
56        }
57    }
58}
59
60fn lock_at(dir: &Path, exclusive: bool) -> Result<WorktreeLock, String> {
61    let path = lock_path(dir)?;
62    let file = std::fs::OpenOptions::new()
63        .create(true)
64        .truncate(false)
65        .write(true)
66        .open(&path)
67        .map_err(|e| format!("open {}: {e}", path.display()))?;
68    let taken = if exclusive {
69        file.try_lock()
70    } else {
71        file.try_lock_shared()
72    };
73    match taken {
74        Ok(()) => Ok(WorktreeLock { _file: file }),
75        Err(std::fs::TryLockError::WouldBlock) => {
76            Err("another myco process is writing — retry".into())
77        }
78        Err(std::fs::TryLockError::Error(e)) => Err(format!("lock {}: {e}", path.display())),
79    }
80}

82/// The exclusive mutation lock — every load→mutate→save→commit
83/// verb holds it for the verb's whole life (including the
84/// autocommit). Fail-fast on contention: concurrent writers
85/// corrupt the ledger (dup ids, last-writer-wins), so the second
86/// one gets a retry hint instead.
87///
88/// `dir` is any path inside the repo — the worktree or the repo
89/// root; the lock lands on the shared git dir either way.
90pub fn lock_worktree(dir: &Path) -> Result<WorktreeLock, String> {
91    lock_at(dir, true)
92}
93
94/// The shared read lock — concurrent readers don't block each
95/// other, but a reader contending with a writer fails fast
96/// rather than rendering a mid-verb ledger.
97pub fn lock_shared(dir: &Path) -> Result<WorktreeLock, String> {
98    lock_at(dir, false)
99}

116/// Create every missing scaffold piece in the worktree; existing
117/// files are never overwritten. Returns the rel paths created.
118pub fn scaffold(store: &MdStore, name: &str) -> Result<Vec<String>, StoreError> {
119    let mut created = Vec::new();
120    let root = store.root();
121    let missing = |rel: &str| !root.join(rel).exists();
122    if missing("meta.toml") {
123        store.save_meta(&ProjectMeta::new(name))?;
124        created.push("meta.toml".to_string());
125    }
126    if missing("TODO.md") {
127        store.save_todo(&TodoDoc::new(name))?;
128        created.push("TODO.md".to_string());
129    }
130    if missing("BACKLOG.md") {
131        store.save_backlog(&BacklogDoc::default())?;
132        created.push("BACKLOG.md".to_string());
133    }
134    if missing("index.md") {
135        store.save_index(&index_template(name))?;
136        created.push("index.md".to_string());
137    }
138    for dir in SCAFFOLD_DIRS {
139        let keep = root.join(dir).join(".gitkeep");
140        let empty_dir = match std::fs::read_dir(root.join(dir)) {
141            Ok(mut it) => it.next().is_none(),
142            Err(_) => true,
143        };
144        if empty_dir && !keep.exists() {
145            std::fs::create_dir_all(root.join(dir))?;
146            std::fs::write(&keep, "")?;
147            created.push(format!("{dir}/.gitkeep"));
148        }
149    }
150    Ok(created)
151}

154/// (any value) suppresses them.

159/// One atomic commit of everything under `dir` — the
160/// auto-commit-per-verb path. No commit when disabled or when the
161/// tree is clean. Returns whether a commit was made.
162pub fn autocommit(dir: &Path, msg: &str, enabled: bool) -> Result<bool, String> {
163    if !enabled || !myco_git::has_changes(dir).map_err(|e| e.to_string())? {
164        return Ok(false);
165    }
166    myco_git::commit_all(dir, msg)
167        .map_err(|e| format!("git commit failed in {}: {e}", dir.display()))?;
168    Ok(true)
169}

171/// Commit only `paths` inside `dir` — used for repo-root edits
172/// (`.gitignore`, `pm/` removal) so an unrelated dirty tree isn't
173/// swept into the verb's commit. Returns whether a commit was made.
174///
175/// Scoping is per *file*, not per hunk: callers must pass only
176/// paths myco itself wrote this invocation. If the user had
177/// pre-existing uncommitted edits inside one of those files, they
178/// land in the commit under myco's message — an accepted
179/// limitation (hunk-level staging via `git apply --cached` was
180/// judged disproportionate for the managed-file call sites).
181pub fn commit_paths(dir: &Path, msg: &str, paths: &[&str], enabled: bool) -> Result<bool, String> {
182    if !enabled {
183        return Ok(false);
184    }
185    let mut args = vec!["status", "--porcelain", "--"];
186    args.extend_from_slice(paths);
187    let dirty = myco_git::git_out(dir, &args).map_err(|e| e.to_string())?;
188    if dirty.is_empty() {
189        return Ok(false);
190    }
191    let mut add = vec!["add", "--"];
192    add.extend_from_slice(paths);
193    myco_git::git_out(dir, &add).map_err(|e| e.to_string())?;
194    let mut commit = vec!["commit", "-m", msg, "--"];
195    commit.extend_from_slice(paths);
196    myco_git::git_ident(dir, &commit)
197        .map_err(|e| e.to_string())
198        .map(|_| true)
199}

209/// Default pager — `-F` quits instantly under one screen, `-R`
210/// keeps ANSI colors, `-S` chops instead of wrapping rows, `-X`
211/// leaves output in the scrollback.
212const DEFAULT_PAGER: &str = "less -FRSX";
213
214/// Pager program: `$MYCO_PAGER` → `$PAGER` → [`DEFAULT_PAGER`].
215/// An explicitly empty value disables paging.
216fn pager_program() -> Option<String> {
217    for var in ["MYCO_PAGER", "PAGER"] {
218        if let Ok(v) = std::env::var(var) {
219            return if v.trim().is_empty() { None } else { Some(v) };
220        }
221    }
222    Some(DEFAULT_PAGER.to_string())
223}
224
225/// `sh -c <pager>` with `text` on its stdin — the pager inherits
226/// our stdout/stderr, so scrolling and colors work as usual.
227/// `$MYCO_PAGER`/`$PAGER` are executed verbatim (git-parity
228/// behavior) — an env var is a trusted setting, never user data.
229fn spawn_pager(pager: &str, text: &str) -> std::io::Result<()> {
230    use std::io::Write;
231    let mut child = std::process::Command::new("sh")
232        .args(["-c", pager])
233        .stdin(std::process::Stdio::piped())
234        .spawn()?;
235    if let Some(mut stdin) = child.stdin.take() {
236        // quitting early (`q`) closes the pipe — with SIGPIPE at
237        // its default disposition we die like any unix tool, which
238        // is what the user expects; ignore the write error for the
239        // SIG_IGN case
240        let _ = stdin.write_all(text.as_bytes());
241    }
242    child.wait().map(|_| ())
243}
244
245/// Print `text`, paging it through the user's pager when stdout is
246/// a tty — `less -F` exits instantly for one-screen output, so
247/// callers can page unconditionally. Non-tty output, a disabled or
248/// unspawnable pager all fall back to `println!`.
249pub fn print_paged(text: &str) {
250    use std::io::IsTerminal;
251    if std::io::stdout().is_terminal() {
252        if let Some(pager) = pager_program() {
253            if spawn_pager(&pager, text).is_ok() {
254                return;
255            }
256        }
257    }
258    println!("{text}");
259}

268    fn pager_program_env_precedence_and_disable() {
269        temp_env::with_vars(
270            [("MYCO_PAGER", None::<&str>), ("PAGER", None::<&str>)],
271            || assert_eq!(pager_program().as_deref(), Some("less -FRSX")),
272        );
273        temp_env::with_var("MYCO_PAGER", Some("cat"), || {
274            assert_eq!(pager_program().as_deref(), Some("cat"));
275        });
276        // empty = explicitly disabled
277        temp_env::with_var("MYCO_PAGER", Some("  "), || {
278            assert_eq!(pager_program(), None);
279        });
280        // PAGER wins over nothing but loses to MYCO_PAGER
281        temp_env::with_vars(
282            [("MYCO_PAGER", Some("most")), ("PAGER", Some("more"))],
283            || assert_eq!(pager_program().as_deref(), Some("most")),

298    #[test]
299    fn lock_serializes_writers_and_reads() {
300        let d = tempfile::tempdir().unwrap();
301        // first writer takes it; a second writer and a reader
302        // both fail fast with the retry hint
303        let w = lock_worktree(d.path()).unwrap();
304        let err = lock_worktree(d.path()).unwrap_err();
305        assert!(err.contains("another myco process is writing"), "{err}");
306        let err = lock_shared(d.path()).unwrap_err();
307        assert!(err.contains("another myco process"), "{err}");
308        // drop releases — a new writer takes over, readers now
309        // share, and a third writer fails against the reader
310        drop(w);
311        let r = lock_shared(d.path()).unwrap();
312        let _r2 = lock_shared(d.path()).unwrap();
313        assert!(lock_worktree(d.path()).is_err());
314        drop((r, _r2));
315        assert!(lock_worktree(d.path()).is_ok());
316    }

318    #[test]
319    fn lock_lives_in_the_shared_git_dir() {
320        // a repo with a linked project worktree — the lock file
321        // must land in <root>/.git so `git add -A` can't sweep it
322        // into a ledger commit
323        let d = tempfile::tempdir().unwrap();
324        git(d.path(), &["init", "-q"]);
325        git(d.path(), &["config", "user.email", "t@t"]);
326        git(d.path(), &["config", "user.name", "t"]);
327        myco_git::create_orphan_branch(d.path(), "project", "init").unwrap();
328        git(d.path(), &["worktree", "add", "project", "project"]);
329        // locking from the worktree and from the repo root
330        // contend — same lock file
331        let w = lock_worktree(d.path()).unwrap();
332        assert!(lock_worktree(&d.path().join("project")).is_err());
333        drop(w);
334        assert!(lock_worktree(&d.path().join("project")).is_ok());
335        assert!(d.path().join(".git/myco.lock").exists());
336        assert!(!d.path().join("project/myco.lock").exists());
337        git(d.path(), &["worktree", "remove", "--force", "project"]);
338    }

340    #[test]
341    fn commit_paths_clean_tree_is_noop() {
342        let d = tempfile::tempdir().unwrap();
343        git(d.path(), &["init", "-q"]);
344        git(d.path(), &["config", "user.email", "t@t"]);
345        git(d.path(), &["config", "user.name", "t"]);
346        // enabled but nothing dirty under the path → false, no commit
347        assert!(!commit_paths(d.path(), "msg", &[".gitignore"], true).unwrap());
348        // disabled short-circuits before even checking
349        assert!(!commit_paths(d.path(), "msg", &[".gitignore"], false).unwrap());
350        std::fs::write(d.path().join("other.txt"), "x").unwrap();
351        // dirty elsewhere doesn't count — only the named paths do
352        assert!(!commit_paths(d.path(), "msg", &[".gitignore"], true).unwrap());
353    }
