1//! Shared plumbing for the lifecycle verbs — the `project/`
2//! scaffold and the auto-commit-per-verb machinery.
3//!
4//! Every mutating verb ends in one atomic commit on the `project`
5//! branch (`--no-commit` or `$MYCO_NO_COMMIT` opts out, e.g. for
6//! tests and dry runs). Commits happen only when the worktree
7//! actually changed — a no-op verb leaves no noise in the audit
8//! trail.
9

15/// Dirs `myco init` / `check --fix` scaffold (with `.gitkeep`).
16const SCAFFOLD_DIRS: [&str; 5] = ["issues", "proposals", "adrs", "plans", "handoffs"];
17
18/// Name of the lock file — inside the repo's shared git dir
19/// (`<main>/.git/myco.lock`) so it never lands in a ledger commit
20/// (`git add -A` can't see it) and covers the `project/` worktree
21/// of every checkout of the repo.
22const LOCK_FILE: &str = "myco.lock";
23
24/// RAII guard for a worktree lock — an OS advisory lock (`flock`)
25/// on the repo's lock file. The lock releases when the guard
26/// drops (or the process dies — the OS cleans up).
27#[derive(Debug)]
28pub struct WorktreeLock {
29    _file: std::fs::File,
30}
31
32/// The file the lock lives on: `<git-common-dir>/myco.lock` — the
33/// common dir is shared by a repo's worktrees and exists before
34/// `myco init` creates the worktree, so one path covers init and
35/// every verb. A dir outside any repo falls back to a dotfile
36/// inside itself (nothing can autocommit there anyway, and the
37/// ledger spec ignores dotfiles).
38fn lock_path(dir: &Path) -> Result<PathBuf, String> {
39    match myco_git::common_dir(dir) {
40        Ok(git) => Ok(git.join(LOCK_FILE)),
41        Err(_) => {
42            if dir.is_dir() {
43                Ok(dir.join(".myco.lock"))
44            } else {
45                Err(format!("{} is not a directory", dir.display()))
46            }
47        }
48    }
49}
50
51fn lock_at(dir: &Path, exclusive: bool) -> Result<WorktreeLock, String> {
52    let path = lock_path(dir)?;
53    let file = std::fs::OpenOptions::new()
54        .create(true)
55        .truncate(false)
56        .write(true)
57        .open(&path)
58        .map_err(|e| format!("open {}: {e}", path.display()))?;
59    let taken = if exclusive {
60        file.try_lock()
61    } else {
62        file.try_lock_shared()
63    };
64    match taken {
65        Ok(()) => Ok(WorktreeLock { _file: file }),
66        Err(std::fs::TryLockError::WouldBlock) => {
67            Err("another myco process is writing — retry".into())
68        }
69        Err(std::fs::TryLockError::Error(e)) => Err(format!("lock {}: {e}", path.display())),
70    }
71}

73/// The exclusive mutation lock — every load→mutate→save→commit
74/// verb holds it for the verb's whole life (including the
75/// autocommit). Fail-fast on contention: concurrent writers
76/// corrupt the ledger (dup ids, last-writer-wins), so the second
77/// one gets a retry hint instead.
78///
79/// `dir` is any path inside the repo — the worktree or the repo
80/// root; the lock lands on the shared git dir either way.
81pub fn lock_worktree(dir: &Path) -> Result<WorktreeLock, String> {
82    lock_at(dir, true)
83}
84
85/// The shared read lock — concurrent readers don't block each
86/// other, but a reader contending with a writer fails fast
87/// rather than rendering a mid-verb ledger.
88pub fn lock_shared(dir: &Path) -> Result<WorktreeLock, String> {
89    lock_at(dir, false)
90}

107/// Create every missing scaffold piece in the worktree; existing
108/// files are never overwritten. `root` is the store's worktree
109/// dir (the `.gitkeep` sweep is raw fs — the port has no "list
110/// dir" verb). Returns the rel paths created.
111pub fn scaffold(store: &impl Store, root: &Path, name: &str) -> Result<Vec<String>, StoreError> {
112    let mut created = Vec::new();
113    let missing = |rel: &str| !root.join(rel).exists();
114    if missing("meta.toml") {
115        store.save_meta(&ProjectMeta::new(name))?;
116        created.push("meta.toml".to_string());
117    }
118    if missing("TODO.md") {
119        store.save_todo(&TodoDoc::new(name))?;
120        created.push("TODO.md".to_string());
121    }
122    if missing("BACKLOG.md") {
123        store.save_backlog(&BacklogDoc::default())?;
124        created.push("BACKLOG.md".to_string());
125    }
126    if missing("index.md") {
127        store.save_index(&index_template(name))?;
128        created.push("index.md".to_string());
129    }
130    for dir in SCAFFOLD_DIRS {
131        let keep = root.join(dir).join(".gitkeep");
132        let empty_dir = match std::fs::read_dir(root.join(dir)) {
133            Ok(mut it) => it.next().is_none(),
134            Err(_) => true,
135        };
136        if empty_dir && !keep.exists() {
137            std::fs::create_dir_all(root.join(dir))?;
138            std::fs::write(&keep, "")?;
139            created.push(format!("{dir}/.gitkeep"));
140        }
141    }
142    Ok(created)
143}

145/// Are verb commits enabled? `--no-commit` or `$MYCO_NO_COMMIT`
146/// (any value) suppresses them.
147pub fn commits_enabled(no_commit: bool) -> bool {
148    !no_commit && std::env::var_os("MYCO_NO_COMMIT").is_none()
149}

151/// Fallback actor for the `Actor:` commit trailer — the local
152/// CLI wrote it. Verbs that name an actor (`--actor` on
153/// `handoff`/`interrupt`, `--owner` on `add`/`new`/`set`/`--claim`)
154/// stamp that id instead; the same default those flags already
155/// fall back to.
156pub const DEFAULT_ACTOR: &str = "cli";
157
158/// `msg` + the `Actor:` trailer — every myco commit names the
159/// verb's actor in the message body (T-011: actor-tagged
160/// trailers, not per-actor git authors — author/committer stay
161/// the repo's stable `user.name`/`user.email` identity, or the
162/// `myco@localhost` fallback). `None` resolves to
163/// [`DEFAULT_ACTOR`]. The subject line is untouched, so `%s`
164/// parses (e.g. `report`'s velocity walk) never see it.
165pub fn commit_msg(msg: &str, actor: Option<&Actor>) -> String {
166    let actor = actor.map(Actor::as_str).unwrap_or(DEFAULT_ACTOR);
167    format!("{msg}\n\nActor: {actor}")
168}

170/// Mutating verbs must not write a ledger they cannot commit. A
171/// detached `project/` worktree is the failure mode that costs the
172/// most: every write lands, no branch records it, and the audit
173/// trail is gone the moment the worktree moves. `check` reports and
174/// repairs it; every write path refuses it first.
175pub fn ensure_committable(dir: &Path) -> Result<(), String> {
176    if myco_git::is_detached(dir) == Some(true) {
177        return Err(format!(
178            "{} is on a detached HEAD — refusing to write a ledger whose commits would be \
179             unreachable; run `myco check --fix` to reattach it",
180            dir.display()
181        ));
182    }
183    Ok(())
184}

186/// One atomic commit of everything under `dir` — the
187/// auto-commit-per-verb path. `msg` gains the `Actor:` trailer
188/// ([`commit_msg`]; `actor` is the verb's actor, `None` →
189/// [`DEFAULT_ACTOR`]). No commit when disabled or when the
190/// tree is clean. Returns whether a commit was made.
191pub fn autocommit(
192    dir: &Path,
193    msg: &str,
194    actor: Option<&Actor>,
195    enabled: bool,
196) -> Result<bool, String> {
197    if !enabled || !myco_git::has_changes(dir).map_err(|e| e.to_string())? {
198        return Ok(false);
199    }
200    myco_git::commit_all(dir, &commit_msg(msg, actor))
201        .map_err(|e| format!("git commit failed in {}: {e}", dir.display()))?;
202    Ok(true)
203}

205/// Commit only `paths` inside `dir` — used for repo-root edits
206/// (`.gitignore`, `pm/` removal) so an unrelated dirty tree isn't
207/// swept into the verb's commit. Returns whether a commit was made.
208///
209/// Scoping is per *file*, not per hunk: callers must pass only
210/// paths myco itself wrote this invocation. If the user had
211/// pre-existing uncommitted edits inside one of those files, they
212/// land in the commit under myco's message — an accepted
213/// limitation (hunk-level staging via `git apply --cached` was
214/// judged disproportionate for the managed-file call sites).
215pub fn commit_paths(
216    dir: &Path,
217    msg: &str,
218    paths: &[&str],
219    actor: Option<&Actor>,
220    enabled: bool,
221) -> Result<bool, String> {
222    if !enabled {
223        return Ok(false);
224    }
225    let mut args = vec!["status", "--porcelain", "--"];
226    args.extend_from_slice(paths);
227    let dirty = myco_git::git_out(dir, &args).map_err(|e| e.to_string())?;
228    if dirty.is_empty() {
229        return Ok(false);
230    }
231    let mut add = vec!["add", "--"];
232    add.extend_from_slice(paths);
233    myco_git::git_out(dir, &add).map_err(|e| e.to_string())?;
234    let msg = commit_msg(msg, actor);
235    let mut commit = vec!["commit", "-m", msg.as_str(), "--"];
236    commit.extend_from_slice(paths);
237    myco_git::git_ident(dir, &commit)
238        .map_err(|e| e.to_string())
239        .map(|_| true)
240}

250/// Default pager — `-F` quits instantly under one screen, `-R`
251/// keeps ANSI colors, `-S` chops instead of wrapping rows, `-X`
252/// leaves output in the scrollback.
253const DEFAULT_PAGER: &str = "less -FRSX";
254
255/// Pager program: `$MYCO_PAGER` → `$PAGER` → [`DEFAULT_PAGER`].
256/// An explicitly empty value disables paging.
257fn pager_program() -> Option<String> {
258    for var in ["MYCO_PAGER", "PAGER"] {
259        if let Ok(v) = std::env::var(var) {
260            return if v.trim().is_empty() { None } else { Some(v) };
261        }
262    }
263    Some(DEFAULT_PAGER.to_string())
264}
265
266/// `sh -c <pager>` with `text` on its stdin — the pager inherits
267/// our stdout/stderr, so scrolling and colors work as usual.
268/// `$MYCO_PAGER`/`$PAGER` are executed verbatim (git-parity
269/// behavior) — an env var is a trusted setting, never user data.
270fn spawn_pager(pager: &str, text: &str) -> std::io::Result<()> {
271    use std::io::Write;
272    let mut child = std::process::Command::new("sh")
273        .args(["-c", pager])
274        .stdin(std::process::Stdio::piped())
275        .spawn()?;
276    if let Some(mut stdin) = child.stdin.take() {
277        // quitting early (`q`) closes the pipe — with SIGPIPE at
278        // its default disposition we die like any unix tool, which
279        // is what the user expects; ignore the write error for the
280        // SIG_IGN case
281        let _ = stdin.write_all(text.as_bytes());
282    }
283    child.wait().map(|_| ())
284}
285
286/// Print `text`, paging it through the user's pager when stdout is
287/// a tty — `less -F` exits instantly for one-screen output, so
288/// callers can page unconditionally. Non-tty output, a disabled or
289/// unspawnable pager all fall back to `println!`.
290pub fn print_paged(text: &str) {
291    use std::io::IsTerminal;
292    if std::io::stdout().is_terminal() {
293        if let Some(pager) = pager_program() {
294            if spawn_pager(&pager, text).is_ok() {
295                return;
296            }
297        }
298    }
299    println!("{text}");
300}

310        temp_env::with_vars(
311            [("MYCO_PAGER", None::<&str>), ("PAGER", None::<&str>)],

339    #[test]
340    fn lock_serializes_writers_and_reads() {
341        let d = tempfile::tempdir().unwrap();
342        // first writer takes it; a second writer and a reader
343        // both fail fast with the retry hint
344        let w = lock_worktree(d.path()).unwrap();
345        let err = lock_worktree(d.path()).unwrap_err();
346        assert!(err.contains("another myco process is writing"), "{err}");
347        let err = lock_shared(d.path()).unwrap_err();
348        assert!(err.contains("another myco process"), "{err}");
349        // drop releases — a new writer takes over, readers now
350        // share, and a third writer fails against the reader
351        drop(w);
352        let r = lock_shared(d.path()).unwrap();
353        let _r2 = lock_shared(d.path()).unwrap();
354        assert!(lock_worktree(d.path()).is_err());
355        drop((r, _r2));
356        assert!(lock_worktree(d.path()).is_ok());
357    }

359    #[test]
360    fn lock_lives_in_the_shared_git_dir() {
361        // a repo with a linked project worktree — the lock file
362        // must land in <root>/.git so `git add -A` can't sweep it
363        // into a ledger commit
364        let d = tempfile::tempdir().unwrap();
365        git(d.path(), &["init", "-q"]);
366        git(d.path(), &["config", "user.email", "t@t"]);
367        git(d.path(), &["config", "user.name", "t"]);
368        myco_git::create_orphan_branch(d.path(), "project", "init").unwrap();
369        git(d.path(), &["worktree", "add", "project", "project"]);
370        // locking from the worktree and from the repo root
371        // contend — same lock file
372        let w = lock_worktree(d.path()).unwrap();
373        assert!(lock_worktree(&d.path().join("project")).is_err());
374        drop(w);
375        assert!(lock_worktree(&d.path().join("project")).is_ok());
376        assert!(d.path().join(".git/myco.lock").exists());
377        assert!(!d.path().join("project/myco.lock").exists());
378        git(d.path(), &["worktree", "remove", "--force", "project"]);
379    }

445                .unwrap()
