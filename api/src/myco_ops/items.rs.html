28/// Load both ledger docs through the store.
29pub fn load_ledger(store: &impl Store) -> Result<Ledger, String> {
30    Ok(Ledger::new(
31        store.todo().map_err(|e| e.to_string())?,
32        store.backlog().map_err(|e| e.to_string())?,
33    ))
34}

36/// Write both ledger docs back through the store.
37pub fn save_ledger(store: &impl Store, ledger: &Ledger) -> Result<(), String> {
38    store.save_todo(&ledger.todo).map_err(|e| e.to_string())?;
39    store
40        .save_backlog(&ledger.backlog)
41        .map_err(|e| e.to_string())
42}

44/// Load a file-backed item for mutation. Archived items are
45/// refused — `save_item` writes to the live dir, so editing one
46/// would silently un-archive it.
47pub fn item_for_edit(store: &impl Store, id: ItemId) -> Result<ItemFile, String> {
48    if store.is_archived(id).map_err(|e| e.to_string())? {
49        return Err(format!("{id} is archived — mutating it is refused"));
50    }
51    store.item(id).map_err(|e| e.to_string())
52}

59/// A store handle holding the worktree lock — exclusive from
60/// [`open_with`] (mutating verbs), shared from [`open_read_with`].
61/// The lock releases when this drops. Derefs to the inner store,
62/// so `&locked` coerces to `&S` at every call site. Generic over
63/// the store — the front-end (composition root) names the impl.
64pub struct LockedStore<S> {
65    store: S,
66    _lock: Option<ops::WorktreeLock>,
67}
68
69impl<S> LockedStore<S> {
70    /// A store holding no lock — for tests and callers that
71    /// guarantee a single writer by construction.
72    pub fn unlocked(store: S) -> Self {
73        LockedStore { store, _lock: None }
74    }
75}
76
77impl<S> std::ops::Deref for LockedStore<S> {
78    type Target = S;
79
80    fn deref(&self) -> &S {
81        &self.store
82    }
83}

158/// Resolve the project + open the store `mk` builds under the
159/// exclusive mutation lock — every verb that writes the ledger
160/// takes this path. `mk` gets the discovered worktree and names
161/// the concrete impl (`myco`'s `ops::store_at` → `MdStore`), so
162/// this crate stays generic over [`Store`]; `cfg` is the same
163/// story for config (discovery reads `project_dir`). The lock
164/// spans discovery → mutation → the per-verb autocommit;
165/// contention fails fast (`another myco process is writing —
166/// retry`) instead of corrupting the ledger.
167pub fn open_with<S: Store>(
168    dir: Option<&Path>,
169    root: Option<&Path>,
170    cfg: crate::config::ConfigLoader,
171    mk: impl FnOnce(&Path) -> Result<S, String>,
172) -> Result<(discover::Project, LockedStore<S>), String> {
173    let proj = discover::discover(dir, root, cfg)?;
174    let lock = ops::lock_worktree(&proj.worktree)?;
175    let store = mk(&proj.worktree)?;
176    Ok((
177        proj,
178        LockedStore {
179            store,
180            _lock: Some(lock),
181        },
182    ))
183}

185/// [`open_with`] under the shared read lock — read-only verbs
186/// (`status`, `view`, `list`, `resume`, …). Readers don't block
187/// each other; a reader meeting a live writer fails fast.
188pub fn open_read_with<S: Store>(
189    dir: Option<&Path>,
190    root: Option<&Path>,
191    cfg: crate::config::ConfigLoader,
192    mk: impl FnOnce(&Path) -> Result<S, String>,
193) -> Result<(discover::Project, LockedStore<S>), String> {
194    let proj = discover::discover(dir, root, cfg)?;
195    let lock = ops::lock_shared(&proj.worktree)?;
196    let store = mk(&proj.worktree)?;
197    Ok((
198        proj,
199        LockedStore {
200            store,
201            _lock: Some(lock),
202        },
203    ))
204}

211/// Standard mutation output: `--json` gets `payload` plus a
212/// `committed` key; human output is one line.
213pub fn report(
214    mut payload: serde_json::Value,
215    committed: bool,
216    json_out: bool,
217    human: &str,
218) -> Result<i32, String> {
219    if json_out {

228/// `--status/--owner/--label` list filter + `--limit` — the parsed
229/// form of the CLI's `ListFlags`, usable by any front-end.
230#[derive(Debug, Clone, Default)]
231pub struct Filter {
232    /// Keep only items with this status.
233    pub status: Option<Status>,
234    /// Keep only items owned by this actor string.
235    pub owner: Option<String>,
236    /// Keep only items carrying all of these labels.
237    pub labels: Vec<String>,
238    /// Keep only the first N items after filtering.
239    pub limit: Option<usize>,
240}
241
242impl Filter {
243    /// Does `item` pass the filters?
244    pub fn matches(&self, item: &Item) -> bool {
245        self.status.is_none_or(|s| item.status == s)
246            && self
247                .owner
248                .as_deref()
249                .is_none_or(|o| item.owner.as_ref().map(myco_core::Actor::as_str) == Some(o))
250            && self.labels.iter().all(|l| item.labels.contains(l))
251    }
252
253    /// `limit` as a `.take()` bound.
254    pub fn take(&self) -> usize {
255        self.limit.unwrap_or(usize::MAX)
256    }
257}

259/// `--json` → a bare array of item objects; else compact lines.
260pub fn print_list(items: &[Item], json_out: bool) -> Result<i32, String> {
261    if json_out {
262        let v: Vec<serde_json::Value> = items
263            .iter()
264            .map(|i| serde_json::to_value(i).unwrap_or(json!(null)))
265            .collect();
266        ops::print_json(&json!(v));
267    } else {
268        for i in items {
269            println!("{}", compact(i));
270        }
271    }
272    Ok(0)
273}

275/// Like [`print_list`] for depth-tagged items — human output
276/// indents children 2 spaces per level under their parent.
277pub fn print_tree(pairs: &[(usize, &Item)], json_out: bool) -> Result<i32, String> {
278    if json_out {
279        let v: Vec<serde_json::Value> = pairs
280            .iter()
281            .map(|(_, i)| serde_json::to_value(i).unwrap_or(json!(null)))
282            .collect();
283        ops::print_json(&json!(v));
284    } else {
285        for (depth, i) in pairs {
286            for _ in 0..*depth {
287                print!("  ");
288            }
289            println!("{}", compact(i));
290        }
291    }
292    Ok(0)
293}

295/// Input bundle for [`new_file`] — one per `<kind> new` call.
296/// `word` is the commit-message noun (`chore: new issue I-001`).
297pub struct NewItem<'a> {
298    /// The kind of item to create.
299    pub kind: ItemKind,
300    /// One-line title (drives the filename slug).
301    pub title: String,
302    /// Body markdown — `None` → `template` fills in.
303    pub body: Option<&'a str>,
304    /// Optional metadata (`pri:`/`owner:`/`#label`/…).
305    pub meta: &'a NewItemMeta,
306    /// The kind's default body template.
307    pub template: fn(&Item) -> String,
308    /// Commit-message noun — `chore: new <word> <id>`.
309    pub word: &'a str,
310    /// `--json` output.
311    pub json_out: bool,
312    /// `--no-commit`.
313    pub no_commit: bool,
314}
315
316/// The shared `<kind> new` body — alloc id → apply meta → body or
317/// `template` → `save_item` → commit + report.
318pub fn new_file(
319    proj: &discover::Project,
320    store: &impl Store,
321    spec: &NewItem<'_>,
322) -> Result<i32, String> {
323    let id = store.alloc_id(spec.kind).map_err(|e| e.to_string())?;
324    let mut item = Item::new(id, &spec.title).created(today());
325    spec.meta.apply(&mut item)?;
326    let body = spec
327        .body
328        .map(str::to_string)
329        .unwrap_or_else(|| (spec.template)(&item));
330    let file = ItemFile::new(item.clone()).body(body);
331    store.save_item(&file).map_err(|e| e.to_string())?;
332    let rel = item_rel_path(store, &item)?;
333    let committed = commit(
334        &proj.worktree,
335        &format!("chore: new {} {id}", spec.word),
336        spec.no_commit,
337    )?;
338    report(
339        json!({"item": item, "path": rel}),
340        committed,
341        spec.json_out,
342        &format!("created {id} {rel}"),
343    )
344}

346/// The read behind [`list_files`] — live item files (plus
347/// `archive/` entries when `include_archived`), filtered, as
348/// `(item, archived)` pairs. Non-CLI front-ends (`myco serve`)
349/// shape it themselves — [`marked_json`] is the list shape.
350pub fn query_files(
351    store: &impl Store,
352    kind: ItemKind,
353    include_archived: bool,
354    filter: &Filter,
355) -> Result<Vec<(Item, bool)>, String> {
356    let live = store.list_items(kind).map_err(|e| e.to_string())?;
357    let mut found: Vec<(Item, bool)> = live.into_iter().map(|f| (f.meta, false)).collect();
358    if include_archived {
359        found.extend(
360            store
361                .list_archived(kind)
362                .map_err(|e| e.to_string())?
363                .into_iter()
364                .map(|f| (f.meta, true)),
365        );
366    }
367    Ok(found
368        .into_iter()
369        .filter(|(i, _)| filter.matches(i))
370        .take(filter.take())
371        .collect())
372}

374/// `(item, archived)` pairs → a bare array of item objects, each
375/// tagged `"archived": bool` — the `--json` list shape. Live and
376/// archived items share one array; the flag is the marker.
377pub fn marked_json(items: &[(Item, bool)]) -> serde_json::Value {
378    json!(items
379        .iter()
380        .map(|(i, archived)| {
381            let mut v = serde_json::to_value(i).unwrap_or(json!(null));
382            if let Some(obj) = v.as_object_mut() {
383                obj.insert("archived".into(), json!(archived));
384            }
385            v
386        })
387        .collect::<Vec<_>>())
388}

390/// The shared `<kind> list` — live item files, plus `archive/`
391/// entries when `include_archived`.
392pub fn list_files(
393    store: &impl Store,
394    kind: ItemKind,
395    include_archived: bool,
396    filter: &Filter,
397    json_out: bool,
398) -> Result<i32, String> {
399    let found = query_files(store, kind, include_archived, filter)?;
400    if json_out {
401        ops::print_json(&marked_json(&found));
402        return Ok(0);
403    }
404    let items: Vec<Item> = found.into_iter().map(|(i, _)| i).collect();
405    print_list(&items, false)
406}

408/// `<dir>/<filename>` for a file-backed item — the store impl's
409/// own naming grammar ([`Store::item_path`], archive scope
410/// included), falling back to the bare id for kinds with no file
411/// form.
412pub fn item_rel_path(store: &impl Store, item: &Item) -> Result<String, String> {
413    Ok(store
414        .item_path(item)
415        .map_err(|e| e.to_string())?
416        .unwrap_or_else(|| item.id.to_string()))
417}

419/// Locate an item anywhere in the ledger — TODO.md, BACKLOG.md, or
420/// an item file (live or `archive/`). Returns the item, its
421/// location label, and the body for file-backed items.
422pub fn find_item(
423    store: &impl Store,
424    id: ItemId,
425) -> Result<Option<(Item, String, Option<String>)>, String> {
426    let ledger = load_ledger(store)?;
427    if let Some(loc) = ledger.locate(id) {
428        let file = if loc.backlog { "BACKLOG.md" } else { "TODO.md" };
429        if let Some(item) = ledger.item(&loc) {
430            return Ok(Some((item.clone(), file.to_string(), None)));
431        }
432    }
433    match store.item(id) {
434        Ok(f) => {
435            let loc = item_rel_path(store, &f.meta)?;
436            Ok(Some((f.meta.clone(), loc, Some(f.body))))
437        }
438        Err(myco_store::Error::NotFound { .. }) => Ok(None),
439        Err(e) => Err(e.to_string()),
440    }
441}

501        let (_p, store) = open_with(

526    #[test]
527    fn concurrent_open_fails_fast_no_dup_ids() {
528        let (_d, wt) = project_dir();
529        let (_p, s1) = open_with(
530            Some(wt.as_path()),
531            None,
532            crate::config::test_loader(),
533            md_store,
534        )
535        .unwrap();
536        let id1 = s1.alloc_id(ItemKind::Task).unwrap();
537        // a racing verb can't even open — prove the failure is
538        // the lock, not the store, by retrying after the drop
539        let wt2 = wt.clone();
540        let (tx, rx) = std::sync::mpsc::channel();
541        let racing = std::thread::spawn(move || {
542            tx.send(
543                open_with(
544                    Some(wt2.as_path()),
545                    None,
546                    crate::config::test_loader(),
547                    md_store,
548                )
549                .map(|_| ()),
550            )
551            .unwrap();
552        });
553        let err = rx.recv().unwrap().unwrap_err();
554        assert!(err.contains("another myco process"), "{err}");
555        drop(s1);
556        // after release, a fresh verb allocates the *next* id —
557        // the lock is what kept two alloc_id RMWs from both
558        // reading "next = id1" and handing out a duplicate
559        let (_p, s2) = open_with(
560            Some(wt.as_path()),
561            None,
562            crate::config::test_loader(),
563            md_store,
564        )
565        .unwrap();
566        let id2 = s2.alloc_id(ItemKind::Task).unwrap();
567        assert_ne!(id1, id2);
568        racing.join().unwrap();
569    }
