29/// Load both ledger docs through the store.
30pub fn load_ledger(store: &MdStore) -> Result<Ledger, String> {
31    Ok(Ledger::new(
32        store.todo().map_err(|e| e.to_string())?,
33        store.backlog().map_err(|e| e.to_string())?,
34    ))
35}

37/// Write both ledger docs back through the store.
38pub fn save_ledger(store: &MdStore, ledger: &Ledger) -> Result<(), String> {
39    store.save_todo(&ledger.todo).map_err(|e| e.to_string())?;
40    store
41        .save_backlog(&ledger.backlog)
42        .map_err(|e| e.to_string())
43}

45/// Load a file-backed item for mutation. Archived items are
46/// refused — `save_item` writes to the live dir, so editing one
47/// would silently un-archive it.
48pub fn item_for_edit(store: &MdStore, id: ItemId) -> Result<ItemFile, String> {
49    if store.is_archived(id).map_err(|e| e.to_string())? {
50        return Err(format!("{id} is archived — mutating it is refused"));
51    }
52    store.item(id).map_err(|e| e.to_string())
53}

60/// A store handle holding the worktree lock — exclusive from
61/// [`open`] (mutating verbs), shared from [`open_read`]. The lock
62/// releases when this drops. Derefs to [`MdStore`], so `&locked`
63/// coerces to `&MdStore` at every call site.
64pub struct LockedStore {
65    store: MdStore,
66    _lock: Option<ops::WorktreeLock>,
67}
68
69impl LockedStore {
70    /// A store holding no lock — for tests and callers that
71    /// guarantee a single writer by construction.
72    pub fn unlocked(store: MdStore) -> Self {
73        LockedStore { store, _lock: None }
74    }
75}
76
77impl std::ops::Deref for LockedStore {
78    type Target = MdStore;
79
80    fn deref(&self) -> &MdStore {
81        &self.store
82    }
83}

155/// Resolve the project + open the store under the exclusive
156/// mutation lock — every verb that writes the ledger takes this
157/// path. The lock spans discovery → mutation → the per-verb
158/// autocommit; contention fails fast (`another myco process is
159/// writing — retry`) instead of corrupting the ledger.
160pub fn open(
161    dir: Option<&Path>,
162    root: Option<&Path>,
163) -> Result<(discover::Project, LockedStore), String> {
164    let proj = discover::discover(dir, root)?;
165    let lock = ops::lock_worktree(&proj.worktree)?;
166    let store = ops::store_at(&proj.worktree)?;
167    Ok((
168        proj,
169        LockedStore {
170            store,
171            _lock: Some(lock),
172        },
173    ))
174}
175
176/// Resolve + open under the shared read lock — read-only verbs
177/// (`status`, `board`, `list`, `resume`, …). Readers don't block
178/// each other; a reader meeting a live writer fails fast.
179pub fn open_read(
180    dir: Option<&Path>,
181    root: Option<&Path>,
182) -> Result<(discover::Project, LockedStore), String> {
183    let proj = discover::discover(dir, root)?;
184    let lock = ops::lock_shared(&proj.worktree)?;
185    let store = ops::store_at(&proj.worktree)?;
186    Ok((
187        proj,
188        LockedStore {
189            store,
190            _lock: Some(lock),
191        },
192    ))
193}

200/// Standard mutation output: `--json` gets `payload` plus a
201/// `committed` key; human output is one line.
202pub fn report(
203    mut payload: serde_json::Value,
204    committed: bool,
205    json_out: bool,
206    human: &str,
207) -> Result<i32, String> {
208    if json_out {
209        payload["committed"] = json!(committed);
210        ops::print_json(&payload);
211    } else {
212        println!("{human}");
213    }
214    Ok(0)
215}

217/// `--status/--owner/--label` list filter + `--limit` — the parsed
218/// form of the CLI's `ListFlags`, usable by any front-end.
219#[derive(Debug, Clone, Default)]
220pub struct Filter {
221    /// Keep only items with this status.
222    pub status: Option<Status>,
223    /// Keep only items owned by this actor string.
224    pub owner: Option<String>,
225    /// Keep only items carrying all of these labels.
226    pub labels: Vec<String>,
227    /// Keep only the first N items after filtering.
228    pub limit: Option<usize>,
229}
230
231impl Filter {
232    /// Does `item` pass the filters?
233    pub fn matches(&self, item: &Item) -> bool {
234        self.status.is_none_or(|s| item.status == s)
235            && self
236                .owner
237                .as_deref()
238                .is_none_or(|o| item.owner.as_ref().map(myco_core::Actor::as_str) == Some(o))
239            && self.labels.iter().all(|l| item.labels.contains(l))
240    }
241
242    /// `limit` as a `.take()` bound.
243    pub fn take(&self) -> usize {
244        self.limit.unwrap_or(usize::MAX)
245    }
246}

248/// `--json` → a bare array of item objects; else compact lines.
249pub fn print_list(items: &[Item], json_out: bool) -> Result<i32, String> {
250    if json_out {
251        let v: Vec<serde_json::Value> = items
252            .iter()
253            .map(|i| serde_json::to_value(i).unwrap_or(json!(null)))
254            .collect();
255        ops::print_json(&json!(v));
256    } else {
257        for i in items {
258            println!("{}", compact(i));
259        }
260    }
261    Ok(0)
262}
263
264/// Like [`print_list`] for depth-tagged items — human output
265/// indents children 2 spaces per level under their parent.
266pub fn print_tree(pairs: &[(usize, &Item)], json_out: bool) -> Result<i32, String> {
267    if json_out {
268        let v: Vec<serde_json::Value> = pairs
269            .iter()
270            .map(|(_, i)| serde_json::to_value(i).unwrap_or(json!(null)))
271            .collect();
272        ops::print_json(&json!(v));
273    } else {
274        for (depth, i) in pairs {
275            for _ in 0..*depth {
276                print!("  ");
277            }
278            println!("{}", compact(i));
279        }
280    }
281    Ok(0)
282}

284/// Input bundle for [`new_file`] — one per `<kind> new` call.
285/// `word` is the commit-message noun (`chore: new issue I-001`).
286pub struct NewItem<'a> {
287    /// The kind of item to create.
288    pub kind: ItemKind,
289    /// One-line title (drives the filename slug).
290    pub title: String,
291    /// Body markdown — `None` → `template` fills in.
292    pub body: Option<&'a str>,
293    /// Optional metadata (`pri:`/`owner:`/`#label`/…).
294    pub meta: &'a NewItemMeta,
295    /// The kind's default body template.
296    pub template: fn(&Item) -> String,
297    /// Commit-message noun — `chore: new <word> <id>`.
298    pub word: &'a str,
299    /// `--json` output.
300    pub json_out: bool,
301    /// `--no-commit`.
302    pub no_commit: bool,
303}
304
305/// The shared `<kind> new` body — alloc id → apply meta → body or
306/// `template` → `save_item` → commit + report.
307pub fn new_file(
308    proj: &discover::Project,
309    store: &MdStore,
310    spec: &NewItem<'_>,
311) -> Result<i32, String> {
312    let id = store.alloc_id(spec.kind).map_err(|e| e.to_string())?;
313    let mut item = Item::new(id, &spec.title).created(today());
314    spec.meta.apply(&mut item)?;
315    let body = spec
316        .body
317        .map(str::to_string)
318        .unwrap_or_else(|| (spec.template)(&item));
319    let file = ItemFile::new(item.clone()).body(body);
320    store.save_item(&file).map_err(|e| e.to_string())?;
321    let rel = item_rel_path(&item);
322    let committed = commit(
323        &proj.worktree,
324        &format!("chore: new {} {id}", spec.word),
325        spec.no_commit,
326    )?;
327    report(
328        json!({"item": item, "path": rel}),
329        committed,
330        spec.json_out,
331        &format!("created {id} {rel}"),
332    )
333}

335/// The read behind [`list_files`] — live item files (plus
336/// `archive/` entries when `include_archived`), filtered, as
337/// `(item, archived)` pairs. Non-CLI front-ends (`myco serve`)
338/// shape it themselves — [`marked_json`] is the list shape.
339pub fn query_files(
340    store: &MdStore,
341    kind: ItemKind,
342    include_archived: bool,
343    filter: &Filter,
344) -> Result<Vec<(Item, bool)>, String> {
345    let live = store.list_items(kind).map_err(|e| e.to_string())?;
346    let mut found: Vec<(Item, bool)> = live.into_iter().map(|f| (f.meta, false)).collect();
347    if include_archived {
348        found.extend(
349            store
350                .list_archived(kind)
351                .map_err(|e| e.to_string())?
352                .into_iter()
353                .map(|f| (f.meta, true)),
354        );
355    }
356    Ok(found
357        .into_iter()
358        .filter(|(i, _)| filter.matches(i))
359        .take(filter.take())
360        .collect())
361}

363/// `(item, archived)` pairs → a bare array of item objects, each
364/// tagged `"archived": bool` — the `--json` list shape. Live and
365/// archived items share one array; the flag is the marker.
366pub fn marked_json(items: &[(Item, bool)]) -> serde_json::Value {
367    json!(items
368        .iter()
369        .map(|(i, archived)| {
370            let mut v = serde_json::to_value(i).unwrap_or(json!(null));
371            if let Some(obj) = v.as_object_mut() {
372                obj.insert("archived".into(), json!(archived));
373            }
374            v
375        })
376        .collect::<Vec<_>>())
377}

379/// The shared `<kind> list` — live item files, plus `archive/`
380/// entries when `include_archived`.
381pub fn list_files(
382    store: &MdStore,
383    kind: ItemKind,
384    include_archived: bool,
385    filter: &Filter,
386    json_out: bool,
387) -> Result<i32, String> {
388    let found = query_files(store, kind, include_archived, filter)?;
389    if json_out {
390        ops::print_json(&marked_json(&found));
391        return Ok(0);
392    }
393    let items: Vec<Item> = found.into_iter().map(|(i, _)| i).collect();
394    print_list(&items, false)
395}

397/// `<dir>/<filename>` for a file-backed item.
398pub fn item_rel_path(item: &Item) -> String {
399    let dir = myco_store_md::item_dir(item.id.kind()).unwrap_or("");
400    let name =
401        myco_store_md::item_filename(&item.id, &item.title).unwrap_or_else(|| item.id.to_string());
402    format!("{dir}/{name}")
403}

405/// Locate an item anywhere in the ledger — TODO.md, BACKLOG.md, or
406/// an item file (live or `archive/`). Returns the item, its
407/// location label, and the body for file-backed items.
408pub fn find_item(
409    store: &MdStore,
410    id: ItemId,
411) -> Result<Option<(Item, String, Option<String>)>, String> {
412    let ledger = load_ledger(store)?;
413    if let Some(loc) = ledger.locate(id) {
414        let file = if loc.backlog { "BACKLOG.md" } else { "TODO.md" };
415        if let Some(item) = ledger.item(&loc) {
416            return Ok(Some((item.clone(), file.to_string(), None)));
417        }
418    }
419    match store.item(id) {
420        Ok(f) => {
421            let mut loc = item_rel_path(&f.meta);
422            if store.is_archived(id).map_err(|e| e.to_string())? {
423                if let Some((d, n)) = loc.split_once('/') {
424                    loc = format!("{d}/archive/{n}");
425                }
426            }
427            Ok(Some((f.meta.clone(), loc, Some(f.body))))
428        }
429        Err(myco_store::Error::NotFound { .. }) => Ok(None),
430        Err(e) => Err(e.to_string()),
431    }
432}

440    /// A plain `project/` dir that qualifies via `TODO.md` — the
441    /// no-repo fallback lock path.
442    fn project_dir() -> (tempfile::TempDir, std::path::PathBuf) {
443        let dir = tempfile::tempdir().unwrap();
444        let wt = dir.path().join("project");
445        std::fs::create_dir_all(&wt).unwrap();
446        std::fs::write(wt.join("TODO.md"), "# TODO — x\n").unwrap();
447        (dir, wt)

450    #[test]
451    fn open_serializes_mutating_verbs() {
452        let (_d, wt) = project_dir();
453        let (_p, store) = open(Some(wt.as_path()), None).unwrap();
454        // a second writer (even in-process — the lock is per
455        // open-file-description) fails fast instead of racing
456        // the load→mutate→save
457        let err = open(Some(wt.as_path()), None).err().unwrap();
458        assert!(err.contains("another myco process is writing"), "{err}");
459        // readers fail fast while a writer holds the lock…
460        assert!(open_read(Some(wt.as_path()), None).is_err());
461        drop(store);
462        // …and everyone succeeds once it drops — a fresh writer,
463        // then concurrent readers
464        let (_p, store) = open(Some(wt.as_path()), None).unwrap();
465        drop(store);
466        let (_p2, r2) = open_read(Some(wt.as_path()), None).unwrap();
467        let (_p3, r3) = open_read(Some(wt.as_path()), None).unwrap();
468        drop((r2, r3));
469    }

471    #[test]
472    fn concurrent_open_fails_fast_no_dup_ids() {
473        let (_d, wt) = project_dir();
474        let (_p, s1) = open(Some(wt.as_path()), None).unwrap();
475        let id1 = s1.alloc_id(ItemKind::Task).unwrap();
476        // a racing verb can't even open — prove the failure is
477        // the lock, not the store, by retrying after the drop
478        let wt2 = wt.clone();
479        let (tx, rx) = std::sync::mpsc::channel();
480        let racing = std::thread::spawn(move || {
481            tx.send(open(Some(wt2.as_path()), None).map(|_| ()))
482                .unwrap();
483        });
484        let err = rx.recv().unwrap().unwrap_err();
485        assert!(err.contains("another myco process"), "{err}");
486        drop(s1);
487        // after release, a fresh verb allocates the *next* id —
488        // the lock is what kept two alloc_id RMWs from both
489        // reading "next = id1" and handing out a duplicate
490        let (_p, s2) = open(Some(wt.as_path()), None).unwrap();
491        let id2 = s2.alloc_id(ItemKind::Task).unwrap();
492        assert_ne!(id1, id2);
493        racing.join().unwrap();
494    }
