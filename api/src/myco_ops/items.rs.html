17/// `--pri/--owner/--label/--blocks/--epic/--milestone` — the
18/// metadata flags shared by every `add`/`new` verb.
19#[derive(Debug, Default, clap::Args)]
20pub struct MetaFlags {
21    /// Priority: low|med|high|crit.
22    #[arg(long, visible_alias = "priority")]
23    pub pri: Option<Priority>,
24    /// Owner actor id (`devin`, `human:luis`, …).
25    #[arg(long)]
26    pub owner: Option<Actor>,
27    /// `#label` — repeatable.
28    #[arg(long)]
29    pub label: Vec<String>,
30    /// Ids this item blocks — comma-separated or repeatable.
31    #[arg(long, value_delimiter = ',')]
32    pub blocks: Vec<ItemId>,
33    /// Epic name — the delivery grouping this item belongs to
34    /// (`[A-Za-z0-9_.-]+`).
35    #[arg(long)]
36    pub epic: Option<String>,
37    /// Milestone name — the target this item ships under
38    /// (`[A-Za-z0-9_.-]+`).
39    #[arg(long)]
40    pub milestone: Option<String>,
41}
42
43impl MetaFlags {
44    /// Apply the flags to a freshly-created item.
45    pub fn apply(&self, item: &mut Item) -> Result<(), String> {
46        if let Some(p) = self.pri {
47            item.priority = p;
48        }
49        if let Some(o) = &self.owner {
50            item.owner = Some(o.clone());
51        }
52        for l in &self.label {
53            if !myco_core::is_valid_label(l) {
54                return Err(format!("invalid label {l:?}"));
55            }
56            item.labels.insert(l.clone());
57        }
58        for b in &self.blocks {
59            item.blocks.insert(*b);
60        }
61        for (what, name) in [("epic", &self.epic), ("milestone", &self.milestone)] {
62            if let Some(name) = name {
63                if !myco_core::is_valid_group_name(name) {
64                    return Err(format!("invalid {what} name {name:?}"));
65                }
66            }
67        }
68        item.epic = self.epic.clone();
69        item.milestone = self.milestone.clone();
70        Ok(())
71    }
72}

74/// `--status/--owner/--label` list filters, shared by every `list`.
75#[derive(Debug, Default, clap::Args)]
76pub struct ListFlags {
77    /// Keep only items with this status.
78    #[arg(long)]
79    pub status: Option<Status>,
80    /// Keep only items owned by this actor.
81    #[arg(long)]
82    pub owner: Option<String>,
83    /// Keep only items carrying all of these labels (repeatable).
84    #[arg(long)]
85    pub label: Vec<String>,
86    /// Keep only the first N items after filtering.
87    #[arg(long, value_name = "N")]
88    pub limit: Option<usize>,
89}
90
91impl ListFlags {
92    /// Does `item` pass the filters?
93    pub fn matches(&self, item: &Item) -> bool {
94        self.status.is_none_or(|s| item.status == s)
95            && self
96                .owner
97                .as_deref()
98                .is_none_or(|o| item.owner.as_ref().map(Actor::as_str) == Some(o))
99            && self.label.iter().all(|l| item.labels.contains(l))
100    }
101
102    /// `--limit N` as a `.take()` bound.
103    pub fn take(&self) -> usize {
104        self.limit.unwrap_or(usize::MAX)
105    }
106}

108/// A body-source flag pair — `--body FILE|-` reads a file or stdin,
109/// `--text` is inline. Mutually exclusive.
110#[derive(Debug, Default, clap::Args)]
111pub struct BodyFlags {
112    /// Body markdown from a file, or `-` for stdin.
113    #[arg(long, conflicts_with = "text", value_name = "FILE|-")]
114    pub body: Option<String>,
115    /// Body markdown inline.
116    #[arg(long)]
117    pub text: Option<String>,
118}
119
120impl BodyFlags {
121    /// The supplied body, or `None` → the caller uses the template.
122    pub fn read(&self) -> Result<Option<String>, String> {
123        if let Some(t) = &self.text {
124            return Ok(Some(t.clone()));
125        }
126        match self.body.as_deref() {
127            None => Ok(None),
128            Some("-") => {
129                use std::io::Read;
130                let mut s = String::new();
131                std::io::stdin()
132                    .read_to_string(&mut s)
133                    .map_err(|e| format!("read --body from stdin: {e}"))?;
134                Ok(Some(s))
135            }
136            Some(path) => std::fs::read_to_string(path)
137                .map(Some)
138                .map_err(|e| format!("read --body {path}: {e}")),
139        }
140    }
141}

143/// TODO.md + BACKLOG.md loaded together — task items move between
144/// them on status changes (`deferred` parks, `done` deletes).
145pub struct Ledger {
146    /// `TODO.md`.
147    pub todo: TodoDoc,
148    /// `BACKLOG.md`.
149    pub backlog: BacklogDoc,
150}

152/// Which doc a task lives in — and its position inside. `path` is
153/// the index trail into the section's item tree: `[2]` is the
154/// third top-level item, `[2,0]` its first child.
155#[derive(Debug, Clone, PartialEq, Eq)]
156pub struct TaskLoc {
157    /// `true` = BACKLOG.md, `false` = TODO.md.
158    pub backlog: bool,
159    /// Section index.
160    pub section: usize,
161    /// Index path within the section — len ≥ 1; deeper than 1
162    /// means a nested child.
163    pub path: Vec<usize>,
164}
165
166impl TaskLoc {
167    /// Is this a nested child (not a top-level line)?
168    pub fn is_child(&self) -> bool {
169        self.path.len() > 1
170    }
171}
172
173impl Ledger {
174    /// Load both docs.
175    pub fn load(store: &MdStore) -> Result<Self, String> {
176        Ok(Ledger {
177            todo: store.todo().map_err(|e| e.to_string())?,
178            backlog: store.backlog().map_err(|e| e.to_string())?,
179        })
180    }
181
182    /// Locate a task in either doc — top-level lines and nested
183    /// children alike (`None` when it's not a task or doesn't
184    /// exist).
185    pub fn locate(&self, id: ItemId) -> Option<TaskLoc> {
186        find_pos(&self.todo.sections, id)
187            .map(|(section, path)| TaskLoc {
188                backlog: false,
189                section,
190                path,
191            })
192            .or_else(|| {
193                find_pos(&self.backlog.sections, id).map(|(section, path)| TaskLoc {
194                    backlog: true,
195                    section,
196                    path,
197                })
198            })
199    }
200
201    /// The item at `loc`.
202    pub fn item(&self, loc: &TaskLoc) -> &Item {
203        let sections = if loc.backlog {
204            &self.backlog.sections
205        } else {
206            &self.todo.sections
207        };
208        item_at(&sections[loc.section].items, &loc.path)
209    }
210
211    /// Mutable access to the item at `loc`.
212    pub fn item_mut(&mut self, loc: &TaskLoc) -> &mut Item {
213        let sections = if loc.backlog {
214            &mut self.backlog.sections
215        } else {
216            &mut self.todo.sections
217        };
218        item_at_mut(&mut sections[loc.section].items, &loc.path)
219    }
220
221    /// Remove the item at `loc` — the whole subtree goes with it
222    /// (`done` on a parent removes its children too). A section
223    /// the removal emptied is dropped (an empty unsectioned run
224    /// can't even be emitted).
225    pub fn take(&mut self, loc: &TaskLoc) -> Item {
226        let sections = if loc.backlog {
227            &mut self.backlog.sections
228        } else {
229            &mut self.todo.sections
230        };
231        let mut level = &mut sections[loc.section].items;
232        for &idx in &loc.path[..loc.path.len() - 1] {
233            level = &mut level[idx].children;
234        }
235        let item = level.remove(loc.path[loc.path.len() - 1]);
236        if loc.path.len() == 1 && sections[loc.section].items.is_empty() {
237            sections.remove(loc.section);
238        }
239        item
240    }
241
242    /// Swap the item at `loc` with a sibling — `delta` -1 moves up,
243    /// +1 down. Works on nested children (siblings share the
244    /// parent). Returns false at the list edges.
245    pub fn move_within(&mut self, loc: &mut TaskLoc, delta: isize) -> bool {
246        let sections = if loc.backlog {
247            &mut self.backlog.sections
248        } else {
249            &mut self.todo.sections
250        };
251        let mut level = &mut sections[loc.section].items;
252        for &idx in &loc.path[..loc.path.len() - 1] {
253            level = &mut level[idx].children;
254        }
255        let i = loc.path[loc.path.len() - 1] as isize;
256        let j = i + delta;
257        if j < 0 || j as usize >= level.len() {
258            return false;
259        }
260        level.swap(i as usize, j as usize);
261        let last = loc.path.len() - 1;
262        loc.path[last] = j as usize;
263        true
264    }
265
266    /// Append to a TODO.md section: `--section` names one (created
267    /// when absent); default = the first section, else an
268    /// unsectioned run.
269    pub fn put_todo(&mut self, item: Item, section: Option<&str>) {
270        push_item(&mut self.todo.sections, item, section);
271    }
272
273    /// Append to BACKLOG.md — same placement rule as [`put_todo`].
274    pub fn put_backlog(&mut self, item: Item, section: Option<&str>) {
275        push_item(&mut self.backlog.sections, item, section);
276    }
277
278    /// Write both docs back.
279    pub fn save(&self, store: &MdStore) -> Result<(), String> {
280        store.save_todo(&self.todo).map_err(|e| e.to_string())?;
281        store.save_backlog(&self.backlog).map_err(|e| e.to_string())
282    }
283}

285fn find_pos(sections: &[Section], id: ItemId) -> Option<(usize, Vec<usize>)> {
286    sections.iter().enumerate().find_map(|(si, s)| {
287        let mut path = Vec::new();
288        find_path(&s.items, id, &mut path).then_some((si, path))
289    })
290}

292/// Depth-first id search — pushes the index trail onto `path`.
293fn find_path(items: &[Item], id: ItemId, path: &mut Vec<usize>) -> bool {
294    for (i, item) in items.iter().enumerate() {
295        path.push(i);
296        if item.id == id || find_path(&item.children, id, path) {
297            return true;
298        }
299        path.pop();
300    }
301    false
302}

304fn item_at<'a>(mut items: &'a [Item], path: &[usize]) -> &'a Item {
305    for &idx in &path[..path.len() - 1] {
306        items = &items[idx].children;
307    }
308    &items[path[path.len() - 1]]
309}
310
311fn item_at_mut<'a>(mut items: &'a mut [Item], path: &[usize]) -> &'a mut Item {
312    for &idx in &path[..path.len() - 1] {
313        items = &mut items[idx].children;
314    }
315    &mut items[path[path.len() - 1]]
316}

318fn push_item(sections: &mut Vec<Section>, item: Item, section: Option<&str>) {
319    match section {
320        Some(name) => {
321            if let Some(s) = sections
322                .iter_mut()
323                .find(|s| s.name.as_deref() == Some(name))
324            {
325                s.items.push(item);
326            } else {
327                sections.push(Section {
328                    name: Some(name.to_string()),
329                    items: vec![item],
330                });
331            }
332        }
333        None => {
334            if sections.is_empty() {
335                sections.push(Section::unsectioned());
336            }
337            sections[0].items.push(item);
338        }
339    }
340}

342/// The shortest legal path `from → to` through the state machine
343/// (BFS over [`Status::transitions`]). Convenience verbs — `done`,
344/// `defer`, `promote`, `resolve`, proposal `status`, ADR
345/// `supersede` — walk multi-step paths (e.g. `open → in-progress →
346/// done`), so every hop stays machine-validated.
347pub fn status_path(id: ItemId, from: Status, to: Status) -> Result<Vec<Status>, String> {
348    if from == to {
349        return Err(format!("{id} is already {to}"));
350    }
351    let mut prev: BTreeMap<Status, Status> = BTreeMap::new();
352    let mut q = VecDeque::from([from]);
353    while let Some(cur) = q.pop_front() {
354        if cur == to {
355            break;
356        }
357        for &next in cur.transitions() {
358            if next != from && !prev.contains_key(&next) {
359                prev.insert(next, cur);
360                q.push_back(next);
361            }
362        }
363    }
364    if !prev.contains_key(&to) {
365        return Err(format!(
366            "{id}: invalid status transition {from} → {to} (valid from {from}: {})",
367            targets(from)
368        ));
369    }
370    let mut path = vec![to];
371    while let Some(&cur) = path.last() {
372        if cur == from {
373            break;
374        }
375        path.push(prev[&cur]);
376    }
377    path.pop();
378    path.reverse();
379    Ok(path)
380}

382/// `"claimed, in-progress, deferred"` / `"none (terminal)"` — the
383/// valid-targets suffix on transition errors.
384pub fn targets(from: Status) -> String {
385    let t: Vec<&str> = from.transitions().iter().map(|s| s.token()).collect();
386    if t.is_empty() {
387        "none (terminal)".into()
388    } else {
389        t.join(", ")
390    }
391}
392
393/// Walk `item` along [`status_path`] to `to`, validating each hop.
394pub fn walk_to(item: &mut Item, to: Status) -> Result<(), String> {
395    for step in status_path(item.id, item.status, to)? {
396        item.status = item.status.transition_to(step).map_err(|e| e.to_string())?;
397    }
398    Ok(())
399}

401/// Walk every descendant to `to` — subtree moves keep the ledger
402/// consistent: a deferred parent's children can't stay open in
403/// BACKLOG.md, and children park/unpark with their parent. Nodes
404/// already at `to` are left alone (identity isn't a transition).
405pub fn walk_descendants_to(item: &mut Item, to: Status) -> Result<(), String> {
406    for c in &mut item.children {
407        if c.status != to {
408            walk_to(c, to)?;
409        }
410        walk_descendants_to(c, to)?;
411    }
412    Ok(())
413}

415/// Verbs are kind-locked — `myco issue resolve T-001` is a usage
416/// error, not a lookup failure.
417pub fn require_kind(id: ItemId, kind: ItemKind, verb: &str) -> Result<(), String> {
418    if id.kind() == kind {
419        Ok(())
420    } else {
421        Err(format!(
422            "{verb} takes {}-NNN ids — {id} is a {} item",
423            kind.prefix(),
424            id.kind().prefix()
425        ))
426    }
427}

429/// Load a file-backed item for mutation. Archived items are
430/// refused — `save_item` writes to the live dir, so editing one
431/// would silently un-archive it.
432pub fn item_for_edit(store: &MdStore, id: ItemId) -> Result<ItemFile, String> {
433    if store.is_archived(id).map_err(|e| e.to_string())? {
434        return Err(format!("{id} is archived — mutating it is refused"));
435    }
436    store.item(id).map_err(|e| e.to_string())
437}

476/// `T-001 open pri:high owner:devin #core title` — the compact
477/// human list line.
478pub fn compact(item: &Item) -> String {
479    let mut s = format!("{} {} pri:{}", item.id, item.status, item.priority.token());
480    if let Some(o) = &item.owner {
481        s += &format!(" owner:{o}");
482    }
483    for l in &item.labels {
484        s += &format!(" #{l}");
485    }
486    if let Some(e) = &item.epic {
487        s += &format!(" epic:{e}");
488    }
489    if let Some(m) = &item.milestone {
490        s += &format!(" ms:{m}");
491    }
492    if !item.title.is_empty() {
493        s += &format!(" {}", item.title);
494    }
495    s
496}

498/// `--json` → a bare array of item objects; else compact lines,
499/// paged when tall.
500pub fn print_list(items: &[Item], json_out: bool) -> Result<i32, String> {
501    if json_out {
502        let v: Vec<serde_json::Value> = items
503            .iter()
504            .map(|i| serde_json::to_value(i).unwrap_or(json!(null)))
505            .collect();
506        ops::print_json(&json!(v));
507    } else {
508        let out: Vec<String> = items.iter().map(compact).collect();
509        ops::print_paged(&out.join("\n"));
510    }
511    Ok(0)
512}

514/// Depth-tagged pre-order view — task list verbs indent children
515/// under their parent by it.
516pub fn tree_view(items: &[Item]) -> Vec<(usize, &Item)> {
517    fn go<'a>(item: &'a Item, depth: usize, out: &mut Vec<(usize, &'a Item)>) {
518        out.push((depth, item));
519        for c in &item.children {
520            go(c, depth + 1, out);
521        }
522    }
523    let mut out = Vec::new();
524    for i in items {
525        go(i, 0, &mut out);
526    }
527    out
528}

530/// Like [`print_list`] for depth-tagged items — human output
531/// indents children 2 spaces per level under their parent.
532pub fn print_tree(pairs: &[(usize, &Item)], json_out: bool) -> Result<i32, String> {
533    if json_out {
534        let v: Vec<serde_json::Value> = pairs
535            .iter()
536            .map(|(_, i)| serde_json::to_value(i).unwrap_or(json!(null)))
537            .collect();
538        ops::print_json(&json!(v));
539    } else {
540        let out: Vec<String> = pairs
541            .iter()
542            .map(|(depth, i)| format!("{}{}", "  ".repeat(*depth), compact(i)))
543            .collect();
544        ops::print_paged(&out.join("\n"));
545    }
546    Ok(0)
547}

549/// Input bundle for [`new_file`] — one per `<kind> new` call.
550/// `word` is the commit-message noun (`chore: new issue I-001`).
551pub struct NewItem<'a> {
552    /// The kind of item to create.
553    pub kind: ItemKind,
554    /// One-line title (drives the filename slug).
555    pub title: String,
556    /// `--body/--text` flags — the template fills in when absent.
557    pub body: &'a BodyFlags,
558    /// `--pri/--owner/--label/--blocks` flags.
559    pub meta: &'a MetaFlags,
560    /// The kind's default body template.
561    pub template: fn(&Item) -> String,
562    /// Commit-message noun — `chore: new <word> <id>`.
563    pub word: &'a str,
564    /// `--json` output.
565    pub json_out: bool,
566    /// `--no-commit`.
567    pub no_commit: bool,
568}

570/// The shared `<kind> new` body — alloc id → apply meta flags →
571/// body from `--body/--text` or `template` → `save_item` → commit
572/// + report.
573pub fn new_file(
574    proj: &discover::Project,
575    store: &MdStore,
576    spec: &NewItem<'_>,
577) -> Result<i32, String> {
578    let id = store.alloc_id(spec.kind).map_err(|e| e.to_string())?;
579    let mut item = Item::new(id, &spec.title).created(today());
580    spec.meta.apply(&mut item)?;
581    let body = spec.body.read()?.unwrap_or_else(|| (spec.template)(&item));
582    let file = ItemFile::new(item.clone()).body(body);
583    store.save_item(&file).map_err(|e| e.to_string())?;
584    let rel = item_rel_path(&item);
585    let committed = commit(
586        &proj.worktree,
587        &format!("chore: new {} {id}", spec.word),
588        spec.no_commit,
589    )?;
590    report(
591        json!({"item": item, "path": rel}),
592        committed,
593        spec.json_out,
594        &format!("created {id} {rel}"),
595    )
596}

598/// The shared `<kind> list` — live item files, plus `archive/`
599/// entries when `include_archived`.
600pub fn list_files(
601    store: &MdStore,
602    kind: ItemKind,
603    include_archived: bool,
604    list: &ListFlags,
605    json_out: bool,
606) -> Result<i32, String> {
607    let mut found = store.list_items(kind).map_err(|e| e.to_string())?;
608    if include_archived {
609        found.extend(store.list_archived(kind).map_err(|e| e.to_string())?);
610    }
611    let found: Vec<Item> = found
612        .into_iter()
613        .map(|f| f.meta)
614        .filter(|i| list.matches(i))
615        .take(list.take())
616        .collect();
617    print_list(&found, json_out)
618}

628/// Locate an item anywhere in the ledger — TODO.md, BACKLOG.md, or
629/// an item file (live or `archive/`). Returns the item, its
630/// location label, and the body for file-backed items.
631pub fn find_item(
632    store: &MdStore,
633    id: ItemId,
634) -> Result<Option<(Item, String, Option<String>)>, String> {
635    let ledger = Ledger::load(store)?;
636    if let Some(loc) = ledger.locate(id) {
637        let file = if loc.backlog { "BACKLOG.md" } else { "TODO.md" };
638        return Ok(Some((ledger.item(&loc).clone(), file.to_string(), None)));
639    }
640    match store.item(id) {
641        Ok(f) => {
642            let mut loc = item_rel_path(&f.meta);
643            if store.is_archived(id).map_err(|e| e.to_string())? {
644                if let Some((d, n)) = loc.split_once('/') {
645                    loc = format!("{d}/archive/{n}");
646                }
647            }
648            Ok(Some((f.meta.clone(), loc, Some(f.body))))
649        }
650        Err(myco_store::Error::NotFound { .. }) => Ok(None),
651        Err(e) => Err(e.to_string()),
652    }
653}

733    fn ledger_locates_and_takes_children() {
734        let dir = tempfile::tempdir().unwrap();
735        let root = camino::Utf8PathBuf::from_path_buf(dir.path().join("p")).unwrap();
736        std::fs::create_dir_all(&root).unwrap();
737        let store = MdStore::new(root);
738        let t = |n: u32| Item::new(ItemId::new(ItemKind::Task, n), format!("t{n}"));
739        let mut ledger = Ledger::load(&store).unwrap();
740        ledger.put_todo(t(1).child(t(2).child(t(4))).child(t(3)), None);
741        ledger.put_todo(t(5), None);
742        // locate walks the whole tree — the path says where
743        let loc = ledger.locate(ItemId::new(ItemKind::Task, 4)).unwrap();
744        assert_eq!(loc.path, vec![0, 0, 0]);
745        assert!(loc.is_child());
746        assert_eq!(ledger.item(&loc).id.number(), 4);
747        // take on a child removes just that node
748        ledger.take(&loc);
749        let parent = &ledger.todo.sections[0].items[0];
750        assert_eq!(parent.children[0].children.len(), 0);
751        // take on a parent removes the whole subtree; sections
752        // stay (a sibling remains)
753        let loc = ledger.locate(ItemId::new(ItemKind::Task, 1)).unwrap();
754        let gone = ledger.take(&loc);
755        assert_eq!(gone.children.len(), 2);
756        assert!(ledger.locate(ItemId::new(ItemKind::Task, 2)).is_none());
757        assert_eq!(ledger.todo.sections[0].items.len(), 1);
758        // mutation in place reaches children too
759        let loc = ledger.locate(ItemId::new(ItemKind::Task, 5)).unwrap();
760        ledger.item_mut(&loc).children.push(t(6));
761        assert!(ledger.locate(ItemId::new(ItemKind::Task, 6)).is_some());
762        ledger.save(&store).unwrap();
763        assert!(Ledger::load(&store)
764            .unwrap()
765            .locate(ItemId::new(ItemKind::Task, 6))
766            .is_some());
767    }

770    fn move_within_swaps_siblings() {
771        let dir = tempfile::tempdir().unwrap();
772        let root = camino::Utf8PathBuf::from_path_buf(dir.path().join("p")).unwrap();
773        std::fs::create_dir_all(&root).unwrap();

780        // edges don't move
781        let mut loc = ledger.locate(ItemId::new(ItemKind::Task, 1)).unwrap();

787        // nested children reorder among themselves
788        let mut loc = ledger.locate(ItemId::new(ItemKind::Task, 4)).unwrap();
789        assert!(!ledger.move_within(&mut loc, -1));
790        ledger.save(&store).unwrap();
791        let reloaded = Ledger::load(&store).unwrap();
792        assert_eq!(reloaded.todo.sections[0].items[1].id.number(), 1);
