1//! OTLP/HTTP (JSON) trace export for audit events — the
2//! observability wire that sits beside stderr ndjson.
3//!
4//! Enabled by `OTEL_EXPORTER_OTLP_ENDPOINT` (standard OTel env;
5//! `/v1/traces` is appended unless the endpoint already ends with
6//! it). `OTEL_EXPORTER_OTLP_HEADERS` carries `k=v,k2=v2` (values
7//! percent-decoded per spec), `OTEL_EXPORTER_OTLP_TIMEOUT` is
8//! milliseconds (default 10s), `OTEL_SERVICE_NAME` names the
9//! resource (default `cuora`), and `OTEL_SDK_DISABLED=true` turns
10//! the whole export off.
11//!
12//! Each [`AuditEvent`] becomes one internal span named
13//! `cuora.<op>` — instant (start == end) — with `cuora.*`
14//! attributes. Export is best-effort: a failed POST warns once on
15//! stderr and never fails the op — audit must not take the broker
16//! down, same posture as [`super::audit::emit`].
17//!
18//! Sync only (ureq, no async runtime) per AGENTS.md. The wire is
19//! OTLP/HTTP JSON — hand-rolled payload, no opentelemetry crates,
20//! keeping the dep tree light.
21

29/// Resolved once per process — env doesn't change mid-run.
30struct Otel {
31    /// Base endpoint, e.g. `http://localhost:4318`.
32    endpoint: String,
33    /// `OTEL_EXPORTER_OTLP_HEADERS`, decoded.
34    headers: Vec<(String, String)>,
35    /// `service.name` resource attribute.
36    service: String,
37    /// `OTEL_EXPORTER_OTLP_TIMEOUT` (ms), spec default 10s.
38    timeout: Duration,
39}
40
41static CONFIG: OnceLock<Option<Otel>> = OnceLock::new();
42
43fn config() -> Option<&'static Otel> {
44    CONFIG
45        .get_or_init(|| {
46            let get = |k: &str| std::env::var(k).ok().filter(|v| !v.is_empty());
47            otel_from(
48                get("OTEL_EXPORTER_OTLP_ENDPOINT").as_deref(),
49                get("OTEL_EXPORTER_OTLP_HEADERS").as_deref(),
50                get("OTEL_SERVICE_NAME").as_deref(),
51                get("OTEL_EXPORTER_OTLP_TIMEOUT").as_deref(),
52                get("OTEL_SDK_DISABLED").as_deref(),
53            )
54        })
55        .as_ref()
56}
57
58/// Whether OTLP export is configured — `audit::emit` buffers only
59/// when this is true, so the no-endpoint path allocates nothing.
60pub(crate) fn enabled() -> bool {
61    config().is_some()
62}

64/// Resolve the config from raw env values — the testable core;
65/// `None` endpoint or `OTEL_SDK_DISABLED` disables export.
66fn otel_from(
67    endpoint: Option<&str>,
68    headers: Option<&str>,
69    service: Option<&str>,
70    timeout_ms: Option<&str>,
71    sdk_disabled: Option<&str>,
72) -> Option<Otel> {
73    if sdk_disabled.is_some_and(|v| matches!(v.to_ascii_lowercase().as_str(), "true" | "1")) {
74        return None;
75    }
76    let endpoint = endpoint?.trim_end_matches('/').to_string();
77    if endpoint.is_empty() {
78        return None;
79    }
80    // Same policy as cf vault base URLs: `http://` only to a
81    // loopback collector — these spans carry secret *names*.
82    if !endpoint_allowed(&endpoint) {
83        cuora_notice::note(
84            "OTEL_EXPORTER_OTLP_ENDPOINT uses http:// to a non-loopback \
85             host — export disabled (use https:// or a loopback collector)",
86        );
87        return None;
88    }
89    Some(Otel {
90        endpoint,
91        headers: parse_headers(headers.unwrap_or("")),
92        service: service.unwrap_or("cuora").to_string(),
93        timeout: timeout_ms
94            .and_then(|v| v.parse::<u64>().ok())
95            .map(Duration::from_millis)
96            .unwrap_or(Duration::from_secs(10)),
97    })
98}

100/// `https://` anywhere, `http://` only to loopback — mirrors the
101/// base-URL rule enforced for cf vaults and deploy sinks.
102fn endpoint_allowed(endpoint: &str) -> bool {
103    let Some(rest) = endpoint.strip_prefix("http://") else {
104        return true; // https:// or a scheme we don't police
105    };
106    let authority = rest.split('/').next().unwrap_or("");
107    let hostport = authority.rsplit('@').next().unwrap_or(authority);
108    let host = if let Some(b) = hostport.strip_prefix('[') {
109        b.split(']').next().unwrap_or("")
110    } else {
111        hostport.split(':').next().unwrap_or("")
112    };
113    matches!(host, "localhost" | "127.0.0.1" | "::1")
114}

116/// `k=v,k2=v2` — spec says values may be percent-encoded; decode
117/// both sides leniently (a malformed pair is skipped, never fatal).
118fn parse_headers(raw: &str) -> Vec<(String, String)> {
119    raw.split(',')
120        .filter_map(|pair| {
121            let (k, v) = pair.split_once('=')?;
122            let k = percent_decode(k.trim());
123            let v = percent_decode(v.trim());
124            (!k.is_empty() && !v.is_empty()).then_some((k, v))
125        })
126        .collect()
127}
128
129/// Minimal `%XX` decode — leaves non-escape bytes (and a stray
130/// `%`) untouched.
131fn percent_decode(s: &str) -> String {
132    let b = s.as_bytes();
133    let mut out = Vec::with_capacity(b.len());
134    let mut i = 0;
135    while i < b.len() {
136        if b[i] == b'%' && i + 2 < b.len() {
137            if let Ok(v) = u8::from_str_radix(&s[i + 1..i + 3], 16) {
138                out.push(v);
139                i += 3;
140                continue;
141            }
142        }
143        out.push(b[i]);
144        i += 1;
145    }
146    String::from_utf8_lossy(&out).into_owned()
147}

149/// `{endpoint}/v1/traces` — unless the endpoint already carries
150/// the signal path (some collectors are configured per-signal).
151fn trace_url(endpoint: &str) -> String {
152    let base = endpoint.trim_end_matches('/');
153    if base.ends_with("/v1/traces") {
154        base.to_string()
155    } else {
156        format!("{base}/v1/traces")
157    }
158}

160/// POST the buffered events as one `ExportTraceServiceRequest`.
161/// Any failure is the caller's warning, not an op failure.
162pub(crate) fn export(events: &[AuditEvent]) -> Result<(), String> {
163    let cfg = config().ok_or("otel export disabled")?;
164    let body = serde_json::to_string(&trace_payload(cfg, events))
165        .map_err(|e| format!("serialize: {e}"))?;
166    let agent: ureq::Agent = ureq::Agent::config_builder()
167        .timeout_global(Some(cfg.timeout))
168        .build()
169        .into();
170    let mut rb = agent.post(&trace_url(&cfg.endpoint));
171    for (k, v) in &cfg.headers {
172        rb = rb.header(k.as_str(), v.as_str());
173    }
174    let mut res = rb
175        .config()
176        .http_status_as_error(false)
177        .build()
178        .content_type("application/json")
179        .send(body)
180        .map_err(|e| e.to_string())?;
181    let status = res.status().as_u16();
182    // Drain a bounded body so the connection can be reused; the
183    // payload is a collector ack, never interesting.
184    let _ = res
185        .body_mut()
186        .with_config()
187        .limit(64 * 1024)
188        .read_to_string();
189    if !(200..300).contains(&status) {
190        return Err(format!("HTTP {status} from {}", trace_url(&cfg.endpoint)));
191    }
192    Ok(())
193}

195/// The OTLP/HTTP JSON payload — pure, so tests pin the wire shape
196/// without a collector.
197fn trace_payload(cfg: &Otel, events: &[AuditEvent]) -> Value {
198    let spans: Vec<Value> = events.iter().map(span_json).collect();
199    json!({
200        "resourceSpans": [{
201            "resource": {
202                "attributes": [{
203                    "key": "service.name",
204                    "value": { "stringValue": cfg.service },
205                }],
206            },
207            "scopeSpans": [{
208                "scope": {
209                    "name": "cuora-ops",
210                    "version": env!("CARGO_PKG_VERSION"),
211                },
212                "spans": spans,
213            }],
214        }],
215    })
216}

218/// One audit event → one instant internal span. `traceId`/`spanId`
219/// are random per event — audit events have no parent context.
220fn span_json(ev: &AuditEvent) -> Value {
221    let nanos = ev.ts.unix_timestamp_nanos().to_string();
222    let mut attrs = vec![
223        json!({"key": "cuora.op", "value": {"stringValue": ev.op.as_str()}}),
224        json!({"key": "cuora.env", "value": {"stringValue": ev.env}}),
225        json!({"key": "cuora.actor", "value": {"stringValue": ev.actor}}),
226        json!({"key": "cuora.outcome", "value": {"stringValue": outcome_str(ev.outcome)}}),
227    ];
228    if let Some(rref) = &ev.rref {
229        attrs.push(json!({"key": "cuora.rref", "value": {"stringValue": rref}}));
230    }
231    if let Some(p) = &ev.policy {
232        attrs.push(json!({"key": "cuora.policy.allow", "value": {"boolValue": p.allow}}));
233        if let Some(reason) = &p.reason {
234            attrs.push(json!({"key": "cuora.policy.reason", "value": {"stringValue": reason}}));
235        }
236        if let Some(obs) = &p.obligations {
237            attrs.push(json!({
238                "key": "cuora.policy.obligations",
239                "value": {"stringValue": obs.join(",")},
240            }));
241        }
242    }
243    let mut span = json!({
244        "traceId": rand_hex(32),
245        "spanId": rand_hex(16),
246        "name": format!("cuora.{}", ev.op.as_str()),
247        "kind": "SPAN_KIND_INTERNAL",
248        "startTimeUnixNano": nanos,
249        "endTimeUnixNano": nanos,
250        "attributes": attrs,
251    });
252    if ev.outcome == Outcome::Error {
253        span["status"] = json!({"code": "STATUS_CODE_ERROR"});
254    }
255    span
256}
257
258fn outcome_str(o: Outcome) -> &'static str {
259    match o {
260        Outcome::Ok => "ok",
261        Outcome::Denied => "denied",
262        Outcome::Error => "error",
263    }
264}

266/// `digits` hex chars from a per-process random state plus a
267/// counter — trace/span ids need uniqueness, not secrecy.
268fn rand_hex(digits: usize) -> String {
269    use std::collections::hash_map::RandomState;
270    use std::hash::{BuildHasher, Hasher};
271    // Fresh keys per call — `RandomState::new` draws OS entropy, so
272    // ids differ across processes and calls.
273    let state = RandomState::new();
274    let mut out = String::with_capacity(digits);
275    let mut i = 0u64;
276    while out.len() < digits {
277        let mut h = state.build_hasher();
278        h.write_u64(i);
279        h.write_u128(
280            std::time::SystemTime::now()
281                .duration_since(std::time::UNIX_EPOCH)
282                .map(|d| d.as_nanos())
283                .unwrap_or(0),
284        );
285        out.push_str(&format!("{:016x}", h.finish()));
286        i += 1;
287    }
288    out.truncate(digits);
289    out
290}

298    #[test]
299    fn disabled_without_endpoint_or_when_sdk_disabled() {
300        assert!(otel_from(None, None, None, None, None).is_none());
301        assert!(otel_from(Some(""), None, None, None, None).is_none());
302        assert!(otel_from(Some("http://x:4318"), None, None, None, Some("true")).is_none());
303        assert!(otel_from(Some("http://x:4318"), None, None, None, Some("1")).is_none());
304        assert!(otel_from(Some("https://x:4318"), None, None, None, Some("false")).is_some());
305    }
306
307    #[test]
308    fn http_endpoints_must_be_loopback() {
309        // secret names ride these spans — plaintext http off-box
310        // is refused, same policy as cf vault base URLs.
311        assert!(otel_from(Some("http://collector:4318"), None, None, None, None).is_none());
312        assert!(otel_from(Some("http://evil.example.com"), None, None, None, None).is_none());
313        assert!(otel_from(Some("http://user@collector:4318"), None, None, None, None).is_none());
314        for ok in [
315            "http://localhost:4318",
316            "http://127.0.0.1:4318",
317            "http://[::1]:4318",
318            "https://collector.example.com",
319        ] {
320            assert!(
321                otel_from(Some(ok), None, None, None, None).is_some(),
322                "{ok}"
323            );
324        }
325    }

353    #[test]
354    fn payload_is_otlp_json_with_cuora_attrs() {
355        let cfg = otel_from(
356            Some("https://otel:4318"),
357            Some("k=v"),
358            Some("cuora-test"),
359            Some("2500"),
360            None,
361        )
362        .unwrap();
363        assert_eq!(cfg.timeout, Duration::from_millis(2500));
364        let ev = AuditEvent::op(
365            "devin",
366            cuora_core::Op::Get,
367            Some("billing/stripe"),
368            "prod",
369            Outcome::Denied,
370        )
371        .with_policy(Some(PolicyVerdict {
372            allow: false,
373            reason: Some("no prod".into()),
374            obligations: None,
375        }));
376        let v = trace_payload(&cfg, &[ev]);
377        let s = &v["resourceSpans"][0]["scopeSpans"][0]["spans"][0];
378        assert_eq!(s["name"], "cuora.get");
379        assert_eq!(s["kind"], "SPAN_KIND_INTERNAL");
380        assert_eq!(s["traceId"].as_str().unwrap().len(), 32);
381        assert_eq!(s["spanId"].as_str().unwrap().len(), 16);
382        let attrs = s["attributes"].as_array().unwrap();
383        let get = |k: &str| {
384            attrs
385                .iter()
386                .find(|a| a["key"] == k)
387                .map(|a| a["value"].clone())
388                .unwrap()
389        };
390        assert_eq!(get("cuora.op")["stringValue"], "get");
391        assert_eq!(get("cuora.rref")["stringValue"], "billing/stripe");
392        assert_eq!(get("cuora.outcome")["stringValue"], "denied");
393        assert_eq!(get("cuora.policy.allow")["boolValue"], false);
394        assert_eq!(get("cuora.policy.reason")["stringValue"], "no prod");
395        assert_eq!(
396            v["resourceSpans"][0]["resource"]["attributes"][0]["value"]["stringValue"],
397            "cuora-test"
398        );
399        // error outcome marks the span
400        let ev = AuditEvent::op("a", cuora_core::Op::List, None, "dev", Outcome::Error);
401        let v = trace_payload(&cfg, &[ev]);
402        assert_eq!(
403            v["resourceSpans"][0]["scopeSpans"][0]["spans"][0]["status"]["code"],
404            "STATUS_CODE_ERROR"
405        );
406    }
