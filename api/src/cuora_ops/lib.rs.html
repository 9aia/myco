20/// Read a `CUORA_*` env var with the pre-rename `ARCA_*` name as
21/// fallback — the rename keeps existing exports working. Only
22/// meaningful for `CUORA_`-prefixed names; anything else reads
23/// literally.
24pub fn env_var(name: &str) -> Option<String> {
25    let read = |n: &str| std::env::var(n).ok().filter(|v| !v.is_empty());
26    read(name).or_else(|| {
27        name.strip_prefix("CUORA_")
28            .and_then(|s| read(&format!("ARCA_{s}")))
29    })
30}

185    /// Load config, resolve env + actor.
186    ///
187    /// Config search: explicit path → `CUORA_CONFIG` → `./cuora.toml`
188    /// → `~/.config/cuora/cuora.toml`. No file on the default search
189    /// means defaults (local vault, `dev`) — cuora works without a
190    /// config; an explicit `--config`/`CUORA_CONFIG` that fails is an
191    /// error, as is a file that fails to parse.
192    pub fn load(opts: LoadOpts) -> Result<Self, OpsError> {
193        let src = TomlSource {
194            path: opts.config_path.clone(),
195        };
196        let located = src.locate();
197        let config = match &located {
198            Some(_) => src.load().map_err(OpsError::Config)?,
199            None if opts.config_path.is_some() => {
200                return Err(OpsError::Config(
201                    // `locate()` found nothing — a load can't succeed.
202                    src.load().err().unwrap_or("unreadable".into()),
203                ));
204            }
205            None => Config::default(),
206        };
207        let env = resolve_env(opts.env.as_deref())?;
208        let actor = resolve_actor(opts.actor.as_deref())?;
209        Ok(Ops {
210            config,
211            config_path: located,
212            env,
213            actor,
214            policy: None,
215        })
216    }

280    /// Open a vault by `[vaults]` name (`local` resolves to its
281    /// implicit defaults when unconfigured).
282    pub fn open_vault_named(&self, name: &str) -> Result<Box<dyn Vault>, OpsError> {
283        let cfg = self.config.vault_config(name);
284        let kind = cfg.map(|c| c.kind.as_str()).unwrap_or("age");
285        match kind {
286            "age" => {
287                let vault_path = cfg
288                    .and_then(|c| c.path.as_ref())
289                    .map(PathBuf::from)
290                    .unwrap_or_else(AgeVault::default_vault_path);
291                let recipients = cfg.map(|c| c.recipients.as_slice()).unwrap_or(&[]);
292                let identity = self.resolve_identity(cfg)?;
293                let v = AgeVault::open(&vault_path, identity, recipients)?;
294                self.wrap_cache(name, cfg, Box::new(v))
295            }
296            "cf" => {
297                // `kind` was read off `cfg` — "cf" here means the
298                // vault is configured, so this expect documents the
299                // invariant rather than guarding a reachable path.
300                #[allow(clippy::unwrap_used)]
301                let c = cfg.expect("kind cf implies a configured vault");
302                let v = self.open_cf_vault(name, c)?;
303                self.wrap_cache(name, cfg, Box::new(v))
304            }
305            other => Err(OpsError::Config(format!(
306                "vault {name:?} has unknown kind {other:?} (have: age, cf)"
307            ))),
308        }
309    }

311    /// Wrap `inner` in the read-through local cache when
312    /// `[vaults.<name>] cache` scopes it on (P-007). The cache is an
313    /// age vault at `$XDG_CACHE_HOME/cuora/<name>.age`, encrypted to
314    /// the same identity + recipients as the vault itself — cached
315    /// rows are ciphertext like any vault. A cache file that won't
316    /// open is moved aside and rebuilt; one that still fails
317    /// degrades to a remote-only pass-through — the remote is
318    /// always the source of truth.
319    fn wrap_cache(
320        &self,
321        name: &str,
322        cfg: Option<&VaultConfig>,
323        inner: Box<dyn Vault>,
324    ) -> Result<Box<dyn Vault>, OpsError> {
325        let scope = match cfg.and_then(|c| c.cache.as_ref()) {
326            Some(s) if s.enabled() => s,
327            _ => return Ok(inner),
328        };
329        let identity = self.resolve_identity(cfg)?;
330        let recipients = cfg.map(|c| c.recipients.as_slice()).unwrap_or(&[]);
331        let cache_path = CachedVault::default_cache_file(name);
332        let cache: Option<AgeVault> =
333            match AgeVault::open(&cache_path, identity.clone(), recipients) {
334                Ok(v) => Some(v),
335                Err(e) => {
336                    fallback_note(&format!(
337                        "cache for vault {name:?} unreadable ({e}) — rebuilding it"
338                    ));
339                    let _ = std::fs::rename(&cache_path, cache_path.with_extension("corrupt"));
340                    match AgeVault::open(&cache_path, identity, recipients) {
341                        Ok(v) => Some(v),
342                        Err(e2) => {
343                            fallback_note(&format!(
344                                "cache for vault {name:?} still unavailable ({e2}) — \
345                             remote-only reads"
346                            ));
347                            None
348                        }
349                    }
350                }
351            };
352        let ttl = Duration::from_secs(
353            cfg.and_then(|c| c.cache_ttl_secs)
354                .unwrap_or_else(|| cuora_vault_cached::DEFAULT_TTL.as_secs()),
355        );
356        let policy = CachePolicy {
357            envs: scope.envs(),
358            ttl,
359        };
360        Ok(Box::new(CachedVault::new(
361            inner,
362            cache.map(|v| Box::new(v) as Box<dyn Vault>),
363            Some(CachedVault::default_meta_file(name)),
364            policy,
365        )))
366    }

368    /// Build the concrete [`CfVault`] for a configured
369    /// `[vaults.<name>] kind = "cf"` — scoped token refs resolved,
370    /// identity resolved through the `cuora-identity` port,
371    /// `private_refs` carried through. Used by both
372    /// `open_vault_named` and `migrate` (which needs the concrete
373    /// type, not the port).
374    fn open_cf_vault(&self, name: &str, c: &cuora_core::VaultConfig) -> Result<CfVault, OpsError> {
375        let url = c
376            .url
377            .as_deref()
378            .ok_or_else(|| OpsError::Config(format!("[vaults.{name}] kind cf needs a `url`")))?;
379        let scoped = self.scoped_tokens(name, &c.tokens)?;
380        let identity = self.resolve_identity(Some(c))?;
381        let v = CfVault::new(&CfVaultConfig {
382            url: url.to_string(),
383            token_env: Some(
384                c.token_env
385                    .clone()
386                    .unwrap_or_else(|| "CUORA_CF_TOKEN".into()),
387            ),
388            scoped,
389            identity,
390            extra_recipients: c.recipients.clone(),
391            private_refs: c.private_refs,
392        })?;
393        Ok(v)
394    }

396    /// Resolve `[vaults.<name>.tokens]` — every `cuora://` ref is a
397    /// *token ref*: the token itself lives as a secret in the
398    /// `local` vault, under the env the entry scopes. The ref path
399    /// (env pin stripped) is the wire `token_id`; the stored value
400    /// is the HMAC secret, so the client sends `CuoraHmac`
401    /// signatures — the token never travels (P-003 phase 2).
402    /// Each resolution is a materialization and is audited as such.
403    fn scoped_tokens(
404        &self,
405        vault_name: &str,
406        tokens: &BTreeMap<Env, Ref>,
407    ) -> Result<BTreeMap<String, CfAuth>, OpsError> {
408        let mut out = BTreeMap::new();
409        if tokens.is_empty() {
410            return Ok(out);
411        }
412        if self
413            .config
414            .vault_config(cuora_core::config::LOCAL_VAULT)
415            .map(|c| c.kind.as_str())
416            == Some("cf")
417        {
418            return Err(OpsError::Config(format!(
419                "[vaults.{vault_name}.tokens] resolve against the local vault, \
420                 but [vaults.local] is itself kind \"cf\" — move the tokens \
421                 or the loop never terminates"
422            )));
423        }
424        let local = self.open_vault_named(cuora_core::config::LOCAL_VAULT)?;
425        for (env, rref) in tokens {
426            let bare = rref
427                .env_pin(|s| self.config.is_known_env(s))
428                .map(|(_, r)| r)
429                .unwrap_or_else(|| rref.clone());
430            let rec = local.get(env, &bare)?.ok_or_else(|| {
431                OpsError::Config(format!(
432                    "[vaults.{vault_name}.tokens.{env}] → {rref} is not stored \
433                     in the local vault env {env}"
434                ))
435            })?;
436            self.audit(Op::Get, &bare, env);
437            out.insert(
438                env.as_str().to_string(),
439                CfAuth::Hmac {
440                    id: bare.path(),
441                    secret: rec.value.expose().to_string().into(),
442                },
443            );
444        }
445        Ok(out)
446    }

448    /// `cuora migrate --private-refs` — re-key every plaintext-named
449    /// row under its derived ref_key, in each env that routes to a
450    /// cf vault with `private_refs = true` (P-004). Deliberate, not
451    /// ambient: when nothing qualifies, that's a config error with
452    /// a pointer, not a silent no-op.
453    pub fn migrate_private_refs(&self) -> Result<Vec<MigrateReport>, OpsError> {
454        let mut reports = Vec::new();
455        let mut saw_cf = false;
456        for env in self.config.envs.keys() {
457            let name = self.config.vault_for(env);
458            let cfg = match self.config.vault_config(name) {
459                Some(c) if c.kind == "cf" => c,
460                _ => continue,
461            };
462            saw_cf = true;
463            if !cfg.private_refs {
464                continue;
465            }
466            let vault = self.open_cf_vault(name, cfg)?;
467            let rep = vault.migrate_private_refs(env)?;
468            for r in &rep.rewritten {
469                self.audit(Op::Migrate, r, env);
470            }
471            reports.push(MigrateReport {
472                vault: name.to_string(),
473                env: env.clone(),
474                rewritten: rep.rewritten,
475                already_private: rep.already_private,
476            });
477        }
478        if reports.is_empty() {
479            let msg = if saw_cf {
480                "no routed cf vault has `private_refs = true` — \
481                 enable it in [vaults.*] before migrating"
482            } else {
483                "no cf vault is routed in [envs] — nothing to migrate"
484            };
485            return Err(OpsError::Config(msg.into()));
486        }
487        Ok(reports)
488    }

558    /// Identity resolution for vault opens —
559    ///
560    /// 1. an **explicit file override** (`CUORA_IDENTITY_FILE`, then
561    ///    `[vaults.*].identity`) — always honored, always required;
562    /// 2. the backend `[identity] backend` / `CUORA_IDENTITY_BACKEND`
563    ///    picks the mode: `keyring` (default) fails closed —
564    ///    unreachable store or missing identity is an error, the
565    ///    keyfile is never read; `auto` falls back to the file
566    ///    with a stderr note — a headless box is a degraded
567    ///    session, not a crash; `file` skips the store probe
568    ///    entirely.
569    fn resolve_identity(&self, cfg: Option<&VaultConfig>) -> Result<x25519::Identity, OpsError> {
570        // An explicit file pins the source — skip the store probe
571        // entirely (no dbus round-trip, no misleading fallback note).
572        if explicit_identity_file(cfg).is_some() {
573            // The pin beats even fail-closed `keyring` — say so
574            // once, since it looks like it should have errored.
575            if self.identity_backend() == IdentityBackend::Keyring {
576                fallback_note("explicit identity file pin overrides backend=keyring");
577            }
578            return self.resolve_identity_in(cfg, None);
579        }
580        match self.identity_backend() {
581            // `file` — keyfile only; never probe the session store.
582            IdentityBackend::File => self.resolve_identity_in(cfg, None),
583            // `keyring` — fail closed: unreachable store or missing
584            // identity is an error, never a silent file read.
585            IdentityBackend::Keyring => keyring_required(&*strict_keyring()?),
586            IdentityBackend::Auto => {
587                let keyring = auto_keyring_probe();
588                self.resolve_identity_in(cfg, keyring.as_deref())
589            }
590        }
591    }

593    /// Which identity backend this run uses — `CUORA_IDENTITY_BACKEND`
594    /// env wins over `[identity] backend`, absent both → `keyring`
595    /// (fail closed). An unrecognized env value is ignored with
596    /// a stderr note (config/default apply).
597    fn identity_backend(&self) -> IdentityBackend {
598        env_var("CUORA_IDENTITY_BACKEND")
599            .and_then(|v| match v.to_ascii_lowercase().as_str() {
600                "auto" => Some(IdentityBackend::Auto),
601                "keyring" => Some(IdentityBackend::Keyring),
602                "file" => Some(IdentityBackend::File),
603                _ => {
604                    // A typo'd opt-out must not silently fail
605                    // closed — say the value was ignored.
606                    fallback_note(&format!(
607                        "unrecognized CUORA_IDENTITY_BACKEND {v:?} — ignoring"
608                    ));
609                    None
610                }
611            })
612            .or_else(|| self.config.identity.as_ref().map(|i| i.backend))
613            .unwrap_or_default()
614    }

616    /// The injectable core of [`Self::resolve_identity`] — tests
617    /// pass a mock-backed `IdentitySource` through `keyring`.
618    fn resolve_identity_in(
619        &self,
620        cfg: Option<&VaultConfig>,
621        keyring: Option<&dyn IdentitySource>,

638    /// `cuora init` — store the identity per `[identity] backend`
639    /// (default `keyring`: fail closed when no session store is
640    /// reachable; `auto` falls back to the keyfile; `file` forces
641    /// it), importing an existing keyfile identity rather than
642    /// orphaning its ciphertext; then verify the vault opens.
643    /// `opts.with_file` forces the keyfile path.
644    pub fn init(&self, opts: &InitOpts) -> Result<InitInfo, OpsError> {
645        let name = self.config.vault_for(&self.env);
646        let cfg = self.config.vault_config(name);
647        let keyring: Option<Arc<dyn IdentitySource>> =
648            if opts.with_file || explicit_identity_file(cfg).is_some() {
649                if !opts.with_file
650                    && explicit_identity_file(cfg).is_some()
651                    && self.identity_backend() == IdentityBackend::Keyring
652                {
653                    fallback_note("explicit identity file pin overrides backend=keyring");
654                }
655                None
656            } else {
657                match self.identity_backend() {
658                    IdentityBackend::File => None,
659                    // Fail closed — `init` can't store into a
660                    // session store that isn't there.
661                    IdentityBackend::Keyring => Some(strict_keyring()?),
662                    IdentityBackend::Auto => auto_keyring_probe(),
663                }
664            };
665        self.init_in(
666            opts,
667            keyring.as_deref(),
668            self.identity_backend() == IdentityBackend::Keyring,
669        )
670    }

672    /// [`Self::init`] with the keyring injected — the testable core.
673    /// `strict_keyring` marks a `backend=keyring` store: a load
674    /// error there is fail-closed, never a file fallback.
675    fn init_in(
676        &self,
677        opts: &InitOpts,
678        keyring: Option<&dyn IdentitySource>,
679        strict_keyring: bool,
680    ) -> Result<InitInfo, OpsError> {
681        let (vault_path, _cfg_identity, recipients) = self.vault_paths(&self.env)?;
682        let name = self.config.vault_for(&self.env);
683        let cfg = self.config.vault_config(name);
684        let explicit = explicit_identity_file(cfg);
685        let (identity, backend, locator, created, migrated, file_remains) =
686            if let Some(src) = explicit {
687                // An explicit file pins the source — like a
688                // `--with-file` user, just configured.
689                let (id, created) = ensure_on(&src)?;
690                (id, "file", src.describe(), created, false, None)
691            } else if opts.with_file {
692                let src = FileIdentity::default();
693                let (id, created) = ensure_on(&src)?;
694                (id, "file", src.describe(), created, false, None)
695            } else if let Some(k) = keyring {
696                match k.load() {
697                    Ok(Some(id)) => (id, "keyring", k.describe(), false, false, None),
698                    Ok(None) => {
699                        // Migration: an existing keyfile identity is
700                        // imported, never orphaned — its ciphertext
701                        // must keep decrypting. The file stays as a
702                        // backup.
703                        let f = FileIdentity::default();
704                        match f.load().map_err(OpsError::Identity)? {
705                            Some(id) => {
706                                k.store(&id).map_err(OpsError::Identity)?;
707                                (
708                                    id,
709                                    "keyring",
710                                    k.describe(),
711                                    false,
712                                    true,
713                                    Some(f.path().to_path_buf()),
714                                )
715                            }
716                            None => {
717                                let (id, created) = ensure_on(k)?;
718                                (id, "keyring", k.describe(), created, false, None)
719                            }
720                        }
721                    }
722                    // Store reachable but the read failed. Under
723                    // `backend=keyring` that's still fail-closed —
724                    // falling back could mint a divergent file
725                    // identity the keyring can't decrypt. `auto`
726                    // treats it as a degraded session: bootstrap on
727                    // the file.
728                    Err(e) => {
729                        if strict_keyring {
730                            return Err(OpsError::Identity(e));
731                        }
732                        fallback_note(&format!(
733                            "keyring identity unreadable ({e}) — using the identity file"
734                        ));
735                        let src = FileIdentity::default();
736                        let (id, created) = ensure_on(&src)?;
737                        (id, "file", src.describe(), created, false, None)
738                    }
739                }
740            } else {
741                // No session store (headless, no dbus) — the file
742                // path, with a clear note rather than a crash.
743                let src = FileIdentity::default();
744                let (id, created) = ensure_on(&src)?;
745                (id, "file", src.describe(), created, false, None)
746            };
747        let vault = AgeVault::open(&vault_path, identity, &recipients)?;
748        Ok(InitInfo {
749            vault_path,
750            identity: locator,
751            identity_backend: backend,
752            identity_created: created,
753            migrated,
754            file_remains,
755            recipient: vault.public_recipient(),
756        })
757    }

884    /// Resolve every declared + ad-hoc `VAR = cuora://ref` pair into
885    /// a `VAR → Secret` map for injection. Emits one
886    /// materialization event per resolved ref. Missing refs and bad
887    /// var names fail the whole operation — partial injection is
888    /// worse than none.
889    pub fn resolve_for_injection(
890        &self,
891        op: Op,
892        extra: &[(String, String)],
893    ) -> Result<BTreeMap<String, Secret>, OpsError> {
894        // `[guardrails] deny_adhoc` closes the `deny = ["get"]`
895        // bypass: ad-hoc -s pairs can expand the injectable surface
896        // arbitrarily — under the flag they're refused (MED-5).
897        if !extra.is_empty()
898            && self
899                .config
900                .guardrails
901                .as_ref()
902                .is_some_and(|g| g.deny_adhoc)
903        {
904            self.emit(op, None, Outcome::Denied);
905            return Err(OpsError::Msg(
906                "ad-hoc -s injection is denied by [guardrails] deny_adhoc".into(),
907            ));
908        }
909        // declared surface first, then ad-hoc overrides/additions
910        let mut wanted: BTreeMap<String, Ref> = self.config.secrets.clone();
911        for (var, raw) in extra {
912            if !is_valid_env_var(var) {
913                return Err(OpsError::Msg(format!("invalid env var name {var:?}")));
914            }
915            wanted.insert(var.clone(), raw.parse::<Ref>()?);
916        }
917        // Loader-hook names execute code inside the child that also
918        // holds every other injected secret — legitimate secrets can
919        // carry them (PYTHONPATH et al), so warn rather than reject.
920        // One note listing them all; `cuora_notice::note` directly —
921        // `fallback_note`'s once-per-process slot may already be
922        // spent on an unrelated notice.
923        let hooks: Vec<String> = wanted
924            .keys()
925            .filter(|v| loader_hook_name(v))
926            .map(|v| format!("{v:?}"))
927            .collect();
928        if !hooks.is_empty() {
929            cuora_notice::note(&format!(
930                "env var(s) {} are loader-hook names — they can run \
931                 code in the injected child",
932                hooks.join(", ")
933            ));
934        }
935        let mut vaults: BTreeMap<String, Box<dyn Vault>> = BTreeMap::new();
936        let mut out = BTreeMap::new();
937        for (var, rref) in wanted {
938            if !is_valid_env_var(&var) {
939                return Err(OpsError::Msg(format!(
940                    "[secrets] key {var:?} is not a valid env var name"
941                )));
942            }
943            let (env, r) = self.resolve(&rref);
944            let vault = match vaults.entry(env.as_str().to_string()) {
945                std::collections::btree_map::Entry::Occupied(e) => e.into_mut(),
946                std::collections::btree_map::Entry::Vacant(e) => e.insert(self.open_vault(&env)?),
947            };
948            let rec = vault.get(&env, &r)?.ok_or_else(|| OpsError::NotStored {
949                var: var.clone(),
950                rref: rref.to_string(),
951                env: env.as_str().to_string(),
952            })?;
953            self.audit(op, &r, &env);
954            out.insert(var, rec.value);
955        }
956        Ok(out)
957    }

959    /// `cuora check` — validate the config, the declared surface,
960    /// and vault readability. Opens vaults read-only (no identity
961    /// generation, no writes); declared secrets that don't resolve
962    /// and undecryptable vaults are findings.
963    pub fn check(&self) -> CheckReport {
964        let mut problems = Vec::new();
965        for var in self.config.secrets.keys() {
966            if !is_valid_env_var(var) {
967                problems.push(format!("[secrets] key {var:?} is not a valid env var name"));
968            }
969        }
970        for (env, route) in &self.config.envs {
971            let name = route.vault();
972            if name != cuora_core::config::LOCAL_VAULT && !self.config.vaults.contains_key(name) {
973                problems.push(format!(
974                    "[envs] {env} routes to unconfigured vault {name:?}"
975                ));
976            }
977        }
978        for (name, v) in &self.config.vaults {
979            if !matches!(v.kind.as_str(), "age" | "cf") {
980                problems.push(format!("[vaults.{name}] kind {:?} — have: age, cf", v.kind));
981            }
982            if v.kind == "cf" && v.url.is_none() {
983                problems.push(format!("[vaults.{name}] kind cf needs a `url`"));
984            }
985            if v.private_refs && v.kind != "cf" {
986                problems.push(format!(
987                    "[vaults.{name}] private_refs only applies to kind \"cf\""
988                ));
989            }
990            if !v.tokens.is_empty() && v.kind != "cf" {
991                problems.push(format!("[vaults.{name}] tokens only apply to kind \"cf\""));
992            }
993            if v.cache_ttl_secs.is_some() && v.cache.is_none() {
994                problems.push(format!(
995                    "[vaults.{name}] cache_ttl_secs is set but cache is not"
996                ));
997            }
998        }
999        for (name, d) in &self.config.deploy {
1000            if d.kind != "cf-secrets-store" {
1001                problems.push(format!(
1002                    "[deploy.{name}] kind {:?} — have: cf-secrets-store",
1003                    d.kind
1004                ));
1005            }
1006            if d.account_id.is_none() || d.store_id.is_none() {
1007                problems.push(format!("[deploy.{name}] needs `account_id` and `store_id`"));
1008            }
1009        }
1010
1011        // Every vault worth reporting on: configured names plus
1012        // every route's target (the implicit `local` included).
1013        let mut vault_names: Vec<String> = self.config.vaults.keys().cloned().collect();
1014        for route in self.config.envs.values() {
1015            let name = route.vault().to_string();
1016            if !vault_names.contains(&name) {
1017                vault_names.push(name);
1018            }
1019        }
1020        let selected_vault = self.config.vault_for(&self.env).to_string();
1021        if !vault_names.contains(&selected_vault) {
1022            vault_names.push(selected_vault.clone());
1023        }
1024        vault_names.sort();
1025        vault_names.dedup();
1026
1027        let mut vaults = Vec::new();
1028        let mut opened: BTreeMap<String, Box<dyn Vault>> = BTreeMap::new();
1029        let mut failures: BTreeMap<String, String> = BTreeMap::new();
1030        for name in &vault_names {
1031            let cfg = self.config.vault_config(name);
1032            let kind = cfg.map(|c| c.kind.clone()).unwrap_or_else(|| "age".into());
1033            let path = cfg.and_then(|c| c.path.as_ref()).map(PathBuf::from);
1034            match self.open_vault_named(name).and_then(|v| {
1035                // Probe: construction is cheap (no I/O); a `list`
1036                // proves a remote vault is actually reachable.
1037                match v.list(&self.env) {
1038                    Ok(_) => Ok(v),
1039                    Err(e) => Err(e.into()),
1040                }
1041            }) {
1042                Ok(v) => {
1043                    opened.insert(name.clone(), v);
1044                    vaults.push(VaultCheck {
1045                        name: name.clone(),
1046                        kind,
1047                        path,
1048                        ok: true,
1049                        error: None,
1050                    });
1051                }
1052                Err(e) => {
1053                    problems.push(format!("vault {name:?} is not readable: {e}"));
1054                    failures.insert(name.clone(), e.to_string());
1055                    vaults.push(VaultCheck {
1056                        name: name.clone(),
1057                        kind,
1058                        path,
1059                        ok: false,
1060                        error: Some(e.to_string()),
1061                    });
1062                }
1063            }
1064        }
1065
1066        // Declared surface: does each VAR's ref resolve in its env?
1067        let mut secrets = Vec::new();
1068        for (var, rref) in &self.config.secrets {
1069            let (env, r) = self.resolve(rref);
1070            let vault_name = self.config.vault_for(&env);
1071            let resolves = match opened.get(vault_name) {
1072                Some(v) => v.get(&env, &r).map(|rec| rec.is_some()).unwrap_or(false),
1073                None => {
1074                    // The vault probe above already tried this name
1075                    // and it failed — a second open would fail
1076                    // identically, so report the recorded error.
1077                    let err = failures
1078                        .get(vault_name)
1079                        .map(String::as_str)
1080                        .unwrap_or("unknown error");
1081                    problems.push(format!(
1082                        "{var} → {rref}: env {env} vault {vault_name:?} not readable: {err}"
1083                    ));
1084                    false
1085                }
1086            };
1087            if !resolves && is_valid_env_var(var) {
1088                problems.push(format!("{var} → {rref} is not stored in env {env}"));
1089            }
1090            secrets.push(SecretCheck {
1091                var: var.clone(),
1092                rref: r.to_string(),
1093                env: env.as_str().to_string(),
1094                resolves,
1095            });
1096        }
1097
1098        CheckReport {
1099            config_path: self.config_path.clone(),
1100            env: self.env.clone(),
1101            vault: selected_vault,
1102            vaults,
1103            secrets,
1104            guardrails: self.config.guardrails.clone(),
1105            problems,
1106        }
1107    }

1261/// `--env` → `CUORA_ENV` → `dev`.
1262fn resolve_env(flag: Option<&str>) -> Result<Env, OpsError> {
1263    let name = flag
1264        .map(str::to_string)
1265        .or_else(|| env_var("CUORA_ENV"))
1266        .unwrap_or_else(|| "dev".into());
1267    Ok(Env::new(&name)?)
1268}

1270/// `--actor`/`CUORA_ACTOR` (must validate) → `$USER`/`$LOGNAME`
1271/// (best-effort) → `anon`.
1272fn resolve_actor(flag: Option<&str>) -> Result<Actor, OpsError> {
1273    let claimed = flag.map(str::to_string).or_else(|| env_var("CUORA_ACTOR"));
1274    if let Some(a) = claimed {
1275        return Ok(Actor::new(&a)?);
1276    }
1277    for var in ["USER", "LOGNAME"] {
1278        if let Ok(u) = std::env::var(var) {
1279            if let Ok(a) = Actor::new(&u) {
1280                return Ok(a);
1281            }
1282        }
1283    }
1284    Actor::new("anon").map_err(OpsError::Actor)
1285}
