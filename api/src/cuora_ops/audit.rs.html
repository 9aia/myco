1//! Audit events — the observability wire.
2//!
3//! ADR-0001 moved audit ownership to peregrin: no local
4//! `audit.jsonl`. Instead cuora emits ndjson lines on **stderr** —
5//! a peregrin/sepia wrapper can capture them, a human sees a
6//! breadcrumb — and, when `OTEL_EXPORTER_OTLP_ENDPOINT` is set,
7//! buffers each event for an OTLP/HTTP JSON span export on
8//! [`flush`] (see [`crate::otel`]).
9//!
10//! stderr stays the default human wire; `CUORA_AUDIT_STDERR=0`
11//! silences it (the OTLP-only mode). [`mute_stderr`] suspends
12//! live stderr writes for surfaces that own the terminal, and
13//! [`unmute_stderr`] replays the held events afterward, so
14//! nothing is lost, just deferred.
15//!
16//! Keep the struct small and serializable: it's the wire
17//! contract on both sinks.
18//!
19//! Every verb invocation leaves a trail, uniformly:
20//!
21//! - **ok** — one event per ref materialized or mutated
22//!   (`put`/`get`/`rotate`/`rm`, and per-ref inside `run`/`env`),
23//!   or one op-scoped event for ref-less verbs
24//!   (`init`/`list`/`check`).
25//! - **denied** — the op was refused by `[guardrails]` (advisory).
26//! - **error** — the op ran but failed (missing secret, vault
27//!   error, bad ref).
28//!
29//! `manifest`/`schema` run before config load — pure
30//! introspection, no event. A `run` child's non-zero exit is the
31//! child's outcome, not cuora's — no `error` event for it.
32

42/// Process-wide audit sink — the OTLP buffer plus the stderr
43/// mute a terminal-owning surface can engage. A poisoned lock is
44/// recovered rather than
45/// panicking: audit must never take the broker down.
46struct Sink {
47    /// Events held for the next [`flush`] — only populated when
48    /// `OTEL_EXPORTER_OTLP_ENDPOINT` is configured.
49    otel: Vec<AuditEvent>,
50    /// Muted mode — live stderr writes suspended; events collect
51    /// in `held` for [`unmute_stderr`] to replay once the surface
52    /// releases the terminal.
53    muted: bool,
54    held: Vec<AuditEvent>,
55}
56
57static SINK: Mutex<Sink> = Mutex::new(Sink {
58    otel: Vec::new(),
59    muted: false,
60    held: Vec::new(),
61});
62
63fn sink() -> std::sync::MutexGuard<'static, Sink> {
64    SINK.lock().unwrap_or_else(|e| e.into_inner())
65}
66
67/// `CUORA_AUDIT_STDERR` — `0`/`false`/`off`/`no` silences the
68/// stderr wire; anything else (or unset) keeps it. Resolved once:
69/// env doesn't change mid-process.
70fn stderr_enabled() -> bool {
71    static ENABLED: OnceLock<bool> = OnceLock::new();
72    *ENABLED.get_or_init(|| {
73        !matches!(
74            crate::env_var("CUORA_AUDIT_STDERR")
75                .map(|v| v.to_ascii_lowercase())
76                .as_deref(),
77            Some("0" | "false" | "off" | "no")
78        )
79    })
80}

82/// Warn-once flag for OTLP export failures — a dead collector
83/// shouldn't spam stderr once per flush.
84static EXPORT_WARNED: AtomicBool = AtomicBool::new(false);

86/// What the policy decision point said — recorded on serve-path
87/// audit events so the stream shows the verdict behind the
88/// outcome, not just the outcome. `None` on every other path:
89/// standalone CLI ops have no PDP, and the field stays absent.
90#[derive(Debug, Clone, Serialize)]
91pub struct PolicyVerdict {
92    /// The verdict — `false` is what a `denied` outcome carries.
93    pub allow: bool,
94    /// The PDP's reason, when it gave one (a deny's "why", an
95    /// allow-on-error's "the PDP was unreachable").
96    #[serde(skip_serializing_if = "Option::is_none")]
97    pub reason: Option<String>,
98    /// Conditions the PDP attached (`allow-with-conditions` —
99    /// e.g. redact, session-bound). Advisory: serve surfaces them
100    /// in the response and records them here; enforcing them is
101    /// the requester's job.
102    #[serde(skip_serializing_if = "Option::is_none")]
103    pub obligations: Option<Vec<String>>,
104}
105
106/// How an audited op ended.
107#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
108#[serde(rename_all = "lowercase")]
109pub enum Outcome {
110    /// The op did what was asked (a ref materialized/mutated, or a
111    /// ref-less verb completed).
112    Ok,
113    /// Refused by `[guardrails]` — the op never touched the vault.
114    Denied,
115    /// The op ran and failed (missing secret, vault i/o, bad ref).
116    Error,
117}
118
119/// One materialization, mutation, or op outcome record.
120#[derive(Debug, Clone, Serialize)]
121pub struct AuditEvent {
122    /// RFC 3339 timestamp.
123    #[serde(with = "time::serde::rfc3339")]
124    pub ts: OffsetDateTime,
125    /// Self-claimed actor (`--actor`/`CUORA_ACTOR`) — attribution,
126    /// not identity.
127    pub actor: String,
128    /// The verb.
129    pub op: Op,
130    /// The resolved ref path (no env pin — `env` carries that).
131    /// Absent for op-scoped events (`list`, denials before parse).
132    #[serde(skip_serializing_if = "Option::is_none")]
133    pub rref: Option<String>,
134    /// The env the op ran under.
135    pub env: String,
136    /// How it ended.
137    pub outcome: Outcome,
138    /// The PDP verdict behind this event — serve-path only.
139    #[serde(skip_serializing_if = "Option::is_none")]
140    pub policy: Option<PolicyVerdict>,
141}
142
143impl AuditEvent {
144    /// A ref-scoped success — one materialization or mutation.
145    pub fn materialized(actor: &str, op: Op, rref: &str, env: &str) -> Self {
146        Self::new(actor, op, Some(rref), env, Outcome::Ok)
147    }
148
149    /// An op-scoped event — verbs with no single ref, denials, and
150    /// failures. `rref` carries the raw requested ref when known
151    /// (e.g. a denied `get cuora://x/y`).
152    pub fn op(actor: &str, op: Op, rref: Option<&str>, env: &str, outcome: Outcome) -> Self {
153        Self::new(actor, op, rref, env, outcome)
154    }
155
156    fn new(actor: &str, op: Op, rref: Option<&str>, env: &str, outcome: Outcome) -> Self {
157        AuditEvent {
158            ts: OffsetDateTime::now_utc(),
159            actor: actor.to_string(),
160            op,
161            rref: rref.map(str::to_string),
162            env: env.to_string(),
163            outcome,
164            policy: None,
165        }
166    }
167
168    /// Attach the PDP verdict that gated this event (serve path).
169    pub fn with_policy(mut self, verdict: Option<PolicyVerdict>) -> Self {
170        self.policy = verdict;
171        self
172    }
173}

175/// Route one event to the live sinks: stderr ndjson (unless muted
176/// or `CUORA_AUDIT_STDERR` silences it) and the OTLP buffer (when
177/// `OTEL_EXPORTER_OTLP_ENDPOINT` is configured). A broken stderr
178/// must never take the broker down — write failures are
179/// deliberately swallowed, same for the buffer.
180pub fn emit(event: &AuditEvent) {
181    let stderr_on = {
182        let mut s = sink();
183        if crate::otel::enabled() {
184            s.otel.push(event.clone());
185        }
186        if s.muted {
187            s.held.push(event.clone());
188            false
189        } else {
190            stderr_enabled()
191        }
192    };
193    if stderr_on {
194        let _ = emit_to(&mut io::stderr(), event);
195    }
196}
197
198/// Suspend live stderr audit writes — for surfaces that own the
199/// terminal (an alt-screen UI, for instance). Events keep
200/// buffering for OTLP and collect in `held` for
201/// [`unmute_stderr`].
202pub fn mute_stderr() {
203    sink().muted = true;
204}
205
206/// Resume stderr writes and replay every event held while muted —
207/// call after the terminal is restored so the breadcrumb lands
208/// below the UI, not inside it.
209pub fn unmute_stderr() {
210    let held = {
211        let mut s = sink();
212        s.muted = false;
213        std::mem::take(&mut s.held)
214    };
215    if !stderr_enabled() {
216        return;
217    }
218    let mut err = io::stderr();
219    for ev in &held {
220        let _ = emit_to(&mut err, ev);
221    }
222}
223
224/// Export every buffered audit event as OTLP spans, then clear the
225/// buffer. No-op when `OTEL_EXPORTER_OTLP_ENDPOINT` is unset or
226/// the buffer is empty. Export failures warn once on stderr —
227/// audit transport must never fail the op that produced it.
228pub fn flush() {
229    let events = {
230        let mut s = sink();
231        std::mem::take(&mut s.otel)
232    };
233    if events.is_empty() {
234        return;
235    }
236    if let Err(e) = crate::otel::export(&events) {
237        if !EXPORT_WARNED.swap(true, Ordering::Relaxed) {
238            cuora_notice::note(&format!("otel audit export failed: {e}"));
239        }
240    }
241}

243/// Write `event` as one ndjson line to `w` — the testable core.
244///
245/// Serializing this POD event cannot fail in practice — every
246/// field is a string/enum — so the `expect` documents an invariant
247/// rather than guards a reachable panic.
248#[allow(clippy::unwrap_used)]
249pub fn emit_to(w: &mut impl Write, event: &AuditEvent) -> io::Result<()> {
250    let line = serde_json::to_string(event).expect("audit event serializes");
251    writeln!(w, "{line}")
252}

259    #[test]
260    fn event_is_ndjson() {
261        let ev = AuditEvent::materialized("devin", Op::Run, "billing/stripe", "prod");
262        let mut buf = Vec::new();
263        emit_to(&mut buf, &ev).unwrap();
264        let line = String::from_utf8(buf).unwrap();
265        assert!(line.ends_with('\n'));
266        let v: serde_json::Value = serde_json::from_str(&line).unwrap();
267        assert_eq!(v["actor"], "devin");
268        assert_eq!(v["op"], "run");
269        assert_eq!(v["rref"], "billing/stripe");
270        assert_eq!(v["env"], "prod");
271        assert_eq!(v["outcome"], "ok");
272        assert!(v["ts"].is_string());
273        assert!(!line.contains("value"), "audit never carries values");
274    }

334    #[test]
335    fn a_failing_writer_propagates_the_io_error() {
336        struct Dead;
337        impl Write for Dead {
338            fn write(&mut self, _: &[u8]) -> io::Result<usize> {
339                Err(io::Error::new(io::ErrorKind::BrokenPipe, "dead"))
340            }
341            fn flush(&mut self) -> io::Result<()> {
342                Ok(())
343            }
344        }
345        let mut dead = Dead;
346        // a flush on the dead writer is a no-op — only writes fail
347        dead.flush().unwrap();
348        let ev = AuditEvent::op("devin", Op::List, None, "dev", Outcome::Ok);
349        assert!(emit_to(&mut dead, &ev).is_err());
350        // and `emit` swallows the same failure — stderr can be a
351        // broken pipe, cuora must not crash
352        emit(&ev);
353    }
