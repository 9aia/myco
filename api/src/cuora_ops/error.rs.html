1//! `OpsError` — one error type for the front-ends to match.
2
3use cuora_core::{ParseActorError, ParseEnvError, ParseRefError};
4use cuora_deploy::DeployError;
5use cuora_vault::VaultError;
6
7/// Everything ops can fail with — config, resolution, vault.
8#[derive(Debug, thiserror::Error)]
9pub enum OpsError {
10    /// Config file load/parse/vault-kind failure.
11    #[error("config: {0}")]
12    Config(String),
13    /// Bad env name (`--env`/`CUORA_ENV`).
14    #[error(transparent)]
15    Env(#[from] ParseEnvError),
16    /// Bad `cuora://` ref.
17    #[error(transparent)]
18    Ref(#[from] ParseRefError),
19    /// Bad actor attribution (`--actor`/`CUORA_ACTOR`).
20    #[error(transparent)]
21    Actor(#[from] ParseActorError),
22    /// Vault backend failure.
23    #[error(transparent)]
24    Vault(#[from] VaultError),
25    /// Deploy target failure.
26    #[error(transparent)]
27    Deploy(#[from] DeployError),
28    /// Identity backend failure (keyfile, session keyring).
29    #[error(transparent)]
30    Identity(#[from] cuora_identity::IdentityError),
31    /// A declared/ad-hoc secret isn't in the vault.
32    #[error("{var} → {rref} is not stored in env {env}")]
33    NotStored {
34        /// The env-var name that wanted it.
35        var: String,
36        /// The ref it pointed at.
37        rref: String,
38        /// The env that was queried.
39        env: String,
40    },
41    /// stdin/stdout/process I/O failure.
42    #[error("i/o: {0}")]
43    Io(#[from] std::io::Error),
44    /// Catch-all with a ready-made message.
45    #[error("{0}")]
46    Msg(String),
47}
